    2 from dataclasses import dataclass, field

    6 @dataclass(frozen=True)
    7 class MeterCandidate:
    8     status: str
    9     value: int = None
   10     lower_value: float = None
   11     upper_value: float = None
   12     # Neither geometric inputs nor the selected equipment are verified.
   13     report_eligible: bool = field(default=False, init=False)
   14     scale_calibrated: bool = field(default=False, init=False)

   17 def read_angle(center, pointer_tip, zero_angle_rad, sweep_rad, angular_uncertainty_rad):
   18     """Map a manually specified 0..50 dial arc to an integer candidate.
   19 
   20     Angles use image x-right/y-down coordinates, with signed sweep. The input
   21     pointer, center, endpoints, and uncertainty must come from the same image;
   22     no association, image detection, calibration, or reporting is implied.
   23     """
   24     try:
   25         cx,cy=center;px,py=pointer_tip
   26         vals=(cx,cy,px,py,zero_angle_rad,sweep_rad,angular_uncertainty_rad)
   27         valid=(all(math.isfinite(v) for v in vals) and 0<abs(sweep_rad)<2*math.pi
   28                and 0<=angular_uncertainty_rad<abs(sweep_rad)/2 and math.hypot(px-cx,py-cy)>=2.)
   29     except (TypeError,ValueError):
   30         valid=False
   31     if not valid:raise ValueError('invalid dial geometry')
   32     direction=1 if sweep_rad>0 else -1
   33     angle=math.atan2(py-cy,px-cx)
   34     offset=(direction*(angle-zero_angle_rad))%(2*math.pi)
   35     if offset-angular_uncertainty_rad<0 or offset+angular_uncertainty_rad>abs(sweep_rad):
   36         return MeterCandidate('OUTSIDE_CALIBRATED_ARC')
   37     lower=(offset-angular_uncertainty_rad)*50/abs(sweep_rad)
   38     upper=(offset+angular_uncertainty_rad)*50/abs(sweep_rad)
   39     # Half-integer boundaries are ambiguous, including exact endpoints.
   40     first=math.floor(lower+.5);last=math.floor(upper+.5)
   41     if first!=last or abs((lower+.5)-round(lower+.5))<1e-12 or abs((upper+.5)-round(upper+.5))<1e-12:
   42         return MeterCandidate('AMBIGUOUS',None,lower,upper)
   43     return MeterCandidate('CANDIDATE',first,lower,upper)
