   32     def reset(self,epoch):
   33         if isinstance(epoch,bool) or not isinstance(epoch,int) or epoch<=self.epoch:raise ValueError('epoch must increase')

   36     def observe(self,meta,arrival,now,wall,device,lights,candidate):
   37         if (not isinstance(meta,Metadata) or not meta.current(now,self.epoch)
   38                 or not isinstance(device,str) or not device or len(device)>128
   39                 or not isinstance(lights,tuple) or len(lights)!=3 or any(x is not True and x is not False and x is not None for x in lights)
   40                 or not isinstance(candidate,MeterCandidate) or not all(math.isfinite(x) for x in (arrival,wall))
   41                 or arrival<0 or not 0<=wall-arrival<self.wall_ttl_s or (self._last_wall is not None and arrival<=self._last_wall)
   42                 or (self._last_stamp is not None and meta.stamp<=self._last_stamp)):
   43             raise ValueError('invalid or duplicate observation')
   44         self._last_stamp,self._last_wall=meta.stamp,arrival
   45         if (lights != (True,True,True) or candidate.status!='CANDIDATE'
   46                 or isinstance(candidate.value,bool) or not isinstance(candidate.value,int) or not 0<=candidate.value<=50):
   47             self._items.clear()

   49         if self._items and (self._items[-1][2]!=device or self._items[-1][3]!=candidate.value

   52         while self._items and (meta.stamp-self._items[0][0].stamp>self.history_s
   53                                or arrival-self._items[0][1]>self.history_s):self._items.popleft()

   56     def snapshot(self,now,wall):
   57         if (not self._items or not all(math.isfinite(x) for x in (now,wall))):
   58             return MeterConsensus('UNAVAILABLE',epoch=self.epoch)
   59         meta,arrival,device,value=self._items[-1]
   60         if (not meta.current(now,self.epoch) or not 0<=wall-arrival<self.wall_ttl_s
   61                 or now-self._items[0][0].stamp>self.history_s
   62                 or wall-self._items[0][1]>self.history_s):
   63             return MeterConsensus('UNAVAILABLE',epoch=self.epoch)
   64         return MeterConsensus('CONSENSUS' if len(self._items)>=self.required else 'PENDING',
   65                               device,value,len(self._items),meta.stamp,meta.valid_until,self.epoch)
