    1 """Conservative pre-control resting-window check for local takeoff experiments.
    2 
    3 Stationary IMU alone cannot distinguish constant velocity. This helper also
    4 requires GPS position consistency and must only run before control is acquired
    5 in a fresh ground-start scene. It is not an in-flight reset or outage recovery.
    6 """
    7 import numpy as np
    8 from estimation.geometry import unit
    9 
   10 
   11 def resting_window(imu,gps,initial_attitude,stamp):
   12     imu=[r for r in imu if stamp-1.1<=r['stamp']<=stamp]
   13     gps=[r for r in gps if stamp-1.1<=r[0]<=stamp]
   14     if len(imu)<65 or len(gps)<7:return None
   15     times=np.array([r['stamp'] for r in imu])
   16     if (not np.all(np.isfinite(times)) or times[-1]-times[0]<1.
   17             or np.any(np.diff(times)<=0) or np.max(np.diff(times))>.04):return None
   18     gt=np.array([r[0] for r in gps])
   19     if (not np.all(np.isfinite(gt)) or np.any(np.diff(gt)<=0)
   20             or gt[-1]-gt[0]<.6 or stamp-gt[-1]>.15):return None
   21     positions=np.array([r[1] for r in gps])
   22     if not np.all(np.isfinite(positions)) or np.any(np.all(positions==0,axis=1)):return None
   23     if np.max(np.linalg.norm(positions-positions.mean(0),axis=1))>.2:return None
   24     acceleration=np.array([r['acceleration_body'] for r in imu])
   25     gyro=np.array([r['omega_body'] for r in imu])
   26     if (acceleration.shape != (len(imu),3) or gyro.shape != (len(imu),3)
   27             or not np.all(np.isfinite(acceleration)) or not np.all(np.isfinite(gyro))
   28             or np.max(np.linalg.norm(gyro,axis=1))>.025
   29             or np.max(np.std(acceleration,axis=0))>.08
   30             or abs(np.linalg.norm(acceleration.mean(0))-9.81)>.15):return None
   31     q0=unit(initial_attitude)
   32     for r in imu:
   33         q=r.get('imu_orientation_wxyz')
   34         if q is None:return None
   35         q=np.array(q,dtype=float)
   36         if not np.all(np.isfinite(q)) or abs(np.linalg.norm(q)-1)>.01:return None
   37         if 2*np.arccos(np.clip(abs(unit(q)@q0),0.,1.))>.02:return None
   38     return {'position':positions.mean(0).tolist(),'gyro_bias':gyro.mean(0).tolist(),
   39             'imu_samples':len(imu),'gps_samples':len(gps),'span_s':float(times[-1]-times[0]),
   40             'gps_position_std_m':positions.std(0).tolist(),
   41             'acceleration_std_m_s2':acceleration.std(0).tolist(),
   42             'assumption':'fresh ground-start scene, before any control publisher',
   43             'imu_orientation_use':'resting alignment check only; initial_pose supplies attitude'}
