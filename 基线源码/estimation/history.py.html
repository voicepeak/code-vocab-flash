   51 def slerp(first, second, fraction):
   52     a, b = unit(first), unit(second)
   53     dot = float(a @ b)
   54     if dot < 0:
   55         b, dot = -b, -dot
   56     dot = min(1., max(-1., dot))
   57     if dot > .9995:
   58         return unit((1 - fraction) * a + fraction * b)
   59     angle = math.acos(dot)
   60     return unit((math.sin((1 - fraction) * angle) * a + math.sin(fraction * angle) * b) / math.sin(angle))

   63 class PoseHistory:
   64     def __init__(self, epoch=0, horizon_s=1., capacity=256, max_bracket_s=.03):
   65         if (not isinstance(epoch, int) or epoch < 0 or not math.isfinite(horizon_s)
   66                 or not .2 <= horizon_s <= 2 or not isinstance(capacity, int)
   67                 or not 4 <= capacity <= 512 or not math.isfinite(max_bracket_s)
   68                 or not .005 <= max_bracket_s <= .05):
   69             raise ValueError('invalid historical pose limits')
   70         self.epoch, self.horizon_s = epoch, horizon_s
   71         self.capacity, self.max_bracket_s = capacity, max_bracket_s
   72         self._states = deque()
   73         self._arrivals = deque()
   74         self._last_arrival = None
   75         self._lock = threading.RLock()
   76 
   77     def reset(self, epoch):
   78         with self._lock:
   79             if not isinstance(epoch, int) or epoch <= self.epoch:
   80                 raise ValueError('history epoch must increase')
   81             self.epoch = epoch
   82             self._states.clear()
   83             self._arrivals.clear()
   84             self._last_arrival = None
   85 
   86     def append(self, state, arrival_wall):
   87         with self._lock:
   88             if state.metadata.epoch != self.epoch or state.metadata.frame_id != 'world_ned':
   89                 raise ValueError('historical source frame or epoch mismatch')
   90             if (not math.isfinite(arrival_wall) or arrival_wall < 0
   91                     or (self._last_arrival is not None and arrival_wall < self._last_arrival)):
   92                 raise ValueError('invalid historical source arrival')
   93             bounds = state.metadata.covariance_or_bounds
   94             if len(bounds) != 225 or any(bounds[i * 15 + i] < 0 for i in range(15)):
   95                 raise ValueError('history requires a finite 15-state ESKF covariance')
   96             if self._states and state.metadata.stamp <= self._states[-1].metadata.stamp:
   97                 return False  # Duplicates never refresh the wall deadline.
   98             self._states.append(state)
   99             self._arrivals.append(arrival_wall)
  100             self._last_arrival = arrival_wall
  101             cutoff = state.metadata.stamp - self.horizon_s
  102             while len(self._states) > 1 and self._states[1].metadata.stamp < cutoff:
  103                 self._states.popleft()
  104                 self._arrivals.popleft()
  105             while len(self._states) > self.capacity:
  106                 self._states.popleft()
  107                 self._arrivals.popleft()
  108             return True
  109 
  110     def snapshot_sources(self):
  111         with self._lock:
  112             return tuple(zip(self._states, self._arrivals))
  113 
  114     def lookup(self, stamp, now, wall_now, epoch):
  115         with self._lock:
  116             if not all(math.isfinite(x) and x >= 0 for x in (stamp, now, wall_now)):
  117                 raise HistoryUnavailable('INVALID_QUERY_TIME')
  118             if epoch != self.epoch:
  119                 raise HistoryUnavailable('OTHER_EPOCH')
  120             if not self._states:
  121                 raise HistoryUnavailable('NO_HISTORY')
  122             latest = self._states[-1]
  123             if not latest.metadata.current(now, epoch):
  124                 raise HistoryUnavailable('LATEST_STATE_TIME')
  125             if not 0 <= wall_now - self._last_arrival <= .1:
  126                 raise HistoryUnavailable('STALE_STATE_ARRIVAL')
  127             if latest.health != Health.OK:
  128                 raise HistoryUnavailable('LATEST_STATE_HEALTH')
  129             if not 0 <= now - stamp <= self.horizon_s:
  130                 raise HistoryUnavailable('OUTSIDE_HISTORY_HORIZON')
  131             lower = upper = None
  132             for state in self._states:
  133                 if state.metadata.stamp <= stamp:
  134                     lower = state
  135                 if state.metadata.stamp >= stamp:
  136                     upper = state
  137                     break
  138             if lower is None or upper is None:
  139                 raise HistoryUnavailable('NO_BRACKET')
  140             if lower.health != Health.OK or upper.health != Health.OK:
  141                 raise HistoryUnavailable('BRACKET_HEALTH')
  142             gap = upper.metadata.stamp - lower.metadata.stamp
  143             if gap > self.max_bracket_s or lower.metadata.valid_until <= stamp:
  144                 raise HistoryUnavailable('BRACKET_GAP')
  145             fraction = (stamp - lower.metadata.stamp) / gap if gap else 0.
  146             p0, p1 = np.array(lower.position_ned), np.array(upper.position_ned)
  147             predicted = .5 * (np.array(lower.velocity_ned) + upper.velocity_ned) * gap
  148             position_residual = float(np.linalg.norm(p1 - p0 - predicted))
  149             q0, q1 = unit(lower.orientation_wxyz), unit(upper.orientation_wxyz)
  150             delta = unit(multiply(q0 * (1., -1., -1., -1.), q1))
  151             if delta[0] < 0:
  152                 delta = -delta
  153             length = float(np.linalg.norm(delta[1:]))
  154             angle = delta[1:] * (2 * math.atan2(length, float(delta[0])) / length) if length > 1e-12 else np.zeros(3)
  155             expected = .5 * (np.array(lower.angular_velocity_body) + upper.angular_velocity_body) * gap
  156             rotation_residual = float(np.linalg.norm(angle - expected))
  157             # Local plausibility guards, not certified interpolation error bounds.
  158             if position_residual > .1 or rotation_residual > .05:
  159                 raise HistoryUnavailable('ESTIMATE_DISCONTINUITY')
  160             diagonals = tuple(tuple(s.metadata.covariance_or_bounds[i * 15 + i] for i in range(15))
  161                               for s in (lower, upper))
  162             return HistoricalPose(stamp, epoch, tuple((1 - fraction) * p0 + fraction * p1),
  163                 tuple(slerp(q0, q1, fraction)), (lower.metadata.stamp, upper.metadata.stamp), diagonals,
  164                 fraction, position_residual, rotation_residual)
