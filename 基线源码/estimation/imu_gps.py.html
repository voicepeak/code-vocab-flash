    1 """15-state error-state Kalman filter. Provisional noise, pending calibration.
    2 
    3 Nominal state p_NED, v_NED, q_body_to_NED, accel bias, gyro bias.
    4 Error order p,v,local rotation,ba,bg. No debug or prerecorded route input.
    5 GPS outages retain inertial propagation and explicitly degrade health.
    6 """
    7 import math
    8 import numpy as np
    9 from interfaces.contracts import Health, Metadata, StateEstimate
   10 from estimation.geometry import delta_quaternion, multiply, rotation, skew, unit
   11 
   12 
   13 def vector(value):
   14     result = np.asarray(value, dtype=float)
   15     if result.shape != (3,) or not np.all(np.isfinite(result)):
   16         raise ValueError("invalid vector")
   17     return result
   18 
   19 
   20 class ImuGpsEstimator:
   21     def __init__(self, epoch=0):
   22         self.epoch = epoch
   23         self.p = np.zeros(3)
   24         self.v = np.zeros(3)
   25         self.q = np.array((1., 0., 0., 0.))
   26         self.ba = np.zeros(3)
   27         self.bg = np.zeros(3)
   28         self.P = np.diag([.1]*3 + [1.]*3 + [.01]*3 + [.01]*3 + [.001]*3)
   29         self.stamp = None
   30         self.last_gps = None
   31         self.omega = np.zeros(3)
   32         self.initialized = False
   33         self.gap = False
   34         self.rejections = {}
   35 
   36     def initialize(self, position, attitude):
   37         if self.initialized:
   38             return False
   39         self.p, self.q = vector(position).copy(), unit(attitude)
   40         self.initialized = True
   41         return True
   42 
   43     def propagate(self, stamp, acceleration_body, angular_velocity_body):
   44         if not math.isfinite(stamp) or stamp <= 0:
   45             raise ValueError("invalid IMU stamp")
   46         acceleration, omega = vector(acceleration_body), vector(angular_velocity_body)
   47         if self.stamp is None:
   48             self.omega = omega - self.bg
   49             self.stamp = stamp
   50             return
   51         dt = stamp - self.stamp
   52         if dt <= 0:
   53             raise ValueError("IMU must increase within epoch")
   54         self.omega = omega - self.bg
   55         self.stamp = stamp
   56         if dt > .1:
   57             # Missing integration interval cannot be silently treated as fresh.
   58             self.P += np.eye(15) * dt
   59             self.gap = True
   60             return
   61         R = rotation(self.q)
   62         specific_force = acceleration - self.ba
   63         world_acceleration = R @ specific_force + (0., 0., 9.81)
   64         self.p += self.v * dt + .5 * world_acceleration * dt**2
   65         self.v += world_acceleration * dt
   66         self.q = unit(multiply(self.q, delta_quaternion(self.omega*dt)))
   67         F = np.zeros((15, 15))
   68         F[:3, 3:6] = np.eye(3)
   69         F[3:6, 6:9] = -R @ skew(specific_force)
   70         F[3:6, 9:12] = -R
   71         F[6:9, 6:9] = -skew(self.omega)
   72         F[6:9, 12:15] = -np.eye(3)
   73         A = np.eye(15) + F*dt
   74         Q = np.diag([1e-8]*3 + [.15**2]*3 + [.01**2]*3 + [1e-6]*3 + [1e-8]*3)*dt
   75         self.P = A @ self.P @ A.T + Q
   76 
   77     def reject(self, reason):
   78         self.rejections[reason] = self.rejections.get(reason, 0) + 1
   79         return False
   80 
   81     def update_gps(self, stamp, position, attitude):
   82         position = vector(position)
   83         # Factory packets can zero BOTH position and quaternion. Reject the
   84         # sentinel before trying to normalize an invalid all-zero quaternion.
   85         if np.all(position == 0):
   86             return self.reject("GPS_ZERO")
   87         if not self.initialized or self.stamp is None:
   88             return self.reject("NOT_INITIALIZED")
   89         if not math.isfinite(stamp) or not 0 <= self.stamp-stamp <= .15:
   90             return self.reject("GPS_TIME")
   91         if self.last_gps is not None and stamp <= self.last_gps:
   92             return self.reject("GPS_REORDERED")
   93         attitude = unit(attitude)
   94         conjugate = self.q * (1., -1., -1., -1.)
   95         error_q = multiply(conjugate, attitude)
   96         if error_q[0] < 0:
   97             error_q = -error_q
   98         residual = np.r_[position-self.p, 2*error_q[1:]]
   99         H = np.zeros((6, 15))
  100         H[:3, :3] = np.eye(3)
  101         H[3:, 6:9] = np.eye(3)
  102         # GPS documented position <= .1m / attitude <= .2rad, conservative baseline.
  103         noise = np.diag([.1**2]*3 + [.2**2]*3)
  104         S = H @ self.P @ H.T + noise
  105         if residual @ np.linalg.solve(S, residual) > 30.:
  106             return self.reject("GPS_INNOVATION")
  107         K = np.linalg.solve(S, H @ self.P).T
  108         error = K @ residual
  109         self.p += error[:3]
  110         self.v += error[3:6]
  111         self.q = unit(multiply(self.q, delta_quaternion(error[6:9])))
  112         self.ba += error[9:12]
  113         self.bg += error[12:15]
  114         I_KH = np.eye(15) - K @ H
  115         self.P = I_KH @ self.P @ I_KH.T + K @ noise @ K.T
  116         # Reset covariance after injecting local attitude error.
  117         G = np.eye(15)
  118         G[6:9, 6:9] -= .5*skew(error[6:9])
  119         self.P = G @ self.P @ G.T
  120         self.P = (self.P+self.P.T)*.5
  121         self.last_gps = stamp
  122         self.gap = False
  123         return True
  124 
  125     def snapshot(self):
  126         if not self.initialized or self.stamp is None:
  127             raise ValueError("estimator not initialized")
  128         age = math.inf if self.last_gps is None else self.stamp-self.last_gps
  129         health = Health.OK
  130         if age > .3 or self.gap:
  131             health = Health.DEGRADED
  132         if age > 2. or np.max(np.diag(self.P)[:3]) > 4.:
  133             health = Health.LOST
  134         metadata = Metadata(self.stamp, self.stamp+.05, self.epoch, "imu_gps_eskf_v1",
  135                             covariance_or_bounds=tuple(self.P.flatten()))
  136         return StateEstimate(metadata, tuple(self.p), tuple(self.v), tuple(self.q),
  137                              tuple(self.omega), health)
