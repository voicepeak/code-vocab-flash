   18 class LidarAssociationBuffer:
   19     def __init__(self, epoch=0, capacity=8, retention_s=.6, max_skew_s=.02):
   20         if (not isinstance(epoch, int) or epoch < 0 or not isinstance(capacity, int)
   21                 or not 1 <= capacity <= 16 or not math.isfinite(retention_s)
   22                 or not .2 <= retention_s <= 1. or not math.isfinite(max_skew_s)
   23                 or not 0 < max_skew_s <= .05):
   24             raise ValueError('invalid joint association limits')
   25         self.epoch, self.capacity = epoch, capacity
   26         self.retention_s, self.max_skew_s = retention_s, max_skew_s
   27         self.pending = deque()
   28         self.last_stamp = self.last_arrival = None
   29         self.last_pair = None
   30         self.events = Counter()
   31 
   32     def reset(self, epoch):
   33         if not isinstance(epoch, int) or epoch <= self.epoch:
   34             raise ValueError('association epoch must increase')
   35         self.epoch = epoch
   36         self.pending.clear()
   37         self.last_stamp = self.last_arrival = self.last_pair = None
   38 
   39     def add(self, cloud, arrival_wall):
   40         if cloud.metadata.frame_id != 'lidar' or not math.isfinite(arrival_wall) or arrival_wall < 0:
   41             raise ValueError('invalid LiDAR source or arrival')
   42         if cloud.metadata.epoch != self.epoch:
   43             self.events['OTHER_EPOCH'] += 1
   44             return False
   45         if self.last_stamp is not None and cloud.metadata.stamp <= self.last_stamp:
   46             self.events['NONINCREASING'] += 1
   47             return False
   48         if self.last_arrival is not None and arrival_wall < self.last_arrival:
   49             raise ValueError('LiDAR arrivals must not go backwards')
   50         self.last_stamp, self.last_arrival = cloud.metadata.stamp, arrival_wall
   51         self.pending.append((cloud, arrival_wall))
   52         while len(self.pending) > self.capacity:
   53             self.pending.popleft()
   54             self.events['CAPACITY'] += 1
   55         return True
   56 
   57     def select(self, pair, now, wall_now, epoch):
   58         if not all(math.isfinite(v) and v >= 0 for v in (now, wall_now)):
   59             raise ValueError('invalid joint query clock')
   60         if epoch != self.epoch or not pair.metadata.current(now, epoch):
   61             self.events['INVALID_STEREO_TIME_OR_EPOCH'] += 1
   62             return None
   63         key = (pair.left.metadata.stamp, pair.right.metadata.stamp)
   64         if self.last_pair is not None and (key[0] <= self.last_pair[0] or key[1] <= self.last_pair[1]):
   65             self.events['REUSED_OR_OLDER_PAIR'] += 1
   66             return None
   67         self.pending = deque((cloud, arrival) for cloud, arrival in self.pending
   68             if 0 <= wall_now - arrival < self.retention_s
   69             and 0 <= now - cloud.metadata.stamp < self.retention_s)
   70         candidates = []
   71         for index, (cloud, arrival) in enumerate(self.pending):
   72             dl = abs(cloud.metadata.stamp - key[0])
   73             dr = abs(cloud.metadata.stamp - key[1])
   74             if max(dl, dr) <= self.max_skew_s:
   75                 candidates.append((max(dl, dr), index, dl, dr))
   76         if not candidates:
   77             self.events['NO_LIDAR_WITHIN_WINDOW'] += 1
   78             return None
   79         _, index, dl, dr = min(candidates)
   80         cloud, arrival = self.pending[index]
   81         del self.pending[index]
   82         self.last_pair = key
   83         return LidarAssociation(cloud, arrival, dl, dr, cloud.metadata.current(now, epoch))
