   25 def _vertical_edges(image, encoding, cv2):
   26     if encoding == 'bgr8': bgr = image
   27     elif encoding == 'rgb8': bgr = cv2.cvtColor(image, cv2.COLOR_RGB2BGR)
   28     elif encoding == 'bgra8': bgr = cv2.cvtColor(image, cv2.COLOR_BGRA2BGR)
   29     elif encoding == 'rgba8': bgr = cv2.cvtColor(image, cv2.COLOR_RGBA2BGR)
   30     else: raise ValueError('gate candidate requires color image')
   31     hsv = cv2.cvtColor(bgr, cv2.COLOR_BGR2HSV)
   32     mask = cv2.inRange(hsv, (0, 100, 75), (18, 255, 255)) | cv2.inRange(hsv, (170, 100, 75), (179, 255, 255))
   33     vertical = cv2.morphologyEx(mask, cv2.MORPH_OPEN, cv2.getStructuringElement(cv2.MORPH_RECT, (1, 9)))
   34     _, _, stats, _ = cv2.connectedComponentsWithStats(vertical, 8)
   35     # Image borders can truncate a frame; never promote such an edge.
   36     h, w = image.shape[:2]
   37     edges = []
   38     for x, y, width, height, area in stats[1:]:
   39         x, y, width, height, area = map(int, (x, y, width, height, area))
   40         if (height < 20 or area < 18 or height < 2 * width or x < 4 or y < 4
   41                 or x + width >= w - 4 or y + height >= h - 4):
   42             continue
   43         edges.append((x + (width - 1) / 2., float(y), float(y + height - 1), width))
   44     return sorted(edges, key=lambda e: (e[0], e[1]))[:128]

   66 def stereo_gate_candidates(pair, disparity, rig, now, epoch, prepared=None):
   67     """Produce nominal 3D proposals, never a gate passage or obstacle clearance.
   68 
   69     Conservative source-age checks apply to the original image/derived disparity.
   70     Edge depth is a regional median, not a measured precise opening polygon.
   71     """
   72     import cv2
   73     if (not pair.metadata.current(now, epoch) or not disparity.metadata.current(now, epoch)
   74             or disparity.metadata.epoch != pair.metadata.epoch
   75             or disparity.metadata.stamp != pair.metadata.stamp
   76             or disparity.metadata.valid_until != pair.metadata.valid_until
   77             or disparity.left_measurement_stamp != pair.left.metadata.stamp
   78             or disparity.right_measurement_stamp != pair.right.metadata.stamp
   79             or disparity.metadata.frame_id != 'front_left_image'
   80             or pair.left.pixels.shape[:2] != (rig.left.height, rig.left.width)
   81             or disparity.disparity_px.shape != (rig.left.height, rig.left.width)):
   82         raise ValueError('gate inputs do not share a current source pair')
   83     edges = (_vertical_edges(pair.left.pixels, pair.left.encoding, cv2) if prepared is None
   84              else checked_gate_edges(pair, prepared))
   85     h, w = disparity.disparity_px.shape
   86     supported = []
   87     for x, top, bottom, width in edges:
   88         x0, x1 = max(0, int(x - width / 2 - 4)), min(w, int(x + width / 2 + 5))
   89         y0, y1 = max(0, int(top + .15 * (bottom-top))), min(h, int(bottom - .15 * (bottom-top)) + 1)
   90         mask = disparity.valid[y0:y1, x0:x1]
   91         values = disparity.disparity_px[y0:y1, x0:x1][mask]
   92         if len(values) < 8: continue
   93         median = float(np.median(values)); deviation = float(np.median(np.abs(values - median)))
   94         if not math.isfinite(median) or median <= 0 or deviation > max(.75, .15 * median): continue
   95         supported.append((x, top, bottom, width, median, len(values)))
   96     candidates = []
   97     for index, left in enumerate(supported):
   98         for right in supported[index+1:]:
   99             lx, lt, lb, _, ld, lc = left; rx, rt, rb, _, rd, rc = right
  100             lh, rh = lb-lt, rb-rt
  101             if (rx-lx < .7 * max(lh, rh) or rx-lx > 6 * min(lh, rh)
  102                     or min(lh, rh) / max(lh, rh) < .65
  103                     or abs((lt+lb)-(rt+rb)) > .4 * max(lh, rh)
  104                     or abs(ld-rd) > max(.75, .2 * max(ld, rd))):
  105                 continue
  106             left_pixels = np.array(((lx, lt), (lx, lb), (rx, rt), (rx, rb)))
  107             right_pixels = left_pixels.copy()
  108             right_pixels[:, 0] -= (ld, ld, rd, rd)
  109             body, valid, _ = rig.triangulate_body(left_pixels, right_pixels, max_depth_m=30.)
  110             if not valid.all(): continue
  111             lcenter, rcenter = (body[0]+body[1])/2., (body[2]+body[3])/2.
  112             width_m = float(np.linalg.norm(rcenter-lcenter))
  113             height_m = float((np.linalg.norm(body[1]-body[0])+np.linalg.norm(body[3]-body[2])) / 2.)
  114             if not (1. <= width_m <= 30. and 1. <= height_m <= 20.): continue
  115             meta = Metadata(pair.metadata.stamp, pair.metadata.valid_until, epoch, 'nominal_stereo_gate_candidate', 'body_frd')
  116             candidates.append(GateCandidate(meta, ((lx, lt, lb), (rx, rt, rb)),
  117                 tuple((lcenter+rcenter)/2.), width_m, height_m, (ld, rd), (lc, rc)))
  118     return tuple(sorted(candidates, key=lambda item: (item.center_body_frd_m[0], -item.nominal_width_m))[:32])
