    1 """Link timely online gate proposals to historical pose and world tracks."""
    2 from dataclasses import asdict
    3 
    4 from estimation.history import HistoryUnavailable
    5 from interfaces.contracts import Metadata
    6 from perception.gate_geometry.stereo_gate import GateCandidate
    7 from perception.stereo_depth.transport import freshness_reason
    8 from world_model.gate_landmarks.tracker import GateTracker
    9 
   10 
   11 class OnlineGateTracks:
   12     def __init__(self, history, tracker=None):
   13         self.history = history
   14         self.tracker = tracker or GateTracker(epoch=history.epoch)
   15 
   16     def process(self, row, pair, arrival, clock_snapshot, wall_clock):
   17         """Associate only an unexpired candidate using a causal history lookup."""
   18         if row['status'] != 'GATE_CANDIDATES' or not row['timing_current']:
   19             return {'status':'SOURCE_UNAVAILABLE','control_eligible':False}
   20         now, epoch, fresh = clock_snapshot(); wall = wall_clock()
   21         reason = freshness_reason(pair.metadata, now, epoch, fresh, arrival, wall)
   22         if reason:
   23             return {'status':'REJECTED','reason':'INPUT_'+reason,'control_eligible':False}
   24         try:
   25             pose = self.history.lookup(pair.metadata.stamp, now, wall, epoch)
   26         except HistoryUnavailable as error:
   27             return {'status':'REJECTED','reason':'POSE_'+str(error),'control_eligible':False}
   28         candidates=[]
   29         try:
   30             for data in row['candidates']:
   31                 meta = dict(data['metadata']); meta['covariance_or_bounds']=tuple(meta['covariance_or_bounds'])
   32                 candidates.append(GateCandidate(Metadata(**meta),tuple(tuple(x) for x in data['image_edges_px']),
   33                     tuple(data['center_body_frd_m']),data['nominal_width_m'],data['nominal_height_m'],
   34                     tuple(data['edge_disparities_px']),tuple(data['edge_support'])))
   35             result = self.tracker.ingest(tuple(candidates), pose, now, epoch)
   36             # Recheck time after pose lookup and association; do not publish a
   37             # snapshot based on a lease that elapsed while processing.
   38             check_now, check_epoch, check_fresh = clock_snapshot(); check_wall = wall_clock()
   39             reason = freshness_reason(pair.metadata, check_now, check_epoch, check_fresh, arrival, check_wall)
   40             if reason:
   41                 return {'status':'REJECTED','reason':'OUTPUT_'+reason,'control_eligible':False}
   42             tracks=self.tracker.snapshot(check_now,check_epoch)
   43         except (ValueError,ArithmeticError) as error:
   44             return {'status':'REJECTED','reason':'TRACK_'+str(error),'control_eligible':False}
   45         return {'status':'TRACKED','association':result,'tracks':[asdict(track) for track in tracks],
   46                 'pose_stamp':pose.stamp,'pose_bracket_stamps':pose.bracket_stamps,
   47                 'source_stamp':pair.metadata.stamp,'source_valid_until':pair.metadata.valid_until,
   48                 'source_epoch':pair.metadata.epoch,'source_arrival_wall':arrival,
   49                 'check_imu_stamp':check_now,'check_epoch':check_epoch,'check_wall':check_wall,
   50                 'clock_fresh_at_check':check_fresh,'control_eligible':False}
