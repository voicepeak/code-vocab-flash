   10 def gate_region_disparity(pair, config=None, prepared=None):
   11     """Match in an edge-bounded crop, returning UNKNOWN everywhere else.
   12 
   13     Both images use the same crop origin so disparity remains in original pixel
   14     units. SGBM may differ from a full-image computation due to image context.
   15     The ROI is only a diagnostic gate proposal input, never free-space evidence.
   16     """
   17     import cv2
   18     config = config or SgbmConfig()
   19     if pair.skew_s > .005:
   20         raise ValueError('incompatible diagnostic stereo input')
   21     edges = (_vertical_edges(pair.left.pixels, pair.left.encoding, cv2) if prepared is None
   22              else checked_gate_edges(pair, prepared))
   23     height, width = pair.left.pixels.shape[:2]
   24     values = np.full((height, width), np.nan, dtype=np.float32)
   25     valid = np.zeros((height, width), dtype=bool)
   26     if edges:
   27         # Give both directions a full disparity search margin in x, and avoid
   28         # letting ROI edges pretend to be image borders in y.
   29         padding = config.num_disparities + config.block_size + 5
   30         ypad = max(16, config.block_size * 3)
   31         x0 = max(0, int(min(x-w/2 for x,_,_,w in edges)) - padding)
   32         x1 = min(width, int(max(x+w/2 for x,_,_,w in edges)) + padding + 1)
   33         y0 = max(0, int(min(top for _,top,_,_ in edges)) - ypad)
   34         y1 = min(height, int(max(bottom for _,_,bottom,_ in edges)) + ypad + 1)
   35         if x1-x0 > config.num_disparities + config.block_size and y1-y0 > 2*config.block_size:
   36             def crop(frame):
   37                 return ImageFrame(frame.metadata, frame.encoding, frame.pixels[y0:y1, x0:x1])
   38             cropped = StereoPair(pair.metadata, crop(pair.left), crop(pair.right), pair.skew_s)
   39             sub = compute_diagnostic_disparity(cropped, config)
   40             # Only the actual edge support windows can be consumed. Other
   41             # pixels, including the rest of the crop, stay explicitly UNKNOWN.
   42             keep = np.zeros((height, width), dtype=bool)
   43             for x, top, bottom, edge_width in edges:
   44                 ax0=max(0,int(x-edge_width/2-4)); ax1=min(width,int(x+edge_width/2+5))
   45                 ay0=max(0,int(top+.15*(bottom-top))); ay1=min(height,int(bottom-.15*(bottom-top))+1)
   46                 keep[ay0:ay1,ax0:ax1]=True
   47             crop_keep = keep[y0:y1,x0:x1] & sub.valid
   48             valid[y0:y1,x0:x1] = crop_keep
   49             region = values[y0:y1,x0:x1]
   50             region[crop_keep] = sub.disparity_px[crop_keep]
   51     meta=Metadata(pair.metadata.stamp,pair.metadata.valid_until,pair.metadata.epoch,
   52                   'gate_roi_sgbm_diagnostic_pixels','front_left_image')
   53     return DisparityObservation(meta,values,valid,pair.left.metadata.stamp,pair.right.metadata.stamp)
