   16 class OnlineDynamicLidar:
   17     def __init__(self, history, sensor_pose, planning_diagnostic=False):
   18         self.history=history;self.sensor_pose=sensor_pose;self.tracker=DynamicTracker(epoch=history.epoch)
   19         # Explicit uncalibrated assumptions for a stationary diagnostic probe.
   20         self.body=SweepBounds((.2,.2,.1),.05,0.)
   21         self.bounds=DynamicBounds(.2,2.,4.,20.,max_horizon_s=.3)
   22         self.map=RollingEndpointMap(epoch=history.epoch) if planning_diagnostic else None
   23         self.plan_limits=LocalPlannerLimits(SweepBounds((.2,.2,.1),.05,2.),1.,
   24                          segment_durations_s=(.05,.1),lateral_offsets_m=())
   25 
   26     def process(self,cloud,arrival,snapshot,wall_clock,goal_snapshot=None):
   27         """Snapshot returns (imu stamp, epoch, fresh, latest state)."""
   28         def check():
   29             now,epoch,fresh,state=snapshot();wall=wall_clock()
   30             return now,epoch,fresh,state,wall,freshness_reason(cloud.metadata,now,epoch,fresh,arrival,wall)
   31         now,epoch,fresh,state,wall,reason=check()
   32         if reason:return {'status':'REJECTED','reason':'INPUT_'+reason,'control_eligible':False,
   33                           'source_stamp':cloud.metadata.stamp,'source_valid_until':cloud.metadata.valid_until,
   34                           'check_imu_stamp':now,'check_epoch':epoch,'source_epoch':cloud.metadata.epoch,
   35                           'source_arrival_wall':arrival,'check_wall':wall}
   36         try:
   37             pose=self.history.lookup(cloud.metadata.stamp,now,wall,epoch)
   38         except HistoryUnavailable as error:
   39             return {'status':'REJECTED','reason':'POSE_'+str(error),'control_eligible':False}
   40         try:
   41             world=world_endpoints(cloud,pose,self.sensor_pose)
   42             clusters=cluster_endpoints(world,now,epoch)
   43             now,epoch,fresh,state,wall,reason=check()
   44             if reason:return {'status':'REJECTED','reason':'AFTER_CLUSTER_'+reason,'control_eligible':False}
   45             if epoch > self.tracker.epoch:self.tracker.reset(epoch)
   46             association=self.tracker.ingest(clusters,now,epoch)
   47             tracks=self.tracker.snapshot(now,epoch)
   48             if state is None or state.health!=Health.OK or not state.metadata.current(now,epoch):
   49                 return {'status':'REJECTED','reason':'STATE_UNAVAILABLE','control_eligible':False}
   50             # This is a held-position geometry probe, not a command or claim
   51             # that the vehicle can stop here. No candidate establishes FREE.
   52             sample=lambda stamp:TrajectorySample(stamp,state.position_ned,(0.,0.,0.),(0.,0.,0.),0.)
   53             trajectory=Trajectory(Metadata(now,now+.2,epoch,'stationary_dynamic_probe'),(sample(now),sample(now+.2)))
   54             dynamic=check_dynamic_candidates(trajectory,DynamicScene(clusters.metadata,tracks),now,epoch,self.body,self.bounds)
   55             planner = None
   56             if self.map is not None:
   57                 if epoch > self.map.epoch:self.map.reset(epoch)
   58                 if not self.map.integrate(world,now,epoch):raise ValueError('duplicate map scan')
   59                 grid=self.map.snapshot(now,epoch)
   60                 goal=goal_snapshot() if goal_snapshot is not None else None
   61                 if goal is None or goal.status!='OBSERVED' or goal.epoch!=epoch:
   62                     planner={'reason':'GOAL_UNAVAILABLE','map_cells':len(grid.cells)}
   63                 else:
   64                     delta=tuple(b-a for a,b in zip(state.position_ned,goal.position))
   65                     length=math.hypot(*delta);factor=min(1.,1./length) if length else 0.
   66                     target=tuple(a+factor*d for a,d in zip(state.position_ned,delta))
   67                     initial=TrajectorySample(now,state.position_ned,state.velocity_ned,(0.,0.,0.),0.)
   68                     plan=plan_static_local(state,initial,target,grid,now,epoch,self.plan_limits,
   69                                            DynamicScene(clusters.metadata,tracks),self.bounds)
   70                     # Diagnostic only: a millimetre-scale probe can expose the
   71                     # subsequent map check; it is not a meaningful flight step.
   72                     short_distance=min(length,.001)
   73                     short_factor=short_distance/length if length else 0.
   74                     short_target=tuple(a+short_factor*d for a,d in zip(state.position_ned,delta))
   75                     short=plan_static_local(state,initial,short_target,grid,now,epoch,self.plan_limits,
   76                                             DynamicScene(clusters.metadata,tracks),self.bounds)
   77                     planner={'reason':plan.reason,'attempts':[asdict(c) for c in plan.checks],
   78                              'map_cells':len(grid.cells),'goal':asdict(goal),'target':target,
   79                              'trajectory_produced':plan.trajectory is not None,
   80                              'short_probe':{'target':short_target,'reason':short.reason,
   81                                  'attempts':[asdict(c) for c in short.checks],
   82                                  'trajectory_produced':short.trajectory is not None},
   83                              'goal_frame_calibrated':False,'control_eligible':False}
   84             check_now,check_epoch,check_fresh,_,check_wall,reason=check()
   85             if reason:return {'status':'REJECTED','reason':'OUTPUT_'+reason,'control_eligible':False}
   86             if planner and 'goal' in planner and (planner['goal']['epoch'] != check_epoch
   87                     or not 0 <= check_wall-planner['goal']['last_arrival_wall'] < .5):
   88                 return {'status':'REJECTED','reason':'OUTPUT_GOAL_STALE','control_eligible':False}
   89             if not state.metadata.current(check_now,check_epoch) or any(
   90                     t.epoch != check_epoch or not t.stamp <= check_now < t.valid_until for t in tracks):
   91                 return {'status':'REJECTED','reason':'OUTPUT_STATE_OR_TRACK_EXPIRED','control_eligible':False}
   92             return {'status':'PROCESSED','source_stamp':cloud.metadata.stamp,'source_valid_until':cloud.metadata.valid_until,
   93                     'source_epoch':cloud.metadata.epoch,'source_arrival_wall':arrival,'check_imu_stamp':check_now,
   94                     'check_epoch':check_epoch,'check_wall':check_wall,'clock_fresh_at_check':check_fresh,
   95                     'pose_stamp':pose.stamp,'pose_bracket_stamps':pose.bracket_stamps,
   96                     'occupied_voxels':clusters.occupied_voxels,'cluster_count':len(clusters.clusters),
   97                     'rejected_components':clusters.rejected_components,'association':association,
   98                     'tracks':[asdict(track) for track in tracks],'dynamic_check':asdict(dynamic), 'planner':planner,
   99                     'control_eligible':False}
  100         except (ValueError,ArithmeticError) as error:
  101             return {'status':'REJECTED','reason':'PIPELINE_'+str(error),'control_eligible':False}
