   27 def cluster_endpoints(observation, now, epoch, voxel_m=.5, min_points=8, max_voxels=50000, max_clusters=128, center_mode="bbox"):

   29 
   30     Large connected components and sparse points are omitted from the object
   31     candidate list; they remain obstacles in the separate occupancy model.
   32     No candidate is a complete object boundary, dynamic class, or free space.
   33     """
   34     if (not observation.metadata.current(now,epoch) or observation.metadata.frame_id!='world_ned'
   35             or observation.pose_bracket_stamps[1]>now or not math.isfinite(voxel_m)
   36             or not .1<=voxel_m<=2. or isinstance(min_points,bool) or not isinstance(min_points,int) or min_points<3
   37             or isinstance(max_voxels,bool) or not isinstance(max_voxels,int) or not 1<=max_voxels<=100000
   38             or isinstance(max_clusters,bool) or not isinstance(max_clusters,int) or not 1<=max_clusters<=256
   39             or center_mode not in ('bbox','centroid')):
   40         raise ValueError('invalid cluster input or limits')
   41     points=observation.world_points
   42     with np.errstate(over='ignore',invalid='ignore'):
   43         coords=points/voxel_m
   44     if not np.isfinite(coords).all() or np.any(np.abs(coords)>2**45):raise ValueError('cluster coordinates exceed range')
   45     if not len(coords):return ClusterFrame(observation.metadata,(),0,0)
   46     keys,inverse,counts=np.unique(np.floor(coords).astype(np.int64),axis=0,return_inverse=True,return_counts=True)
   47     sums=np.column_stack([np.bincount(inverse,weights=points[:,axis],minlength=len(keys)) for axis in range(3)]) if center_mode=='centroid' else None
   48     if len(keys)>max_voxels:raise ValueError('cluster voxel budget exceeded')
   49     # Every selected cluster encloses its contributing occupied voxels, rather
   50     # than only centroids, and is therefore no smaller than the sampled points.
   51     mapping={tuple(int(v) for v in key):i for i,key in enumerate(keys)}
   52     unvisited=set(mapping);components=[];rejected=0
   53     while unvisited:

   57             for dx in (-1,0,1):
   58                 for dy in (-1,0,1):
   59                     for dz in (-1,0,1):
   60                         if dx==dy==dz==0:continue
   61                         neighbor=(key[0]+dx,key[1]+dy,key[2]+dz)
   62                         if neighbor in unvisited:unvisited.remove(neighbor);queue.append(neighbor)

   66             rejected+=1;continue
   67         center=(lower+upper)/2 if center_mode=='bbox' else sums[indices].sum(axis=0)/total
   68         # Ensure the box still encloses every selected voxel when its center
   69         # moves from the midpoint to the measured centroid.
   70         extents=np.maximum(center-lower,upper-center)
   71         components.append(Cluster(tuple(center),tuple(extents),total))
   72         if len(components)>max_clusters:raise ValueError('cluster count budget exceeded')
   73     components.sort(key=lambda c:(-c.points,c.center_ned))
