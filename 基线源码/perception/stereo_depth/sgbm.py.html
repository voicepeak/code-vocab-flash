   13 @dataclass(frozen=True)
   14 class SgbmConfig:
   15     num_disparities: int = 128
   16     block_size: int = 5
   17     uniqueness_ratio: int = 10
   18     consistency_px: float = 1.
   19     min_texture_std: float = 2.

   21     def __post_init__(self):
   22         if (not isinstance(self.num_disparities, int) or not 16 <= self.num_disparities <= 256
   23                 or self.num_disparities % 16 or not isinstance(self.block_size, int)
   24                 or not 3 <= self.block_size <= 11 or self.block_size % 2 != 1
   25                 or not isinstance(self.uniqueness_ratio, int) or not 5 <= self.uniqueness_ratio <= 30
   26                 or not math.isfinite(self.consistency_px) or not 0 < self.consistency_px <= 1.5
   27                 or not math.isfinite(self.min_texture_std) or not 1 <= self.min_texture_std <= 20):
   28             raise ValueError('invalid bounded SGBM parameters')

   42     def __post_init__(self):
   43         values, valid = np.asarray(self.disparity_px), np.asarray(self.valid)
   44         if (values.ndim != 2 or not np.issubdtype(values.dtype, np.floating)
   45                 or valid.shape != values.shape or valid.dtype != np.bool_
   46                 or not np.isfinite(values[valid]).all() or np.any(values[valid] <= 0)
   47                 or np.any(values[valid] > np.finfo(np.float32).max)
   48                 or not np.isnan(values[~valid]).all()):
   49             raise ValueError('invalid disparity mask or unknown pixels')
   50         object.__setattr__(self, 'disparity_px', np.frombuffer(
   51             values.astype(np.float32).tobytes(), dtype=np.float32).reshape(values.shape))
   52         object.__setattr__(self, 'valid', np.frombuffer(valid.tobytes(), dtype=np.bool_).reshape(valid.shape))

   55 def _gray(frame, cv2):
   56     if frame.encoding == 'mono8':
   57         return frame.pixels[:, :, 0].copy()
   58     conversion = {'bgr8': cv2.COLOR_BGR2GRAY, 'rgb8': cv2.COLOR_RGB2GRAY,
   59                   'bgra8': cv2.COLOR_BGRA2GRAY, 'rgba8': cv2.COLOR_RGBA2GRAY}[frame.encoding]
   60     return cv2.cvtColor(frame.pixels, conversion)

   63 def compute_diagnostic_disparity(pair, config=None):
   64     import cv2
   65 
   66     config = config or SgbmConfig()
   67     left, right = _gray(pair.left, cv2), _gray(pair.right, cv2)
   68     if (left.shape != right.shape or left.shape[1] <= config.num_disparities + config.block_size
   69             or left.shape[0] <= 2 * config.block_size
   70             or pair.left.metadata.epoch != pair.right.metadata.epoch
   71             or pair.left.metadata.epoch != pair.metadata.epoch
   72             or abs(pair.left.metadata.stamp - pair.right.metadata.stamp) > .005):
   73         raise ValueError('incompatible diagnostic stereo input')
   74 
   75     def matcher(minimum):
   76         block = config.block_size
   77         return cv2.StereoSGBM_create(minDisparity=minimum,
   78             numDisparities=config.num_disparities, blockSize=block,
   79             P1=8 * block * block, P2=32 * block * block,
   80             disp12MaxDiff=1, preFilterCap=31, uniquenessRatio=config.uniqueness_ratio,
   81             speckleWindowSize=100, speckleRange=2, mode=cv2.STEREO_SGBM_MODE_SGBM)
   82 
   83     minimum_right = 1 - config.num_disparities
   84     forward = matcher(0).compute(left, right).astype(np.float32) / 16.
   85     reverse = matcher(minimum_right).compute(right, left).astype(np.float32) / 16.
   86     width = left.shape[1]
   87     right_x = np.arange(width, dtype=np.float32)[None, :] - forward
   88     lower = np.floor(right_x).astype(np.int32)
   89     clipped = np.clip(lower, 0, width - 2)
   90     r0 = np.take_along_axis(reverse, clipped, axis=1)
   91     r1 = np.take_along_axis(reverse, clipped + 1, axis=1)
   92     weight = right_x - lower
   93     interpolated = (1. - weight) * r0 + weight * r1
   94 
   95     def texture(image):
   96         image = image.astype(np.float32)
   97         size = (config.block_size, config.block_size)
   98         mean = cv2.boxFilter(image, cv2.CV_32F, size, borderType=cv2.BORDER_REPLICATE)
   99         square = cv2.boxFilter(image * image, cv2.CV_32F, size, borderType=cv2.BORDER_REPLICATE)
  100         return np.maximum(square - mean * mean, 0) >= config.min_texture_std ** 2
  101 
  102     right_texture = texture(right)
  103     valid = ((forward > 0) & (forward < config.num_disparities)
  104              & (lower >= 0) & (lower + 1 < width)
  105              & (r0 >= minimum_right) & (r0 < 0) & (r1 >= minimum_right) & (r1 < 0)
  106              & (np.abs(forward + interpolated) <= config.consistency_px)
  107              & texture(left) & np.take_along_axis(right_texture, clipped, axis=1)
  108              & np.take_along_axis(right_texture, clipped + 1, axis=1))
  109     radius = config.block_size // 2
  110     valid[:radius] = False
  111     valid[-radius:] = False
  112     valid[:, :radius] = False
  113     valid[:, -radius:] = False
  114     disparity = np.where(valid, forward, np.nan).astype(np.float32)
  115     metadata = Metadata(pair.metadata.stamp, pair.metadata.valid_until, pair.metadata.epoch,
  116                         'opencv_sgbm_diagnostic_pixels', 'front_left_image')
  117     return DisparityObservation(metadata, disparity, valid,
  118                                 pair.left.metadata.stamp, pair.right.metadata.stamp)
