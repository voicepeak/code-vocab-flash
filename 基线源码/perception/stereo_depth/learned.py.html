   33 def filter_learned_disparity(forward, reverse_positive, left_gray, right_gray,
   34                              consistency_px=1., min_texture_std=2.):
   35     """Full-resolution pixel units; right map has positive disparity magnitude.
   36 
   37     Returns masked disparity, shared texture/LR mask, and LR-only mask.
   38     None of these masks is a calibrated confidence or free-space certificate.
   39     """
   40     import cv2
   41 
   42     forward = np.asarray(forward, dtype=np.float32)
   43     reverse_positive = np.asarray(reverse_positive, dtype=np.float32)
   44     if (forward.ndim != 2 or forward.shape != reverse_positive.shape
   45             or forward.shape != left_gray.shape or left_gray.shape != right_gray.shape
   46             or left_gray.dtype != np.uint8 or right_gray.dtype != np.uint8
   47             or min(forward.shape) < 5 or not math.isfinite(consistency_px)
   48             or not 0 < consistency_px <= 1.5 or not math.isfinite(min_texture_std)
   49             or not 1 <= min_texture_std <= 20):
   50         raise ValueError('invalid learned disparity fields or rejection limits')
   51     h, w = forward.shape
   52     finite_forward = np.isfinite(forward)
   53     finite_forward[finite_forward] = (forward[finite_forward] > 0) & (forward[finite_forward] < w)
   54     safe = np.where(finite_forward, forward, 0.)
   55     right_x = np.arange(w, dtype=np.float32)[None, :] - safe
   56     # Only potentially in-image coordinates enter integer conversion.
   57     inside = (right_x >= 0) & (right_x < w - 1)
   58     lower = np.floor(np.where(inside, right_x, 0.)).astype(np.int64)
   59     weight = np.where(inside, right_x - lower, 0.)
   60     reverse_valid = np.isfinite(reverse_positive)
   61     reverse_valid[reverse_valid] = ((reverse_positive[reverse_valid] > 0)
   62                                   & (reverse_positive[reverse_valid] < w))
   63     reverse_safe = np.where(reverse_valid, reverse_positive, 0.)
   64     r0 = np.take_along_axis(reverse_safe, lower, axis=1)
   65     r1 = np.take_along_axis(reverse_safe, lower + 1, axis=1)
   66     right_ok = (np.take_along_axis(reverse_valid, lower, axis=1)
   67                 & np.take_along_axis(reverse_valid, lower + 1, axis=1))
   68     difference = np.abs(safe - ((1-weight)*r0 + weight*r1))
   69     lr_valid = finite_forward & (safe > 0) & inside & right_ok & (difference <= consistency_px)
   70     lr_valid[:2] = False;lr_valid[-2:] = False
   71     lr_valid[:, :2] = False;lr_valid[:, -2:] = False

   73     def texture(gray):
   74         gray = gray.astype(np.float32)
   75         mean = cv2.boxFilter(gray, cv2.CV_32F, (5,5), borderType=cv2.BORDER_REPLICATE)
   76         square = cv2.boxFilter(gray*gray, cv2.CV_32F, (5,5), borderType=cv2.BORDER_REPLICATE)
   77         return np.maximum(square-mean*mean,0.) >= min_texture_std**2
   78     right_texture = texture(right_gray)
   79     common_valid = (lr_valid & texture(left_gray)
   80         & np.take_along_axis(right_texture,lower,axis=1)
   81         & np.take_along_axis(right_texture,lower+1,axis=1))
   82     return np.where(common_valid,forward,np.nan).astype(np.float32), common_valid, lr_valid

  119     def predict(self, left_rgb, right_rgb):
  120         import cv2
  121 
  122         if (left_rgb.dtype != np.uint8 or right_rgb.dtype != np.uint8
  123                 or left_rgb.ndim != 3 or left_rgb.shape[2] != 3 or left_rgb.shape != right_rgb.shape):
  124             raise ValueError('matching RGB byte images required')
  125         torch = self.torch
  126         torch.cuda.synchronize()
  127         begin = time.monotonic()
  128         height, width = left_rgb.shape[:2]
  129         size = (int(width*self.scale),int(height*self.scale))
  130         if min(size) < 32 or size[0]/width != self.scale or size[1]/height != self.scale:
  131             raise ValueError('profile requires exact image scaling')
  132         left = cv2.resize(left_rgb,size,interpolation=cv2.INTER_AREA) if self.scale != 1 else left_rgb
  133         right = cv2.resize(right_rgb,size,interpolation=cv2.INTER_AREA) if self.scale != 1 else right_rgb
  134         def tensor(image):
  135             return torch.as_tensor(np.ascontiguousarray(image),device='cuda').float()[None].permute(0,3,1,2).contiguous()
  136         a,b = tensor(left),tensor(right)
  137         padder = self.padder(a.shape,divis_by=32,force_square=False)
  138         def one(x,y):
  139             x,y = padder.pad(x,y)
  140             output = self.model.forward(x,y,iters=self.iterations,test_mode=True,optimize_build_volume='pytorch1')
  141             return padder.unpad(output.float())
  142         with torch.inference_mode(),torch.amp.autocast('cuda',enabled=self.mixed_precision,dtype=torch.float16):
  143             forward = one(a,b)
  144             reverse = one(torch.flip(b,(-1,)),torch.flip(a,(-1,))).flip(-1)
  145         forward = forward.cpu().numpy().reshape(size[1],size[0])
  146         reverse = reverse.cpu().numpy().reshape(size[1],size[0])
  147         forward = full_resolution_disparity(forward,width,height,self.scale)
  148         reverse = full_resolution_disparity(reverse,width,height,self.scale)
  149         torch.cuda.synchronize()
  150         return forward.astype(np.float32),reverse.astype(np.float32),time.monotonic()-begin
