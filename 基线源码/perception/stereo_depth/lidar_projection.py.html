    8 def project_lidar_into_stereo(sensor_points, scan_pose, left_pose, right_pose, rig):
    9     poses = (scan_pose, left_pose, right_pose)
   10     if (len({pose.epoch for pose in poses}) != 1
   11             or any(pose.frame_id != 'world_ned' for pose in poses)
   12             or max(abs(scan_pose.stamp - pose.stamp) for pose in poses[1:]) > .05):
   13         raise ValueError('incompatible historical poses for nominal projection')
   14     sensor_points = points(sensor_points, 3)
   15     ranges = np.hypot(np.hypot(sensor_points[:, 0], sensor_points[:, 1]), sensor_points[:, 2])
   16     finite = np.isfinite(ranges)
   17     keep = np.zeros(len(sensor_points), dtype=bool)
   18     keep[finite] = (ranges[finite] > 0) & (ranges[finite] < 30)
   19     world = np.full(sensor_points.shape, np.nan)
   20     body_at_scan = rig.lidar_pose.to_body(sensor_points[keep])
   21     world[keep] = body_at_scan @ rotation(scan_pose.orientation_wxyz).T + scan_pose.position_ned
   22     outputs = []
   23     valid = keep.copy()
   24     for camera, pose in ((rig.left, left_pose), (rig.right, right_pose)):
   25         body = np.full(sensor_points.shape, np.nan)
   26         body[keep] = (world[keep] - pose.position_ned) @ rotation(pose.orientation_wxyz)
   27         uv, depth, inside = camera.project_body(body)
   28         valid &= inside
   29         outputs.extend((uv, depth))
   30     return outputs[0], outputs[1], outputs[2], outputs[3], valid

   33 def sparse_visible(uv, depth, valid, width, height):
   34     """Nearest retained return per pixel; not a complete visibility proof."""
   35     visible = np.zeros(len(uv), dtype=bool)
   36     indices = np.flatnonzero(valid)
   37     if not len(indices):return visible
   38     pixel = np.floor(uv[indices]).astype(np.int64)
   39     if np.any(pixel < 0) or np.any(pixel[:, 0] >= width) or np.any(pixel[:, 1] >= height):
   40         raise ValueError('out-of-view samples cannot enter sparse z-buffer')
   41     keys = pixel[:, 1] * width + pixel[:, 0]
   42     nearest = np.full(width * height, np.inf)
   43     np.minimum.at(nearest, keys, depth[indices])
   44     visible[indices] = depth[indices] <= nearest[keys] + 1e-8
   45     return visible

   48 def sample_disparity(uv, disparity, valid_pixels, candidates):
   49     """Bilinear sampling only when all four neighbors are valid."""
   50     if disparity.ndim != 2 or disparity.shape != valid_pixels.shape:
   51         raise ValueError('disparity and mask shapes differ')
   52     h, w = disparity.shape
   53     valid = np.zeros(len(uv), dtype=bool)
   54     initial = np.flatnonzero(candidates)
   55     if len(initial):
   56         valid[initial] = ((uv[initial, 0] >= 0) & (uv[initial, 0] < w - 1)
   57                           & (uv[initial, 1] >= 0) & (uv[initial, 1] < h - 1))
   58     indices = np.flatnonzero(valid)
   59     values = np.full(len(uv), np.nan)
   60     if not len(indices):return values, valid
   61     xy = np.floor(uv[indices]).astype(np.int64)
   62     x, y = xy[:, 0], xy[:, 1]
   63     good = valid_pixels[y, x] & valid_pixels[y, x + 1] & valid_pixels[y + 1, x] & valid_pixels[y + 1, x + 1]
   64     valid[indices] = good
   65     indices, x, y = indices[good], x[good], y[good]
   66     dx, dy = uv[indices, 0] - x, uv[indices, 1] - y
   67     values[indices] = ((1-dx)*(1-dy)*disparity[y,x] + dx*(1-dy)*disparity[y,x+1]
   68                        + (1-dx)*dy*disparity[y+1,x] + dx*dy*disparity[y+1,x+1])
   69     return values, valid
