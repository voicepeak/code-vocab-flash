   51 class StereoPairBuffer:
   52     def __init__(self, epoch=0, max_skew_s=.005, capacity=4, wall_ttl_s=.2):
   53         if (not isinstance(epoch, int) or epoch < 0 or not isinstance(capacity, int)
   54                 or not 1 <= capacity <= 8 or not math.isfinite(max_skew_s)
   55                 or not 0 <= max_skew_s <= .02 or not math.isfinite(wall_ttl_s)
   56                 or not 0 < wall_ttl_s <= .5):
   57             raise ValueError('invalid stereo pairing limits')
   58         self.epoch = epoch
   59         self.max_skew_s = max_skew_s
   60         self.wall_ttl_s = wall_ttl_s
   61         self.capacity = capacity
   62         self.pending = {'left': deque(), 'right': deque()}
   63         self.last = {'left': None, 'right': None}
   64         self.events = Counter()
   65 
   66     def reset(self, epoch):
   67         if not isinstance(epoch, int) or epoch <= self.epoch:
   68             raise ValueError('new epoch must increase')
   69         for queue in self.pending.values(): queue.clear()
   70         self.last = {'left': None, 'right': None}
   71         self.epoch = epoch
   72         self.events['EPOCH_RESET'] += 1
   73 
   74     def add(self, side, frame, arrival):
   75         if side not in self.pending or not math.isfinite(arrival) or arrival < 0:
   76             raise ValueError('invalid stereo source or arrival')
   77         expected = 'front_' + side + '_image'
   78         if frame.metadata.frame_id != expected:
   79             raise ValueError('stereo source frame mismatch')
   80         if frame.metadata.epoch != self.epoch:
   81             self.events['OTHER_EPOCH'] += 1
   82             return False
   83         previous = self.last[side]
   84         if previous is not None and frame.metadata.stamp <= previous:
   85             self.events['NONINCREASING_' + side] += 1
   86             return False
   87         self.last[side] = frame.metadata.stamp
   88         queue = self.pending[side]
   89         if len(queue) == self.capacity:
   90             queue.popleft()
   91             self.events['CAPACITY_' + side] += 1
   92         queue.append((frame, arrival))
   93         return True
   94 
   95     def pop(self, now, wall_now):
   96         if not all(math.isfinite(x) and x >= 0 for x in (now, wall_now)):
   97             raise ValueError('invalid pairing clock')
   98         for side, queue in self.pending.items():
   99             keep = deque()
  100             for frame, arrival in queue:
  101                 if frame.metadata.epoch != self.epoch:
  102                     self.events['OTHER_EPOCH_' + side] += 1
  103                 elif frame.metadata.valid_until <= now:
  104                     self.events['SENSOR_EXPIRED_' + side] += 1
  105                 elif not 0 <= wall_now - arrival < self.wall_ttl_s:
  106                     self.events['WALL_EXPIRED_' + side] += 1
  107                 elif frame.metadata.stamp > now + .15:
  108                     self.events['FUTURE_' + side] += 1
  109                 else:
  110                     keep.append((frame, arrival))
  111             self.pending[side] = keep
  112         candidates = []
  113         for i, (left, _) in enumerate(self.pending['left']):
  114             for j, (right, _) in enumerate(self.pending['right']):
  115                 skew = abs(left.metadata.stamp - right.metadata.stamp)
  116                 if (left.metadata.current(now, self.epoch) and right.metadata.current(now, self.epoch)
  117                         and skew <= self.max_skew_s and left.pixels.shape == right.pixels.shape
  118                         and left.encoding == right.encoding):
  119                     candidates.append((skew, i, j))
  120         if not candidates:
  121             return None
  122         skew, i, j = min(candidates)
  123         selected = {}
  124         for side, index in (('left', i), ('right', j)):
  125             self.events['SUPERSEDED_' + side] += index
  126             for _ in range(index + 1):
  127                 selected[side], _ = self.pending[side].popleft()
  128         left, right = selected['left'], selected['right']
  129         metadata = Metadata(max(left.metadata.stamp, right.metadata.stamp),
  130                             min(left.metadata.valid_until, right.metadata.valid_until), self.epoch,
  131                             'front_stereo_approximate_pair', 'front_stereo', (skew,))
