   24 @dataclass(frozen=True)
   25 class SensorPose:
   26     translation_body_frd: tuple
   27     rotation_body_from_sensor_frd: tuple

   29     def __post_init__(self):
   30         finite_vec3(self.translation_body_frd)
   31         rows = self.rotation_body_from_sensor_frd
   32         if not isinstance(rows, tuple) or len(rows) != 3:
   33             raise ValueError('immutable rotation rows required')
   34         for row in rows:
   35             finite_vec3(row)
   36         rotation = np.array(rows)
   37         if not np.allclose(rotation.T @ rotation, np.eye(3), atol=1e-10, rtol=0) or abs(np.linalg.det(rotation) - 1) > 1e-10:
   38             raise ValueError('proper sensor rotation required')
   39 
   40     def to_body(self, sensor_points):
   41         return points(sensor_points, 3) @ np.array(self.rotation_body_from_sensor_frd).T + self.translation_body_frd
   42 
   43     def from_body(self, body_points):
   44         return (points(body_points, 3) - self.translation_body_frd) @ np.array(self.rotation_body_from_sensor_frd)

   47 def sensor_pose(settings):
   48     translation = tuple(float(settings.get(k, 0.)) for k in ('X', 'Y', 'Z'))
   49     roll, pitch, yaw = (math.radians(float(settings.get(k, 0.))) for k in ('Roll', 'Pitch', 'Yaw'))
   50     if not all(math.isfinite(v) for v in (roll, pitch, yaw)):
   51         raise ValueError('nonfinite sensor angle')
   52     cr, sr, cp, sp, cy, sy = math.cos(roll), math.sin(roll), math.cos(pitch), math.sin(pitch), math.cos(yaw), math.sin(yaw)
   53     rotation = ((cy*cp, cy*sp*sr-sy*cr, cy*sp*cr+sy*sr),
   54                 (sy*cp, sy*sp*sr+cy*cr, sy*sp*cr-cy*sr),
   55                 (-sp, cp*sr, cp*cr))
   56     return SensorPose(translation, rotation)

   77     def body_to_optical(self, body_points):
   78         frd = self.pose.from_body(body_points)
   79         return frd[:, (1, 2, 0)]

   81     def optical_to_body(self, optical_points):
   82         return self.pose.to_body(points(optical_points, 3)[:, (2, 0, 1)])

   84     def project_body(self, body_points):
   85         body_points = points(body_points, 3)
   86         finite = np.isfinite(body_points).all(axis=1)
   87         optical = np.full(body_points.shape, np.nan)
   88         optical[finite] = self.body_to_optical(body_points[finite])
   89         z = optical[:, 2]
   90         valid = np.zeros(len(z), dtype=bool)
   91         valid[finite] = z[finite] > 0
   92         uv = np.full((len(z), 2), np.nan)
   93         uv[valid, 0] = self.fx * optical[valid, 0] / z[valid] + self.cx
   94         uv[valid, 1] = self.fy * optical[valid, 1] / z[valid] + self.cy
   95         valid[valid] = ((uv[valid, 0] >= 0) & (uv[valid, 0] < self.width)
   96                        & (uv[valid, 1] >= 0) & (uv[valid, 1] < self.height))
   97         uv[~valid] = np.nan
   98         return uv, z, valid

  110     def __post_init__(self):
  111         a, b = self.left, self.right
  112         ra = np.array(a.pose.rotation_body_from_sensor_frd)
  113         rb = np.array(b.pose.rotation_body_from_sensor_frd)
  114         offset = ra.T @ (np.array(b.pose.translation_body_frd) - a.pose.translation_body_frd)
  115         if (not np.allclose(ra, rb, atol=1e-10, rtol=0)
  116                 or abs(offset[0]) > 1e-10 or abs(offset[2]) > 1e-10 or offset[1] <= 0
  117                 or (a.width, a.height, a.fx, a.fy, a.cx, a.cy) != (b.width, b.height, b.fx, b.fy, b.cx, b.cy)
  118                 or len(self.settings_sha256) != 64
  119                 or any(c not in '0123456789abcdef' for c in self.settings_sha256)):
  120             raise ValueError('nominal stereo requires parallel matched cameras and a positive right baseline')

  122     @property
  123     def baseline_m(self):
  124         rotation = np.array(self.left.pose.rotation_body_from_sensor_frd)
  125         offset = np.array(self.right.pose.translation_body_frd) - self.left.pose.translation_body_frd
  126         return float((rotation.T @ offset)[1])

  128     def triangulate_body(self, left_uv, right_uv, max_depth_m=30., max_vertical_px=1.):
  129         """Nominal points and |dZ/dd|, not calibrated uncertainty or safe space.
  130 
  131         Depth is optical forward Z, not Euclidean distance. Invalid/too distant
  132         correspondences stay NaN. Range and vertical limits are explicit.
  133         """
  134         left_uv, right_uv = points(left_uv, 2), points(right_uv, 2)
  135         if (left_uv.shape != right_uv.shape or not math.isfinite(max_depth_m) or not 0 < max_depth_m <= 100
  136                 or not math.isfinite(max_vertical_px) or not 0 <= max_vertical_px <= 2):
  137             raise ValueError('invalid nominal triangulation limits')
  138         valid = np.isfinite(left_uv).all(axis=1) & np.isfinite(right_uv).all(axis=1)
  139         for uv, camera in ((left_uv, self.left), (right_uv, self.right)):
  140             valid[valid] = ((uv[valid, 0] >= 0) & (uv[valid, 0] < camera.width)
  141                             & (uv[valid, 1] >= 0) & (uv[valid, 1] < camera.height))
  142         disparity = np.full(len(left_uv), np.nan)
  143         disparity[valid] = left_uv[valid, 0] - right_uv[valid, 0]
  144         valid[valid] = ((disparity[valid] > 0)
  145                        & (np.abs(left_uv[valid, 1] - right_uv[valid, 1]) <= max_vertical_px))
  146         z = np.full(len(left_uv), np.nan)
  147         z[valid] = self.left.fx * self.baseline_m / disparity[valid]
  148         valid[valid] = np.isfinite(z[valid]) & (z[valid] <= max_depth_m)
  149         optical = np.full((len(z), 3), np.nan)
  150         optical[valid, 2] = z[valid]
  151         optical[valid, 0] = (left_uv[valid, 0] - self.left.cx) * z[valid] / self.left.fx
  152         optical[valid, 1] = (left_uv[valid, 1] - self.left.cy) * z[valid] / self.left.fy
  153         sensitivity = np.full(len(z), np.nan)
  154         sensitivity[valid] = z[valid] ** 2 / (self.left.fx * self.baseline_m)
  155         body = np.full(optical.shape, np.nan)
  156         body[valid] = self.left.optical_to_body(optical[valid])
  157         return body, valid, sensitivity

  160 def load_nominal_rig(path, direction='front'):
  161     if direction not in ('front', 'back'):
  162         raise ValueError('unsupported stereo direction')
  163     data = Path(path).read_bytes()
  164     settings = json.loads(data)
  165     vehicle = settings['Vehicles']['drone_1']

  167     def scene(entries):
  168         matches = [s for s in entries if s['ImageType'] == 0]
  169         if len(matches) != 1:
  170             raise ValueError('exactly one Scene capture setting required')
  171         return matches[0]
  172 
  173     defaults = scene(settings['CameraDefaults']['CaptureSettings'])
  174     def camera(side):
  175         name = direction + '_' + side
  176         entry = vehicle['Cameras'][name]
  177         capture = {**defaults, **scene(entry['CaptureSettings'])}
  178         width, height, fov = capture['Width'], capture['Height'], float(capture['FOV_Degrees'])
  179         if not math.isfinite(fov) or not 5 < fov < 175:
  180             raise ValueError('invalid camera horizontal FOV')
  181         focal = width / (2 * math.tan(math.radians(fov) / 2))
  182         return NominalCamera(name, width, height, focal, focal, width / 2, height / 2, sensor_pose(entry))
  183 
  184     lidar = vehicle['Sensors']['lidar']
  185     if lidar['DataFrame'] != 'SensorLocalFrame':
  186         raise ValueError('nominal lidar transform requires SensorLocalFrame')
  187     return NominalStereoRig(camera('left'), camera('right'), sensor_pose(lidar), hashlib.sha256(data).hexdigest())
