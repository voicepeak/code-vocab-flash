    5 def measure_sparse_epipolar(left_gray, right_gray):
    6     import cv2

    8     if (left_gray.dtype != np.uint8 or right_gray.dtype != np.uint8

   10         raise ValueError('matching grayscale byte images required')
   11     orb = cv2.ORB_create(nfeatures=2000, fastThreshold=10)

   13     kr, dr = orb.detectAndCompute(right_gray, None)
   14     matcher = cv2.BFMatcher(cv2.NORM_HAMMING)
   15 
   16     def reciprocal_candidates(a, b):
   17         if a is None or b is None or len(b) < 2:
   18             return {}
   19         return {m.queryIdx: m for group in matcher.knnMatch(a, b, k=2) if len(group) == 2
   20                 for m, n in [group] if m.distance < .75 * n.distance}
   21 
   22     forward, backward = reciprocal_candidates(dl, dr), reciprocal_candidates(dr, dl)
   23     matches = [m for m in forward.values() if m.trainIdx in backward and backward[m.trainIdx].trainIdx == m.queryIdx]
   24     left = np.asarray([kl[m.queryIdx].pt for m in matches], dtype=np.float32).reshape(-1, 2)
   25     right = np.asarray([kr[m.trainIdx].pt for m in matches], dtype=np.float32).reshape(-1, 2)
   26     # Do not prefilter vertical residuals: that would make the check circular.
   27     fundamental, mask = (None, None)
   28     if len(matches) >= 8:
   29         fundamental, mask = cv2.findFundamentalMat(left, right, cv2.FM_RANSAC, 1., .999)

   31     disparity = left[:, 0] - right[:, 0]
   32     vertical = np.abs(left[:, 1] - right[:, 1])
   33     def quantiles(values):
   34         return [float(x) for x in np.percentile(values, (50, 95, 99, 100))] if len(values) else None
   35     h, w = left_gray.shape
   36     bins = {(min(3, int(x * 4 / w)), min(2, int(y * 3 / h))) for x, y in left[inliers]}
   37     report = {'left_features': len(kl), 'right_features': len(kr), 'reciprocal_ratio_matches': len(matches),
   38               'fundamental_inliers': int(inliers.sum()), 'inlier_spatial_bins_4x3': len(bins),
   39               'vertical_residual_all_p50_p95_p99_max_px': quantiles(vertical),
   40               'vertical_residual_inliers_p50_p95_p99_max_px': quantiles(vertical[inliers]),
   41               'positive_disparity_inliers': int(((disparity > 0) & inliers).sum()),
   42               'zero_or_negative_disparity_inliers': int(((disparity <= 0) & inliers).sum()),
   43               'vertical_prefilter_applied': False, 'rectification_verified': False,
   44               'metric_calibration_verified': False, 'acceptance_eligible': False}
   45     return report, left, right, inliers, fundamental
