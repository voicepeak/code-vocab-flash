   19 def send_packet(connection, header, payload):
   20     encoded = json.dumps(header,allow_nan=False,separators=(',',':')).encode()
   21     if not 0 < len(encoded) <= MAX_HEADER or len(payload) > MAX_PAYLOAD:
   22         raise ValueError('oversized stereo IPC packet')
   23     connection.sendall(struct.pack('!II',len(encoded),len(payload)))
   24     connection.sendall(encoded)
   25     connection.sendall(payload)

   28 def receive_packet(connection):
   29     def exact(size):
   30         parts=[]
   31         while size:
   32             part=connection.recv(min(size,1024*1024))
   33             if not part:raise EOFError('stereo IPC closed mid-packet')
   34             parts.append(part);size-=len(part)
   35         return b''.join(parts)
   36     header_size,payload_size=struct.unpack('!II',exact(8))
   37     if not 0 < header_size <= MAX_HEADER or payload_size > MAX_PAYLOAD:
   38         raise ValueError('oversized stereo IPC descriptor')
   39     header=json.loads(exact(header_size))
   40     if not isinstance(header,dict):raise ValueError('stereo IPC requires object header')
   41     return header,exact(payload_size)

   57 def encode_disparity(disparity,valid):
   58     if (disparity.shape!=(720,960) or valid.shape!=disparity.shape or valid.dtype!=np.bool_
   59             or not np.isfinite(disparity[valid]).all() or np.any(disparity[valid]<=0)
   60             or not np.isnan(disparity[~valid]).all()):
   61         raise ValueError('invalid diagnostic disparity IPC output')
   62     return disparity.astype('<f4').tobytes()+valid.astype(np.uint8).tobytes()

   65 def decode_disparity(payload):
   66     count=720*960
   67     if len(payload)!=count*5:raise ValueError('invalid diagnostic disparity IPC size')
   68     disparity=np.frombuffer(payload[:count*4],dtype='<f4').reshape(720,960)
   69     mask=np.frombuffer(payload[count*4:],dtype=np.uint8).reshape(720,960)
   70     if np.any(mask>1):raise ValueError('invalid diagnostic validity mask')
   71     valid=mask.astype(bool)
   72     # Same finite/UNKNOWN contract as the sender; no implicit conversion to FREE.
   73     encode_disparity(disparity,valid)
   74     return disparity,valid

   77 def freshness_reason(metadata,now,epoch,clock_fresh,source_arrival,wall_now,wall_ttl=.2):
   78     """Timing only. A timely diagnostic still has no control eligibility."""
   79     if not clock_fresh:return 'STALE_IMU_CLOCK'
   80     if metadata.epoch!=epoch:return 'OTHER_EPOCH'
   81     if not metadata.current(now,epoch):return 'SOURCE_LEASE_NOT_CURRENT'
   82     if not all(math.isfinite(v) for v in (source_arrival,wall_now)) or not 0<=wall_now-source_arrival<wall_ttl:
   83         return 'WALL_EXPIRED'

   87 class LatestSlot:
   88     """One consumer owns its in-flight item; producers replace only the pending one."""
   89     def __init__(self):
   90         self._lock=threading.Lock();self._pending=None;self.closed=False
   91         self.submitted=0;self.superseded=0;self.taken=0;self.discarded_on_close=0
   92 
   93     def put(self,item):
   94         if item is None:raise ValueError('None is reserved for an empty pending slot')
   95         with self._lock:
   96             if self.closed:return False
   97             self.submitted+=1
   98             if self._pending is not None:self.superseded+=1
   99             self._pending=item
  100             return True
  101 
  102     def take(self):
  103         with self._lock:
  104             item,self._pending=self._pending,None
  105             if item is not None:self.taken+=1
  106             return item
  107 
  108     def close(self):
  109         with self._lock:
  110             self.closed=True
  111             if self._pending is not None:self.discarded_on_close+=1
  112             self._pending=None
  113 
  114     def stats(self):
  115         with self._lock:
  116             return {'capacity':1,'pending':int(self._pending is not None),'submitted':self.submitted,
  117                 'superseded':self.superseded,'taken':self.taken,'discarded_on_close':self.discarded_on_close}
