   18 class VisualTrajectory:
   19     """A missing relative step ends the chain; no implicit relocalization."""
   20     def __init__(self, stamp, epoch, max_step_s=1.):
   21         if (not math.isfinite(stamp) or stamp < 0 or isinstance(epoch, bool) or not isinstance(epoch, int)
   22                 or epoch < 0 or not math.isfinite(max_step_s) or not 0 < max_step_s <= 2):
   23             raise ValueError('invalid visual anchor')
   24         self.stamp, self.epoch, self.max_step_s = stamp, epoch, max_step_s
   25         self.position = np.zeros(3); self.rotation = np.eye(3); self.steps = 0; self.broken = False
   26 
   27     def invalidate(self):
   28         self.broken = True
   29 
   30     def append(self, motion):
   31         if self.broken:
   32             raise ValueError('visual chain already broken')
   33         try:
   34             p = np.asarray(motion.translation_previous_body_m, dtype=float)
   35             r = np.asarray(motion.rotation_previous_from_current_body, dtype=float)
   36             valid = (motion.epoch == self.epoch and math.isfinite(motion.previous_stamp)
   37                      and math.isfinite(motion.current_stamp) and abs(motion.previous_stamp-self.stamp) <= 1e-6
   38                      and 0 < motion.current_stamp - motion.previous_stamp <= self.max_step_s
   39                      and p.shape == (3,) and r.shape == (3,3)
   40                      and np.isfinite(p).all() and np.isfinite(r).all()
   41                      and np.allclose(r.T @ r, np.eye(3), rtol=0, atol=1e-5)
   42                      and abs(np.linalg.det(r)-1.) <= 1e-5)
   43         except (TypeError, ValueError, AttributeError):
   44             valid = False
   45         if not valid:
   46             self.broken = True
   47             raise ValueError('invalid or discontinuous relative motion')
   48         position = self.position + self.rotation @ p
   49         orientation = self.rotation @ r
   50         if not np.isfinite(position).all() or not np.isfinite(orientation).all():
   51             self.broken = True
   52             raise ValueError('visual trajectory numerical range')
   53         self.position, self.rotation = position, orientation
   54         self.stamp = motion.current_stamp; self.steps += 1
   55         return self.snapshot()
   56 
   57     def snapshot(self):
   58         return LocalVisualPose(self.stamp, self.epoch, tuple(self.position),
   59                                tuple(tuple(float(v) for v in row) for row in self.rotation), self.steps)
