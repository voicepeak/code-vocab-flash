    2 from dataclasses import asdict

    4 from perception.stereo_depth.transport import freshness_reason

    8 class MotionWindow:
    9     def __init__(self, rig, method='orb_lk', estimator=estimate_stereo_motion):
   10         if method not in ('orb', 'orb_lk', 'orb_lk_half'):
   11             raise ValueError('unknown motion method')
   12         self.rig, self.method, self.estimator = rig, method, estimator
   13         self.previous = None
   14 
   15     def process(self, pair, source_arrival, clock_snapshot, wall_clock):
   16         """Consume one pair; previous pair is historical and never gets a new lease.
   17 
   18         clock_snapshot() returns (sim_time, epoch, fresh), using an independent
   19         IMU clock. A returned estimate is diagnostic even if timing is current.
   20         """
   21         now, epoch, healthy = clock_snapshot()
   22         reason = freshness_reason(pair.metadata, now, epoch, healthy, source_arrival, wall_clock())
   23         if reason:
   24             self.previous = None
   25             return {'status': 'REJECTED', 'reason': 'INPUT_' + reason, 'control_eligible': False}
   26         previous, self.previous = self.previous, pair
   27         if previous is None or previous.metadata.epoch != pair.metadata.epoch:
   28             return {'status': 'WARMUP', 'control_eligible': False}
   29         try:
   30             motion = self.estimator(previous, pair, self.rig, method=self.method)
   31         except (ValueError, ArithmeticError) as error:
   32             return {'status': 'REJECTED', 'reason': 'MOTION_' + str(error), 'control_eligible': False}
   33         now, epoch, healthy = clock_snapshot()
   34         checked_wall = wall_clock()
   35         reason = freshness_reason(pair.metadata, now, epoch, healthy, source_arrival, checked_wall)
   36         # A stale current frame must not serve as an anchor for the next estimate.
   37         if reason:
   38             self.previous = None
   39         return {'status': 'ESTIMATED', 'motion': asdict(motion), 'timing_current': reason is None,
   40                 'timing_rejection': reason, 'check_imu_stamp': now, 'check_epoch': epoch,
   41                 'source_stamp': pair.metadata.stamp, 'source_valid_until': pair.metadata.valid_until,
   42                 'source_arrival_wall': source_arrival, 'check_wall': checked_wall,
   43                 'clock_fresh_at_check': healthy, 'control_eligible': False}
