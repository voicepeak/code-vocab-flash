    8 def gyro_rotation(samples, start, end, max_gap_s=.04):
    9     """Integrate unbiased-or-raw body gyro; no extrapolation or gap bridging.
   10 
   11     samples are (stamp, angular_velocity_body), already restricted to one epoch.
   12     The returned previous-from-current orientation is diagnostic and has no
   13     calibrated error bound. Bias correction, if any, is the caller's input.
   14     """
   15     if (not math.isfinite(start) or not math.isfinite(end) or not 0 <= start < end
   16             or not math.isfinite(max_gap_s) or not 0 < max_gap_s <= .1):
   17         raise ValueError('invalid gyro interval')
   18     stamps = [x[0] for x in samples]
   19     if (len(samples)<2 or any(not math.isfinite(s) for s in stamps)
   20             or any(b <= a for a,b in zip(stamps,stamps[1:]))):
   21         raise ValueError('invalid gyro timestamps')
   22     left=bisect.bisect_right(stamps,start)-1;right=bisect.bisect_left(stamps,end)
   23     if left<0 or right>=len(stamps):raise ValueError('gyro interval not bracketed')
   24     selection = samples[left:right+1]
   25     if any(b[0]-a[0]>max_gap_s for a,b in zip(selection,selection[1:])):
   26         raise ValueError('gyro sample gap')
   27     rates=[]
   28     for _, rate in selection:
   29         v=np.asarray(rate,dtype=float)
   30         if v.shape!=(3,) or not np.isfinite(v).all():raise ValueError('invalid gyro rate')
   31         rates.append(v)
   32     def interpolate(t):
   33         i=min(bisect.bisect_right(stamps,t)-1,right-1)
   34         a,b=stamps[i],stamps[i+1];f=(t-a)/(b-a)
   35         return (1-f)*rates[i-left]+f*rates[i-left+1]
   36     times=[start]+[s for s in stamps[left+1:right] if start<s<end]+[end]
   37     result=np.eye(3)
   38     for a,b in zip(times,times[1:]):
   39         result=result @ rotation(delta_quaternion((interpolate(a)+interpolate(b))*.5*(b-a)))
   40     return result

   43 def rotation_residual(visual_rotation, gyro):
   44     visual=np.asarray(visual_rotation,dtype=float);gyro=np.asarray(gyro,dtype=float)
   45     if (visual.shape!=(3,3) or gyro.shape!=(3,3) or not np.isfinite(visual).all() or not np.isfinite(gyro).all()
   46             or any(not np.allclose(r.T@r,np.eye(3),rtol=0,atol=1e-5) or abs(np.linalg.det(r)-1)>1e-5
   47                    for r in (visual,gyro))):
   48         raise ValueError('invalid rotation')
   49     return float(np.arccos(np.clip((np.trace(gyro.T @ visual)-1)/2,-1,1)))
