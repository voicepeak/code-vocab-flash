    9 class InertialVisualTrajectory:
   10     """Initial-body local position, with explicitly uncalibrated blend gains.
   11 
   12     IMU orientation is taken from the official sensor, not inferred truth.
   13     The caller supplies visual relative displacement in the previous body frame.
   14     Initial velocity is a zero assumption; correlations, bias, uncertainty and
   15     delayed updates are not modeled. Any missing interval breaks the chain.
   16     """
   17     def __init__(self, stamp, epoch, anchor_orientation, position_gain=.5, velocity_gain=.3):
   18         if (not math.isfinite(stamp) or stamp<0 or isinstance(epoch,bool) or not isinstance(epoch,int) or epoch<0
   19                 or not all(math.isfinite(x) and 0<=x<=1 for x in (position_gain,velocity_gain))):
   20             raise ValueError('invalid fusion parameters')
   21         self.stamp,self.epoch=stamp,epoch
   22         self.anchor_rotation=rotation(unit(anchor_orientation));self.p=np.zeros(3);self.v=np.zeros(3)
   23         self.position_gain,self.velocity_gain=position_gain,velocity_gain;self.steps=0;self.broken=False
   24 
   25     def append(self, current_stamp, visual_translation, samples, max_gap_s=.04):
   26         if self.broken:raise ValueError('fusion chain broken')
   27         try:
   28             return self._append(current_stamp,visual_translation,samples,max_gap_s)
   29         except (ValueError,ArithmeticError,TypeError):
   30             self.broken=True
   31             raise
   32 
   33     def _append(self, end, visual_translation, samples, max_gap_s):
   34         start=self.stamp;delta=np.asarray(visual_translation,dtype=float)
   35         if (delta.shape!=(3,) or not np.isfinite(delta).all() or not math.isfinite(end)
   36                 or not .02<=end-start<=1. or not math.isfinite(max_gap_s) or not 0<max_gap_s<=.1):
   37             raise ValueError('invalid fusion interval')
   38         times=[s[0] for s in samples]
   39         if (len(times)<2 or any(not math.isfinite(x) for x in times)
   40                 or any(b<=a for a,b in zip(times,times[1:]))):raise ValueError('invalid imu timestamps')
   41         left=bisect.bisect_right(times,start)-1;right=bisect.bisect_left(times,end)
   42         if left<0 or right>=len(times):raise ValueError('imu interval not bracketed')
   43         portion=samples[left:right+1]
   44         if any(b[0]-a[0]>max_gap_s for a,b in zip(portion,portion[1:])):raise ValueError('imu sample gap')
   45         values=[]
   46         for _,acc,q in portion:
   47             acc=np.asarray(acc,dtype=float)
   48             if acc.shape!=(3,) or not np.isfinite(acc).all():raise ValueError('invalid acceleration')
   49             values.append((acc,unit(q)))
   50         def imu(t):
   51             i=min(bisect.bisect_right(times,t)-1,right-1);f=(t-times[i])/(times[i+1]-times[i])
   52             a,q=values[i-left];b,r=values[i-left+1]
   53             return (1-f)*a+f*b,rotation(slerp(q,r,f))
   54         points=[start]+[t for t in times[left+1:right] if start<t<end]+[end]
   55         anchor_p=self.p.copy();p=self.p.copy();v=self.v.copy();_,start_r=imu(start)
   56         for first,last in zip(points,points[1:]):
   57             a0,r0=imu(first);a1,r1=imu(last);dt=last-first
   58             accel=self.anchor_rotation.T @ (.5*(r0@a0+r1@a1)+np.array((0.,0.,9.81)))
   59             p+=v*dt+.5*accel*dt*dt;v+=accel*dt
   60         visual=anchor_p+self.anchor_rotation.T@start_r@delta
   61         residual=visual-p
   62         p+=self.position_gain*residual;v+=self.velocity_gain*residual/(end-start)
   63         if not np.isfinite(p).all() or not np.isfinite(v).all():raise ValueError('fusion numerical range')
   64         _,end_r=imu(end);self.p,self.v,self.stamp=p,v,end;self.steps+=1
   65         return {'stamp':end,'epoch':self.epoch,'position_initial_body_m':tuple(p),
   66                 'velocity_initial_body_m_s':tuple(v),
   67                 'rotation_initial_from_body':tuple(tuple(float(x) for x in row) for row in self.anchor_rotation.T@end_r),
   68                 'steps':self.steps,'control_eligible':False,'covariance_calibrated':False}
