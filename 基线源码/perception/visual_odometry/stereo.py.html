   26 def _reciprocal(first, second, cv2):
   27     if first is None or second is None or len(first) < 2 or len(second) < 2:
   28         return []
   29     matcher = cv2.BFMatcher(cv2.NORM_HAMMING)
   30     def good(a, b):
   31         return {m.queryIdx: m for group in matcher.knnMatch(a, b, k=2) if len(group) == 2
   32                 for m, n in [group] if m.distance < .75 * n.distance}
   33     forward, backward = good(first, second), good(second, first)
   34     return [m for m in forward.values() if m.trainIdx in backward
   35             and backward[m.trainIdx].trainIdx == m.queryIdx]

   38 def solve_relative(points_body_previous, pixels_current, rig, *, min_inliers=12, reprojection_limit_px=2.):
   39     """Estimate body-current pose in body-previous, using nominal stereo geometry.
   40 
   41     PnP maps previous optical points into the current optical frame; we invert
   42     that transform, then conjugate through the fixed camera-to-body mount. A
   43     RANSAC result is rejected for poor coverage, inliers or reprojection error.
   44     The result is diagnostic until calibration, timing and moving-scene checks.
   45     """
   46     import cv2
   47     body = np.asarray(points_body_previous, dtype=np.float64)
   48     pixels = np.asarray(pixels_current, dtype=np.float64)
   49     if (body.ndim != 2 or body.shape[1:] != (3,) or pixels.shape != (len(body), 2)
   50             or len(body) < min_inliers or not np.isfinite(body).all() or not np.isfinite(pixels).all()
   51             or not isinstance(min_inliers, int) or isinstance(min_inliers, bool) or min_inliers < 6
   52             or not math.isfinite(reprojection_limit_px) or not 0 < reprojection_limit_px <= 2.):
   53         raise ValueError('invalid visual odometry correspondences')
   54     camera = rig.left
   55     if (np.any(pixels[:, 0] < 0) or np.any(pixels[:, 0] >= camera.width)
   56             or np.any(pixels[:, 1] < 0) or np.any(pixels[:, 1] >= camera.height)):
   57         raise ValueError('image points outside camera')
   58     optical = camera.body_to_optical(body)
   59     if np.any(optical[:, 2] <= 0):
   60         raise ValueError('nonpositive previous optical depth')
   61     # Reject concentrated or effectively collinear observations.
   62     singular = np.linalg.svd(optical - optical.mean(axis=0), compute_uv=False)
   63     if len(singular) < 2 or singular[1] < .05:
   64         raise ValueError('degenerate geometry')
   65     k = np.array(((camera.fx, 0., camera.cx), (0., camera.fy, camera.cy), (0., 0., 1.)))
   66     ok, rvec, tvec, inliers = cv2.solvePnPRansac(optical, pixels, k, None,
   67         iterationsCount=150, reprojectionError=reprojection_limit_px, confidence=.99, flags=cv2.SOLVEPNP_EPNP)
   68     if not ok or inliers is None or len(inliers) < min_inliers or len(inliers) / len(body) < .5:
   69         raise ValueError('insufficient PnP inliers')
   70     indices = inliers.ravel()
   71     bins = {(min(3, int(u * 4 / camera.width)), min(2, int(v * 3 / camera.height)))
   72             for u, v in pixels[indices]}
   73     if len(bins) < 4:
   74         raise ValueError('insufficient image coverage')
   75     rotation, _ = cv2.Rodrigues(rvec)
   76     projected, _ = cv2.projectPoints(optical[indices], rvec, tvec, k, None)
   77     errors = np.linalg.norm(projected.reshape(-1, 2) - pixels[indices], axis=1)
   78     p95 = float(np.percentile(errors, 95))
   79     if not math.isfinite(p95) or p95 > reprojection_limit_px or np.any((optical[indices] @ rotation.T + tvec.ravel())[:, 2] <= 0):
   80         raise ValueError('invalid PnP reprojection')
   81     # Optical (right, down, forward) -> sensor FRD (forward, right, down).
   82     optical_to_frd = np.array(((0., 0., 1.), (1., 0., 0.), (0., 1., 0.)))
   83     b_from_c = np.array(camera.pose.rotation_body_from_sensor_frd) @ optical_to_frd
   84     mount = np.asarray(camera.pose.translation_body_frd)
   85     prev_from_cur_opt = rotation.T
   86     prev_from_cur_body = b_from_c @ prev_from_cur_opt @ b_from_c.T
   87     # Invert x_cur = R x_prev + t; account for the rotating camera lever arm.
   88     displacement = b_from_c @ (-rotation.T @ tvec.ravel()) + mount - prev_from_cur_body @ mount
   89     return tuple(displacement), tuple(tuple(float(v) for v in row) for row in prev_from_cur_body), len(indices), p95

   92 def refine_correspondences(source_image, target_image, source_points, target_points, cv2, pixel_scale=1.):
   93     """Refine descriptor matches with forward/backward pyramidal LK.
   94 
   95     Returns refined target pixels and a mask; invalid, drifting or inconsistent
   96     patches never contribute to triangulation or PnP. Not a calibration claim.
   97     """
   98     if pixel_scale not in (1., .5):
   99         raise ValueError("unsupported image scale")
  100     source = np.asarray(source_points, dtype=np.float32).reshape(-1, 1, 2)
  101     initial = np.asarray(target_points, dtype=np.float32).reshape(-1, 1, 2)
  102     if source.shape != initial.shape or not np.isfinite(source).all() or not np.isfinite(initial).all():
  103         raise ValueError('invalid feature coordinates')
  104     if not len(source):
  105         return np.empty((0, 2)), np.zeros(0, dtype=bool)
  106     criteria = (cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 30, .01)
  107     parameters = dict(winSize=(21, 21), maxLevel=3, criteria=criteria,
  108                       flags=cv2.OPTFLOW_USE_INITIAL_FLOW, minEigThreshold=1e-4)
  109     forward, good_forward, _ = cv2.calcOpticalFlowPyrLK(source_image, target_image,
  110         source, initial.copy(), **parameters)
  111     if forward is None or good_forward is None:
  112         return np.full((len(source), 2), np.nan), np.zeros(len(source), dtype=bool)
  113     backward, good_backward, _ = cv2.calcOpticalFlowPyrLK(target_image, source_image,
  114         forward, source.copy(), **parameters)
  115     if backward is None or good_backward is None:
  116         return np.full((len(source), 2), np.nan), np.zeros(len(source), dtype=bool)
  117     forward = forward.reshape(-1, 2); backward = backward.reshape(-1, 2)
  118     original = source.reshape(-1, 2); guessed = initial.reshape(-1, 2)
  119     height, width = target_image.shape
  120     valid = ((good_forward.ravel() != 0) & (good_backward.ravel() != 0)
  121              & np.isfinite(forward).all(axis=1) & np.isfinite(backward).all(axis=1)
  122              & (np.linalg.norm(backward - original, axis=1) <= .75 * pixel_scale)
  123              & (np.linalg.norm(forward - guessed, axis=1) <= 3. * pixel_scale)
  124              & (forward[:, 0] >= 0) & (forward[:, 0] < width)
  125              & (forward[:, 1] >= 0) & (forward[:, 1] < height))
  126     return forward.astype(np.float64), valid

  130 def half_scale_rig(rig):
  131     """Pixel-center-preserving geometry for OpenCV 2x area downsampling."""
  132     if rig.left.width % 2 or rig.left.height % 2:
  133         raise ValueError('half scale requires even dimensions')
  134     def scaled(camera):
  135         return replace(camera, width=camera.width // 2, height=camera.height // 2,
  136                        fx=camera.fx * .5, fy=camera.fy * .5,
  137                        cx=(camera.cx + .5) * .5 - .5, cy=(camera.cy + .5) * .5 - .5)
  138     return replace(rig, left=scaled(rig.left), right=scaled(rig.right))

  141 def estimate_stereo_motion(previous, current, rig, max_gap_s=2., method="orb"):
  142     """Extract stereo/temporal ORB tracks and estimate nominal relative motion."""
  143     import cv2
  144     if (previous.metadata.epoch != current.metadata.epoch
  145             or previous.skew_s > .005 or current.skew_s > .005
  146             or not math.isfinite(max_gap_s) or not 0 < max_gap_s <= 5
  147             or not 0 < current.metadata.stamp - previous.metadata.stamp <= max_gap_s):
  148         raise ValueError('stereo pair epoch or time mismatch')
  149     shape = (rig.left.height, rig.left.width)
  150     if any(frame.pixels.shape[:2] != shape for frame in (previous.left, previous.right, current.left)):
  151         raise ValueError('stereo dimensions differ from camera model')
  152     if method not in ("orb", "orb_lk", "orb_lk_half"):
  153         raise ValueError("unknown visual odometry method")
  154     scale = .5 if method == "orb_lk_half" else 1.
  155     orb = cv2.ORB_create(nfeatures=1200 if scale == .5 else 2000, fastThreshold=10)
  156     images = [_gray(frame, cv2) for frame in (previous.left, previous.right, current.left)]
  157     if scale == .5:
  158         images = [cv2.resize(image, (rig.left.width // 2, rig.left.height // 2), interpolation=cv2.INTER_AREA) for image in images]
  159         rig = half_scale_rig(rig)
  160     features = [orb.detectAndCompute(image, None) for image in images]
  161     (kl, dl), (kr, dr), (kc, dc) = features
  162     stereo, temporal = _reciprocal(dl, dr, cv2), _reciprocal(dl, dc, cv2)
  163     stereo_by_left = {m.queryIdx: m.trainIdx for m in stereo}
  164     joined = [m for m in temporal if m.queryIdx in stereo_by_left]
  165     left = np.array([kl[m.queryIdx].pt for m in joined], dtype=np.float64).reshape(-1, 2)
  166     right = np.array([kr[stereo_by_left[m.queryIdx]].pt for m in joined], dtype=np.float64).reshape(-1, 2)
  167     target = np.array([kc[m.trainIdx].pt for m in joined], dtype=np.float64).reshape(-1, 2)
  168     if method in ("orb_lk", "orb_lk_half"):
  169         right, good_stereo = refine_correspondences(images[0], images[1], left, right, cv2, pixel_scale=scale)
  170         target, good_temporal = refine_correspondences(images[0], images[2], left, target, cv2, pixel_scale=scale)
  171         keep = good_stereo & good_temporal
  172         left, right, target = left[keep], right[keep], target[keep]
  173     body, valid, _ = rig.triangulate_body(left, right, max_depth_m=30., max_vertical_px=scale)
  174     translation, rotation, inliers, p95 = solve_relative(body[valid], target[valid], rig, reprojection_limit_px=2. * scale)
  175     return RelativeMotion(translation, rotation, len(stereo), len(joined), inliers, p95,
  176                           previous.metadata.stamp, current.metadata.stamp, current.metadata.epoch)
