   26     def __init__(self,epoch=0,max_gap_s=.5,max_speed_m_s=20.,association_margin_m=1.,capacity=128,
   27                  association_mode="strict", nearest_margin_m=.5):

   30                 or max_gap_s>2 or max_speed_m_s>50 or association_margin_m>5
   31                 or isinstance(capacity,bool) or not isinstance(capacity,int) or not 1<=capacity<=256
   32                 or association_mode not in ("strict","mutual")
   33                 or not math.isfinite(nearest_margin_m) or not 0<nearest_margin_m<=3):
   34             raise ValueError('invalid dynamic tracker limits')

   36         self.association_margin_m=association_margin_m;self.capacity=capacity
   37         self.association_mode=association_mode;self.nearest_margin_m=nearest_margin_m

   40     def reset(self,epoch):
   41         with self._lock:
   42             if isinstance(epoch,bool) or not isinstance(epoch,int) or epoch<=self.epoch:raise ValueError('epoch must increase')
   43             self.epoch=epoch;self._tracks={};self._last_stamp=None;self._last_now=None;self._next_id=0

   45     def ingest(self,frame,now,epoch):
   46         with self._lock:
   47             meta=frame.metadata
   48             if (epoch!=self.epoch or meta.frame_id!='world_ned' or not meta.current(now,epoch)
   49                     or (self._last_now is not None and now<self._last_now)
   50                     or (self._last_stamp is not None and meta.stamp<=self._last_stamp)
   51                     or not isinstance(frame.clusters,tuple) or len(frame.clusters)>256):
   52                 raise ValueError('invalid, expired or duplicate cluster frame')
   53             for obj in frame.clusters:
   54                 finite_vec3(obj.center_ned);finite_vec3(obj.half_extents_m)
   55                 if any(v<=0 for v in obj.half_extents_m):raise ValueError('invalid cluster extents')
   56             live={k:t for k,t in self._tracks.items() if 0 < meta.stamp-t.stamp <= self.max_gap_s}
   57             edges=[];distances={}
   58             for i,obj in enumerate(frame.clusters):
   59                 for track_id,t in live.items():
   60                     dt=meta.stamp-t.stamp
   61                     predicted=tuple(p+v*dt for p,v in zip(t.center_ned,t.velocity_ned_m_s))
   62                     if (math.dist(predicted,obj.center_ned)<=self.association_margin_m+self.max_speed_m_s*dt
   63                             and all(abs(a-b)<=max(2.,a,b) for a,b in zip(obj.half_extents_m,t.half_extents_m))):
   64                         edges.append((i,track_id));distances[(i,track_id)]=math.dist(predicted,obj.center_ned)
   65             degree={k:sum(b==k for _,b in edges) for k in live};updated=dict(live);next_id=self._next_id
   66             matched=created=ambiguous=0
   67             def mutual_choice(i, track_id):
   68                 ordered_candidate=sorted((distances[(a,b)],b) for a,b in edges if a==i)
   69                 ordered_track=sorted((distances[(a,b)],a) for a,b in edges if b==track_id)
   70                 def unique(items, expected):
   71                     return (items and items[0][1]==expected and
   72                             (len(items)==1 or items[1][0]-items[0][0]>=self.nearest_margin_m))
   73                 return unique(ordered_candidate,track_id) and unique(ordered_track,i)
   74             for i,obj in enumerate(frame.clusters):
   75                 possible=[k for a,k in edges if a==i]
   76                 matched_id = None
   77                 if self.association_mode=='strict':
   78                     if len(possible)==1 and degree[possible[0]]==1:matched_id=possible[0]
   79                 elif possible:
   80                     best=min(possible,key=lambda k:distances[(i,k)])
   81                     if mutual_choice(i,best):matched_id=best
   82                 if matched_id is not None:
   83                     t=updated[matched_id];dt=meta.stamp-t.stamp
   84                     velocity=tuple((a-b)/dt for a,b in zip(obj.center_ned,t.center_ned))
   85                     if math.hypot(*velocity)>self.max_speed_m_s:
   86                         ambiguous+=1;continue
   87                     updated[t.track_id]=replace(t,center_ned=obj.center_ned,half_extents_m=obj.half_extents_m,
   88                         velocity_ned_m_s=velocity,observations=t.observations+1,stamp=meta.stamp,valid_until=meta.valid_until)
   89                     matched+=1
   90                 elif not possible:
   91                     if len(updated)>=self.capacity:raise ValueError('dynamic track capacity exceeded')
   92                     updated[next_id]=ObjectTrack(next_id,epoch,obj.center_ned,obj.half_extents_m,(0.,0.,0.),1,meta.stamp,meta.valid_until)
   93                     next_id+=1;created+=1
   94                 else:ambiguous+=1
   95             self._tracks=updated;self._next_id=next_id;self._last_stamp=meta.stamp;self._last_now=now
   96             return {'matched':matched,'created':created,'ambiguous':ambiguous}

   98     def snapshot(self,now,epoch):
   99         with self._lock:
  100             if epoch!=self.epoch or not math.isfinite(now) or now<0 or (self._last_now is not None and now<self._last_now):
  101                 raise ValueError('dynamic track snapshot clock mismatch')
  102             self._last_now=now
  103             return tuple(t for _,t in sorted(self._tracks.items()) if t.stamp<=now<t.valid_until)

  119 def predict_candidate(track, now, stamp, epoch, *, position_error_m, velocity_error_m_s, acceleration_bound_m_s2):
  120     """Predict a provisional axis-aligned envelope under explicit error bounds.
  121 
  122     Source geometry and stated error bounds are not calibrated. This function
  123     does not establish that an object, including unseen portions, fits the box.
  124     """
  125     if (epoch!=track.epoch or not all(math.isfinite(x) for x in
  126             (now,stamp,position_error_m,velocity_error_m_s,acceleration_bound_m_s2))
  127             or min(position_error_m,velocity_error_m_s,acceleration_bound_m_s2)<0
  128             or not track.stamp<=now<track.valid_until or not now<=stamp<=now+.5
  129             or track.observations<2):
  130         raise ValueError('invalid prediction source, horizon or bounds')
  131     dt=stamp-track.stamp
  132     center=tuple(p+v*dt for p,v in zip(track.center_ned,track.velocity_ned_m_s))
  133     padding=position_error_m+velocity_error_m_s*dt+.5*acceleration_bound_m_s2*dt*dt
  134     extents=tuple(v+padding for v in track.half_extents_m)
  135     finite_vec3(center);finite_vec3(extents)
  136     return PredictedOccupancy(track.track_id,epoch,stamp,center,extents,track.stamp,track.valid_until)
