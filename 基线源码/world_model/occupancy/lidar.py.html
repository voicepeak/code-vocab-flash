    9 def endpoint_snapshot(cloud, now, epoch, resolution_m=.25, max_range_m=30.):
   10     """Voxelize returns in the scan's frozen sensor frame, never world_ned.
   11 
   12     No per-point times or calibrated poses are available in this contract.
   13     This snapshot cannot certify collision clearance for a moving vehicle.
   14     Missing returns, out-of-range returns and all unvisited cells stay UNKNOWN.
   15     """
   16     if not math.isfinite(resolution_m) or not .05 <= resolution_m <= 2.:
   17         raise ValueError('endpoint resolution must be in [.05,2] meters')
   18     if not math.isfinite(max_range_m) or not 0 < max_range_m <= 30.:
   19         raise ValueError('endpoint range must be in (0,30] meters')
   20     if not cloud.metadata.current(now, epoch):
   21         raise ValueError('stale, future or other-epoch lidar cloud')
   22     if cloud.metadata.frame_id != 'lidar':
   23         raise ValueError('endpoint map requires the frozen lidar sensor frame')
   24     ranges = np.hypot(np.hypot(cloud.points[:, 0], cloud.points[:, 1]), cloud.points[:, 2])
   25     # Zero and max-range returns have no confirmed hit semantics.
   26     hits = cloud.points[(ranges > 0) & (ranges < max_range_m)]
   27     # Noetic's NumPy 1.17 cannot unique(axis=0) an empty (0,3) array.
   28     keys = (np.unique(np.floor(hits / resolution_m).astype(np.int64), axis=0)
   29             if len(hits) else np.empty((0, 3), dtype=np.int64))
   30     cells = tuple(Cell(tuple(int(k) for k in key), Occupancy.OCCUPIED, cloud.metadata)
   31                   for key in keys)
   32     return OccupancySnapshot(cloud.metadata, resolution_m, cells)
