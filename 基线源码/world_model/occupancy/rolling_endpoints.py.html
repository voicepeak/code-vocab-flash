   40 def world_endpoints(cloud, pose, sensor_pose, max_range_m=30.):
   41     """Apply mount and estimated scan-header pose, preserving the source lease.
   42 
   43     This is NOT per-return scan deskew or verified metric calibration. Invalid,
   44     zero and maximum-range returns do not provide hit or free-space evidence.
   45     """
   46     if (cloud.metadata.frame_id != 'lidar' or pose.frame_id != 'world_ned'
   47             or pose.stamp != cloud.metadata.stamp or pose.epoch != cloud.metadata.epoch
   48             or pose.position_correction_residual_m > .1 or pose.rotation_correction_residual_rad > .05
   49             or not math.isfinite(max_range_m) or not 0 < max_range_m <= 30):
   50         raise ValueError('incompatible scan and historical pose')
   51     ranges = np.hypot(np.hypot(cloud.points[:, 0], cloud.points[:, 1]), cloud.points[:, 2])
   52     hits = cloud.points[(ranges > 0) & (ranges < max_range_m)]
   53     body_to_world = rotation(pose.orientation_wxyz)
   54     world = sensor_pose.to_body(hits) @ body_to_world.T + pose.position_ned
   55     origin = sensor_pose.to_body(np.zeros((1, 3)))[0] @ body_to_world.T + pose.position_ned
   56     metadata = replace(cloud.metadata, frame_id='world_ned', source='nominal_lidar_header_pose_no_deskew')
   57     return WorldEndpointObservation(metadata, cloud.metadata, world, tuple(origin), pose.bracket_stamps)

   60 class RollingEndpointMap:
   61     """Thread-safe occupied-only map. Expired/cropped cells become UNKNOWN."""
   62     def __init__(self, epoch=0, resolution_m=.25, capacity=50000, window_radius_m=40.):
   63         if (isinstance(epoch, bool) or not isinstance(epoch, int) or epoch < 0
   64                 or not math.isfinite(resolution_m) or not .05 <= resolution_m <= 2
   65                 or isinstance(capacity, bool) or not isinstance(capacity, int) or not 1 <= capacity <= 100000
   66                 or not math.isfinite(window_radius_m) or not 1 <= window_radius_m <= 100):
   67             raise ValueError('invalid rolling endpoint map limits')
   68         self.epoch, self.resolution_m = epoch, resolution_m
   69         self.capacity, self.window_radius_m = capacity, window_radius_m
   70         self._cells = {}
   71         self._latest = None
   72         self._last_now = None
   73         self._lock = threading.RLock()
   74 
   75     def reset(self, epoch):
   76         with self._lock:
   77             if isinstance(epoch, bool) or not isinstance(epoch, int) or epoch <= self.epoch:
   78                 raise ValueError('map epoch must increase')
   79             self.epoch = epoch
   80             self._cells.clear()
   81             self._latest = self._last_now = None
   82 
   83     def _check_time(self, now, epoch):
   84         if (not math.isfinite(now) or now < 0 or epoch != self.epoch
   85                 or (self._last_now is not None and now < self._last_now)):
   86             raise ValueError('map clock or epoch mismatch')
   87 
   88     def integrate(self, observation, now, epoch):
   89         with self._lock:
   90             self._check_time(now, epoch)
   91             meta = observation.metadata
   92             if not meta.current(now, epoch) or meta.frame_id != 'world_ned':
   93                 raise ValueError('endpoint observation is not current')
   94             if observation.pose_bracket_stamps[1] > now:
   95                 raise ValueError('historical pose not yet available at query time')
   96             if self._latest is not None and meta.stamp <= self._latest.stamp:
   97                 return False  # No change to data, lease or clock on duplicates.
   98             origin = np.asarray(observation.sensor_origin_ned)
   99             with np.errstate(over='ignore', invalid='ignore'):
  100                 normalized = observation.world_points / self.resolution_m
  101             if not np.isfinite(normalized).all() or np.any(np.abs(normalized) > 2**50):
  102                 raise ValueError('endpoint voxel coordinates exceed numerical range')
  103             keys = (np.unique(np.floor(normalized).astype(np.int64), axis=0)
  104                     if len(normalized) else np.empty((0, 3), dtype=np.int64))
  105             def within_window(values):
  106                 centers = (values.astype(float) + .5) * self.resolution_m
  107                 return np.linalg.norm(centers - origin, axis=1) <= self.window_radius_m
  108             live = [(key, value) for key, value in self._cells.items() if value.current(now, epoch)]
  109             if live:
  110                 keep = within_window(np.array([key for key, _ in live], dtype=float))
  111                 candidate = {key: value for (key, value), selected in zip(live, keep) if selected}
  112             else:
  113                 candidate = {}
  114             for raw in keys[within_window(keys)]:
  115                 key = tuple(int(value) for value in raw)
  116                 candidate[key] = meta
  117             if len(candidate) > self.capacity:
  118                 raise ValueError('rolling endpoint map capacity exceeded')
  119             # Commit only after complete validation; never publish a partial scan.
  120             self._cells = candidate
  121             self._latest, self._last_now = meta, now
  122             return True
  123 
  124     def snapshot(self, now, epoch):
  125         with self._lock:
  126             self._check_time(now, epoch)
  127             metadata = self._latest or Metadata(now, now, epoch, 'empty_rolling_endpoints')
  128             cells = tuple(Cell(key, Occupancy.OCCUPIED, value)
  129                           for key, value in sorted(self._cells.items()) if value.current(now, epoch))
  130             self._last_now = now
  131             return OccupancySnapshot(metadata, self.resolution_m, cells)
