   28 @dataclass(frozen=True)
   29 class OccupancySnapshot:
   30     metadata: Metadata
   31     resolution_m: float
   32     cells: Tuple[Cell, ...]
   33 
   34     def __post_init__(self):
   35         if not math.isfinite(self.resolution_m) or self.resolution_m <= 0:
   36             raise ValueError("positive resolution required")
   37         if not isinstance(self.cells, tuple):
   38             raise ValueError("cells must be immutable")
   39         if len({c.key for c in self.cells}) != len(self.cells):
   40             raise ValueError("duplicate voxel keys")
   41 
   42     def query(self, position: Vec3, now: float, epoch: int) -> Occupancy:
   43         finite_vec3(position)
   44         if not self.metadata.current(now, epoch):
   45             return Occupancy.UNKNOWN
   46         key = tuple(math.floor(x / self.resolution_m) for x in position)
   47         for cell in self.cells:
   48             if cell.key == key:
   49                 return (cell.status if cell.metadata.frame_id == self.metadata.frame_id
   50                         and cell.metadata.current(now, epoch) else Occupancy.UNKNOWN)
   51         return Occupancy.UNKNOWN
