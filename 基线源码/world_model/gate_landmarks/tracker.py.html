   29 class GateTracker:
   30     """Nearest unambiguous associations; historical tracks never renew source leases."""
   31     def __init__(self, epoch=0, max_gap_s=4., max_distance_m=2., size_tolerance_m=2., capacity=64):
   32         if (isinstance(epoch, bool) or not isinstance(epoch, int) or epoch < 0
   33                 or not all(math.isfinite(x) and x > 0 for x in (max_gap_s,max_distance_m,size_tolerance_m))
   34                 or max_gap_s > 10 or max_distance_m > 10 or size_tolerance_m > 10
   35                 or isinstance(capacity,bool) or not isinstance(capacity,int) or not 1 <= capacity <= 256):
   36             raise ValueError('invalid gate tracking limits')
   37         self.epoch=epoch; self.max_gap_s=max_gap_s; self.max_distance_m=max_distance_m
   38         self.size_tolerance_m=size_tolerance_m; self.capacity=capacity
   39         self._tracks={}; self._last_stamp=None; self._last_now=None; self._next_id=0
   40         self._lock=threading.RLock()

   42     def reset(self, epoch):
   43         with self._lock:
   44             if isinstance(epoch,bool) or not isinstance(epoch,int) or epoch <= self.epoch:
   45                 raise ValueError('gate epoch must increase')
   46             self.epoch=epoch; self._tracks={}; self._last_stamp=None; self._last_now=None; self._next_id=0

   48     def ingest(self, candidates, pose, now, epoch):
   49         """Associate one frame atomically, using a same-stamp historical pose.
   50 
   51         An empty frame does not refresh any existing track. Each observation may
   52         match at most one track and vice versa; ambiguous associations are held.
   53         """
   54         with self._lock:
   55             if (epoch != self.epoch or pose.epoch != epoch or pose.frame_id != 'world_ned'
   56                     or not math.isfinite(now) or now < pose.stamp or pose.bracket_stamps[1] > now
   57                     or (self._last_now is not None and now < self._last_now)
   58                     or (self._last_stamp is not None and pose.stamp <= self._last_stamp)
   59                     or pose.position_correction_residual_m > .1 or pose.rotation_correction_residual_rad > .05
   60                     or not isinstance(candidates,tuple) or len(candidates)>32):
   61                 raise ValueError('invalid gate frame, epoch, history or clock')
   62             prepared=[]
   63             for candidate in candidates:
   64                 meta=candidate.metadata
   65                 finite_vec3(candidate.center_body_frd_m)
   66                 if (meta.frame_id!='body_frd' or meta.stamp!=pose.stamp or not meta.current(now,epoch)
   67                         or not all(math.isfinite(x) and x>0 for x in (candidate.nominal_width_m,candidate.nominal_height_m))):
   68                     raise ValueError('gate candidate source invalid')
   69                 point=np.asarray(pose.position_ned)+rotation(pose.orientation_wxyz)@np.asarray(candidate.center_body_frd_m)
   70                 if not np.isfinite(point).all():raise ValueError('nonfinite world gate')
   71                 prepared.append((candidate,tuple(float(x) for x in point)))
   72             live={k:v for k,v in self._tracks.items() if now-v.last_stamp <= self.max_gap_s}
   73             # Build the whole bipartite compatibility graph. Only degree-1 on
   74             # both sides is associated; this refuses accidental many-to-one.
   75             edges=[]
   76             for index,(candidate,point) in enumerate(prepared):
   77                 for track_id,track in live.items():
   78                     if (math.dist(point,track.center_world_ned)<=self.max_distance_m
   79                             and abs(candidate.nominal_width_m-track.width_m)<=self.size_tolerance_m
   80                             and abs(candidate.nominal_height_m-track.height_m)<=self.size_tolerance_m):
   81                         edges.append((index,track_id))
   82             degrees_track={k:sum(b==k for _,b in edges) for k in live}
   83             updates=dict(live); next_id=self._next_id; matched=created=ambiguous=0
   84             for i,(candidate,point) in enumerate(prepared):
   85                 possible=[k for a,k in edges if a==i]
   86                 if len(possible)==1 and degrees_track[possible[0]]==1:
   87                     t=updates[possible[0]]; n=t.observations+1
   88                     mean=tuple((a*t.observations+b)/n for a,b in zip(t.center_world_ned,point))
   89                     updates[t.track_id]=replace(t,center_world_ned=mean,
   90                         width_m=(t.width_m*t.observations+candidate.nominal_width_m)/n,
   91                         height_m=(t.height_m*t.observations+candidate.nominal_height_m)/n,
   92                         last_stamp=pose.stamp,last_valid_until=candidate.metadata.valid_until,observations=n)
   93                     matched+=1
   94                 elif not possible:
   95                     if len(updates)>=self.capacity:raise ValueError('gate track capacity exceeded')
   96                     updates[next_id]=GateTrack(next_id,epoch,point,candidate.nominal_width_m,
   97                         candidate.nominal_height_m,pose.stamp,pose.stamp,candidate.metadata.valid_until,1)
   98                     next_id+=1;created+=1
   99                 else:ambiguous+=1
  100             self._tracks=updates; self._next_id=next_id; self._last_stamp=pose.stamp;self._last_now=now
  101             return {'matched':matched,'created':created,'ambiguous':ambiguous}

  103     def snapshot(self, now, epoch):
  104         with self._lock:
  105             if (epoch!=self.epoch or not math.isfinite(now) or now<0
  106                     or (self._last_now is not None and now<self._last_now)):
  107                 raise ValueError('gate snapshot clock or epoch mismatch')
  108             self._last_now=now
  109             return tuple(t for _,t in sorted(self._tracks.items())
  110                          if t.last_stamp<=now<t.last_valid_until and now-t.last_stamp<=self.max_gap_s)
