   16 @dataclass(frozen=True)
   17 class Metadata:
   18     stamp: float
   19     valid_until: float
   20     epoch: int
   21     source: str
   22     frame_id: str = "world_ned"
   23     covariance_or_bounds: Tuple[float, ...] = ()
   24 
   25     def __post_init__(self):
   26         if not all(math.isfinite(v) for v in (self.stamp, self.valid_until)):
   27             raise ValueError("timestamps must be finite")
   28         if self.stamp < 0 or self.valid_until < self.stamp or self.epoch < 0:
   29             raise ValueError("invalid validity interval or epoch")
   30         if not self.source or not self.frame_id:
   31             raise ValueError("source and frame required")
   32         if not isinstance(self.covariance_or_bounds, tuple) or not all(
   33                 math.isfinite(v) for v in self.covariance_or_bounds):
   34             raise ValueError("bounds must be an immutable finite tuple")
   35 
   36     def current(self, now: float, epoch: int) -> bool:
   37         return (math.isfinite(now) and self.epoch == epoch
   38                 and self.stamp <= now < self.valid_until)

   41 def finite_vec3(value):
   42     if not isinstance(value, tuple) or len(value) != 3 or not all(math.isfinite(v) for v in value):
   43         raise ValueError("expected immutable finite Vec3")

   47 class StateEstimate:

   52     angular_velocity_body: Vec3

   54 
   55     def __post_init__(self):
   56         finite_vec3(self.position_ned)
   57         finite_vec3(self.velocity_ned)
   58         finite_vec3(self.angular_velocity_body)
   59         q = self.orientation_wxyz
   60         if (not isinstance(q, tuple) or len(q) != 4 or not all(math.isfinite(v) for v in q)
   61                 or abs(sum(v * v for v in q) - 1.0) > 1e-6):
   62             raise ValueError("orientation must be an immutable unit quaternion (w,x,y,z)")
   63         if not isinstance(self.health, Health):
   64             raise ValueError("invalid health")

   87     def __post_init__(self):
   88         if not isinstance(self.samples, tuple) or len(self.samples) < 2:
   89             raise ValueError("trajectory needs at least two immutable samples")
   90         times = tuple(s.stamp for s in self.samples)
   91         if any(a >= b for a, b in zip(times, times[1:])):
   92             raise ValueError("trajectory timestamps must increase")
   93         if times[0] > self.metadata.stamp or times[-1] < self.metadata.valid_until:
