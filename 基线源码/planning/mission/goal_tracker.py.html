    9 @dataclass(frozen=True)
   10 class GoalSnapshot:
   11     status: str
   12     position: tuple = None
   13     epoch: int = 0
   14     observation_count: int = 0
   15     last_arrival_wall: float = None
   16     source_stamp: float = None
   17     source_frame: str = ''
   18     # Goal updates alone do not confirm an official stage completion or score.
   19     official_stage_confirmed: bool = False
   20     control_eligible: bool = False
   21 
   22 
   23 class GoalTracker:
   24     """Debounce /end_goal observations with wall freshness and epoch reset.
   25 
   26     The message may have stamp zero; never replace it with a fresh IMU stamp.
   27     A changed or unstable candidate suppresses the prior goal until confirmed.
   28     """
   29     def __init__(self, epoch=0, confirmations=3, tolerance_m=.05, wall_ttl_s=.5):
   30         if (isinstance(epoch, bool) or not isinstance(epoch, int) or epoch < 0
   31                 or isinstance(confirmations, bool) or not isinstance(confirmations, int) or not 2 <= confirmations <= 10
   32                 or not math.isfinite(tolerance_m) or not 0 < tolerance_m <= 1
   33                 or not math.isfinite(wall_ttl_s) or not 0 < wall_ttl_s <= 2):
   34             raise ValueError('invalid goal tracker settings')
   35         self.epoch, self.confirmations = epoch, confirmations
   36         self.tolerance_m, self.wall_ttl_s = tolerance_m, wall_ttl_s
   37         self._lock = threading.RLock()
   38         self._candidate = self._stable = None
   39         self._count = 0
   40         self._arrival = self._stamp = None
   41         self._last_timed_stamp = None
   42         self._source_frame = ''
   43 
   44     def reset(self, epoch):
   45         with self._lock:
   46             if isinstance(epoch, bool) or not isinstance(epoch, int) or epoch <= self.epoch:
   47                 raise ValueError('goal epoch must increase')
   48             self.epoch = epoch
   49             self._candidate = self._stable = None
   50             self._count = 0; self._arrival = self._stamp = None; self._last_timed_stamp = None; self._source_frame = ''
   51 
   52     def observe(self, position, source_stamp, source_frame, arrival_wall, imu_stamp, epoch):
   53         with self._lock:
   54             finite_vec3(position)
   55             if (epoch != self.epoch or not isinstance(source_frame, str)
   56                     or not all(math.isfinite(t) and t >= 0 for t in (source_stamp, arrival_wall, imu_stamp))
   57                     or imu_stamp == 0 or (self._arrival is not None and arrival_wall <= self._arrival)
   58                     or (source_stamp != 0 and not 0 <= imu_stamp - source_stamp <= .3)
   59                     or (source_stamp != 0 and self._last_timed_stamp is not None
   60                         and source_stamp <= self._last_timed_stamp)):
   61                 raise ValueError('invalid goal source, clock, or epoch')
   62             if (self._arrival is not None and arrival_wall - self._arrival >= self.wall_ttl_s) or source_frame != self._source_frame:
   63                 self._candidate = self._stable = None; self._count = 0
   64             if self._candidate is None or math.dist(position, self._candidate) > self.tolerance_m:
   65                 self._candidate = position; self._count = 1
   66             else:
   67                 self._count += 1
   68             self._arrival, self._stamp, self._source_frame = arrival_wall, source_stamp, source_frame
   69             if source_stamp != 0:
   70                 self._last_timed_stamp = source_stamp
   71             if self._count >= self.confirmations:
   72                 self._stable = self._candidate
   73             return self.snapshot(arrival_wall, epoch, True)
   74 
   75     def snapshot(self, wall_now, epoch, imu_fresh):
   76         with self._lock:
   77             if (epoch != self.epoch or not imu_fresh or not math.isfinite(wall_now)
   78                     or self._arrival is None or not 0 <= wall_now - self._arrival < self.wall_ttl_s):
   79                 return GoalSnapshot('UNAVAILABLE', epoch=self.epoch)
   80             if self._count < self.confirmations:
   81                 return GoalSnapshot('PENDING', epoch=self.epoch, observation_count=self._count,
   82                                     last_arrival_wall=self._arrival, source_stamp=self._stamp,
   83                                     source_frame=self._source_frame)
   84             return GoalSnapshot('OBSERVED', self._stable, self.epoch, self._count,
   85                                 self._arrival, self._stamp, self._source_frame)
