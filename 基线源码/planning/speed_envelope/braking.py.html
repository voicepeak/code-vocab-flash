   20 def speed_limit(free_distance, delay, deceleration, margin):
   21     _validate(free_distance, delay, deceleration, margin)
   22     available = max(0.0, free_distance - margin)

   24         return 0.
   25     # Rationalized root avoids cancellation at small distances / long delays.
   26     root = math.hypot(deceleration * delay, math.sqrt(2 * deceleration * available))

   28     result = 0.0 if denominator == 0 else 2 * deceleration * available / denominator
   29     if not math.isfinite(result):
