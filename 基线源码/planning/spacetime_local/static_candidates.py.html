   16     return tuple(a - b for a, b in zip(first, second))

   19 def _evaluate(controls, fraction):
   20     values = controls
   21     while len(values) > 1:
   22         values = tuple(tuple((1 - fraction) * a + fraction * b for a, b in zip(left, right))
   23                        for left, right in zip(values, values[1:]))
   24     return values[0]

   27 def _derivative(controls, duration):
   28     factor = (len(controls) - 1) / duration
   29     return tuple(tuple(factor * (b - a) for a, b in zip(left, right))
   30                  for left, right in zip(controls, controls[1:]))

   33 def _continuous_bound(controls, depth=3):
   34     # Each Bezier curve lies in its control hull. Subdivision tightens the bound
   35     # while retaining coverage of all times, including between emitted samples.
   36     if depth == 0:
   37         return max(_norm(value) for value in controls)
   38     levels = controls
   39     left, right = [levels[0]], [levels[-1]]
   40     while len(levels) > 1:
   41         levels = tuple(tuple(.5 * a + .5 * b for a, b in zip(first, second))
   42                        for first, second in zip(levels, levels[1:]))
   43         left.append(levels[0]); right.append(levels[-1])
   44     return max(_continuous_bound(tuple(left), depth - 1),
   45                _continuous_bound(tuple(reversed(right)), depth - 1))

   55     def __post_init__(self):
   56         if (not all(math.isfinite(v) for v in (self.stamp, self.duration_s, self.yaw))
   57                 or self.stamp < 0 or not .05 <= self.duration_s <= 10
   58                 or not math.isfinite(self.end_stamp) or self.end_stamp <= self.stamp
   59                 or not isinstance(self.controls, tuple) or len(self.controls) != 6):
   60             raise ValueError('invalid quintic reference')
   61         for point in self.controls:
   62             finite_vec3(point)

   64     @classmethod
   65     def to_rest(cls, initial, target, duration_s):
   66         finite_vec3(target)
   67         elapsed = initial.stamp + duration_s - initial.stamp
   68         p, v, a = initial.position_ned, initial.velocity_ned, initial.acceleration_ned
   69         first = tuple(x + elapsed * speed / 5 for x, speed in zip(p, v))
   70         second = tuple(x + 2 * elapsed * speed / 5 + elapsed**2 * accel / 20
   71                        for x, speed, accel in zip(p, v, a))
   72         return cls(initial.stamp, duration_s, (p, first, second, target, target, target), initial.yaw)
   73 
   74     @property
   75     def end_stamp(self):
   76         return self.stamp + self.duration_s
   77 
   78     def derivative_bounds(self):
   79         elapsed = self.end_stamp - self.stamp
   80         velocity = _derivative(self.controls, elapsed)
   81         acceleration = _derivative(velocity, elapsed)
   82         for value in velocity + acceleration:
   83             finite_vec3(value)
   84         # Small outward padding covers floating-point hull arithmetic.
   85         return tuple(0. if value == 0 else value + 1e-12 * max(1., value)
   86                      for value in (_continuous_bound(velocity), _continuous_bound(acceleration)))
   87 
   88     def sample(self, stamp):
   89         if not math.isfinite(stamp) or not self.stamp <= stamp <= self.end_stamp:
   90             raise ValueError('reference query outside segment')
   91         elapsed = self.end_stamp - self.stamp
   92         fraction = (stamp - self.stamp) / elapsed
   93         velocity = _derivative(self.controls, elapsed)
   94         return TrajectorySample(stamp, _evaluate(self.controls, fraction),
   95             _evaluate(velocity, fraction), _evaluate(_derivative(velocity, elapsed), fraction), self.yaw)

  145     def reference_at(self, stamp, epoch):
  146         if self.trajectory is None or not self.trajectory.metadata.current(stamp, epoch):
  147             raise ValueError('local plan lease is not current')
  148         for piece in self.pieces:
  149             if piece.stamp <= stamp <= piece.end_stamp:
  150                 return piece.sample(stamp)
  151         raise ValueError('reference has no covering segment')

  154 def plan_static_local(state, initial_reference, goal, snapshot, now, epoch, limits,
  155                       dynamic_scene=None, dynamic_bounds=None):
  156     """Select a fully checked stop-at-waypoint reference with a short output lease.
  157 
  158     Candidate paths include direct and left/right midpoints relative to the
  159     current goal, without seed maps. All pieces preserve endpoint p/v/a; each
  160     waypoint ends at rest. The WHOLE stopping reference is checked in the map,
  161     then only a source-bounded short lease is issued. This is a static/kinematic
  162     candidate, not proof of actual braking, dynamic clearance or control health.
  163     An optional dynamic candidate veto refuses predicted intersections but
  164     does not certify sensing coverage or dynamic safety.
  165     """
  166     finite_vec3(goal)
  167     if (dynamic_scene is None) != (dynamic_bounds is None):
  168         raise ValueError('dynamic scene and bounds must be supplied together')
  169     def result(reason, trajectory=None, pieces=(), checks=()):
  170         return LocalPlan(trajectory, pieces, tuple(checks), reason, state.metadata, snapshot.metadata,
  171                          dynamic_scene.metadata if dynamic_scene is not None else None)
  172     if not state.metadata.current(now, epoch) or state.health != Health.OK:
  173         return result('STATE_UNAVAILABLE')
  174     if state.metadata.frame_id != 'world_ned' or snapshot.metadata.frame_id != 'world_ned':
  175         return result('WORLD_FRAME_REQUIRED')
  176     if not snapshot.metadata.current(now, epoch):
  177         return result('MAP_UNAVAILABLE')
  178     if (initial_reference.stamp != now
  179             or _norm(_subtract(initial_reference.position_ned, state.position_ned)) > limits.sweep.tracking_error_m + 1e-9
  180             or _norm(_subtract(initial_reference.velocity_ned, state.velocity_ned)) > limits.max_start_velocity_error_m_s):
  181         return result('REFERENCE_START_MISMATCH')
  182     start = initial_reference.position_ned
  183     delta = _subtract(goal, start)
  184     horizontal = math.hypot(delta[0], delta[1])
  185     routes = [(0., (goal,))]
  186     if horizontal > 1e-6:
  187         perpendicular = (-delta[1] / horizontal, delta[0] / horizontal, 0.)
  188         middle = tuple(.5 * a + .5 * b for a, b in zip(start, goal))
  189         for magnitude in limits.lateral_offsets_m:
  190             for offset in (magnitude, -magnitude):
  191                 via = tuple(value + offset * axis for value, axis in zip(middle, perpendicular))
  192                 routes.append((offset, (via, goal)))
  193     best = None
  194     checks = []
  195     for offset, waypoints in routes:
  196         for duration in limits.segment_durations_s:
  197             if len(checks) >= limits.max_attempts:
  198                 break
  199             end_stamp = now + duration * len(waypoints)
  200             if not snapshot.metadata.current(end_stamp, epoch):
  201                 checks.append(CandidateCheck(offset, duration, 'MAP_STOP_HORIZON'))
  202                 continue
  203             steps = math.ceil(duration / limits.sample_dt_s)
  204             if steps * len(waypoints) > limits.sweep.max_segments:
  205                 checks.append(CandidateCheck(offset, duration, 'SAMPLE_BUDGET'))
  206                 continue
  207             pieces, samples = [], []
  208             current = initial_reference
  209             try:
  210                 for waypoint in waypoints:
  211                     piece = QuinticReference.to_rest(current, waypoint, duration)
  212                     pieces.append(piece)
  213                     first_index = 0 if not samples else 1
  214                     samples.extend(piece.sample(piece.end_stamp if i == steps else
  215                         piece.stamp + i * (piece.end_stamp - piece.stamp) / steps)
  216                         for i in range(first_index, steps + 1))
  217                     current = piece.sample(piece.end_stamp)
  218                 bounds = [piece.derivative_bounds() for piece in pieces]
  219                 speed = max(v[0] for v in bounds)
  220                 acceleration = max(v[1] for v in bounds)
  221                 if not all(math.isfinite(v) for v in (speed, acceleration)):
  222                     raise ValueError('nonfinite derivative bounds')
  223             except (ValueError, OverflowError):
  224                 checks.append(CandidateCheck(offset, duration, 'NUMERICAL_RANGE'))
  225                 continue
  226             if speed > limits.speed_limit_m_s or acceleration > limits.sweep.acceleration_bound_m_s2:
  227                 checks.append(CandidateCheck(offset, duration, 'KINEMATIC_LIMIT', speed, acceleration))
  228                 continue
  229             full = Trajectory(Metadata(now, pieces[-1].end_stamp, epoch, 'static_local_quintic'), tuple(samples))
  230             sweep = check_static_sweep(full, snapshot, now, epoch, limits.sweep)
  231             reason = sweep.reason
  232             accepted = sweep.clear
  233             if accepted and dynamic_scene is not None:
  234                 dynamic = check_dynamic_candidates(full, dynamic_scene, now, epoch, limits.sweep, dynamic_bounds)
  235                 reason, accepted = dynamic.reason, dynamic.allowed
  236             checks.append(CandidateCheck(offset, duration, reason, speed, acceleration, sweep.voxel_tests))
  237             if not accepted:
  238                 continue
  239             length = sum(_norm(_subtract(b, a)) for piece in pieces
  240                          for a, b in zip(piece.controls, piece.controls[1:]))
  241             cost = length + .05 * duration * len(pieces)
  242             if best is None or cost < best[0]:
  243                 best = cost, full, tuple(pieces)
  244     if best is None:
  245         return result('NO_STATIC_CANDIDATE', checks=checks)
  246     _, full, pieces = best
  247     lease_end = min(now + limits.lease_s, state.metadata.valid_until, full.metadata.valid_until)
  248     if dynamic_scene is not None:
  249         lease_end = min(lease_end, dynamic_scene.metadata.valid_until,
  250                         *(track.valid_until for track in dynamic_scene.tracks))
  251     leased = replace(full, metadata=replace(full.metadata, valid_until=lease_end))
  252     return result('STATIC_CANDIDATE', leased, pieces, checks)
