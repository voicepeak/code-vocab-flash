   12 @dataclass(frozen=True)
   13 class SweepBounds:
   14     # Must enclose the body/rotors in world axes at every permitted attitude.
   15     half_extents_m: Tuple[float, float, float]
   16     tracking_error_m: float
   17     acceleration_bound_m_s2: float
   18     max_voxel_tests: int = 20000
   19     max_segments: int = 256

   21     def __post_init__(self):
   22         finite_vec3(self.half_extents_m)
   23         if (any(v <= 0 for v in self.half_extents_m)
   24                 or not all(math.isfinite(v) and v >= 0 for v in
   25                            (self.tracking_error_m, self.acceleration_bound_m_s2))
   26                 or isinstance(self.max_voxel_tests, bool) or not isinstance(self.max_voxel_tests, int)
   27                 or not 1 <= self.max_voxel_tests <= 1000000
   28                 or isinstance(self.max_segments, bool) or not isinstance(self.max_segments, int)
   29                 or not 1 <= self.max_segments <= 2048):
   30             raise ValueError('invalid swept-volume bounds or work budget')

   44 def _contact_interval(start, end, lower, upper):
   45     """Slab intersection on [0,1], including tangency at closed voxel faces."""
   46     enter, leave = 0., 1.
   47     for a, b, lo, hi in zip(start, end, lower, upper):
   48         delta = b - a
   49         if delta == 0:
   50             if a < lo or a > hi:
   51                 return None
   52         else:
   53             first, last = (lo - a) / delta, (hi - a) / delta
   54             enter, leave = max(enter, min(first, last)), min(leave, max(first, last))
   55             if enter > leave:
   56                 return None
   57     return enter, leave

   60 def check_static_sweep(trajectory, snapshot, now, epoch, bounds):
   61     """Check the remaining trajectory including its terminal pose.
   62 
   63     A world-axis box is swept continuously along each position chord. Expand it
   64     by the tracking bound plus a_max * sample_dt**2 / 8: the componentwise chord
   65     deviation bound for any C2 reference obeying that continuous acceleration
   66     bound. Sampled accelerations alone do not prove this producer guarantee.
   67 
   68     FREE must cover the whole contact interval. Occupied, absent, future, stale
   69     or mismatched cells reject. This neither creates FREE evidence nor checks
   70     dynamic objects, road boundaries, braking or executable control authority.
   71     """
   72     tested = intersected = 0
   73     def result(clear, reason, segment=None, key=None, stamp=None):
   74         return StaticSweepResult(clear, reason, tested, intersected, segment, key, stamp)
   75     if not trajectory.metadata.current(now, epoch):
   76         return result(False, 'TRAJECTORY_TIME_OR_EPOCH')
   77     if not snapshot.metadata.current(now, epoch):
   78         return result(False, 'MAP_TIME_OR_EPOCH')
   79     if trajectory.metadata.frame_id != 'world_ned' or snapshot.metadata.frame_id != 'world_ned':
   80         return result(False, 'WORLD_FRAME_REQUIRED')
   81     horizon = trajectory.metadata.valid_until
   82     if not snapshot.metadata.current(horizon, epoch):
   83         return result(False, 'MAP_EXPIRES_BEFORE_HORIZON')
   84     samples = trajectory.samples
   85     if len(samples) - 1 > bounds.max_segments or len(snapshot.cells) > 250000:
   86         return result(False, 'INPUT_BUDGET')
   87     if any(math.hypot(*sample.acceleration_ned) > bounds.acceleration_bound_m_s2 + 1e-9
   88            for sample in samples):
   89         return result(False, 'ACCELERATION_BOUND_CONTRADICTION')
   90     cells = {cell.key: cell for cell in snapshot.cells}
   91     resolution = snapshot.resolution_m
   92     for index, (first, second) in enumerate(zip(samples, samples[1:])):
   93         begin, finish = max(now, first.stamp), min(horizon, second.stamp)
   94         if begin > finish:
   95             continue
   96         duration = second.stamp - first.stamp
   97         if not math.isfinite(duration):
   98             return result(False, 'NUMERICAL_RANGE', index)
   99         # Use the original sample interval, even when checking only its suffix.
  100         padding = bounds.tracking_error_m + bounds.acceleration_bound_m_s2 * duration * duration / 8
  101         extents = tuple(v + padding for v in bounds.half_extents_m)
  102         if not all(math.isfinite(v) for v in extents):
  103             return result(False, 'NUMERICAL_RANGE', index)
  104         def at(stamp):
  105             fraction = (stamp - first.stamp) / duration
  106             return tuple((1 - fraction) * a + fraction * b
  107                          for a, b in zip(first.position_ned, second.position_ned))
  108         start, end = at(begin), at(finish)
  109         if not all(math.isfinite(a) and math.isfinite(b) and math.isfinite(b - a)
  110                    for a, b in zip(start, end)):
  111             return result(False, 'NUMERICAL_RANGE', index)
  112         # Round outward rather than turn a grazing contact into a clear result.
  113         scale = max(1., *(abs(v) for v in start + end + extents))
  114         roundoff = 32 * sys.float_info.epsilon * scale
  115         extents = tuple(v + roundoff for v in extents)
  116         ranges = []
  117         for a, b, extent in zip(start, end, extents):
  118             low, high = (min(a, b) - extent) / resolution, (max(a, b) + extent) / resolution
  119             if not all(math.isfinite(v) and abs(v) <= 2**45 for v in (low, high)):
  120                 return result(False, 'NUMERICAL_RANGE', index)
  121             # Include the cell below an exact lower face: tangency is a hit.
  122             ranges.append(range(math.floor(low) - 1, math.floor(high) + 1))
  123         count = math.prod(len(axis) for axis in ranges)
  124         if tested + count > bounds.max_voxel_tests:
  125             return result(False, 'VOXEL_BUDGET', index)
  126         for key in product(*ranges):
  127             tested += 1
  128             lower = tuple(k * resolution - e for k, e in zip(key, extents))
  129             upper = tuple((k + 1) * resolution + e for k, e in zip(key, extents))
  130             if not all(math.isfinite(v) for v in lower + upper):
  131                 return result(False, 'NUMERICAL_RANGE', index)
  132             interval = _contact_interval(start, end, lower, upper)
  133             if interval is None:
  134                 continue
  135             intersected += 1
  136             enter, leave = (begin + t * (finish - begin) for t in interval)
  137             cell = cells.get(key)
  138             if cell is None or cell.status == Occupancy.UNKNOWN:
  139                 return result(False, 'UNKNOWN_SPACE', index, key, enter)
  140             meta = cell.metadata
  141             if meta.frame_id != 'world_ned':
  142                 return result(False, 'CELL_FRAME', index, key, enter)
  143             if meta.stamp > now or not meta.current(enter, epoch) or not meta.current(leave, epoch):
  144                 return result(False, 'CELL_TIME_OR_EPOCH', index, key, enter)
  145             if cell.status != Occupancy.FREE:
  146                 return result(False, 'OCCUPIED_SPACE', index, key, enter)
  147     return result(True, 'CLEAR_STATIC_MODEL')
