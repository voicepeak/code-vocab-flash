   53 def check_dynamic_candidates(trajectory, scene, now, epoch, body, bounds):
   54     """Check full sampled trajectory against linearly predicted candidate boxes.
   55 
   56     Inflate by tracking error, continuous drone-acceleration chord bound and
   57     explicit position/velocity/acceleration uncertainty for the obstacle. The
   58     interval uses its maximum uncertainty and closed faces so crossings and
   59     tangencies between samples are rejected. Missing/uncertified objects cannot
   60     be proved absent; a passing result is only a candidate conflict veto.
   61     """
   62     if not isinstance(scene,DynamicScene) or not isinstance(body,SweepBounds) or not isinstance(bounds,DynamicBounds):
   63         raise ValueError('dynamic scene and bounds required')
   64     if not trajectory.metadata.current(now,epoch) or not scene.metadata.current(now,epoch):
   65         return DynamicCheck(False,'DYNAMIC_SOURCE_UNAVAILABLE')
   66     if trajectory.metadata.frame_id!='world_ned' or scene.metadata.frame_id!='world_ned':
   67         return DynamicCheck(False,'DYNAMIC_FRAME')
   68     horizon=trajectory.metadata.valid_until
   69     if horizon-now > bounds.max_horizon_s + 1e-9:
   70         return DynamicCheck(False,'DYNAMIC_HORIZON')
   71     samples=trajectory.samples
   72     if len(samples)-1>body.max_segments or (len(samples)-1)*len(scene.tracks)>bounds.max_pair_tests:
   73         return DynamicCheck(False,'DYNAMIC_BUDGET')
   74     if any(math.hypot(*sample.acceleration_ned)>body.acceleration_bound_m_s2+1e-9 for sample in samples):
   75         return DynamicCheck(False,'DYNAMIC_ACCELERATION_BOUND')
   76     for track in scene.tracks:
   77         try:
   78             finite_vec3(track.center_ned);finite_vec3(track.velocity_ned_m_s);finite_vec3(track.half_extents_m)
   79             if (track.epoch!=epoch or not track.stamp<=now<track.valid_until or track.observations<1
   80                     or any(v<=0 for v in track.half_extents_m)):
   81                 return DynamicCheck(False,'DYNAMIC_TRACK_UNAVAILABLE',track_id=track.track_id)
   82         except (TypeError,ValueError):
   83             return DynamicCheck(False,'DYNAMIC_TRACK_INVALID')
   84     tested=0
   85     for i,(first,second) in enumerate(zip(samples,samples[1:])):
   86         begin,finish=max(now,first.stamp),min(horizon,second.stamp)
   87         if begin>finish:continue
   88         duration=second.stamp-first.stamp
   89         if not math.isfinite(duration) or duration<=0:return DynamicCheck(False,'DYNAMIC_NUMERICAL_RANGE',tested,segment_index=i)
   90         def drone_at(stamp):
   91             fraction=(stamp-first.stamp)/duration
   92             return tuple((1-fraction)*a+fraction*b for a,b in zip(first.position_ned,second.position_ned))
   93         start,end=drone_at(begin),drone_at(finish)
   94         deviation=body.tracking_error_m+body.acceleration_bound_m_s2*duration*duration/8
   95         for track in scene.tracks:
   96             tested+=1
   97             dt0,dt1=begin-track.stamp,finish-track.stamp
   98             velocity=track.velocity_ned_m_s if track.observations>=2 else (0.,0.,0.)
   99             velocity_error=(bounds.velocity_error_m_s if track.observations>=2
  100                             else bounds.single_observation_speed_bound_m_s)
  101             inflation=bounds.position_error_m+velocity_error*dt1+.5*bounds.acceleration_bound_m_s2*dt1*dt1
  102             relative0=tuple(p-(q+v*dt0) for p,q,v in zip(start,track.center_ned,velocity))
  103             relative1=tuple(p-(q+v*dt1) for p,q,v in zip(end,track.center_ned,velocity))
  104             extents=tuple(a+b+deviation+inflation for a,b in zip(body.half_extents_m,track.half_extents_m))
  105             values=relative0+relative1+extents
  106             if not all(math.isfinite(v) for v in values):return DynamicCheck(False,'DYNAMIC_NUMERICAL_RANGE',tested,track.track_id,i)
  107             scale=max(1.,*(abs(v) for v in values));padding=32*sys.float_info.epsilon*scale
  108             extents=tuple(v+padding for v in extents)
  109             if _contact_interval(relative0,relative1,tuple(-v for v in extents),extents) is not None:
  110                 return DynamicCheck(False,'DYNAMIC_CANDIDATE_CONFLICT',tested,track.track_id,i)
  111     return DynamicCheck(True,'NO_DETECTED_DYNAMIC_CONFLICT_UNCERTIFIED',tested)
