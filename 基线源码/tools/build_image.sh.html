    1 #!/usr/bin/env bash
    2 set -euo pipefail
    3 root=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/.." && pwd)
    4 image=${1:-rmua-next:p1-development}
    5 revision=$(git -C "$root" rev-parse HEAD)
    6 if [[ -n $(git -C "$root" status --porcelain) ]]; then revision="$revision-dirty"; fi
    7 docker build --network=none --build-arg "GIT_REVISION=$revision" \
    8   -f "$root/deployment/Dockerfile" -t "$image" "$root"
    9 size=$(docker image inspect "$image" --format '{{.Size}}')
   10 ((size < 15000000000)) || { echo 'Development image exceeds the official size limit.' >&2; exit 2; }
   11 # A successful docker build alone does not exercise the packaged Python modules.
   12 exec docker run --rm --network none "$image" \
   13   python3 -m unittest discover -s tests/unit -q
