    1 """Exploratory fixed-target gain study, not a recovery-envelope certification.
    2 
    3 Assumes recovery has already selected the fixed target. Actual supervisor
    4 transitions, initialization, ESKF, collision, actuator consumption and GPS errors
    5 are absent. Production state-machine tests cover the retained 2.5rad/s boundary.
    6 """
    7 import argparse,json,itertools,math
    8 from pathlib import Path
    9 import numpy as np
   10 from control.se3_controller.controller import Se3Controller
   11 from control.motor_allocator.allocator import MotorAllocator
   12 from estimation.geometry import rotation,delta_quaternion,multiply,unit
   13 from interfaces.contracts import StateEstimate,Metadata,Health
   14 from evaluation.seed_runner.manifest import file_hash
   15 
   16 def main():

   18     parser=argparse.ArgumentParser(description=__doc__)
   19     parser.add_argument('--output',type=Path,required=True)
   20     args=parser.parse_args()
   21     if root/'runs' not in args.output.resolve().parents:parser.error('output must be inside ignored runs/')
   22     if args.output.exists():parser.error('refusing to overwrite an existing result')
   23     cfg=json.loads((root/'configs/vehicle_identification.json').read_text())
   24     inputs=['control/se3_controller/controller.py','control/motor_allocator/allocator.py','estimation/geometry.py','interfaces/contracts.py','configs/vehicle_identification.json']
   25     source_before={name:file_hash(root/name) for name in inputs}
   26     results=[]
   27     for (attitude_gain,gain),axis,lag,delay,iscale in itertools.product(((1.,2.),(4.,2.),(4.,3.),(6.,3.)),range(4),(.02,.04),(0.,.02),(.8,1.2)):
   28      c=Se3Controller(cfg);c.kw*=gain/2;c.kr*=attitude_gain;alloc=MotorAllocator(cfg)
   29      direction=np.eye(3)[axis] if axis<3 else np.array([1.,1.,0.])/2**.5

   31      actual=np.array([cfg['mass_kg']*9.81,0.,0.,0.]);request=actual.copy()
   32      inertia=np.array(cfg['inertia_kg_m2'])*iscale;dt=.0025;history=[(p.copy(),v.copy(),q.copy(),np.zeros(3)) for _ in range(round(delay/dt))]
   33      peak_tilt=peak_drift=0.;saturated=0;fault=None
   34      for i in range(1600):
   35       history.append((p.copy(),v.copy(),q.copy(),w.copy()))
   36       if i%5==0:
   37        sensed=history[max(0,len(history)-1-round(delay/dt))]
   38        state=StateEstimate(Metadata(1+i*dt,1.05+i*dt,0,'model'),tuple(sensed[0]),tuple(sensed[1]),tuple(sensed[2]),tuple(sensed[3]),Health.OK)
   39        wrench=c.step(state,(0.,0.,-1.),dt=.0125,allow_integral=False,rate_gain_scale=2.)
   40        a=alloc.allocate(wrench.thrust,wrench.moment);request=np.array(a.achieved_wrench);saturated+=a.saturated
   41       actual+=(1-np.exp(-dt/lag))*(request-actual)
   42       R=rotation(q);acceleration=np.array([0.,0.,9.81])-R[:,2]*actual[0]/cfg['mass_kg']
   43       alpha=(actual[1:]-np.cross(w,inertia*w))/inertia
   44       p+=v*dt+.5*acceleration*dt*dt;v+=acceleration*dt
   45       q=unit(multiply(q,delta_quaternion(w*dt+.5*alpha*dt*dt)));w+=alpha*dt
   46       tilt=math.acos(float(np.clip(R[2,2],-1,1)));drift=float(np.linalg.norm(p-[0,0,-1]))
   47       peak_tilt=max(peak_tilt,tilt);peak_drift=max(peak_drift,drift)
   48       if tilt>.35 or drift>.8:
   49        fault='TILT' if tilt>.35 else 'DRIFT';break
   50      result={'gain':gain,'attitude_gain':attitude_gain,'axis':axis,'lag_s':lag,'delay_s':delay,'inertia_scale':iscale,
   51      'impulse_norm_rad_s':4.2,'peak_tilt_rad':peak_tilt,'peak_drift_m':peak_drift,
   52      'final_rate_rad_s':float(np.linalg.norm(w)),'final_speed_m_s':float(np.linalg.norm(v)),
   53      'saturated_control_ticks':int(saturated),'fault':fault,'end_s':i*dt,
   54      'settled':bool(fault is None and np.linalg.norm(w)<.2 and np.linalg.norm(v)<.25 and tilt<.12)}
   55      results.append(result)
   56     report={'exploratory_model_only':True,'safety_supervisor_exercised':False,'source_sha256_before':source_before,'source_sha256_after':{name:file_hash(root/name) for name in inputs},'controller_limits_unchanged':True,'version':'v2_prefault_history','simulation_source_sha256':file_hash(Path(__file__)),
   57      'assumptions':'Instant 4.2rad/s angular impulse, healthy exact position/orientation sensors delayed 0/20ms, including pre-impulse zero-rate history; provisional mixer, no wind/contact/GPS bias, inertia +-20%, actuator first-order lag20/40ms. Not a certified uncertainty envelope.',
   58      'cases':results}
   59     args.output.parent.mkdir(parents=True,exist_ok=True)
   60     with args.output.open('x') as output:output.write(json.dumps(report,indent=2)+'\n')
   61     for attitude_gain,gain in ((1.,2.),(4.,2.),(4.,3.),(6.,3.)):
   62      rows=[r for r in results if r['gain']==gain and r['attitude_gain']==attitude_gain]
   63      print('attitude/rate scales',attitude_gain,gain,'settled',sum(r['settled'] for r in rows),'/',len(rows),'faults',[(r['axis'],r['lag_s'],r['delay_s'],r['inertia_scale'],r['fault']) for r in rows if r['fault']])
   64 
   65 if __name__ == "__main__":
   66     main()
