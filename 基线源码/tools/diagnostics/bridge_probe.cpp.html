   75 [[noreturn]] void fatal(const char* message) {
   76     const char prefix[]="RMUA diagnostic bridge probe: ";
   77     (void)!write(STDERR_FILENO,prefix,sizeof(prefix)-1);
   78     (void)!write(STDERR_FILENO,message,std::strlen(message));
   79     (void)!write(STDERR_FILENO,"\n",1);
   80     _exit(86);
   81 }

   83 std::uint64_t now_ns() {
   84     timespec value{};
   85     if(clock_gettime(CLOCK_MONOTONIC,&value)!=0)fatal("monotonic clock unavailable");
   86     return std::uint64_t(value.tv_sec)*1000000000ULL+value.tv_nsec;
   87 }

   89 void record(const char* event,int function,std::uint64_t begin,std::uint64_t end,long other_thread=0,
   90             std::uintptr_t parameter=0,std::uintptr_t command=0,std::uint64_t span=0,
   91             const char* pwm_hex="",bool read_ok=false,bool call_interval=false) {
   92     if(output<0)return;
   93     char line[768];
   94     const int length=std::snprintf(line,sizeof(line),
   95         "{\"event\":\"%s\",\"mode\":\"%s\",\"function\":\"%s\","
   96         "\"start_monotonic_ns\":%llu,\"end_monotonic_ns\":%llu,\"thread_id\":%ld,\"other_thread_id\":%ld,"
   97         "\"pwm_calls\":%llu,\"bridge_calls\":%llu,\"overlap_entries\":%llu,\"write_errors\":%llu,"
   98         "\"parameter_address\":\"0x%llx\",\"command_address\":\"0x%llx\",\"bridge_span\":%llu,"
   99         "\"parameter_read_ok\":%s,\"parameter_float32_hex\":\"%s\",\"native_call_interval_measured\":%s}\n",
  100         event,mode_name,function==0?"pwm":"bridge",
  101         static_cast<unsigned long long>(begin),static_cast<unsigned long long>(end),syscall(SYS_gettid),other_thread,
  102         static_cast<unsigned long long>(calls[0].load()),static_cast<unsigned long long>(calls[1].load()),
  103         static_cast<unsigned long long>(overlaps.load()),static_cast<unsigned long long>(write_errors.load()),
  104         static_cast<unsigned long long>(parameter),static_cast<unsigned long long>(command),
  105         static_cast<unsigned long long>(span),read_ok?"true":"false",pwm_hex,call_interval?"true":"false");
  106     if(length<=0 || length>=int(sizeof(line)) || write(output,line,length)!=length)++write_errors;
  107 }

  109 void initialize() {
  110     const char* library=std::getenv("RMUA_BRIDGE_PROBE_ORIGINAL");
  111     const char* path=std::getenv("RMUA_BRIDGE_PROBE_OUTPUT");
  112     const char* mode=std::getenv("RMUA_BRIDGE_PROBE_MODE");
  113     if(!library || !path || !mode)fatal("missing explicit diagnostic configuration");
  114     mode_name=mode;
  115 #ifdef RMUA_LIFETIME_PROBE
  116 #ifdef RMUA_NATIVE_INPUT_PROBE
  117     if(std::strcmp(mode,"native-input")!=0)fatal("native input build requires native-input mode");
  118 #else
  119     if(std::strcmp(mode,"lifetime")!=0)fatal("lifetime build requires lifetime mode");
  120 #endif
  121 #else
  122     if(std::strcmp(mode,"serialize")==0)locking=true;
  123     else if(std::strcmp(mode,"observe")!=0)fatal("invalid probe mode");
  124 #endif
  125     // Resolve definitions in the already-loaded, unmodified pinned library.
  126     // Keeping this handle also prevents unloading an in-use original function.
  127     void* handle=dlopen(library,RTLD_NOW|RTLD_NOLOAD);
  128     if(!handle)fatal("original bridge library is not loaded");
  129     original_rotor=reinterpret_cast<RotorFunction>(dlsym(handle,
  130         "_ZN10airsim_ros13commands_impl18ROTOR_PWM_CallbackERKN5boost10shared_ptrIKNS_9RotorPWM_ISaIvEEEEE"));
  131     original_bridge=reinterpret_cast<BridgeFunction>(dlsym(handle,
  132         "_ZN10airsim_ros20PublishAsyncCommandsERKN3ros10TimerEventE"));
  133     if(!original_rotor || !original_bridge
  134             || original_rotor==&airsim_ros::commands_impl::ROTOR_PWM_Callback
  135             || original_bridge==&airsim_ros::PublishAsyncCommands)fatal("original symbols unavailable or recursive");
  136 #ifdef RMUA_LIFETIME_PROBE
  137     original_get=reinterpret_cast<Getter>(dlsym(handle,"_ZNKSt12__shared_ptrIN10airsim_ros16ROTOR_PWM_ParamsELN9__gnu_cxx12_Lock_policyE2EE3getEv"));
  138     original_command_get=reinterpret_cast<Getter>(dlsym(handle,"_ZNKSt12__shared_ptrIN10airsim_ros7CommandELN9__gnu_cxx12_Lock_policyE2EE3getEv"));
  139     original_parameter_from_owner=reinterpret_cast<Getter>(dlsym(handle,"_ZNSt23_Sp_counted_ptr_inplaceIN10airsim_ros16ROTOR_PWM_ParamsESaIS1_ELN9__gnu_cxx12_Lock_policyE2EE6_M_ptrEv"));
  140     original_free=reinterpret_cast<Destroy>(dlsym(handle,"_ZNSt23_Sp_counted_ptr_inplaceIN10airsim_ros16ROTOR_PWM_ParamsESaIS1_ELN9__gnu_cxx12_Lock_policyE2EE10_M_destroyEv"));
  141     original_ctor1=reinterpret_cast<Constructor>(dlsym(handle,"_ZN10airsim_ros7CommandC1EPKcPv"));
  142     original_ctor2=reinterpret_cast<Constructor>(dlsym(handle,"_ZN10airsim_ros7CommandC2EPKcPv"));
  143     if(!original_get || !original_command_get || !original_parameter_from_owner || !original_free
  144             || !original_ctor1 || !original_ctor2 || original_get==&lifetime_get

  148 #endif
  149 #ifdef RMUA_NATIVE_INPUT_PROBE
  150     original_init=reinterpret_cast<InitFunction>(dlsym(handle,
  151         "_ZN10airsim_ros13InitAirSimROSEiPPcPFvPvS2_EbbNS_17AirSimROSGameTypeEPKcS7_S7_i"));

  157     if(output<0)fatal("exclusive probe output unavailable");

  161 class Scope {
  162     const int index;
  163     const std::uint64_t begin;
  164     long other_thread=0;
  165     bool overlap=false;
  166 #ifdef RMUA_LIFETIME_PROBE
  167     Context previous;
  168 #endif
  169 public:
  170     explicit Scope(int value):index(value),begin(now_ns()) {
  171         const long own_thread=syscall(SYS_gettid);
  172         active_thread[index]=own_thread;
  173         ++active[index];++calls[index];
  174         other_thread=active_thread[1-index].load();
  175         overlap=active[1-index].load()!=0 && other_thread!=0 && other_thread!=own_thread;
  176         if(overlap)++overlaps;
  177 #ifdef RMUA_LIFETIME_PROBE
  178         if(index==1){previous=context;context=Context{};context.span=++span_counter;
  179             record("bridge_begin",1,begin,begin,0,0,0,context.span);}
  180 #endif
  181     }
  182     ~Scope() {
  183         const auto end=now_ns();
  184 #ifdef RMUA_LIFETIME_PROBE
  185         if(index==1){record("bridge_end",1,end,end,0,context.parameter,context.command,context.span);
  186             context=previous;}
  187 #endif
  188         if(--active[index]==0)active_thread[index]=0;
  189         if(overlap || calls[index].load()==1 || calls[index].load()%128==0)
  190             record(overlap?"overlap":"periodic",index,begin,end,other_thread);
  191     }
  192 };

  194 __attribute__((destructor)) void finish() {
  195     if(output>=0){const auto stamp=now_ns();record("exit",0,stamp,stamp);close(output);output=-1;}
  196 #ifdef RMUA_NATIVE_INPUT_PROBE
  197     if(self_memory>=0){close(self_memory);self_memory=-1;}
  198 #endif
  199 }

  238     std::call_once(initialized,initialize);

  267 void native_input_callback(void* command,void* result) {
  268     const auto original=native_callback.load();
  269     if(!original)fatal("native callback not bound");
  270     // This TLS context identifies a PWM Command constructed by the original
  271     // bridge. Unknown commands/sensor callbacks are passed through unread.
  272     const Context current=context;
  273     const bool identified=current.span && current.parameter && current.command
  274         && reinterpret_cast<std::uintptr_t>(command)==current.command;
  275     if(!identified){original(command,result);return;}
  276     const auto begin=now_ns();
  277     unsigned char payload[16]{};
  278     // Kernel-mediated bounded read avoids dereferencing potentially freed or
  279     // unmapped storage in this diagnostic. This is a snapshot, not an owning
  280     // copy, lock, or claim about the consumer's eventual read.
  281     const bool read_ok=pread(self_memory,payload,sizeof(payload),static_cast<off_t>(current.parameter))==ssize_t(sizeof(payload));
  282     char encoded[33]{};
  283     if(read_ok)for(unsigned i=0;i<sizeof(payload);++i)std::snprintf(encoded+2*i,3,"%02x",unsigned(payload[i]));
  284     record("native_pwm_entry",1,begin,now_ns(),0,current.parameter,current.command,current.span,encoded,read_ok);
  285     const auto call_begin=now_ns();
  286     original(command,result);
  287     const auto end=now_ns();
  288     record("native_pwm_exit",1,call_begin,end,0,current.parameter,current.command,current.span,"",false,true);
  289     const auto post_begin=now_ns();
  290     const bool post_ok=pread(self_memory,payload,sizeof(payload),static_cast<off_t>(current.parameter))==ssize_t(sizeof(payload));
  291     encoded[0]='\0';
  292     if(post_ok)for(unsigned i=0;i<sizeof(payload);++i)std::snprintf(encoded+2*i,3,"%02x",unsigned(payload[i]));
  293     record("native_pwm_post",1,post_begin,now_ns(),0,current.parameter,current.command,current.span,encoded,post_ok);
  294 }
