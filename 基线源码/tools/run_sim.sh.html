    1 #!/usr/bin/env bash
    2 # One owned scene; the harness starts control separately. Probes are diagnostic.
    3 set -euo pipefail
    4 CONT=${RMUA_CONTAINER:-rmua_noetic}
    5 SEED=${1:-123}
    6 MODE=${2:-offscreen}
    7 DIAGNOSTIC=${3:-}
    8 [[ "$SEED" =~ ^[0-9]+$ ]] || { echo 'Seed must be a nonnegative integer.' >&2; exit 2; }
    9 [[ "$MODE" == offscreen || "$MODE" == render ]] || { echo 'Mode: offscreen|render' >&2; exit 2; }
   10 [[ -z "$DIAGNOSTIC" || "$DIAGNOSTIC" == diagnostic-bridge-probe ]] || { echo 'Unknown diagnostic mode.' >&2; exit 2; }
   11 probe_environment=(-e RMUA_BRIDGE_PROBE_ENABLED=0)
   12 if [[ "$DIAGNOSTIC" == diagnostic-bridge-probe ]]; then
   13   [[ "${RMUA_BRIDGE_PROBE_MODE:-}" == observe || "${RMUA_BRIDGE_PROBE_MODE:-}" == serialize || "${RMUA_BRIDGE_PROBE_MODE:-}" == lifetime || "${RMUA_BRIDGE_PROBE_MODE:-}" == native-input ]] || exit 2
   14   [[ -n "${RMUA_BRIDGE_PROBE_SO:-}" && -n "${RMUA_BRIDGE_PROBE_OUTPUT:-}" ]] || exit 2
   15   probe_environment=(-e RMUA_BRIDGE_PROBE_ENABLED=1 -e "RMUA_BRIDGE_PROBE_SO=$RMUA_BRIDGE_PROBE_SO"
   16                      -e "RMUA_BRIDGE_PROBE_MODE=$RMUA_BRIDGE_PROBE_MODE"
   17                      -e "RMUA_BRIDGE_PROBE_OUTPUT=$RMUA_BRIDGE_PROBE_OUTPUT")
   18 fi
   19 exec 9>/tmp/rmua-next-simulator.lock
   20 flock -n 9 || { echo 'Another main simulator is active.' >&2; exit 1; }
   21 docker start "$CONT" >/dev/null
   22 if docker exec "$CONT" pgrep -f '[R]MUA-Linux|[r]osmaster|[r]oute_follower.py' >/dev/null; then
   23   echo 'Close the existing simulator/ROS/controller before starting a new scene.' >&2
   24   exit 1
   25 fi
   26 exec docker exec -u 1000:1000 -e DISPLAY="${DISPLAY:-:0}" \
   27   "${probe_environment[@]}" \
   28   -w /workspace/main/simulator/simulator_12.0.0.5 "$CONT" \
   29   bash -c '
   30     set -e
   31     source /opt/ros/noetic/setup.bash
   32     export ROS_HOME=/tmp/rmua_next_ros
   33     mkdir -p "$ROS_HOME"
   34     setsid roscore >"$ROS_HOME/roscore.log" 2>&1 &
   35     ros_pid=$!
   36     echo "$ros_pid" >"$ROS_HOME/ros.pid"
   37     sim_pid=
   38     cleanup() {
   39       if [[ -n "$sim_pid" ]]; then kill -- "-$sim_pid" 2>/dev/null || true; fi
   40       kill -- "-$ros_pid" 2>/dev/null || true
   41       rm -f "$ROS_HOME/sim.pid" "$ROS_HOME/ros.pid"
   42     }
   43     trap cleanup EXIT
   44     trap "exit 130" INT TERM
   45     ready=false
   46     for attempt in {1..100}; do
   47       if rosparam list >/dev/null 2>&1; then ready=true; break; fi
   48       sleep .1
   49     done
   50     "$ready" || { echo "ROS master failed to start" >&2; exit 1; }
   51     if [[ "$2" == offscreen ]]; then flags=(-RenderOffscreen); else flags=(-windowed -ResX=1280 -ResY=720); fi
   52     launcher=(./Build/LinuxNoEditor/RMUA.sh seed "$1" "${flags[@]}" -settings=/workspace/main/configs/simulator_settings.json)
   53     if [[ "${RMUA_BRIDGE_PROBE_ENABLED:-0}" == 1 ]]; then
   54       launcher=(env "LD_PRELOAD=$RMUA_BRIDGE_PROBE_SO"
   55         "RMUA_BRIDGE_PROBE_ORIGINAL=/workspace/main/simulator/simulator_12.0.0.5/Build/LinuxNoEditor/RMUA/Plugins/AirSimROS/Binaries/Linux/libairsimros.so"
   56         "RMUA_BRIDGE_PROBE_MODE=$RMUA_BRIDGE_PROBE_MODE"
   57         "RMUA_BRIDGE_PROBE_OUTPUT=$RMUA_BRIDGE_PROBE_OUTPUT" "${launcher[@]}")
   58     fi
   59     setsid "${launcher[@]}" &
   60     sim_pid=$!
   61     echo "$sim_pid" >"$ROS_HOME/sim.pid"
   62     wait "$sim_pid"
   63   ' rmua-next-simulator "$SEED" "$MODE"
