    1 #!/usr/bin/env bash
    2 # Stop only process groups recorded by main/tools/run_sim.sh.
    3 set -euo pipefail
    4 CONT=${RMUA_CONTAINER:-rmua_noetic}
    5 exec docker exec -u 1000:1000 "$CONT" bash -c '
    6   for name in sim ros; do
    7     path=/tmp/rmua_next_ros/$name.pid
    8     [[ -f "$path" ]] || continue
    9     read -r pid <"$path"
   10     [[ "$pid" =~ ^[0-9]+$ && "$pid" -gt 1 ]] || exit 1
   11     [[ -e /proc/$pid/cmdline ]] || continue
   12     command=$(tr "\0" " " </proc/$pid/cmdline)
   13     if [[ "$name" == sim && "$command" != *RMUA* ]] || [[ "$name" == ros && "$command" != *roscore* ]]; then
   14       echo "Recorded PID changed; refusing to stop $pid" >&2
   15       exit 1
   16     fi
   17     kill -- "-$pid" 2>/dev/null || true
   18     stopped=false
   19     for attempt in {1..100}; do
   20       if ! kill -0 -- "-$pid" 2>/dev/null; then stopped=true; break; fi
   21       sleep .1
   22     done
   23     "$stopped" || { echo "Process group $pid did not stop in 10s" >&2; exit 1; }
   24   done
   25 '
