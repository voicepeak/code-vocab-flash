    1 """Run the unit suite with explicit Windows exclusions; Linux runs discovery unchanged."""
    2 import importlib
    3 from pathlib import Path
    4 import sys
    5 import unittest
    6 
    7 ROOT = Path(__file__).resolve().parents[1]
    8 sys.path.insert(0, str(ROOT))
    9 
   10 
   11 def main():
   12     loader = unittest.TestLoader()
   13     if sys.platform != 'win32':
   14         suite = loader.discover(str(ROOT / 'tests' / 'unit'), top_level_dir=str(ROOT))
   15     else:
   16         print('Windows exclusions: test_control_guard (fcntl), reference worker '
   17               'process lifetime (Linux prctl), and source archive executable-mode '
   18               'publication (Windows file permissions).', file=sys.stderr, flush=True)
   19         excluded = {'test_control_guard', 'test_reference_worker', 'test_source_archive'}
   20         suite = unittest.TestSuite()
   21         for path in sorted((ROOT / 'tests' / 'unit').glob('test_*.py')):
   22             if path.stem not in excluded:
   23                 suite.addTests(loader.loadTestsFromModule(
   24                     importlib.import_module('tests.unit.' + path.stem)))
   25         for name in (
   26             'tests.unit.test_reference_worker.ReferenceIsolationTests.test_recovered_reference_does_not_automatically_resume_after_hold',
   27             'tests.unit.test_source_archive.SourceArchiveTests.test_changed_source_or_source_set_aborts_before_publication',
   28             'tests.unit.test_source_archive.SourceArchiveTests.test_symlink_and_escaping_names_are_not_archived',
   29         ):
   30             suite.addTests(loader.loadTestsFromName(name))
   31     return 0 if unittest.TextTestRunner(verbosity=1).run(suite).wasSuccessful() else 1
   32 
   33 
   34 if __name__ == '__main__':
   35     sys.exit(main())
