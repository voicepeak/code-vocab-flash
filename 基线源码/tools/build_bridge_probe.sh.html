    1 #!/usr/bin/env bash
    2 # Diagnostics only. Build with the simulator container's C++/glibc ABI.
    3 set -euo pipefail
    4 root=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/.." && pwd)
    5 variant=${1:-concurrency}
    6 [[ "$variant" == concurrency || "$variant" == lifetime || "$variant" == native-input ]] || { echo 'Variant: concurrency|lifetime|native-input' >&2; exit 2; }
    7 compile_flags=()
    8 if [[ "$variant" == lifetime ]]; then compile_flags=(-DRMUA_LIFETIME_PROBE); fi
    9 if [[ "$variant" == native-input ]]; then compile_flags=(-DRMUA_LIFETIME_PROBE -DRMUA_NATIVE_INPUT_PROBE); fi
   10 source_hash=$(sha256sum "$root/tools/diagnostics/bridge_probe.cpp" | cut -d' ' -f1)
   11 output="$root/runs/probe_builds/$source_hash/$variant"
   12 mkdir -p "$output"
   13 probe_build_dir=$(mktemp -d "$output/.build.XXXXXX")
   14 trap 'rm -rf -- "$probe_build_dir"' EXIT
   15 cp "$root/tools/diagnostics/bridge_probe.cpp" "$probe_build_dir/bridge_probe.cpp"
   16 [[ $(sha256sum "$probe_build_dir/bridge_probe.cpp" | cut -d' ' -f1) == "$source_hash" ]] || {
   17   echo 'Probe source changed during snapshot.' >&2; exit 2;
   18 }
   19 container_build_dir="/workspace/main/${probe_build_dir#"$root/"}"
   20 bash "$root/tools/enter_env.sh" g++ -std=c++14 -O2 -fPIC -shared -Wall -Wextra -Werror \
   21   "${compile_flags[@]}" "$container_build_dir/bridge_probe.cpp" -pthread -ldl -o "$container_build_dir/bridge_probe.so"
   22 # Atomic replacement leaves any already-loaded inode intact. Preserve the exact
   23 # source copy as well as its hash; never rebuild into a running library's inode.
   24 mv -f -- "$probe_build_dir/bridge_probe.so" "$output/bridge_probe.so"
   25 mv -f -- "$probe_build_dir/bridge_probe.cpp" "$output/bridge_probe_source.cpp"
   26 echo "$output/bridge_probe.so"
