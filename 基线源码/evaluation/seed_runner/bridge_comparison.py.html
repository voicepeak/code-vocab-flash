   18 def case_order(protocol):
   19     if (protocol.get('schema_version') != 1
   20             or type(protocol.get('repeats')) is not int or not 1 <= protocol['repeats'] <= 5
   21             or type(protocol.get('hover_seconds')) not in (int, float)
   22             or not 10 <= protocol['hover_seconds'] <= 20
   23             or protocol.get('stationary_bootstrap') is not True):
   24         raise ValueError('invalid bounded bridge protocol')
   25     seeds = protocol.get('seeds')
   26     if (not isinstance(seeds, list) or not 1 <= len(seeds) <= 4
   27             or any(type(seed) is not int or seed < 0 for seed in seeds)):
   28         raise ValueError('invalid development seeds')
   29     modes = tuple(protocol.get('modes', MODES))
   30     if modes not in (MODES, ('none', 'lifetime')):
   31         raise ValueError('unsupported fixed diagnostic comparison')
   32     cases = []
   33     for repeat in range(protocol['repeats']):
   34         for index, seed in enumerate(seeds):
   35             offset = (repeat + index) % len(modes)
   36             for mode in modes[offset:] + modes[:offset]:
   37                 cases.append({'seed': seed, 'repeat': repeat, 'mode': mode})
   38     return cases

   41 def collect_case(run):
   42     manifest = json.loads((run / 'manifest.json').read_text())
   43     summary_path = run / 'pwm_summary.json'
   44     summary = json.loads(summary_path.read_text()) if summary_path.exists() else None
   45     metrics_path = run / 'closed_loop_metrics.json'
   46     metrics = json.loads(metrics_path.read_text()) if metrics_path.exists() else None
   47     trace_path = run / 'command_trace_audit.json'
   48     trace = json.loads(trace_path.read_text()) if trace_path.exists() else None
   49     lifetime_path = run / 'bridge_lifetime_audit.json'
   50     lifetime = json.loads(lifetime_path.read_text()) if lifetime_path.exists() else None
   51     return {'run_directory': str(run), 'run_state': manifest.get('run_state'),
   52             'controller_fault': summary.get('fault') if summary else 'MISSING_SUMMARY',
   53             'controller_completed': bool(summary and summary.get('fault') is None
   54                                          and summary.get('commands', 0) > 0
   55                                          and manifest.get('observer_exit_code') == 0),
   56             'command_trace_audit_exit_code': manifest.get('command_trace_audit_exit_code'),
   57             'evaluation_valid': metrics.get('evaluation_valid', False) if metrics else False,
   58             'summary': summary, 'metrics': metrics, 'command_trace': trace,
   59             'lifetime_audit': lifetime,
   60             'bridge_probe_summary': manifest.get('bridge_probe_summary'),
   61             'source_archive': manifest.get('source_archive'),
   62             'instrumentation': manifest.get('simulator_runtime_instrumentation')}

   65 def main():
   66     root = Path(__file__).resolve().parents[2]
   67     parser = argparse.ArgumentParser(description=__doc__)
   68     parser.add_argument('--protocol', type=Path, default=root / 'configs/bridge_comparison_v1.json')
   69     args = parser.parse_args()
   70     protocol = json.loads(args.protocol.read_text())
   71     planned = case_order(protocol)
   72     frozen = source_hashes(root)
   73     simulator = verify_local_simulator(root)
   74     container_digest = subprocess.check_output(['docker', 'inspect', os.environ.get('RMUA_CONTAINER', 'rmua_noetic'),
   75                                                 '--format', '{{.Image}}'], text=True).strip()
   76     variant = 'lifetime' if any(c['mode'] == 'lifetime' for c in planned) else 'concurrency'
   77     probe = Path(subprocess.check_output(['bash', str(root / 'tools/build_bridge_probe.sh'), variant],
   78                                         cwd=root, text=True).strip())
   79     probe_hash = file_hash(probe)
   80     batch = root / 'runs/batches' / (datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%SZ_')
   81                                     + uuid.uuid4().hex)
   82     batch.mkdir(parents=True, exist_ok=False)
   83     report = {'protocol': protocol, 'planned_cases': planned, 'source_sha256_frozen': frozen,
   84               'source_archive': archive_sources(root, batch, frozen), 'simulator': simulator,
   85               'simulator_container_image_digest': container_digest,
   86               'probe_binary_sha256': probe_hash, 'cases': [], 'status': 'RUNNING',
   87               'probe_build_variant': variant,
   88               'stage_acceptance': False, 'physical_impulse_root_cause_proven': False,
   89               'official_result': 'UNKNOWN'}
   90 
   91     def save():
   92         (batch / 'batch_report.json').write_text(json.dumps(report, indent=2) + '\n')
   93 
   94     save()
   95     print('Batch:', batch, flush=True)
   96     try:
   97         for index, plan in enumerate(planned):
   98             if source_hashes(root) != frozen or file_hash(probe) != probe_hash:
   99                 raise RuntimeError('source or diagnostic binary changed before case')
  100             command = [sys.executable, '-m', 'evaluation.seed_runner.pwm_identification',
  101                        '--seed', str(plan['seed']), '--hover-seconds', str(protocol['hover_seconds']),
  102                        '--evaluate-truth', '--stationary-bootstrap']
  103             if plan['mode'] != 'none':
  104                 command.extend(['--bridge-probe', plan['mode']])
  105             console = batch / ('case_%02d.console.log' % index)
  106             with console.open('x') as output:
  107                 code = subprocess.call(command, cwd=root, stdout=output, stderr=subprocess.STDOUT)
  108             candidates = [Path(line.strip()) for line in console.read_text().splitlines()
  109                           if line.startswith(str(root / 'runs') + '/')]
  110             candidates = [path for path in candidates if path.parent == root / 'runs'
  111                           and (path / 'manifest.json').is_file()]
  112             case = dict(plan, index=index, exit_code=code, console=console.name, command=command)
  113             report['cases'].append(case)
  114             if len(candidates) != 1:
  115                 raise RuntimeError('case has no unique run manifest')
  116             run = candidates[0]
  117             case.update(collect_case(run))
  118             manifest = json.loads((run / 'manifest.json').read_text())
  119             case['provenance_verified'] = (
  120                 manifest.get('source_sha256_before') == frozen and manifest.get('source_unchanged')
  121                 and manifest['simulator_preflight'] == simulator
  122                 and manifest.get('simulator_container_image_digest') == container_digest
  123                 and verify_local_simulator(root) == simulator and source_hashes(root) == frozen
  124                 and file_hash(run / case['source_archive']['path']) == case['source_archive']['sha256'])
  125             if case['instrumentation']:
  126                 case['provenance_verified'] = (case['provenance_verified']
  127                     and case['instrumentation']['binary_sha256'] == probe_hash
  128                     and case['instrumentation']['mode'] == plan['mode']
  129                     and manifest.get('original_simulator_files_unchanged') is True)
  130             elif plan['mode'] != 'none':
  131                 case['provenance_verified'] = False
  132             save()
  133             print('Case', index + 1, plan['seed'], plan['mode'], 'exit', code,
  134                   'fault', case['controller_fault'], 'trace', case['command_trace_audit_exit_code'], flush=True)
  135             if not case['provenance_verified'] or code not in (0, 2):
  136                 raise RuntimeError('case provenance or infrastructure failed; retain evidence and stop batch')
  137         report['status'] = 'COMPLETE'
  138     except Exception as error:
  139         report['status'] = 'ABORTED'
  140         report['error'] = str(error)
  141         raise
  142     finally:
  143         report['completed_cases'] = len(report['cases'])
  144         report['all_cases_completed'] = (report['status'] == 'COMPLETE'
  145                                          and len(report['cases']) == len(planned))
  146         report['source_unchanged'] = source_hashes(root) == frozen
  147         save()
  148     return 0
