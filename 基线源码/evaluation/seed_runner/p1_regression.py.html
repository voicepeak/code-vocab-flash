   11 def load_protocol(path):
   12     protocol=json.loads(path.read_text())
   13     seeds=protocol.get('seeds')
   14     if (protocol.get('schema_version')!=1 or not isinstance(seeds,list) or not 1<=len(seeds)<=50
   15             or any(type(seed) is not int or seed<0 for seed in seeds)
   16             or protocol.get('mode') not in ('PWM','VEL')
   17             or not 10<=protocol.get('hover_seconds',0)<=20):
   18         raise ValueError('invalid bounded reproduction protocol')
   19     for name in ('braking','yaw_motion','snapshot'):
   20         if type(protocol.get(name)) is not bool:
   21             raise ValueError('protocol flags must be booleans')
   22     if type(protocol.get('angular_recovery', False)) is not bool:
   23         raise ValueError('angular recovery flag must be boolean')
   24     if type(protocol.get('stationary_bootstrap', False)) is not bool:
   25         raise ValueError('stationary bootstrap flag must be boolean')
   26     if protocol.get('angular_recovery') and protocol['mode']!='PWM':
   27         raise ValueError('angular recovery requires PWM')
   28     profile=protocol.get('angular_recovery_profile','standard')
   29     if profile not in ('standard','fast') or (profile!='standard' and not protocol.get('angular_recovery')):
   30         raise ValueError('invalid angular recovery profile')
   31     if 'angular_pulse_moments' in protocol:
   32         moments=protocol['angular_pulse_moments']
   33         if (not protocol.get('angular_recovery') or protocol['mode']!='PWM'
   34                 or protocol['braking'] or protocol['yaw_motion']
   35                 or not isinstance(moments,list) or not 1<=len(moments)<=4
   36                 or any(type(v) not in (int,float) or not 0<v<=.8 for v in moments)
   37                 or moments!=sorted(moments)):
   38             raise ValueError('invalid staged angular pulse protocol')
   39     return protocol

   42 def main():
   43     root=Path(__file__).resolve().parents[2]
   44     parser=argparse.ArgumentParser(description=__doc__)
   45     parser.add_argument('--protocol',type=Path,default=root/'configs/p1_development_reproduction_v1.json')
   46     parser.add_argument('--limit',type=int)
   47     args=parser.parse_args()
   48     protocol=load_protocol(args.protocol)
   49     plans=[{'seed':seed,'angular_pulse_moment':moment}
   50            for moment in protocol.get('angular_pulse_moments',[None]) for seed in protocol['seeds']]
   51     if args.limit is not None:
   52         if not 1<=args.limit<=len(plans):parser.error('invalid case limit')
   53         plans=plans[:args.limit]
   54     batch=root/'runs/batches'/(datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%SZ_')+uuid.uuid4().hex)
   55     batch.mkdir(parents=True,exist_ok=False)
   56     frozen=source_hashes(root)
   57     report={'protocol':protocol,'source_sha256_frozen':frozen,'cases':[],
   58             'planned_cases':plans,'source_archive':archive_sources(root,batch,frozen),
   59             'stage_acceptance':False,'official_result':'UNKNOWN','source_unchanged':True}
   60     def save():
   61         (batch/'batch_report.json').write_text(json.dumps(report,indent=2)+'\n')
   62     save()
   63     print('Batch:',batch,flush=True)
   64     for index,plan in enumerate(plans):
   65         seed=plan['seed']
   66         command=['python3','-m','evaluation.seed_runner.pwm_identification','--seed',str(seed),
   67                  '--mode',protocol['mode'],'--hover-seconds',str(protocol['hover_seconds']),'--evaluate-truth']
   68         for key,flag in (('braking','--braking'),('yaw_motion','--yaw-motion'),('snapshot','--snapshot')):
   69             if protocol[key]:command.append(flag)
   70         if protocol.get('angular_recovery'):command.append('--angular-recovery')
   71         if protocol.get('angular_recovery_profile','standard')!='standard':
   72             command.extend(['--angular-recovery-profile',protocol['angular_recovery_profile']])
   73         if protocol.get('stationary_bootstrap'):command.append('--stationary-bootstrap')
   74         if plan['angular_pulse_moment'] is not None:
   75             command.extend(['--angular-pulse','--angular-pulse-moment',str(plan['angular_pulse_moment'])])
   76         output=batch/('case_%02d.console.log'%index)
   77         with output.open('x') as log:
   78             code=subprocess.call(command,cwd=root,stdout=log,stderr=subprocess.STDOUT)
   79         candidates=[Path(line.strip()) for line in output.read_text().splitlines()
   80                     if line.startswith(str(root/'runs')+'/')]
   81         candidates=[p for p in candidates if p.parent==root/'runs' and (p/'manifest.json').exists()]
   82         case={'index':index,'seed':seed,'exit_code':code,'console':output.name,'command':command}
   83         case['angular_pulse_moment']=plan['angular_pulse_moment']
   84         if len(candidates)==1:
   85             run=candidates[0]
   86             prefix=protocol['mode'].lower()
   87             case['run_directory']=str(run)
   88             summary=run/(prefix+'_summary.json')
   89             if summary.exists():case['summary']=json.loads(summary.read_text())
   90             if (run/'fault_observation.json').exists():case['fault_observation']='fault_observation.json'
   91             manifest=json.loads((run/'manifest.json').read_text())
   92             case['run_source_unchanged']=(manifest.get('source_unchanged',False)
   93                                          and manifest['source_sha256_before']==frozen)
   94         report['cases'].append(case)
   95         report['source_unchanged']=source_hashes(root)==frozen
   96         save()
   97         print('Case',index+1,'seed',seed,'exit',code,'fault',case.get('summary',{}).get('fault'),
   98               'run',case.get('run_directory'),flush=True)
   99         if not report['source_unchanged'] or code not in (0,2):break
  100         if 'angular_pulse_moments' in protocol and (code!=0 or not case.get('run_source_unchanged')):
  101             report['staged_pulse_stopped_on_failed_case']=True
  102             break
  103     report['completed_cases']=len(report['cases'])
  104     report['all_cases_completed']=len(report['cases'])==len(plans)
  105     report['control_fault_cases']=sum(c.get('summary',{}).get('fault') is not None for c in report['cases'])
  106     save()
  107     return 0 if report['source_unchanged'] and report['all_cases_completed'] else 1
