   16 def file_hash(path):
   17     value = hashlib.sha256()
   18     with Path(path).open("rb") as handle:
   19         for chunk in iter(lambda: handle.read(1024 * 1024), b""):
   20             value.update(chunk)
   21     return value.hexdigest()

   24 def source_hashes(root):
   25     names = subprocess.check_output(
   26         ["git", "ls-files", "-z", "--cached", "--others", "--exclude-standard"], cwd=root)
   27     result = {}
   28     for name in sorted(set(n.decode() for n in names.split(b"\0") if n)):
   29         path = Path(root) / name
   30         if path.is_file():
   31             result[name] = file_hash(path)
   32     return result

   35 def archive_sources(root, run, expected):
   36     """Retain the exact bytes, including dirty files, before starting a scene.
   37 
   38     Only regular, repository-relative sources are accepted. Atomic exclusive
   39     publication prevents either a partial archive or replacement of old evidence.
   40     Metadata is normalized so the same bytes and modes produce the same archive.
   41     """
   42     root, run = Path(root).resolve(), Path(run)
   43     destination = run / "source_snapshot.tar.gz"
   44     temporary = None
   45     total = 0
   46     try:
   47         with tempfile.NamedTemporaryFile(dir=run, suffix=".tmp", delete=False) as output:
   48             temporary = Path(output.name)
   49             with gzip.GzipFile(filename="", mode="wb", fileobj=output, mtime=0) as compressed:
   50                 with tarfile.open(fileobj=compressed, mode="w", format=tarfile.PAX_FORMAT) as archive:
   51                     for name, expected_hash in sorted(expected.items()):
   52                         relative = Path(name)
   53                         path = root / relative
   54                         if (relative.is_absolute() or ".." in relative.parts
   55                                 or relative.as_posix() != name or path.resolve() != path
   56                                 or path.is_symlink()):
   57                             raise ValueError("source must be a regular repository-relative file: " + name)
   58                         with path.open("rb") as source:
   59                             metadata = os.fstat(source.fileno())
   60                             if not stat.S_ISREG(metadata.st_mode):
   61                                 raise ValueError("nonregular source: " + name)
   62                             data = source.read()
   63                         if hashlib.sha256(data).hexdigest() != expected_hash:
   64                             raise RuntimeError("source changed before archival: " + name)
   65                         member = tarfile.TarInfo(name)
   66                         member.size = len(data)
   67                         member.mode = stat.S_IMODE(metadata.st_mode)
   68                         archive.addfile(member, io.BytesIO(data))
   69                         total += len(data)
   70             output.flush()
   71             os.fsync(output.fileno())
   72         if source_hashes(root) != expected:
   73             raise RuntimeError("source set changed during archival")
   74         temporary.chmod(0o444)
   75         os.link(str(temporary), str(destination))
   76         return {"path": destination.name, "sha256": file_hash(destination),
   77                 "files": len(expected), "uncompressed_bytes": total,
   78                 "scope": "workspace_sources_not_container_or_simulator_binaries"}
   79     finally:
   80         if temporary is not None:
   81             temporary.unlink()

   84 def prepare_run(root, output, seed, image_digest, config):
   85     root = Path(root).resolve()
   86     run = Path(output).resolve() / (datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%SZ_")
   87                                   + uuid.uuid4().hex)
   88     run.mkdir(parents=True, exist_ok=False)
   89     status = subprocess.check_output(["git", "status", "--porcelain"], cwd=root, text=True)
   90     manifest = {
   91         "schema_version": 1, "seed": seed, "seed_evidence": "operator_declared",
   92         "git_sha": subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=root, text=True).strip(),
   93         "git_status": status, "image_digest": image_digest,
   94         "config": config, "source_sha256_before": source_hashes(root),
   95         "model_hashes": {}, "telemetry": [], "safety_events": [],
   96         "official_result": "UNKNOWN", "stage": "P0_READ_ONLY_AUDIT",
   97     }
   98     manifest["source_archive"] = archive_sources(root, run, manifest["source_sha256_before"])
   99     (run / "manifest.json").write_text(json.dumps(manifest, indent=2) + "\n")
  100     return run, manifest
