   11 def compare(estimate_path, telemetry_path):
   12     estimate_path, telemetry_path = Path(estimate_path), Path(telemetry_path)
   13     estimates=json.loads(estimate_path.read_text()); groups={}
   14     for line in telemetry_path.read_text().splitlines():
   15         row=json.loads(line);raw=row['raw'];epoch=row['command']['epoch']
   16         group=groups.setdefault(epoch,{})
   17         # Duplicates do not introduce a new sample or refresh a timestamp.
   18         group.setdefault(raw['stamp'],tuple(raw['omega_body']))
   19     groups={epoch:sorted(samples.items()) for epoch,samples in groups.items()}
   20     rows=[]
   21     for row in estimates['results']:
   22         if row['status']!='ESTIMATED':continue
   23         motion=row['motion'];result={'previous_stamp':motion['previous_stamp'],
   24                                     'current_stamp':motion['current_stamp'],'epoch':motion['epoch']}
   25         try:
   26             gyro=gyro_rotation(groups.get(motion['epoch'],[]),motion['previous_stamp'],motion['current_stamp'])
   27             result.update(status='COMPARED',rotation_residual_rad=rotation_residual(motion['rotation_previous_from_current_body'],gyro))
   28         except ValueError as error:result.update(status='REJECTED',reason=str(error))
   29         rows.append(result)
   30     return {'source':estimates['source'],'estimate_sha256':hashlib.sha256(estimate_path.read_bytes()).hexdigest(),
   31             'imu_telemetry_sha256':hashlib.sha256(telemetry_path.read_bytes()).hexdigest(),
   32             'input_role':'official image VO and recorded official IMU angular velocity only',
   33             'gyro_bias_correction':'none','truth_accessed':False,'control_eligible':False,
   34             'status_counts':dict(Counter(r['status'] for r in rows)), 'results':rows}

   40     print(json.dumps({'output':str(output),'status_counts':result['status_counts']}))
