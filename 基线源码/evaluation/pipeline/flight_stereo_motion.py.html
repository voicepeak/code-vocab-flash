   25 def estimate(source, settings, method="orb"):
   26     """Only scene images/index and simulator settings are read in this step."""
   27     import cv2
   28     cv2.setNumThreads(1)
   29     source = Path(source)
   30     rig = load_nominal_rig(settings)
   31     index_path = source / 'scene_trace/index.json'
   32     index = json.loads(index_path.read_text())
   33     if index['fault'] is not None:
   34         raise ValueError('scene capture reports an error')
   35     images = {side: sorted((row for row in index['frames'] if row['sensor'] == 'front_' + side),
   36                            key=lambda row: row['stamp']) for side in ('left', 'right')}
   37     pairs, unmatched, used = [], [], set()
   38     files = {str(index_path): digest(index_path), str(settings): digest(Path(settings))}
   39     for left in images['left']:
   40         candidates = [(abs(left['stamp'] - row['stamp']), i, row) for i, row in enumerate(images['right']) if i not in used]
   41         if not candidates:
   42             unmatched.append(left['file']); continue
   43         skew, i, right = min(candidates, key=lambda item: item[0])
   44         if skew > .005:
   45             unmatched.append(left['file']); continue
   46         used.add(i)
   47         frames = []
   48         for side, row in (('left', left), ('right', right)):
   49             path = index_path.parent / row['file']
   50             if path.parent != index_path.parent or path.suffix != '.png':
   51                 raise ValueError('invalid scene image path')
   52             data = cv2.imread(str(path), cv2.IMREAD_COLOR)
   53             if data is None:
   54                 raise ValueError('unreadable scene image')
   55             files[str(path)] = digest(path)
   56             stamp = row['stamp']
   57             frames.append(ImageFrame(Metadata(stamp, stamp + .2, 0, 'retained_' + side,
   58                                               'front_' + side + '_image'), 'bgr8', data))
   59         stamp = max(frame.metadata.stamp for frame in frames)
   60         meta = Metadata(stamp, min(frame.metadata.valid_until for frame in frames), 0,
   61                         'offline_pair', 'front_stereo')
   62         pairs.append((left['file'], StereoPair(meta, frames[0], frames[1], skew)))
   63     output = []
   64     for (previous_name, previous), (current_name, current) in zip(pairs, pairs[1:]):
   65         row = {'previous': previous_name, 'current': current_name,
   66                'previous_stamp': previous.left.metadata.stamp, 'current_stamp': current.left.metadata.stamp}
   67         started = time.monotonic()
   68         try:
   69             cv2.setRNGSeed(0)
   70             result = estimate_stereo_motion(previous, current, rig, method=method)
   71             row.update(status='ESTIMATED', motion=asdict(result))
   72         except (ValueError, cv2.error) as error:
   73             row.update(status='REJECTED', reason=str(error))
   74         row['elapsed_s'] = time.monotonic() - started
   75         output.append(row)
   76     return {'source': str(source), 'method': method, 'status_counts': dict(Counter(r['status'] for r in output)),
   77             'stereo_pairs': len(pairs), 'unmatched_left': unmatched,
   78             'unmatched_right_count': len(images['right']) - len(used), 'input_sha256': files,
   79             'input_role': 'official_recorded_images_and_nominal_settings',
   80             'truth_accessed': False, 'control_eligible': False, 'results': output}

   83 def evaluate(estimate_result, truth_path):
   84     """Open truth only after estimates were saved; never feeds the VO algorithm."""
   85     truth_path = Path(truth_path)
   86     records = [json.loads(line) for line in truth_path.read_text().splitlines()]
   87     ordered = sorted({r['stamp']: r for r in records if r['stamp'] > 0}.values(), key=lambda row: row['stamp'])
   88     stamps = [r['stamp'] for r in ordered]
   89     def interpolate(stamp):
   90         i = bisect.bisect_left(stamps, stamp)
   91         if i == 0 or i == len(stamps): return None
   92         a, b = ordered[i-1], ordered[i]
   93         span = b['stamp'] - a['stamp']
   94         if span <= 0 or span > .05: return None
   95         fraction = (stamp - a['stamp']) / span
   96         position = (1-fraction)*np.array(a['position']) + fraction*np.array(b['position'])
   97         attitude = slerp(a['orientation'], b['orientation'], fraction)
   98         return position, rotation(attitude)
   99     result = []
  100     for row in estimate_result['results']:
  101         compared = {'previous': row['previous'], 'current': row['current'], 'status': row['status']}
  102         if row['status'] != 'ESTIMATED':
  103             compared['reason'] = row['reason']; result.append(compared); continue
  104         old, new = interpolate(row['previous_stamp']), interpolate(row['current_stamp'])
  105         if old is None or new is None:
  106             compared['status'] = 'TRUTH_UNAVAILABLE'; result.append(compared); continue
  107         delta = old[1].T @ (new[0] - old[0])
  108         truth_rotation = old[1].T @ new[1]
  109         motion = row['motion']
  110         predicted = np.array(motion['translation_previous_body_m'])
  111         residual = np.array(motion['rotation_previous_from_current_body']).T @ truth_rotation
  112         compared.update(status='COMPARED', truth_displacement_m=delta.tolist(),
  113                         truth_displacement_norm_m=float(np.linalg.norm(delta)),
  114                         translation_error_m=float(np.linalg.norm(predicted-delta)),
  115                         rotation_error_rad=float(np.arccos(np.clip((np.trace(residual)-1)/2, -1., 1.))))
  116         result.append(compared)
  117     return {'source': estimate_result['source'], 'truth_sha256': digest(truth_path),
  118             'role': 'isolated_post_estimate_evaluation', 'status_counts': dict(Counter(r['status'] for r in result)),
  119             'results': result}

  122 def main():
  123     parser = argparse.ArgumentParser(description=__doc__)
  124     sub = parser.add_subparsers(dest='command', required=True)
  125     est = sub.add_parser('estimate'); est.add_argument('source'); est.add_argument('output')
  126     est.add_argument('--settings', default='configs/simulator_settings.json')
  127     est.add_argument('--method', choices=('orb', 'orb_lk', 'orb_lk_half'), default='orb')
  128     ev = sub.add_parser('evaluate'); ev.add_argument('estimate'); ev.add_argument('truth'); ev.add_argument('output')
  129     args = parser.parse_args()
  130     result = (estimate(args.source, args.settings, args.method) if args.command == 'estimate'
  131               else evaluate(json.loads(Path(args.estimate).read_text()), args.truth))
  132     with Path(args.output).open('x') as out: json.dump(result, out, indent=2, allow_nan=False); out.write('\n')
  133     print(json.dumps({'output': args.output, 'status_counts': result['status_counts']}))
