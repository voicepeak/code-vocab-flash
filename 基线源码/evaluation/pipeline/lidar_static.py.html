   11 @dataclass(frozen=True)
   12 class ProbeResult:
   13     reason: str
   14     map_cells: int = 0
   15     candidate_reasons: tuple = ()
   16     # A nominal endpoint map cannot certify FREE or an executable trajectory.
   17     control_eligible: bool = False
   18 
   19 
   20 class LidarStaticProbe:
   21     """Join retained scan, history, rolling occupied map and static planner.
   22 
   23     The caller owns the history and serializes its updates with this probe.
   24     A scan is never re-used after rejection and no control/reference is returned.
   25     """
   26     def __init__(self, history, sensor_pose, limits, epoch=0, **map_limits):
   27         if history.epoch != epoch:
   28             raise ValueError('history and map epoch mismatch')
   29         self.history = history
   30         self.sensor_pose = sensor_pose
   31         self.limits = limits
   32         self.map = RollingEndpointMap(epoch=epoch, **map_limits)
   33         self._lock = threading.RLock()
   34 
   35     def reset(self, epoch):
   36         with self._lock:
   37             # History is reset by its owner before the map, so they cannot silently diverge.
   38             if self.history.epoch != epoch:
   39                 raise ValueError('history must be reset before probe')
   40             self.map.reset(epoch)
   41 
   42     def evaluate(self, cloud, state, initial_reference, goal, now, wall_now, epoch):
   43         with self._lock:
   44             if epoch != self.map.epoch or epoch != self.history.epoch:
   45                 return ProbeResult('EPOCH_MISMATCH')
   46             if not cloud.metadata.current(now, epoch) or cloud.metadata.frame_id != 'lidar':
   47                 return ProbeResult('SCAN_UNAVAILABLE')
   48             sources = self.history.snapshot_sources()
   49             if (not sources or sources[-1][0] != state or not state.metadata.current(now, epoch)
   50                     or state.health != Health.OK):
   51                 return ProbeResult('STATE_UNAVAILABLE')
   52             try:
   53                 pose = self.history.lookup(cloud.metadata.stamp, now, wall_now, epoch)
   54             except HistoryUnavailable as exc:
   55                 return ProbeResult('POSE_' + str(exc))
   56             try:
   57                 observation = world_endpoints(cloud, pose, self.sensor_pose)
   58                 if not self.map.integrate(observation, now, epoch):
   59                     return ProbeResult('SCAN_DUPLICATE')
   60                 snapshot = self.map.snapshot(now, epoch)
   61             except ValueError:
   62                 return ProbeResult('MAP_REJECTED')
   63             try:
   64                 plan = plan_static_local(state, initial_reference, goal, snapshot, now, epoch, self.limits)
   65             except (ValueError, OverflowError):
   66                 return ProbeResult('PLANNER_INPUT_REJECTED', len(snapshot.cells))
   67             return ProbeResult(plan.reason, len(snapshot.cells), tuple(check.reason for check in plan.checks))
