   31 def interpolate_pose(rows, stamp, epoch=0):
   32     """Interpolate recorded controller state, not independently calibrated truth."""
   33     timestamps=[row['raw']['stamp'] for row in rows]
   34     i=bisect.bisect_left(timestamps,stamp)
   35     if i==0 or i==len(rows):return None,'NO_STATE_BRACKET'
   36     a,b=rows[i-1],rows[i];t0,t1=timestamps[i-1],timestamps[i]
   37     if (t1<=t0 or t1-t0>.05 or a['state']['health']!='OK' or b['state']['health']!='OK'
   38             or a['command']['epoch']!=epoch or b['command']['epoch']!=epoch):
   39         return None,'BAD_STATE_BRACKET'
   40     first,second=a['state'],b['state'];dt=t1-t0;fraction=(stamp-t0)/dt
   41     p0,p1=np.asarray(first['position_ned']),np.asarray(second['position_ned'])
   42     predicted=.5*(np.asarray(first['velocity_ned'])+second['velocity_ned'])*dt
   43     position_residual=float(np.linalg.norm(p1-p0-predicted))
   44     q0,q1=unit(first['orientation_wxyz']),unit(second['orientation_wxyz'])
   45     delta=unit(multiply(q0*(1.,-1.,-1.,-1.),q1))
   46     if delta[0]<0:delta=-delta
   47     length=float(np.linalg.norm(delta[1:]))
   48     angle=delta[1:]*(2*math.atan2(length,float(delta[0]))/length) if length>1e-12 else np.zeros(3)
   49     expected=.5*(np.asarray(first['angular_velocity_body'])+second['angular_velocity_body'])*dt
   50     angular_residual=float(np.linalg.norm(angle-expected))
   51     if position_residual>.1 or angular_residual>.05:return None,'STATE_DISCONTINUITY'
   52     return DiagnosticPose(stamp,epoch,tuple((1-fraction)*p0+fraction*p1),tuple(slerp(q0,q1,fraction)),
   53                           (t0,t1),position_residual,angular_residual),None

   56 def replay(gate_file):
   57     gate_file=Path(gate_file); data=json.loads(gate_file.read_text()); source=Path(data['source'])
   58     log=source/('pwm_telemetry.jsonl' if (source/'pwm_telemetry.jsonl').exists() else 'vel_telemetry.jsonl')
   59     states=[json.loads(line) for line in log.read_text().splitlines()]
   60     if any(a['raw']['stamp']>=b['raw']['stamp'] for a,b in zip(states,states[1:])):
   61         raise ValueError('recorded estimator clock not strictly increasing')
   62     tracker=GateTracker();output=[]
   63     for frame in data['results']:
   64         if frame['status']!='PROCESSED':
   65             output.append({'left':frame['left'],'status':frame['status']});continue
   66         if not frame['candidates']:
   67             # Without candidate metadata the saved 51 data does not retain image stamp here.
   68             output.append({'left':frame['left'],'status':'NO_CANDIDATE'});continue
   69         values=frame['candidates'];stamp=values[0]['metadata']['stamp'];epoch=values[0]['metadata']['epoch']
   70         pos,reason=interpolate_pose(states,stamp,epoch)
   71         if pos is None:
   72             output.append({'left':frame['left'],'status':reason});continue
   73         candidates=[]
   74         for value in values:
   75             m=Metadata(**{**value['metadata'],'covariance_or_bounds':tuple(value['metadata']['covariance_or_bounds'])})
   76             candidates.append(GateCandidate(m,tuple(tuple(p) for p in value['image_edges_px']),
   77                 tuple(value['center_body_frd_m']),value['nominal_width_m'],value['nominal_height_m'],
   78                 tuple(value['edge_disparities_px']),tuple(value['edge_support'])))
   79         # This is the earliest time both estimator samples exist, not the actual
   80         # completion time of the expensive disparity/gate computation.
   81         now=pos.bracket_stamps[1]
   82         try:
   83             associations=tracker.ingest(tuple(candidates),pos,now,epoch)
   84             tracks=tracker.snapshot(now,epoch)
   85             output.append({'left':frame['left'],'status':'TRACKED','stamp':stamp,'now':now,
   86                            'associations':associations,'tracks':[asdict(t) for t in tracks],
   87                            'pose_bracket_stamps':pos.bracket_stamps})
   88         except ValueError as error:
   89             output.append({'left':frame['left'],'status':'REJECTED','reason':str(error)})
   90     return {'gate_input':str(gate_file),'gate_input_sha256':hashlib.sha256(gate_file.read_bytes()).hexdigest(),
   91             'state_input':str(log),'state_input_sha256':hashlib.sha256(log.read_bytes()).hexdigest(),
   92             'uses_truth':False,'now_is_earliest_bracket_not_actual_processing_end':True,
   93             'control_eligible':False,'results':output}
