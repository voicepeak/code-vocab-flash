   19 def replay(source, settings, stride=5, depth_method="full", reuse_edges=False):
   20     source = Path(source); directory = source / 'scene_trace'
   21     index = json.loads((directory / 'index.json').read_text())
   22     rig = load_nominal_rig(settings)
   23     manifest = json.loads((source / 'manifest.json').read_text())
   24     if manifest['simulator_preflight']['settings_sha256'] != rig.settings_sha256:
   25         raise ValueError('settings do not match recorded capture')
   26     if index['fault'] is not None: raise ValueError('scene recorder failed')
   27     sides = {side: sorted((r for r in index['frames'] if r['sensor'] == 'front_' + side),
   28                           key=lambda row: row['stamp']) for side in ('left', 'right')}
   29     used = set(); rows = []
   30     for left in sides['left'][::stride]:
   31         choices = [(abs(left['stamp'] - r['stamp']), i, r) for i, r in enumerate(sides['right']) if i not in used]
   32         if not choices: continue
   33         skew, index_right, right = min(choices, key=lambda v:v[0])
   34         if skew > .005:
   35             rows.append({'left': left['file'], 'status': 'UNPAIRED', 'skew_s': skew}); continue
   36         used.add(index_right); frames=[]; hashes={}
   37         for side, row in (('left', left), ('right', right)):
   38             path = directory / row['file']
   39             if path.parent != directory or path.suffix != '.png': raise ValueError('invalid image path')
   40             image=cv2.imread(str(path), cv2.IMREAD_COLOR)
   41             if image is None: raise ValueError('image read failed')
   42             hashes[side]=hashlib.sha256(path.read_bytes()).hexdigest()
   43             meta=Metadata(row['stamp'],row['stamp']+.2,0,'retained_'+side,'front_'+side+'_image')
   44             frames.append(ImageFrame(meta,'bgr8',image))
   45         stamp=max(x.metadata.stamp for x in frames)
   46         pair=StereoPair(Metadata(stamp,min(x.metadata.valid_until for x in frames),0,'offline_pair','front_stereo'),
   47                         frames[0],frames[1],skew)
   48         started=time.monotonic(); prepared=prepare_gate_edges(pair) if reuse_edges else None
   49         depth=(gate_region_disparity(pair,prepared=prepared) if depth_method=="roi" else compute_diagnostic_disparity(pair))
   50         # Query at capture time solely to test geometry; do not claim real-time output.
   51         gates=stereo_gate_candidates(pair,depth,rig,stamp,0,prepared=prepared)
   52         rows.append({'left':left['file'],'right':right['file'],'input_sha256':hashes,'skew_s':skew,
   53                      'status':'PROCESSED','gate_count':len(gates),'candidates':[asdict(g) for g in gates],
   54                      'valid_disparity_fraction':float(depth.valid.mean()),'elapsed_s':time.monotonic()-started,
   55                      'query_at_capture_time_not_output_time':True})
   56     return {'source':str(source),'depth_method':depth_method,'reuse_edges':reuse_edges,'role':'offline_nominal_gate_proposal','settings_sha256':rig.settings_sha256,
   57             'index_sha256':hashlib.sha256((directory/'index.json').read_bytes()).hexdigest(),
   58             'control_published':False,'acceptance_eligible':False,'results':rows}

   61 def main():
   62     parser=argparse.ArgumentParser(description=__doc__);parser.add_argument('source');parser.add_argument('output')
   63     parser.add_argument('--settings',default='configs/simulator_settings.json');parser.add_argument('--stride',type=int,default=5)
   64     parser.add_argument('--depth-method',choices=('full','roi'),default='full')
