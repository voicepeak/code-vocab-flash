   18 def replay(source, settings):
   19     source = Path(source)
   20     rig = load_nominal_rig(settings)
   21     manifest = json.loads((source / 'manifest.json').read_text())
   22     if manifest['simulator_preflight']['settings_sha256'] != rig.settings_sha256:
   23         raise ValueError('settings do not match recorded capture')
   24     summary = json.loads((source / 'joint_summary.json').read_text())
   25     states = [json.loads(line) for line in (source / 'joint_states.jsonl').read_text().splitlines()]
   26     history = PoseHistory()
   27     limits = LocalPlannerLimits(SweepBounds((.1, .1, .1), .02, 2.), 1.,
   28                                 segment_durations_s=(.05,), lateral_offsets_m=())
   29     probe = LidarStaticProbe(history, rig.lidar_pose, limits)
   30     index, rows = 0, []
   31     for capture in summary['captures']:
   32         directory = source / capture['directory']
   33         packet = json.loads((directory / 'packet.json').read_text())
   34         now, wall = packet['query_now'], packet['query_wall']
   35         while index < len(states) and states[index]['arrival_wall_monotonic'] <= wall:
   36             record = states[index]
   37             state = state_from_json(record['state'])
   38             if state.metadata.stamp > now:
   39                 break
   40             history.append(state, record['arrival_wall_monotonic'])
   41             index += 1
   42         sources = history.snapshot_sources()
   43         if not sources:
   44             rows.append({'capture': capture['directory'], 'reason': 'NO_SOURCE_STATE'})
   45             continue
   46         state = sources[-1][0]
   47         cloud = decode_xyz(load_message(directory / 'lidar.json'), packet['epoch'])
   48         ref = TrajectorySample(now, state.position_ned, state.velocity_ned, (0., 0., 0.), 0.)
   49         goal = tuple(a + b for a, b in zip(state.position_ned, (.0001, 0., 0.)))
   50         result = probe.evaluate(cloud, state, ref, goal, now, wall, packet['epoch'])
   51         rows.append({'capture': capture['directory'], 'scan_stamp': cloud.metadata.stamp,
   52                      'query_now': now, 'result': asdict(result)})
   53     return {'source': str(source), 'role': 'read_only_diagnostic', 'control_published': False,
   54             'current_control_eligible': False, 'captures': rows}

   57 def main():
   58     parser = argparse.ArgumentParser()

   60     parser.add_argument('--settings', default='configs/simulator_settings.json')
   61     args = parser.parse_args()
