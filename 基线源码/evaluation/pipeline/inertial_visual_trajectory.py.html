   11 def build(estimate_path,telemetry_path):
   12     estimate_path,telemetry_path=Path(estimate_path),Path(telemetry_path)
   13     estimates=json.loads(estimate_path.read_text());groups={}
   14     for line in telemetry_path.read_text().splitlines():
   15         row=json.loads(line);raw=row['raw'];epoch=row['command']['epoch']
   16         groups.setdefault(epoch,{}).setdefault(raw['stamp'],(tuple(raw['acceleration_body']),tuple(raw['imu_orientation_wxyz'])))
   17     groups={epoch:sorted((stamp,a,q) for stamp,(a,q) in group.items()) for epoch,group in groups.items()}
   18     def anchor(samples,stamp):
   19         stamps=[s[0] for s in samples];i=bisect.bisect_right(stamps,stamp)-1
   20         if i<0 or i+1>=len(samples) or samples[i+1][0]-samples[i][0]>.04:
   21             raise ValueError('initial imu orientation not bracketed')
   22         a,b=samples[i],samples[i+1];f=(stamp-a[0])/(b[0]-a[0]);return slerp(a[2],b[2],f)
   23     segments=[];breaks=[];fusion=None
   24     for row in estimates['results']:
   25         if row['status']!='ESTIMATED':
   26             breaks.append({'previous_stamp':row['previous_stamp'],'current_stamp':row['current_stamp'],
   27                            'reason':row.get('reason','MOTION_REJECTED')});fusion=None;continue
   28         m=row['motion'];start,end,epoch=m['previous_stamp'],m['current_stamp'],m['epoch'];samples=groups.get(epoch,[])
   29         try:
   30             if fusion is None:
   31                 fusion=InertialVisualTrajectory(start,epoch,anchor(samples,start))
   32                 segments.append({'anchor_stamp':start,'epoch':epoch,'poses':[]})
   33             if epoch!=fusion.epoch or abs(start-fusion.stamp)>1e-6:raise ValueError('visual interval discontinuity')
   34             pose=fusion.append(end,m['translation_previous_body_m'],samples)
   35             segments[-1]['poses'].append(pose)
   36         except (ValueError,ArithmeticError) as error:
   37             breaks.append({'previous_stamp':start,'current_stamp':end,'reason':str(error)});fusion=None
   38     return {'source':estimates['source'],'segments':segments,'breaks':breaks,'truth_accessed':False,
   39             'control_eligible':False,'metric_calibration_verified':False,'initial_velocity_assumption':'zero',
   40             'position_gain':.5,'velocity_gain':.3,'imu_bias_correction':'none',
   41             'estimate_sha256':hashlib.sha256(estimate_path.read_bytes()).hexdigest(),
   42             'imu_telemetry_sha256':hashlib.sha256(telemetry_path.read_bytes()).hexdigest()}
