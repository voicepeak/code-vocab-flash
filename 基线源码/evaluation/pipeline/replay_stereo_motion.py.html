   19 def read_pair(directory, epoch):
   20     frames = []
   21     for side in ('left', 'right'):
   22         info, payload = read_payload(directory / (side + '.json'))
   23         channels = {'rgb8': 3, 'bgr8': 3, 'rgba8': 4, 'bgra8': 4, 'mono8': 1}.get(info['encoding'])
   24         if channels is None or info['step'] < info['width'] * channels or len(payload) != info['step'] * info['height']:
   25             raise ValueError('unsupported recorded image')
   26         pixels = np.ndarray((info['height'], info['width'], channels), dtype=np.uint8,
   27                             buffer=payload, strides=(info['step'], channels, 1))
   28         stamp = info['stamp']
   29         frames.append(ImageFrame(Metadata(stamp, stamp + .2, epoch, 'retained_' + side,
   30                                           'front_' + side + '_image'), info['encoding'], pixels))
   31     left, right = frames
   32     stamp = max(left.metadata.stamp, right.metadata.stamp)
   33     meta = Metadata(stamp, min(left.metadata.valid_until, right.metadata.valid_until), epoch,
   34                     'retained_stereo', 'front_stereo')
   35     return StereoPair(meta, left, right, abs(left.metadata.stamp - right.metadata.stamp))

   38 def replay(source, settings):
   39     import cv2
   40     cv2.setNumThreads(1)
   41     source = Path(source)
   42     rig = load_nominal_rig(settings)
   43     manifest = json.loads((source / 'manifest.json').read_text())
   44     if manifest['simulator_preflight']['settings_sha256'] != rig.settings_sha256:
   45         raise ValueError('settings do not match recorded capture')
   46     captures = json.loads((source / 'joint_summary.json').read_text())['captures']
   47     frames = []
   48     for capture in captures:
   49         directory = source / capture['directory']
   50         packet = json.loads((directory / 'packet.json').read_text())
   51         frames.append((capture['directory'], read_pair(directory, packet['epoch']), packet['poses']['left']))
   52     results = []
   53     for (old_dir, old, old_pose), (new_dir, new, new_pose) in zip(frames, frames[1:]):
   54         start = time.monotonic()
   55         record = {'previous': old_dir, 'current': new_dir, 'gap_s': new.metadata.stamp - old.metadata.stamp}
   56         try:
   57             cv2.setRNGSeed(0)
   58             motion = estimate_stereo_motion(old, new, rig)
   59             elapsed = time.monotonic() - start
   60             prev_rotation = rotation(old_pose['orientation_wxyz'])
   61             next_rotation = rotation(new_pose['orientation_wxyz'])
   62             expected = prev_rotation.T @ (np.asarray(new_pose['position_ned']) - old_pose['position_ned'])
   63             observed_rotation = np.asarray(motion.rotation_previous_from_current_body)
   64             difference = observed_rotation.T @ prev_rotation.T @ next_rotation
   65             angle = float(np.arccos(np.clip((np.trace(difference) - 1.) / 2., -1., 1.)))
   66             record.update(status='ESTIMATED', result=asdict(motion), elapsed_s=elapsed,
   67                           eskf_delta_previous_body_m=expected.tolist(),
   68                           translation_difference_m=float(np.linalg.norm(np.asarray(motion.translation_previous_body_m) - expected)),
   69                           rotation_difference_rad=angle)
   70         except (ValueError, cv2.error) as error:
   71             record.update(status='REJECTED', reason=str(error), elapsed_s=time.monotonic() - start)
   72         results.append(record)
   73     return {'source': str(source), 'role': 'offline_diagnostic', 'comparison': 'imu_gps_eskf_not_truth',
   74             'control_eligible': False, 'counts': dict(Counter(row['status'] for row in results)), 'results': results}
