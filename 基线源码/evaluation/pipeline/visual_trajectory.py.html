   16 def build(estimates):
   17     segments = []; breaks = []; chain = None
   18     for row in estimates['results']:
   19         if row['status'] != 'ESTIMATED':
   20             breaks.append({'previous_stamp':row['previous_stamp'], 'current_stamp':row['current_stamp'],
   21                            'reason':row.get('reason', 'MOTION_REJECTED')})
   22             if chain is not None: chain.invalidate()
   23             chain = None
   24             continue
   25         motion = SimpleNamespace(**row['motion'])
   26         if chain is None:
   27             chain = VisualTrajectory(motion.previous_stamp, motion.epoch)
   28             segments.append({'anchor_stamp': motion.previous_stamp, 'epoch': motion.epoch, 'poses': []})
   29         try:
   30             pose = chain.append(motion)
   31         except ValueError as error:
   32             breaks.append({'previous_stamp':motion.previous_stamp, 'current_stamp':motion.current_stamp,
   33                            'reason':str(error)})
   34             chain = None
   35             continue
   36         segments[-1]['poses'].append(asdict(pose))
   37     return {'source': estimates['source'], 'segments': segments, 'breaks': breaks, 'truth_accessed': False,
   38             'control_eligible': False, 'metric_calibration_verified': False}

   41 def evaluate(estimate, truth_path):
   42     path = Path(truth_path)
   43     records = [json.loads(line) for line in path.read_text().splitlines()]
   44     truth = sorted({r['stamp']: r for r in records if r['stamp'] > 0}.values(), key=lambda r: r['stamp'])
   45     stamps = [r['stamp'] for r in truth]
   46     def interpolate(stamp):
   47         i = bisect.bisect_left(stamps, stamp)
   48         if i == 0 or i == len(stamps): return None
   49         a, b = truth[i-1], truth[i]; span = b['stamp'] - a['stamp']
   50         if span <= 0 or span > .05: return None
   51         f = (stamp-a['stamp'])/span
   52         return (1-f)*np.asarray(a['position'])+f*np.asarray(b['position']), rotation(slerp(a['orientation'],b['orientation'],f))
   53     results = []
   54     for segment in estimate['segments']:
   55         anchor = interpolate(segment['anchor_stamp']); comparisons = []
   56         for pose in segment['poses']:
   57             current = interpolate(pose['stamp'])
   58             if anchor is None or current is None: continue
   59             relative = anchor[1].T @ (current[0]-anchor[0])
   60             error = float(np.linalg.norm(np.asarray(pose['position_initial_body_m'])-relative))
   61             comparisons.append({'stamp':pose['stamp'],'elapsed_s':pose['stamp']-segment['anchor_stamp'],
   62                                 'translation_error_m':error})
   63         results.append({'anchor_stamp':segment['anchor_stamp'],'steps':len(segment['poses']), 'comparisons':comparisons})
   64     return {'source':estimate['source'], 'truth_sha256':hashlib.sha256(path.read_bytes()).hexdigest(),
   65             'role':'isolated_post_estimate_evaluation', 'segments':results, 'control_eligible':False}

   68 def main():

   70     b = sub.add_parser('build'); b.add_argument('estimate'); b.add_argument('output')
   71     e = sub.add_parser('evaluate'); e.add_argument('estimate'); e.add_argument('truth'); e.add_argument('output')
