   21 def replay(source,settings,limit=24,center_mode="bbox",association_mode="strict"):
   22     source=Path(source);directory=source/'scene_trace';index_path=directory/'index.json'
   23     index=json.loads(index_path.read_text());rig=load_nominal_rig(settings)
   24     manifest=json.loads((source/'manifest.json').read_text())
   25     if index['fault'] is not None or manifest['simulator_preflight']['settings_sha256']!=rig.settings_sha256:
   26         raise ValueError('capture or simulator settings mismatch')
   27     log=source/('pwm_telemetry.jsonl' if (source/'pwm_telemetry.jsonl').exists() else 'vel_telemetry.jsonl')
   28     states=[json.loads(s) for s in log.read_text().splitlines()]
   29     if any(a['raw']['stamp']>=b['raw']['stamp'] for a,b in zip(states,states[1:])):
   30         raise ValueError('nonmonotonic state recording')
   31     scans=sorted((r for r in index['frames'] if r['sensor']=='lidar'),key=lambda r:r['stamp'])
   32     tracker=DynamicTracker(association_mode=association_mode);rows=[];processed=0
   33     for scan in scans:
   34         if processed>=limit:break
   35         stamp=scan['stamp']; pose,reason=interpolate_pose(states,stamp)
   36         if pose is None:
   37             rows.append({'file':scan['file'],'status':reason});continue
   38         path=directory/scan['file']
   39         if path.parent!=directory or path.suffix!='.npz':raise ValueError('invalid point file')
   40         with np.load(path,allow_pickle=False) as payload:points=np.asarray(payload['points'],dtype=float)
   41         cloud=LidarCloud(Metadata(stamp,stamp+.2,0,'recorded_lidar','lidar'),points,len(points),0)
   42         now=pose.bracket_stamps[1];started=time.monotonic()
   43         try:
   44             world=world_endpoints(cloud,pose,rig.lidar_pose);clusters=cluster_endpoints(world,now,0,center_mode=center_mode)
   45             association=tracker.ingest(clusters,now,0);tracks=tracker.snapshot(now,0)
   46             rows.append({'file':scan['file'],'stamp':stamp,'status':'PROCESSED','input_sha256':hashlib.sha256(path.read_bytes()).hexdigest(),
   47                          'now':now,'duration_s':time.monotonic()-started,'clusters':len(clusters.clusters),
   48                          'occupied_voxels':clusters.occupied_voxels,'rejected_components':clusters.rejected_components,
   49                          'association':association,'tracks':[asdict(t) for t in tracks]})
   50         except (ValueError,ArithmeticError) as error:
   51             rows.append({'file':scan['file'],'stamp':stamp,'status':'REJECTED','reason':str(error)})
   52         processed+=1
   53     return {'source':str(source),'center_mode':center_mode,'association_mode':association_mode,'settings_sha256':rig.settings_sha256,
   54             'scene_index_sha256':hashlib.sha256(index_path.read_bytes()).hexdigest(),
   55             'state_input_sha256':hashlib.sha256(log.read_bytes()).hexdigest(),'statuses':dict(Counter(x['status'] for x in rows)),
   56             'uses_truth':False,'now_is_earliest_bracket_not_processing_completion':True,
   57             'control_eligible':False,'results':rows}
