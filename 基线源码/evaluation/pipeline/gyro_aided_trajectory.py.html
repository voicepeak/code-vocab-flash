   13 def build(estimate_path, telemetry_path):
   14     estimate_path,telemetry_path=Path(estimate_path),Path(telemetry_path)
   15     estimates=json.loads(estimate_path.read_text());groups={}
   16     for line in telemetry_path.read_text().splitlines():
   17         row=json.loads(line);raw=row['raw'];epoch=row['command']['epoch']
   18         groups.setdefault(epoch,{}).setdefault(raw['stamp'],tuple(raw['omega_body']))
   19     groups={e:sorted(group.items()) for e,group in groups.items()}
   20     segments=[];breaks=[];chain=None
   21     for row in estimates['results']:
   22         if row['status']!='ESTIMATED':
   23             breaks.append({'previous_stamp':row['previous_stamp'],'current_stamp':row['current_stamp'],
   24                            'reason':row.get('reason','MOTION_REJECTED')})
   25             chain=None;continue
   26         motion=row['motion'];previous,current,epoch=motion['previous_stamp'],motion['current_stamp'],motion['epoch']
   27         try:
   28             r=gyro_rotation(groups.get(epoch,[]),previous,current)
   29             if chain is None:
   30                 chain=VisualTrajectory(previous,epoch)
   31                 segments.append({'anchor_stamp':previous,'epoch':epoch,'poses':[]})
   32             step=SimpleNamespace(previous_stamp=previous,current_stamp=current,epoch=epoch,
   33                 translation_previous_body_m=motion['translation_previous_body_m'],
   34                 rotation_previous_from_current_body=r)
   35             pose=chain.append(step);segments[-1]['poses'].append(asdict(pose))
   36         except ValueError as error:
   37             breaks.append({'previous_stamp':previous,'current_stamp':current,'reason':str(error)})
   38             chain=None
   39     return {'source':estimates['source'],'segments':segments,'breaks':breaks,'truth_accessed':False,
   40             'control_eligible':False,'metric_calibration_verified':False,'gyro_bias_correction':'none',
   41             'estimate_sha256':hashlib.sha256(estimate_path.read_bytes()).hexdigest(),
   42             'imu_telemetry_sha256':hashlib.sha256(telemetry_path.read_bytes()).hexdigest()}
