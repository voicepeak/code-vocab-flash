   47 def sha256(path):
   48     h = hashlib.sha256()
   49     with Path(path).open("rb") as f:
   50         for block in iter(lambda: f.read(1024 * 1024), b""):
   51             h.update(block)
   52     return h.hexdigest()

   55 def contains_secret(path):
   56     tail = b""
   57     with path.open("rb") as f:
   58         for block in iter(lambda: f.read(1024 * 1024), b""):
   59             combined = tail + block
   60             if SECRET.search(combined):
   61                 return True
   62             tail = combined[-4096:]
   63     return False

   66 def scan(root):
   67     root = Path(root).resolve()
   68     accepted, excluded = [], []
   69     reasons = Counter()
   70     last = time.monotonic()
   71     for top in sorted(ROOTS):
   72         base = root / top
   73         if not base.is_dir():
   74             continue
   75         for directory, dirs, files in os.walk(base, followlinks=False):
   76             current = Path(directory)
   77             kept = []
   78             for name in sorted(dirs):
   79                 p = current / name
   80                 if p.is_symlink() or name.lower() in SKIP_DIRS:
   81                     reasons["excluded_directory"] += 1
   82                 else:
   83                     kept.append(name)
   84             dirs[:] = kept
   85             for name in sorted(files):
   86                 p = current / name
   87                 rel = p.relative_to(root).as_posix()
   88                 reason = None
   89                 if p.is_symlink():
   90                     reason = "symlink"
   91                 elif p.suffix.lower() not in SUFFIXES:
   92                     reason = "not_observation_or_metadata"
   93                 elif name.startswith(".") or any(v in name.lower() for v in (
   94                         "credential", "private_key", "id_rsa", "id_ed25519", ".env")):
   95                     reason = "credential_filename"
   96                 elif p.suffix.lower() in TEXT and contains_secret(p):
   97                     reason = "possible_credential_content"
   98                 if reason:
   99                     reasons[reason] += 1
  100                     excluded.append({"path": rel, "reason": reason})
  101                     continue
  102                 info = p.stat()
  103                 accepted.append({"path": rel, "size": info.st_size, "mtime_ns": info.st_mtime_ns})
  104             if time.monotonic() - last > 20:
  105                 print("Scanned eligible files:", len(accepted), flush=True)
  106                 last = time.monotonic()
  107     return sorted(accepted, key=lambda r: r["path"]), excluded, dict(reasons)

  110 def prepare(root, output, target):
  111     output.mkdir(parents=True, exist_ok=True)
  112     plan_path = output / "plan.json"
  113     if plan_path.exists():
  114         raise FileExistsError("Existing plan: use pack/upload to resume, or choose a new output")
  115     entries, excluded, reasons = scan(root)
  116     if not entries:
  117         raise ValueError("No eligible observations")
  118     shards, group, size = [], [], 0
  119     for entry in entries:
  120         # Estimate tar header, PAX header, padded payload; bounded near target.
  121         estimate = ((entry["size"] + 511) // 512 + 4) * 512
  122         if group and size + estimate > target:
  123             shards.append({"name": "data-%05d.tar" % len(shards), "entries": group})
  124             group, size = [], 0
  125         group.append(entry)
  126         size += estimate
  127     if group:
  128         shards.append({"name": "data-%05d.tar" % len(shards), "entries": group})
  129     plan = {"schema_version": 1, "source_root": str(root.resolve()),
  130             "file_count": len(entries), "source_bytes": sum(e["size"] for e in entries),
  131             "target_shard_bytes": target, "excluded_counts": reasons, "shards": shards}
  132     write_json(plan_path, plan)
  133     write_json(output / "exclusions.local.json", excluded)
  134     public = output / "public"
  135     public.mkdir(exist_ok=True)
  136     write_json(public / "dataset_summary.json", {
  137         k: v for k, v in plan.items() if k not in {"source_root", "shards"}})
  138     write_json(public / "shard_plan.json", [
  139         {"name": s["name"], "file_count": len(s["entries"]),
  140          "source_bytes": sum(e["size"] for e in s["entries"])} for s in shards])
  141     (public / "README.md").write_text("""---

  193 class HashingReader:
  194     def __init__(self, source):
  195         self.source, self.digest = source, hashlib.sha256()
  196 
  197     def read(self, size=-1):
  198         value = self.source.read(size)
  199         self.digest.update(value)
  200         return value

  203 def pack(plan, output, shard):
  204     staging = output / "staging"
  205     staging.mkdir(exist_ok=True)
  206     target = staging / shard["name"]
  207     index = staging / (shard["name"] + ".files.jsonl.gz")
  208     receipt = staging / (shard["name"] + ".receipt.json")
  209     if target.exists() and index.exists() and receipt.exists():
  210         info = json.loads(receipt.read_text())
  211         if sha256(target) == info["sha256"] and sha256(index) == info["index_sha256"]:
  212             return target, index, info
  213         raise ValueError("Existing staging shard does not match its receipt")
  214     required = sum(e["size"] + 2048 for e in shard["entries"])
  215     if shutil.disk_usage(output).free < required + 1024 ** 3:
  216         raise RuntimeError("Insufficient free disk space for staging one shard")
  217     root = Path(plan["source_root"])
  218     tmp = target.with_suffix(".tar.tmp")
  219     index_tmp = index.with_suffix(".gz.tmp")
  220     with tarfile.open(tmp, "w", format=tarfile.PAX_FORMAT) as tar, \
  221             index_tmp.open("wb") as raw, gzip.GzipFile(filename="", mode="wb", fileobj=raw, mtime=0) as z:
  222         for entry in shard["entries"]:
  223             rel = Path(entry["path"])
  224             if rel.is_absolute() or ".." in rel.parts:
  225                 raise ValueError("Invalid archive member path")
  226             source = root / rel
  227             before = source.stat()
  228             if source.is_symlink() or (before.st_size, before.st_mtime_ns) != (entry["size"], entry["mtime_ns"]):
  229                 raise RuntimeError("Source changed since review: " + str(rel))
  230             info = tarfile.TarInfo(entry["path"])
  231             info.size, info.mode, info.mtime = entry["size"], 0o644, 0
  232             with source.open("rb") as handle:
  233                 reader = HashingReader(handle)
  234                 tar.addfile(info, reader)
  235             after = source.stat()
  236             if (after.st_size, after.st_mtime_ns) != (before.st_size, before.st_mtime_ns):
  237                 raise RuntimeError("Source changed during packing: " + str(rel))
  238             z.write((json.dumps({"path": entry["path"], "size": entry["size"],
  239                                 "sha256": reader.digest.hexdigest()}) + "\n").encode())
  240     tmp.replace(target)
  241     index_tmp.replace(index)
  242     result = {"name": shard["name"], "size": target.stat().st_size,
  243               "sha256": sha256(target), "file_count": len(shard["entries"]),
  244               "index_size": index.stat().st_size,
  245               "index_sha256": sha256(index)}
  246     write_json(receipt, result)
  247     return target, index, result

  250 def verify_remote(api, repo_id, path, size, digest, revision="main"):
  251     from huggingface_hub import hf_hub_url
  252     rows = api.get_paths_info(repo_id, [path], repo_type="dataset", revision=revision)
  253     if len(rows) != 1 or rows[0].size != size:
  254         raise RuntimeError("Remote size mismatch: " + path)
  255     lfs = rows[0].lfs
  256     if lfs is not None:
  257         if lfs.sha256 != digest:
  258             raise RuntimeError("Remote SHA256 mismatch: " + path)
  259         return
  260     # Small files may be regular Git blobs. Read them back to verify their content.
  261     import urllib.request
  262     h = hashlib.sha256()
  263     url = hf_hub_url(repo_id, path, repo_type="dataset", revision=revision)
  264     with urllib.request.urlopen(url, timeout=60) as response:
  265         for block in iter(lambda: response.read(1024 * 1024), b""):
  266             h.update(block)
  267     if h.hexdigest() != digest:
  268         raise RuntimeError("Remote content mismatch: " + path)

  271 def publish(output, plan, repo_id):
  272     from huggingface_hub import CommitOperationAdd, HfApi, get_token
  273     if not get_token():
  274         raise RuntimeError("Hugging Face login required; run hf auth login locally")
  275     api = HfApi()
  276     who = api.whoami()
  277     owner = repo_id.split("/")[0]
  278     if owner.lower() != who["name"].lower():
  279         raise RuntimeError("Logged-in account does not match requested owner: " + who["name"])
  280     # Use authenticated canonical casing; never change another existing private repo.
  281     repo_id = who["name"] + "/" + repo_id.split("/", 1)[1]
  282     api.create_repo(repo_id, repo_type="dataset", private=False, exist_ok=True)
  283     if api.repo_info(repo_id, repo_type="dataset").private:
  284         raise RuntimeError("Existing repository is private; refusing to change its visibility implicitly")
  285     public = output / "public"
  286     state_file = output / "upload_state.local.json"
  287     state = json.loads(state_file.read_text()) if state_file.exists() else {"repo_id": repo_id, "verified": {}}
  288     if state["repo_id"] != repo_id:
  289         raise RuntimeError("Existing upload state belongs to another repository")
  290     write_json(public / "shards.json", list(state["verified"].values()))
  291     write_json(public / "publication_status.json", {"state": "UPLOADING", "shards_expected": len(plan["shards"])})
  292     api.upload_folder(repo_id=repo_id, repo_type="dataset", folder_path=public,
  293                       commit_message="Document public legacy simulation dataset and shard plan")
  294     for shard in plan["shards"]:
  295         name = shard["name"]
  296         if name in state["verified"]:
  297             existing = state["verified"][name]
  298             verify_remote(api, repo_id, "data/" + name, existing["size"], existing["sha256"])
  299             verify_remote(api, repo_id, "indices/" + name + ".files.jsonl.gz",
  300                           existing["index_size"], existing["index_sha256"])
  301             print("Already verified:", name, flush=True)
  302             continue
  303         target, index, receipt = pack(plan, output, shard)
  304         commit = api.create_commit(repo_id, repo_type="dataset", operations=[
  305             CommitOperationAdd(path_in_repo="data/" + name, path_or_fileobj=target),
  306             CommitOperationAdd(path_in_repo="indices/" + index.name, path_or_fileobj=index),
  307         ], commit_message="Add observation shard " + name)
  308         verify_remote(api, repo_id, "data/" + name, receipt["size"], receipt["sha256"], commit.oid)
  309         verify_remote(api, repo_id, "indices/" + index.name, index.stat().st_size, receipt["index_sha256"], commit.oid)
  310         receipt["revision"] = commit.oid
  311         state["verified"][name] = receipt
  312         write_json(state_file, state)
  313         write_json(public / "shards.json", list(state["verified"].values()))
  314         api.upload_file(repo_id=repo_id, repo_type="dataset", path_or_fileobj=public / "shards.json",
  315                         path_in_repo="shards.json", commit_message="Record verified shard " + name)
  316         # Only disposable staging copies are removed. Original archived data is untouched.
  317         target.unlink()
  318         index.unlink()
  319         print("Verified remote shard:", name, flush=True)
  320     api.upload_file(repo_id=repo_id, repo_type="dataset", path_or_fileobj=public / "shards.json",
  321                     path_in_repo="shards.json", commit_message="Finalize verified shard index")
  322     write_json(public / "publication_status.json", {"state": "COMPLETE", "shards_expected": len(plan["shards"]),
  323                                                    "shards_verified": len(state["verified"])})
  324     api.upload_file(repo_id=repo_id, repo_type="dataset", path_or_fileobj=public / "publication_status.json",
  325                     path_in_repo="publication_status.json", commit_message="Complete verified archive publication")
  326     print("Published:", "https://huggingface.co/datasets/" + repo_id, flush=True)

  329 def main():
