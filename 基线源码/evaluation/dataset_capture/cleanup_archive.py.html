   23 class PublicArchive:

   36     def read(self, path):

   42     def metadata(self, paths):
   43         rows = self.api.get_paths_info(self.repo_id, paths, repo_type="dataset", revision=self.revision)

   48 def verify_publication(plan, state, remote, output, shard_names=None):
   49     expected = {s["name"] for s in plan["shards"]}
   50     partial = shard_names is not None
   51     selected = set(shard_names) if partial else expected
   52     if not selected or not selected <= expected or not selected <= set(state["verified"]):
   53         raise ValueError("Requested shards are not locally verified")
   54     if len(expected) != len(plan["shards"]) or (not partial and set(state["verified"]) != expected):
   55         raise ValueError("Local upload receipts are incomplete")
   56     status = json.loads(remote.read("publication_status.json"))
   57     if not partial and (status.get("state") != "COMPLETE" or status.get("shards_expected") != len(expected)
   58                         or status.get("shards_verified") != len(expected)):
   59         raise PublicationPending("Remote publication is not COMPLETE")
   60     summary = json.loads(remote.read("dataset_summary.json"))
   61     for key in ("file_count", "source_bytes"):
   62         if summary.get(key) != plan[key]:
   63             raise ValueError("Remote summary does not match reviewed plan")
   64     records = json.loads(remote.read("shards.json"))
   65     by_name = {r["name"]: r for r in records}
   66     if len(by_name) != len(records) or not set(by_name) <= expected:
   67         raise ValueError("Remote shard index is incomplete")
   68     if not selected <= set(by_name):
   69         raise PublicationPending("Remote shard index has not committed requested receipts yet")
   70     paths = []
   71     for name in sorted(selected):
   72         paths.extend(("data/" + name, "indices/" + name + ".files.jsonl.gz"))
   73     metadata = remote.metadata(paths)
   74     indices = output / "cleanup_indices"
   75     indices.mkdir(exist_ok=True)
   76     for shard in plan["shards"]:
   77         name = shard["name"]
   78         if name not in selected:
   79             continue
   80         receipt, published = state["verified"][name], by_name[name]
   81         for key in ("sha256", "size", "index_sha256", "index_size", "file_count"):
   82             if receipt[key] != published[key]:
   83                 raise ValueError("Published receipt differs from local receipt: " + name)
   84         for path, size, digest in (
   85                 ("data/" + name, receipt["size"], receipt["sha256"]),
   86                 ("indices/" + name + ".files.jsonl.gz", receipt["index_size"], receipt["index_sha256"])):
   87             info = metadata.get(path)
   88             if info is None or info["size"] != size:
   89                 raise ValueError("Remote artifact missing or size mismatch: " + path)
   90             if info["sha256"] is not None and info["sha256"] != digest:
   91                 raise ValueError("Remote artifact hash mismatch: " + path)
   92             if path.startswith("data/") and info["sha256"] is None:
   93                 raise ValueError("Missing content-addressed hash for remote data shard")
   94         data = remote.read("indices/" + name + ".files.jsonl.gz")
   95         if hashlib.sha256(data).hexdigest() != receipt["index_sha256"] or len(data) != receipt["index_size"]:
   96             raise ValueError("Downloaded index checksum mismatch: " + name)
   97         rows = (json.loads(line) for line in gzip.decompress(data).decode().splitlines())
   98         for entry, row in itertools.zip_longest(shard["entries"], rows):
   99             if (entry is None or row is None or entry["path"] != row["path"]
  100                     or entry["size"] != row["size"] or len(row["sha256"]) != 64):
  101                 raise ValueError("Per-file index differs from reviewed plan: " + name)
  102         (indices / (name + ".files.jsonl.gz")).write_bytes(data)
  103     return {"repo_id": state["repo_id"], "revision": remote.revision,
  104             "shards_verified": len(selected),
  105             "file_count": sum(len(s["entries"]) for s in plan["shards"] if s["name"] in selected)}

  108 def parent_fd(root_fd, relative):
  109     path = Path(relative)
  110     if path.is_absolute() or not path.parts or any(p in (".", "..") for p in path.parts):
  111         raise ValueError("Unsafe archive member path")
  112     descriptor = os.dup(root_fd)
  113     try:
  114         for name in path.parts[:-1]:
  115             next_fd = os.open(name, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=descriptor)
  116             os.close(descriptor)
  117             descriptor = next_fd
  118         return descriptor, path.name
  119     except BaseException:
  120         os.close(descriptor)
  121         raise

  128 def delete_verified_members(plan, output, progress=lambda **kwargs: None):
  129     root = Path(plan["source_root"])
  130     root_fd = os.open(root, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
  131     report = {"deleted_files": 0, "deleted_bytes": 0, "preserved_changed_files": 0,
  132               "already_missing_files": 0, "removed_empty_directories": 0,
  133               "disk_free_bytes_before": shutil.disk_usage(root).free}
  134     empty_candidates = set()
  135     last_update = time.monotonic()
  136     try:
  137         # All remote artifacts for this selection have already been verified.
  138         # Hash all members of the selected shard before deleting its first member.
  139         with tempfile.TemporaryFile(mode="w+t", dir=output) as candidates, \
  140                 (output / "cleanup_preserved.local.jsonl").open("a") as preserved:
  141             inspected = 0
  142             for shard in plan["shards"]:
  143                 index = output / "cleanup_indices" / (shard["name"] + ".files.jsonl.gz")
  144                 with gzip.open(index, "rt") as stream:
  145                     for line in stream:
  146                         row = json.loads(line)
  147                         directory = None
  148                         try:
  149                             directory, name = parent_fd(root_fd, row["path"])
  150                             fd = os.open(name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=directory)
  151                             with os.fdopen(fd, "rb") as source:
  152                                 before = os.fstat(source.fileno())
  153                                 if not stat.S_ISREG(before.st_mode):
  154                                     raise ValueError("Not a regular file")
  155                                 digest = hashlib.sha256()
  156                                 for block in iter(lambda: source.read(1024 * 1024), b""):
  157                                     digest.update(block)
  158                                 after = os.fstat(source.fileno())
  159                             current = os.stat(name, dir_fd=directory, follow_symlinks=False)
  160                             if (fingerprint(before) != fingerprint(after) or fingerprint(after) != fingerprint(current)
  161                                     or after.st_size != row["size"] or digest.hexdigest() != row["sha256"]):
  162                                 raise ValueError("Local data changed after backup")
  163                             candidates.write(json.dumps({"path": row["path"], "fingerprint": fingerprint(after)}) + "\n")
  164                         except FileNotFoundError:
  165                             report["already_missing_files"] += 1
  166                         except (OSError, ValueError) as error:
  167                             report["preserved_changed_files"] += 1
  168                             preserved.write(json.dumps({"path": row["path"], "reason": str(error)}) + "\n")
  169                         finally:
  170                             if directory is not None:
  171                                 os.close(directory)
  172                         inspected += 1
  173                         if time.monotonic() - last_update > 20:
  174                             progress(phase="HASHING_LOCAL", inspected_files=inspected, **report)
  175                             last_update = time.monotonic()
  176             if inspected != plan["file_count"]:
  177                 raise ValueError("Local candidate count mismatch; nothing deleted")
  178             candidates.flush()
  179             candidates.seek(0)
  180             progress(phase="DELETING_VERIFIED_MEMBERS", inspected_files=inspected, **report)
  181             for line in candidates:
  182                 candidate = json.loads(line)
  183                 directory = None
  184                 try:
  185                     directory, name = parent_fd(root_fd, candidate["path"])
  186                     current = os.stat(name, dir_fd=directory, follow_symlinks=False)
  187                     if fingerprint(current) != candidate["fingerprint"]:
  188                         raise ValueError("File changed between verification and removal")
  189                     os.unlink(name, dir_fd=directory)
  190                     report["deleted_files"] += 1
  191                     report["deleted_bytes"] += current.st_size
  192                     parts = Path(candidate["path"]).parts
  193                     empty_candidates.update("/".join(parts[:i]) for i in range(1, len(parts)))
  194                 except FileNotFoundError:
  195                     report["already_missing_files"] += 1
  196                 except (OSError, ValueError) as error:
  197                     report["preserved_changed_files"] += 1
  198                     preserved.write(json.dumps({"path": candidate["path"], "reason": str(error)}) + "\n")
  199                 finally:
  200                     if directory is not None:
  201                         os.close(directory)
  202                 if time.monotonic() - last_update > 20:
  203                     progress(phase="DELETING_VERIFIED_MEMBERS", **report)
  204                     last_update = time.monotonic()
  205         for relative in sorted(empty_candidates, key=lambda p: len(Path(p).parts), reverse=True):
  206             directory = None
  207             try:
  208                 directory, name = parent_fd(root_fd, relative)
  209                 os.rmdir(name, dir_fd=directory)
  210                 report["removed_empty_directories"] += 1
  211             except OSError:
  212                 pass  # Nonempty, changed or already missing directories stay untouched.
  213             finally:
  214                 if directory is not None:
  215                     os.close(directory)
  216     finally:
  217         os.close(root_fd)
  218     report["disk_free_bytes_after"] = shutil.disk_usage(root).free
  219     return report

  222 def incremental_cleanup(plan, output, update, plan_digest):
  223     journal_path = output / "per_shard_cleanup_state.local.json"
  224     journal = json.loads(journal_path.read_text()) if journal_path.exists() else {
  225         "plan_sha256": plan_digest, "cleaned": {}}
  226     if journal["plan_sha256"] != plan_digest:
  227         raise ValueError("Cleanup journal belongs to another reviewed plan")
  228     shards = {s["name"]: s for s in plan["shards"]}
  229 
  230     def totals():
  231         keys = ("deleted_files", "deleted_bytes", "preserved_changed_files", "already_missing_files")
  232         records = list(journal["cleaned"].values()) + list(journal.get("retry_previous", {}).values())
  233         return {key: sum(r.get(key, 0) for r in records) for key in keys}
  234 
  235     while len(journal["cleaned"]) < len(shards):
  236         if sha256(output / "plan.json") != plan_digest:
  237             raise ValueError("Reviewed upload plan changed")
  238         state_path = output / "upload_state.local.json"
  239         state = json.loads(state_path.read_text()) if state_path.exists() else {"verified": {}}
  240         pending = sorted(set(state["verified"]) - set(journal["cleaned"]))
  241         if not pending:
  242             update(state="WAITING_FOR_NEXT_SHARD", error=None, cleaned_shards=len(journal["cleaned"]), **totals())
  243             time.sleep(10)
  244             continue
  245         for name in pending:
  246             update(state="VERIFYING_SHARD", current_shard=name, cleaned_shards=len(journal["cleaned"]), **totals())
  247             for attempt in range(1, 13):
  248                 try:
  249                     remote = PublicArchive(state["repo_id"])
  250                     verified = verify_publication(plan, state, remote, output, {name})
  251                     break
  252                 except PublicationPending:
  253                     if attempt == 12:
  254                         raise
  255                     time.sleep(5)
  256                 except ValueError:
  257                     raise
  258                 except Exception as error:
  259                     if attempt == 12:
  260                         raise
  261                     update(state="RETRYING_SHARD_VERIFICATION", error=str(error), current_shard=name)
  262                     time.sleep(10)
  263             subset = {"source_root": plan["source_root"], "file_count": len(shards[name]["entries"]),
  264                       "shards": [shards[name]]}
  265             base = totals()
  266 
  267             def progress(**values):
  268                 for key in base:
  269                     if key in values:
  270                         values[key] += base[key]
  271                 update(state="CLEANING_SHARD", current_shard=name, cleaned_shards=len(journal["cleaned"]), **values)
  272 
  273             result = delete_verified_members(subset, output, progress)
  274             previous = journal.get("retry_previous", {}).pop(name, None)
  275             if previous:
  276                 result["already_missing_files"] = max(previous.get("already_missing_files", 0),
  277                     result["already_missing_files"] - previous.get("deleted_files", 0))
  278                 result["deleted_files"] += previous.get("deleted_files", 0)
  279                 result["deleted_bytes"] += previous.get("deleted_bytes", 0)
  280             result["remote_revision"] = verified["revision"]
  281             journal["cleaned"][name] = result
  282             write_json(journal_path, journal)
  283             update(state="SHARD_CLEANED", error=None, current_shard=name, cleaned_shards=len(journal["cleaned"]), **totals())
  284             print(json.dumps({"shard_cleaned": name, "deleted_bytes": result["deleted_bytes"],
  285                               "cleaned_shards": len(journal["cleaned"])}), flush=True)
  286     update(state="COMPLETE", cleaned_shards=len(journal["cleaned"]), **totals())

  289 def main():
  290     parser = argparse.ArgumentParser(description=__doc__)
  291     parser.add_argument("--output", type=Path, default=Path("/home/tianbot/RUMA-hf-public-20261019"))
  292     parser.add_argument("--authorize-cleanup", action="store_true", required=True)
  293     parser.add_argument("--per-shard", action="store_true",
  294                         help="remove each verified shard immediately while the rest uploads")
  295     args = parser.parse_args()
  296     output = args.output.resolve()
  297     with (output / "cleanup_service.lock").open("a") as lock:
  298         fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
  299         plan_path = output / "plan.json"
  300         plan_digest = sha256(plan_path)
  301         plan = json.loads(plan_path.read_text())
  302         source = Path(plan["source_root"])
  303         if (source != Path("/home/tianbot/RUMA-legacy-archive-20261019") or source.is_symlink()
  304                 or source.resolve() == output or source in output.parents):
  305             raise ValueError("Refusing cleanup outside the explicitly authorized historical archive")
  306         status_path = output / "local_cleanup_status.local.json"
  307         status = {"pid": os.getpid(), "source_root": str(source), "plan_sha256": plan_digest,
  308                   "authorized": True, "per_shard": args.per_shard,
  309                   "state": "WAITING_FOR_UPLOAD", "updated_at_unix": time.time()}
  310 
  311         def update(**values):
  312             status.update(values, updated_at_unix=time.time())
  313             write_json(status_path, status)
  314 
  315         update()
  316         if args.per_shard:
  317             try:
  318                 incremental_cleanup(plan, output, update, plan_digest)
  319             except Exception as error:
  320                 update(state="FAILED", error=str(error))
  321                 raise
  322             return
  323         while True:
  324             if sha256(plan_path) != plan_digest:
  325                 update(state="FAILED", error="Reviewed upload plan changed; no cleanup attempted")
  326                 return
  327             state_path = output / "upload_state.local.json"
  328             completion = json.loads((output / "public/publication_status.json").read_text())
  329             state = json.loads(state_path.read_text()) if state_path.exists() else None
  330             if completion.get("state") == "COMPLETE" and state and len(state["verified"]) == len(plan["shards"]):
  331                 break
  332             time.sleep(30)
  333         try:
  334             update(state="VERIFYING_REMOTE")
  335             for attempt in range(1, 13):
  336                 try:
  337                     remote = PublicArchive(state["repo_id"])
  338                     verified = verify_publication(plan, state, remote, output)
  339                     break
  340                 except PublicationPending:
  341                     if attempt == 12:
  342                         raise
  343                     update(state="WAITING_FOR_REMOTE_COMPLETION", verification_attempt=attempt)
  344                     time.sleep(30)
  345                 except ValueError:
  346                     raise  # An integrity mismatch must never authorize deletion.
  347                 except Exception as error:
  348                     if attempt == 12:
  349                         raise
  350                     update(state="RETRYING_REMOTE_VERIFICATION", error=str(error), verification_attempt=attempt)
  351                     time.sleep(30)
  352             write_json(output / "cleanup_remote_verification.local.json", verified)
  353             result = delete_verified_members(plan, output, lambda **values: update(state="VERIFYING_AND_CLEANING", **values))
  354             update(state="COMPLETE", **result, remote_revision=verified["revision"])
  355             print(json.dumps(status), flush=True)
  356         except Exception as error:
  357             update(state="FAILED", error=str(error))
  358             raise
