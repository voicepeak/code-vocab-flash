   20 def benchmark(run):
   21     import cv2
   22 
   23     manifest = json.loads((run / 'manifest.json').read_text())
   24     config = manifest['config']['disparity_experiment']
   25     source = Path('/workspace/main') / config['source_run']
   26     before = {name: digest(source / name) for name in config['input_sha256']}
   27     if before != config['input_sha256']:
   28         raise ValueError('retained stereo input changed')
   29     summary = json.loads((source / 'stereo_summary.json').read_text())
   30     with (source / 'stereo_pairs.jsonl').open() as stream:
   31         row = json.loads(stream.readline())
   32     lifetime = summary.get('image_age_budget_s', .15)
   33     with np.load(source / 'stereo_first_pair.npz', allow_pickle=False) as saved:
   34         def image(side):
   35             stamp = row[side + '_stamp']
   36             return ImageFrame(Metadata(stamp, stamp + lifetime, row['epoch'],
   37                     'retained_official_front_' + side, 'front_' + side + '_image'),
   38                     row['encoding'], saved[side])
   39         pair = StereoPair(Metadata(row['pair_stamp'], row['valid_until'], row['epoch'],
   40                           'retained_stereo_pair', 'front_stereo'), image('left'), image('right'), row['header_skew_s'])
   41     settings = SgbmConfig()
   42     cv2.setNumThreads(1)
   43     compute_diagnostic_disparity(pair, settings)  # Separate warmup, not timed.
   44     durations, hashes = [], []
   45     for _ in range(3):
   46         begin = time.monotonic()
   47         result = compute_diagnostic_disparity(pair, settings)
   48         durations.append(time.monotonic() - begin)
   49         hashes.append(hashlib.sha256(result.disparity_px.tobytes() + result.valid.tobytes()).hexdigest())
   50     after = {name: digest(source / name) for name in before}
   51     if after != before:
   52         raise ValueError('retained stereo input changed during benchmark')
   53     np.savez_compressed(run / 'diagnostic_disparity.npz', disparity_px=result.disparity_px, valid=result.valid)
   54     values = result.disparity_px[result.valid]
   55     report = {'computed': True, 'historical_input_only': True, 'opencv_version': cv2.__version__,
   56               'opencv_threads': cv2.getNumThreads(), 'settings': asdict(settings),
   57               'input_shape': list(pair.left.pixels.shape), 'source_run': config['source_run'],
   58               'input_sha256': before, 'input_unchanged': before == after,
   59               'warmup_runs': 1, 'measured_runs': 3, 'compute_durations_s': durations,
   60               'compute_median_s': float(np.median(durations)), 'compute_max_s': max(durations),
   61               'output_sha256_per_repeat': hashes, 'repeat_deterministic': len(set(hashes)) == 1,
   62               'valid_pixels': int(result.valid.sum()), 'pixels': int(result.valid.size),
   63               'valid_fraction': float(result.valid.mean()),
   64               'disparity_p10_p50_p90_px': [float(v) for v in np.percentile(values, (10, 50, 90))] if len(values) else None,
   65               'source_metadata': asdict(result.metadata),
   66               'left_measurement_stamp': result.left_measurement_stamp,
   67               'right_measurement_stamp': result.right_measurement_stamp,
   68               'unknown_pixels_are_nan': bool(np.isnan(result.disparity_px[~result.valid]).all()),
   69               'rectification_verified': False, 'exposure_sync_verified': False,
   70               'metric_depth_verified': False, 'real_scene_accuracy_measured': False,
   71               'online_latency_verified': False, 'acceptance_eligible': False,
   72               'interpretation': 'Pixel disparity under unverified rectification. Coverage is not accuracy. Compute timing excludes input decoding/transport, simulator contention and GPU inference. Original validity is preserved, not refreshed by offline processing.'}
   73     (run / 'stereo_disparity_report.json').write_text(json.dumps(report, indent=2, allow_nan=False) + '\n')
   74     return 0 if report['repeat_deterministic'] and report['unknown_pixels_are_nan'] else 2
