   16 def normalized_input(image):
   17     import cv2
   18     import torch
   19 
   20     resized=cv2.resize(image,(480,360),interpolation=cv2.INTER_AREA)
   21     value=torch.as_tensor(resized.copy(),device='cuda').float()[None].permute(0,3,1,2).contiguous()
   22     value=torch.nn.functional.pad(value,(0,0,12,12),mode='replicate')
   23     mean=value.new_tensor([.485,.456,.406]).view(1,3,1,1)
   24     std=value.new_tensor([.229,.224,.225]).view(1,3,1,1)
   25     return ((value/255.-mean)/std).cpu().numpy()
