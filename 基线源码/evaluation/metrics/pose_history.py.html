   14 def state_from_json(data):
   15     metadata = dict(data['metadata'])
   16     metadata['covariance_or_bounds'] = tuple(metadata['covariance_or_bounds'])
   17     return StateEstimate(Metadata(**metadata), tuple(data['position_ned']), tuple(data['velocity_ned']),
   18         tuple(data['orientation_wxyz']), tuple(data['angular_velocity_body']), Health(data['health']))

   21 def audit(run):
   22     rows = lambda name: [json.loads(s) for s in (run / name).read_text().splitlines()]
   23     sources = rows('history_states.jsonl')
   24     queries = rows('camera_history.jsonl')
   25     summary = json.loads((run / 'pose_history_summary.json').read_text())
   26     history = PoseHistory()
   27     index = matched = agreed = 0
   28     divergences = []
   29     brackets = []
   30     for number, row in enumerate(queries):
   31         while index < len(sources) and sources[index]['state']['metadata']['stamp'] <= row['query_now']:
   32             source = sources[index]
   33             state = state_from_json(source['state'])
   34             if state.metadata.epoch != history.epoch:
   35                 history.reset(state.metadata.epoch)
   36             history.append(state, source['arrival_wall_monotonic'])
   37             index += 1
   38         expected = None
   39         reason = None
   40         try:
   41             expected = history.lookup(row['image_stamp'], row['query_now'], row['query_wall_monotonic'], row['epoch'])
   42         except HistoryUnavailable as error:
   43             reason = str(error)
   44         encoded = json.loads(json.dumps(asdict(expected))) if expected else None
   45         same = encoded == row['historical_pose'] and reason == row['reason']
   46         if same:
   47             agreed += 1
   48         else:
   49             divergences.append(number)
   50         if expected:
   51             matched += 1
   52             brackets.append(expected.bracket_stamps[1] - expected.bracket_stamps[0])
   53     expected_inputs = sorted(('/airsim_node/initial_pose', '/airsim_node/drone_1/gps',
   54                              '/airsim_node/drone_1/imu/imu', '/airsim_node/drone_1/front_left/Scene'))
   55     graph = summary.get('node_graph') or {}
   56     valid = (bool(queries) and matched > 0 and agreed == len(queries) and summary['fault'] is None
   57              and len(queries) == summary['images'] and matched == summary['historical_matches']
   58              and len(sources) == summary['source_states']
   59              and graph.get('subscriptions') == expected_inputs
   60              and not set(graph.get('publishers', [])) - {'/rosout'}
   61              and not summary['control_published'] and not summary['debug_inputs_used']
   62              and all(not r['current_control_eligible'] and not r['world_projection_applied'] for r in queries))
   63     def percentiles(values):
   64         return [float(x) for x in np.percentile(values, (50, 95, 99, 100))] if values else None
   65     result = {'audit_valid': bool(valid), 'source_states': len(sources), 'camera_queries': len(queries),
   66               'replayed_historical_matches': matched, 'queries_agreeing_with_replay': agreed,
   67               'divergent_query_indices': divergences, 'unavailable_reasons': summary['unavailable_reasons'],
   68               'match_fraction': matched / len(queries) if queries else 0.,
   69               'header_age_p50_p95_p99_max_s': percentiles([r['query_now'] - r['image_stamp'] for r in queries]),
   70               'lookup_elapsed_p50_p95_p99_max_s': percentiles([r['lookup_elapsed_wall_s'] for r in queries]),
   71               'maximum_interpolation_bracket_s': max(brackets) if brackets else None,
   72               'current_control_eligible': False, 'historical_uncertainty_calibrated': False,
   73               'image_exposure_time_calibrated': False, 'world_projection_verified': False,
   74               'acceptance_eligible': False,
   75               'analysis_source_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
   76     (run / 'pose_history_audit.json').write_text(json.dumps(result, indent=2, allow_nan=False) + '\n')
   77     return result
