   10 def evaluate(records, telemetry, control_end=None):
   11     sources=defaultdict(list)
   12     for row in telemetry:
   13         begin=row['publish_begin_wall_monotonic']
   14         end=row['publish_end_wall_monotonic']
   15         if not math.isfinite(begin) or not begin<=end:raise ValueError('invalid sender timing')
   16         sources[struct.pack('<4f',*row['command']['values']).hex()].append(
   17             {'sequence':row['command']['sequence'],'begin':begin,'end':end})
   18     entries=[r for r in records if r['event']=='native_pwm_entry']
   19     exits={r['bridge_span']:r for r in records if r['event']=='native_pwm_exit'}
   20     bindings=sum(r['event']=='native_callback_bound' for r in records)
   21     frees=defaultdict(list)
   22     for r in records:
   23         if r['event']=='parameter_free_end':frees[r['parameter_address']].append(r)
   24     posts={r['bridge_span']:r for r in records if r['event']=='native_pwm_post'}
   25     control_begin=min((r['publish_begin_wall_monotonic'] for r in telemetry),default=0.)
   26     end_basis='controller_end' if control_end is not None else 'last_publisher_call_end'
   27     if control_end is None:
   28         control_end=max((r['publish_end_wall_monotonic'] for r in telemetry),default=0.)
   29     matched=[];unmatched=[];unreadable=[];incomplete=0
   30     latency=[];active_entries=0;freed_during_call=[]
   31     for entry in entries:
   32         start=entry['start_monotonic_ns']*1e-9
   33         end=entry['end_monotonic_ns']*1e-9
   34         if not 0<=start<=end:raise ValueError('invalid native snapshot timing')
   35         close=exits.get(entry['bridge_span'])
   36         if close is None or close['start_monotonic_ns']<entry['end_monotonic_ns']:
   37             incomplete+=1
   38         active=control_begin<=start<=end<=control_end
   39         active_entries+=active
   40         if close and close.get('native_call_interval_measured'):
   41             for free in frees[entry['parameter_address']]:
   42                 if (close['start_monotonic_ns']<=free['start_monotonic_ns']
   43                         and free['end_monotonic_ns']<=close['end_monotonic_ns']):
   44                     freed_during_call.append({'entry':entry,'native_call':close,'free':free,
   45                         'post_snapshot':posts.get(entry['bridge_span']),'in_control_window':active})
   46         if not entry['parameter_read_ok']:
   47             unreadable.append(entry)
   48             continue
   49         payload=entry['parameter_float32_hex']
   50         raw=bytes.fromhex(payload)
   51         if len(raw)!=16:raise ValueError('native snapshot must have four float32 fields')
   52         candidates=[r for r in sources[payload] if r['begin']<=end]
   53         if not candidates:
   54             values=struct.unpack('<4f',raw)
   55             unmatched.append({'entry':entry,'snapshot_values':[v if math.isfinite(v) else None for v in values],
   56                               'finite_unit_range':all(math.isfinite(v) and 0<=v<=1 for v in values)})
   57             continue
   58         source=max(candidates,key=lambda r:r['begin'])
   59         bounds=(start-source['end'],end-source['begin'])
   60         if active:latency.append(bounds)
   61         matched.append({'bridge_span':entry['bridge_span'],'source_sequence':source['sequence'],
   62                         'entry_monotonic_s':start,'parameter_address':entry['parameter_address'],
   63                         'publish_to_snapshot_bounds_s':list(bounds),'in_control_window':active})
   64     result={'native_entries':len(entries),'matched_entries':len(matched),'unmatched_entries':len(unmatched),
   65             'unreadable_entries':len(unreadable),'incomplete_native_calls':incomplete,'callback_bindings':bindings,
   66             'unmatched':unmatched,'unreadable':unreadable,'matched':matched,
   67             'snapshot_coverage_valid':bool(entries) and bindings==1 and not unreadable and not incomplete,
   68             'record_write_errors':max((r['write_errors'] for r in records),default=0),
   69             'control_window_monotonic_s':[control_begin,control_end],'control_window_end_basis':end_basis,
   70             'native_entries_in_control_window':active_entries,
   71             'native_calls_with_exact_intervals':sum(bool(r.get('native_call_interval_measured')) for r in exits.values()),
   72             'native_entries_after_control':sum(r['start_monotonic_ns']*1e-9>control_end for r in entries),
   73             'parameters_freed_during_original_callback':len(freed_during_call),
   74             'freed_during_original_callback':freed_during_call,
   75             'actual_actuator_consumption_verified':False,'physical_impulse_cause_identified':False,
   76             'acceptance_eligible':False,
   77             'interpretation':'Snapshot of four float32 fields at the original native callback entry, not the downstream payload read or motor application. Payloads are compared to sender float32 conversions published before the snapshot ended. Bounds use the publisher-call and snapshot sampling intervals on the same Linux monotonic clock; asynchronous send can overlap the publisher call.'}
   78     if latency:
   79         result['publish_to_snapshot_lower_min_s']=min(x[0] for x in latency)
   80         result['publish_to_snapshot_upper_max_s']=max(x[1] for x in latency)
   81     return result

   84 def main():
   85     parser=argparse.ArgumentParser(description=__doc__)
   86     parser.add_argument('run',type=Path)
   87     args=parser.parse_args()
   88     def rows(name):return [json.loads(x) for x in (args.run/name).read_text().splitlines()]
   89     summary=json.loads((args.run/'pwm_summary.json').read_text())
   90     result=evaluate(rows('bridge_probe.jsonl'),rows('pwm_telemetry.jsonl'),summary.get('control_end_wall_monotonic'))
   91     (args.run/'native_input_audit.json').write_text(json.dumps(result,indent=2)+'\n')
   92     print(json.dumps({k:v for k,v in result.items() if k not in ('matched','unmatched','unreadable')},indent=2))
   93     return 0 if result['snapshot_coverage_valid'] and not result['record_write_errors'] else 2
