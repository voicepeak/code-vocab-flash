    1 """Compare the independent ROS subscriber record with the sender's command log.
    2 
    3 Wire agreement proves serialization/observation, not consumption by a motor.
    4 Debug PWM is checked for observability; it is not assumed to be actual force.
    5 """

   17 def decode(row):
   18     if row.get('md5sum')!=MD5:return None
   19     try:
   20         data=row['data'];header=data['header'];stamp=header['stamp']
   21         return {'sequence':header['seq'],'stamp':stamp['secs']+stamp['nsecs']*1e-9,
   22                 'values':tuple(data['rotorPWM'+str(i)] for i in range(4))}
   23     except (KeyError,TypeError):return None

   26 def packed(values):return struct.pack('<4d',*values)

   29 def evaluate(telemetry,wire,debug):
   30     decoded=[decode(x) for x in wire]
   31     invalid=sum(x is None for x in decoded)
   32     expected={r['command']['sequence']:r['command'] for r in telemetry}
   33     # rospy assigns Header.seq during serialization; it is distinct from the
   34     # sender's logical guard counter. Infer a single offset only from matching
   35     # measurement stamps AND bit-exact official float64 payloads, then audit every row.
   36     offsets=Counter(remote['sequence']-seq for remote in decoded if remote is not None
   37         for seq,command in expected.items() if abs(remote['stamp']-command['stamp'])<=1e-6
   38         and packed(remote['values'])==packed(command['values']))
   39     best=offsets.most_common(2)
   40     offset=best[0][0] if best and (len(best)==1 or best[0][1]>best[1][1]) else None
   41     observed={}
   42     duplicates=0
   43     for row in decoded:
   44         if row is None:continue
   45         seq=row['sequence']-(offset if offset is not None else 0)
   46         if seq in observed:duplicates+=1
   47         else:observed[seq]=row
   48     matched=0;missing=[];mismatch=[]
   49     for seq,command in expected.items():
   50         remote=observed.get(seq)
   51         if remote is None:missing.append(seq)
   52         elif (abs(remote['stamp']-command['stamp'])>1e-6
   53                 or packed(remote['values'])!=packed(command['values'])):mismatch.append(seq)
   54         else:matched+=1
   55     extra=sorted(set(observed)-set(expected))
   56     result={'sender_commands':len(telemetry),'wire_records':len(wire),'matched_commands':matched,
   57             'wire_match_fraction':matched/len(telemetry) if telemetry else 0.,
   58             'missing_sequences':missing,'mismatched_sequences':mismatch,'unexpected_sequences':extra,
   59             'invalid_records':invalid,'duplicate_sequences':duplicates,'actuator_consumption_verified':False,
   60             'debug_pwm_is_verified_actual_motor_output':False,'official_result':'UNKNOWN'}
   61     result['ros_header_sequence_offset']=offset
   62     result['wire_trace_valid']=(bool(telemetry) and matched==len(telemetry)
   63                                 and offset is not None and not (invalid or duplicates or extra))
   64     debug_rows=sorted((r for r in (decode(x) for x in debug) if r is not None),key=lambda r:r['stamp'])
   65     stamps=np.array(sorted(set(x['stamp'] for x in debug_rows)))
   66     if len(stamps)>1:result['debug_message_stamp_interval_median_s']=float(np.median(np.diff(stamps)))
   67     changes=[];previous=None
   68     for row in debug_rows:
   69         if previous is None or np.max(np.abs(np.array(row['values'])-previous))>1e-7:
   70             changes.append(row['stamp']);previous=np.array(row['values'])
   71     if len(changes)>1:result['observed_debug_value_change_interval_p10_p50_p90_s']=np.percentile(np.diff(changes),[10,50,90]).tolist()
   72     pulses=[]
   73     for i,row in enumerate(telemetry):
   74         if not (row.get('collective_pulse_applied') or row.get('angular_pulse_applied')):continue
   75         command=row['command'];end=telemetry[i+1]['command']['stamp'] if i+1<len(telemetry) else command['stamp']+.02
   76         candidates=[r for r in debug_rows if command['stamp']-1e-6<=r['stamp']<=end+.08]
   77         pulses.append({'sequence':command['sequence'],'stamp':command['stamp'],'values':command['values'],
   78             'next_command_after_sensor_s':end-command['stamp'],'debug_samples_in_window':len(candidates),
   79             # Debug values may be float32 internally before their float64 wire
   80             # serialization. Do not confuse quantization with a missed pulse.
   81             'debug_samples_matching_pulse':int(sum(np.max(np.abs(np.array(r['values'])-command['values']))<=1e-6
   82                                               for r in candidates))})
   83     result['experiment_pulses']=pulses
   84     return result

   87 def main():
   88     parser=argparse.ArgumentParser(description=__doc__);parser.add_argument('run',type=Path)
   89     args=parser.parse_args()
   90     def rows(name):return [json.loads(x) for x in (args.run/name).read_text().splitlines()]
   91     report=evaluate(rows('pwm_telemetry.jsonl'),rows('evaluation_pwm_commands.jsonl'),rows('evaluation_actuation.jsonl'))
   92     analysis_hash=hashlib.sha256(Path(__file__).read_bytes()).hexdigest()
   93     report['analysis_code_sha256']=analysis_hash
   94     manifest=args.run/'manifest.json'
   95     if manifest.exists():
   96         original=json.loads(manifest.read_text()).get('source_sha256_before',{}).get('evaluation/metrics/command_trace.py')
   97         report['analysis_matches_run_source_snapshot']=original==analysis_hash
   98     (args.run/'command_trace_audit.json').write_text(json.dumps(report,indent=2)+'\n')
   99     print(json.dumps(report,indent=2))
  100     return 0 if report['wire_trace_valid'] else 2
