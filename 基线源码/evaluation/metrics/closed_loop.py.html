    8 def evaluate(telemetry, truth, settled_after=4.):
    9     ordered = sorted({r['stamp']: r for r in truth if r['stamp'] > 0}.values(),
   10                      key=lambda r: r['stamp'])
   11     stamps = np.array([r['stamp'] for r in ordered])
   12     errors, tracking, attitudes, yaw_errors, imu_attitudes = [], [], [], [], []
   13     matched = 0
   14     for row in telemetry:
   15         stamp = row['command']['stamp']
   16         index = int(np.searchsorted(stamps, stamp))
   17         if index == 0 or index == len(ordered):
   18             continue
   19         a, b = ordered[index-1], ordered[index]
   20         if b['stamp']-a['stamp'] > .05:
   21             continue
   22         t = (stamp-a['stamp'])/(b['stamp']-a['stamp'])
   23         position = (1-t)*np.array(a['position'])+t*np.array(b['position'])
   24         qa, qb = np.array(a['orientation']), np.array(b['orientation'])
   25         if qa @ qb < 0:
   26             qb = -qb
   27         orientation = (1-t)*qa+t*qb
   28         orientation /= np.linalg.norm(orientation)
   29         estimated = np.array(row['state']['position_ned'])
   30         matched += 1
   31         if row['wall_elapsed'] >= settled_after:
   32             errors.append(estimated-position)
   33             dot = abs(np.array(row['state']['orientation_wxyz']) @ orientation)
   34             attitudes.append(2*np.arccos(np.clip(dot, 0., 1.)))
   35             raw_q=row.get('raw',{}).get('imu_orientation_wxyz')
   36             if raw_q is not None:
   37                 q=np.asarray(raw_q,dtype=float)
   38                 norm=np.linalg.norm(q)
   39                 if q.shape==(4,) and np.all(np.isfinite(q)) and abs(norm-1)<.01:
   40                     dot=abs(q @ orientation/norm)
   41                     imu_attitudes.append(2*np.arccos(np.clip(dot,0.,1.)))
   42             if 'reference' in row:
   43                 tracking.append(position-np.array(row['reference']))
   44             if 'yaw_reference' in row:
   45                 w,x,y,z=orientation
   46                 yaw=np.arctan2(2*(w*z+x*y),1-2*(y*y+z*z))
   47                 yaw_errors.append((yaw-row['yaw_reference']+np.pi) % (2*np.pi)-np.pi)
   48     timestamp_key = 'publish_wall_elapsed' if telemetry and all('publish_wall_elapsed' in r for r in telemetry) else 'wall_elapsed'
   49     intervals = np.diff([r[timestamp_key] for r in telemetry])
   50     result = {'matched_commands': matched, 'commands': len(telemetry),
   51               'match_fraction': matched/len(telemetry) if telemetry else 0.,
   52               'settled_after_wall_s': settled_after, 'settled_samples': len(errors),
   53               'command_timing_measurement': 'after_publish' if timestamp_key == 'publish_wall_elapsed' else 'loop_start_before_graph_query',
   54               'official_result': 'UNKNOWN', 'truth_use': 'offline_evaluation_only'}
   55     if len(intervals):
   56         result['command_wall_interval_p50_p95_p99_s'] = np.percentile(intervals, (50, 95, 99)).tolist()
   57         result['minimum_command_wall_interval_s'] = float(intervals.min())
   58     if errors:
   59         result['state_position_rmse_m'] = np.sqrt(np.mean(np.array(errors)**2, axis=0)).tolist()
   60         result['state_attitude_rmse_rad'] = float(np.sqrt(np.mean(np.array(attitudes)**2)))
   61     if tracking:
   62         result['truth_tracking_rmse_m'] = np.sqrt(np.mean(np.array(tracking)**2, axis=0)).tolist()
   63         result['truth_tracking_max_norm_m'] = float(np.linalg.norm(tracking, axis=1).max())
   64     if yaw_errors:
   65         result['truth_yaw_tracking_rmse_rad']=float(np.sqrt(np.mean(np.array(yaw_errors)**2)))
   66         result['truth_yaw_tracking_max_abs_rad']=float(np.abs(yaw_errors).max())
   67     if imu_attitudes:
   68         result['imu_orientation_unit_valid_fraction']=len(imu_attitudes)/len(errors)
   69         result['imu_orientation_truth_rmse_rad']=float(np.sqrt(np.mean(np.array(imu_attitudes)**2)))
   70         result['imu_orientation_fused_in_estimator']=False
   71     result['evaluation_valid'] = result['match_fraction'] >= .9 and len(errors) >= 100
   72     return result

   75 def main():
   76     parser = argparse.ArgumentParser(description=__doc__)
   77     parser.add_argument('run', type=Path)
   78     args = parser.parse_args()
   79     def rows(name):
   80         return [json.loads(line) for line in (args.run/name).read_text().splitlines()]
   81     manifest = json.loads((args.run/'manifest.json').read_text())
   82     mode = manifest['config'].get('control_mode','PWM')
   83     telemetry=rows('vel_telemetry.jsonl' if mode == 'VEL' else 'pwm_telemetry.jsonl')
   84     result = evaluate(telemetry, rows('evaluation_truth.jsonl'))
   85     result['control_mode'] = mode
   86     if manifest['config'].get('experiment',{}).get('braking'):
   87         from evaluation.metrics.braking import measure_braking
   88         result['braking_response']=measure_braking(telemetry,rows('evaluation_truth.jsonl'))
   89     if mode == 'VEL':
   90         samples=[r for r in telemetry if 5.4 <= r.get('sim_elapsed',0) < 6.8
   91                  and r.get('yaw_rate_reference',0) > 0 and abs(r['command']['values'][3]) > 1.]
   92         if len(samples)>20:
   93             command=np.array([r['command']['values'][3] for r in samples])
   94             omega=np.array([r['raw']['omega_body'][2] for r in samples])
   95             gain=float(command @ omega/(command @ command))
   96             result['vel_yaw_response_gain_rad_per_command_unit']=gain
   97             result['vel_yaw_expected_rad_per_degree']=float(np.pi/180)
   98             result['vel_yaw_degrees_response_within_20pct']=abs(gain/(np.pi/180)-1) < .2
   99     (args.run/'closed_loop_metrics.json').write_text(json.dumps(result, indent=2)+'\n')
  100     print(json.dumps(result, indent=2))
  101     return 0 if result['evaluation_valid'] else 2
