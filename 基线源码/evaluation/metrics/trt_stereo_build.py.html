   18 def export(run,vendor,checkpoint,sample,freeze_fourier=False,reference_run=None):

   85     report={'onnx_sha256':file_hash(run/'stereo.onnx'),'opset':17,'image_shape':[1,3,384,480],
   86         'external_imagenet_normalization':True,'volume_normalize':bool(model.args.normalize),

   91         'fourier_inputs_frozen':freeze_fourier,

   96 def build(run,precision,precision_policy='default'):
   97     import tensorrt as trt
   98     import torch
   99 
  100     logger=trt.Logger(trt.Logger.INFO);start=time.monotonic()
  101     builder=trt.Builder(logger);network=builder.create_network(1<<int(trt.NetworkDefinitionCreationFlag.EXPLICIT_BATCH))
  102     parser=trt.OnnxParser(network,logger)
  103     if not parser.parse((run/'stereo.onnx').read_bytes()):
  104         raise RuntimeError('ONNX parse failed: '+str([str(parser.get_error(i)) for i in range(parser.num_errors)]))
  105     config=builder.create_builder_config();config.set_memory_pool_limit(trt.MemoryPoolType.WORKSPACE,512*1024*1024)
  106     config.clear_flag(trt.BuilderFlag.TF32);config.max_aux_streams=0
  107     if precision=='fp16':config.set_flag(trt.BuilderFlag.FP16)
  108     from evaluation.metrics.trt_precision_policy import apply_precision_policy
  109     constraints=apply_precision_policy(network,config,trt,precision_policy,precision)
  110     (run/'precision_constraints.json').write_text(json.dumps({'policy':precision_policy,'layers':constraints,
  111         'obey_precision_constraints':bool(constraints),'selection_is_experimental_not_cause_proof':True},indent=2)+'\n')
  112     serialized=builder.build_serialized_network(network,config)
  113     if serialized is None:raise RuntimeError('TensorRT engine construction failed')
  114     engine_path=run/('stereo_'+precision+'.engine');engine_path.write_bytes(bytes(serialized))
  115     report={'engine_sha256':file_hash(engine_path),'engine_file':engine_path.name,'onnx_sha256':file_hash(run/'stereo.onnx'),
  116         'tensorrt':trt.__version__,'precision':precision,'tf32_enabled':False,
  117         'precision_policy':precision_policy,'fp32_constrained_layers':len(constraints),
  118         'precision_constraints_sha256':file_hash(run/'precision_constraints.json'),
  119         'global_tf32_override':os.environ.get('NVIDIA_TF32_OVERRIDE'),'max_aux_streams':0,
  120         'workspace_limit_bytes':512*1024*1024,'build_s':time.monotonic()-start,
  121         'gpu':torch.cuda.get_device_name(0),'compute_capability':list(torch.cuda.get_device_capability(0)),
  122         'cuda_runtime':torch.version.cuda,'acceptance_eligible':False,'current_control_eligible':False}
  123     (run/('build_'+precision+'.json')).write_text(json.dumps(report,indent=2)+'\n')
  124     if constraints:
  125         runtime=trt.Runtime(logger);engine=runtime.deserialize_cuda_engine(serialized)
  126         if engine is None:raise RuntimeError('built engine cannot be inspected')
  127         inspector=engine.create_engine_inspector()
  128         information=inspector.get_engine_information(trt.LayerInformationFormat.JSON)
  129         (run/'engine_inspector.json').write_text(information+'\n')
  130         if precision_policy=='sensitive_geometry_fp32':
  131             from evaluation.metrics.trt_precision_inspection import inspect_geometry_precision
  132             inspection=inspect_geometry_precision(json.loads(information),constraints)
  133             (run/'geometry_precision_inspection.json').write_text(json.dumps(inspection,indent=2)+'\n')
  134             if inspection['non_fp32_coordinates']:
  135                 raise RuntimeError('visible geometry coordinates still use reduced precision')
