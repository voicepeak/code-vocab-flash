   17 def prepare(run):
   18     from adapters.ros_noetic.capture_io import load_message
   19     from adapters.ros_noetic.lidar_cloud import decode_xyz
   20     from adapters.ros_noetic.stereo_images import decode_image
   21     from evaluation.metrics.joint_sensors import restore_pose
   22     from perception.stereo_depth.geometry import load_nominal_rig
   23     from perception.stereo_depth.learned import rgb_pixels
   24     from perception.stereo_depth.lidar_projection import project_lidar_into_stereo, sparse_visible
   25 
   26     root = Path(__file__).resolve().parents[2]
   27     manifest = json.loads((run/'manifest.json').read_text())
   28     rig = load_nominal_rig(root/'configs/simulator_settings.json')
   29     (run/'inputs').mkdir(); (run/'references').mkdir()
   30     records = []
   31     for source_name in manifest['config']['learned_stereo']['source_runs']:
   32         source = root/source_name
   33         original = json.loads((source/'manifest.json').read_text())
   34         if original['simulator_preflight']['settings_sha256'] != rig.settings_sha256:
   35             raise ValueError('captured geometry settings differ')
   36         summary = json.loads((source/'joint_summary.json').read_text())
   37         audit = json.loads((source/'joint_audit.json').read_text())
   38         if not audit['input_audit_valid'] or summary['fault']:
   39             raise ValueError('source capture has not passed the independent payload audit')
   40         for entry in summary['captures']:
   41             directory = source/entry['directory']
   42             packet = json.loads((directory/'packet.json').read_text())
   43             if file_hash(directory/'packet.json') != entry['packet_sha256']:
   44                 raise ValueError('capture packet changed')
   45             frames = {side: decode_image(load_message(directory/(side+'.json')), side,
   46                 packet['epoch'], packet['image_age_budget_s']) for side in ('left','right')}
   47             for side, frame in frames.items():
   48                 if json.loads(json.dumps(asdict(frame.metadata))) != packet[side+'_source_metadata']:
   49                     raise ValueError('source metadata changed')
   50             key = '%04d' % len(records)
   51             np.savez_compressed(run/'inputs'/(key+'.npz'),
   52                 left=rgb_pixels(frames['left']), right=rgb_pixels(frames['right']))
   53             # Recompute all sparse nominal references, without consulting SGBM masks.
   54             cloud = decode_xyz(load_message(directory/'lidar.json'), packet['epoch'])
   55             poses = {tag: restore_pose(data) for tag, data in packet['poses'].items()}
   56             luv,lz,ruv,rz,in_view = project_lidar_into_stereo(cloud.points,
   57                 poses['lidar'],poses['left'],poses['right'],rig)
   58             visible = (sparse_visible(luv,lz,np.isfinite(luv).all(axis=1),rig.left.width,rig.left.height)
   59                 & sparse_visible(ruv,rz,np.isfinite(ruv).all(axis=1),rig.right.width,rig.right.height))
   60             expected = np.full(len(luv),np.nan)
   61             expected[in_view] = luv[in_view,0]-ruv[in_view,0]
   62             candidates = visible & in_view & np.isfinite(expected)
   63             candidates[candidates] = expected[candidates] > 0
   64             np.savez_compressed(run/'references'/(key+'.npz'),
   65                 left_uv=luv, expected_disparity_px=expected, candidates=candidates)
   66             records.append({'key':key,'source_run':source_name,'packet_directory':entry['directory'],
   67                 'seed':original['seed'],'left_metadata':packet['left_source_metadata'],
   68                 'right_metadata':packet['right_source_metadata'], 'epoch':packet['epoch'],
   69                 'image_sha256':file_hash(run/'inputs'/(key+'.npz')),
   70                 'reference_sha256':file_hash(run/'references'/(key+'.npz'))})
   71     write_json(run/'inputs'/'index.json', records)
   72     return records

   75 def residual_stats(values):
   76     values = np.asarray(values)
   77     return {'count':int(len(values)),
   78         'absolute_median_px':float(np.median(np.abs(values))) if len(values) else None,
   79         'absolute_p90_px':float(np.percentile(np.abs(values),90)) if len(values) else None,
   80         'signed_median_px':float(np.median(values)) if len(values) else None}

   83 def compare_reference(uv, expected, candidates, fields):
   84     """No residual rejection: compare own support and the exact shared support."""
   85     sampled, masks = {}, {}
   86     for name, (disparity, valid) in fields.items():
   87         sampled[name], masks[name] = sample_disparity(uv, disparity, valid, candidates)
   88     shared = np.logical_and.reduce(list(masks.values()))
   89     return {'reference_candidates':int(candidates.sum()), 'methods':{
   90         name:{'own_support':residual_stats(sampled[name][masks[name]]-expected[masks[name]]),
   91               'shared_support':residual_stats(sampled[name][shared]-expected[shared])}
   92         for name in fields}}, masks

   95 def evaluate(run):
   96     records = json.loads((run/'inputs'/'index.json').read_text())
   97     runtime = json.loads((run/'predictions'/'runtime.json').read_text())
   98     packets, pooled = [], {method:[] for method in ('sgbm','ffs')}
   99     for record in records:
  100         key = record['key']
  101         reference_path, image_path = run/'references'/(key+'.npz'), run/'inputs'/(key+'.npz')
  102         if (file_hash(reference_path) != record['reference_sha256']
  103                 or file_hash(image_path) != record['image_sha256']):
  104             raise ValueError('evaluation inputs changed')
  105         with np.load(reference_path,allow_pickle=False) as ref, np.load(run/'predictions'/(key+'.npz'),allow_pickle=False) as pred:
  106             fields = {name:(pred[name+'_disparity'],pred[name+'_valid']) for name in ('sgbm','ffs')}
  107             comparison, masks = compare_reference(ref['left_uv'],ref['expected_disparity_px'],ref['candidates'],fields)
  108             shared = masks['sgbm'] & masks['ffs']
  109             for name,(disp,valid) in fields.items():
  110                 if not (np.isnan(disp[~valid]).all() and np.isfinite(disp[valid]).all() and (disp[valid]>0).all()):
  111                     raise ValueError('invalid unknown or valid disparity pixels')
  112                 samples,_ = sample_disparity(ref['left_uv'],disp,valid,ref['candidates'])
  113                 pooled[name].extend((samples[shared]-ref['expected_disparity_px'][shared]).tolist())
  114             comparison.update(key=key,seed=record['seed'],source_run=record['source_run'],
  115                 packet_directory=record['packet_directory'],
  116                 source_metadata={'left':record['left_metadata'],'right':record['right_metadata']},
  117                 prediction_sha256=file_hash(run/'predictions'/(key+'.npz')),
  118                 coverage={name:float(valid.mean()) for name,(_,valid) in fields.items()},
  119                 raw_ffs_positive_fraction=float((np.isfinite(pred['ffs_forward'])&(pred['ffs_forward']>0)).mean()),
  120                 ffs_lr_only_fraction=float(pred['ffs_lr_valid'].mean()))
  121             packets.append(comparison)
  122     report = {'computed':True,'packets':packets,'runtime':runtime,
  123         'pooled_shared_residuals':{name:residual_stats(values) for name,values in pooled.items()},
  124         'historical_only':True,'reference_type':'nominal_lidar_after_estimated_header_pose_alignment',
  125         'reference_selection_uses_disparity_or_residual':False,
  126         'development_samples_not_heldout':True,'metric_calibration_verified':False,
  127         'exposure_sync_verified':False,'per_point_motion_compensation_applied':False,
  128         'sparse_visibility_is_complete_proof':False,'real_scene_accuracy_verified':False,
  129         'confidence_calibrated':False,'online_latency_verified':False,
  130         'current_control_eligible':False,'acceptance_eligible':False,
  131         'interpretation':'Nominal residuals are not ground-truth accuracy. Shared-support errors use identical reference points without residual filtering. Coverage is not confidence or FREE. Timing is offline without simulator contention or ROS transport. Original source leases are retained.'}
  132     write_json(run/'learned_stereo_report.json',report)
  133     return report
