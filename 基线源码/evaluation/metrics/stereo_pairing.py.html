   16 def audit(run):
   17     import rospy
   18     from cv_bridge import CvBridge
   19     from sensor_msgs.msg import Image
   20 
   21     summary = json.loads((run / 'stereo_summary.json').read_text())
   22     rows = [json.loads(s) for s in (run / 'stereo_pairs.jsonl').read_text().splitlines()]
   23     timing_path = run / 'stereo_image_timing.jsonl'
   24     timing = [json.loads(s) for s in timing_path.read_text().splitlines()] if timing_path.exists() else []
   25     matches = {}
   26     for side in ('left', 'right'):
   27         if not (run / ('stereo_first_' + side + '.json')).exists():
   28             matches[side] = False
   29             continue
   30         header = json.loads((run / ('stereo_first_' + side + '.json')).read_text())
   31         message = Image()
   32         message.header.stamp = rospy.Time.from_sec(header['stamp'])
   33         message.header.frame_id = header['frame_id']
   34         for key in ('width', 'height', 'step', 'encoding', 'is_bigendian'):
   35             setattr(message, key, header[key])
   36         message.data = (run / ('stereo_first_' + side + '.bin')).read_bytes()
   37         reference = CvBridge().imgmsg_to_cv2(message, desired_encoding='passthrough')
   38         if reference.ndim == 2:
   39             reference = reference[:, :, None]
   40         matches[side] = bool(np.array_equal(reference, decode_image(message, side, 0).pixels))
   41     graph = summary.get('node_graph') or {}
   42     budget = summary.get('image_age_budget_s', .15)
   43     graph_valid = (graph.get('subscriptions') == sorted(
   44                  ['/airsim_node/drone_1/front_left/Scene', '/airsim_node/drone_1/front_right/Scene',
   45                   '/airsim_node/drone_1/imu/imu'])
   46                  and not set(graph.get('publishers', [])) - {'/rosout'})
   47     timing_valid = (bool(timing) and len(timing) == sum(summary['decoded_frames'].values())
   48                     and all(r['arrival_wall_monotonic'] <= r['decode_begin_wall_monotonic']
   49                             <= r['decode_end_wall_monotonic'] for r in timing))
   50     pairs_valid = (bool(rows) and summary['pairs'] == len(rows) and summary['max_skew_s'] == .005
   51              and budget in (.15, .2)
   52              and all(0 <= r['header_skew_s'] <= .005
   53                      and abs(abs(r['left_stamp'] - r['right_stamp']) - r['header_skew_s']) < 1e-12
   54                      and r['pair_stamp'] == max(r['left_stamp'], r['right_stamp'])
   55                      and r['pair_stamp'] <= r['imu_clock_stamp'] < r['valid_until']
   56                      and r['valid_until'] == min(r['left_stamp'], r['right_stamp']) + budget
   57                      and r['processing_begin_wall_monotonic'] <= r['emitted_wall_monotonic']
   58                      and not r['exposure_sync_verified'] and not r['calibrated_rectification_verified'] for r in rows)
   59              and all(a[side + '_stamp'] < b[side + '_stamp'] for a, b in zip(rows, rows[1:])
   60                      for side in ('left', 'right'))
   61              )
   62     inputs_valid = (all(matches.values()) and timing_valid and graph_valid
   63                     and not summary['control_published'] and not summary['debug_inputs_used'])
   64     valid = inputs_valid and pairs_valid and summary['fault'] is None
   65     result = {'audit_valid': bool(valid), 'pairs': len(rows), 'cv_bridge_first_images_match': matches,
   66               'input_records_valid': bool(inputs_valid), 'pair_records_valid': bool(pairs_valid),
   67               'image_age_budget_s': budget, 'default_150ms_pipeline_passed': bool(valid and budget == .15),
   68               'source_fault': summary['fault'], 'image_timing_records': len(timing),
   69               'header_skew_p50_p95_p99_max_s': percentiles([r['header_skew_s'] for r in rows]),
   70               'pair_age_on_imu_clock_p50_p95_p99_max_s': percentiles(
   71                   [r['imu_clock_stamp'] - min(r['left_stamp'], r['right_stamp']) for r in rows]),
   72               'pair_processing_p50_p95_p99_max_s': percentiles(
   73                   [r['emitted_wall_monotonic'] - r['processing_begin_wall_monotonic'] for r in rows]),
   74               'pair_processing_interval_scope': 'Emission loop only; images may have been decoded in earlier loops. Not end-to-end latency.',
   75               'image_decode_p50_p95_p99_max_s': percentiles(
   76                   [r['decode_end_wall_monotonic'] - r['decode_begin_wall_monotonic'] for r in timing]),
   77               'image_header_age_at_callback_p50_p95_p99_max_s': percentiles(
   78                   [r['imu_stamp_at_callback'] - r['stamp'] for r in timing if r['imu_stamp_at_callback'] is not None]),
   79               'image_header_age_at_processing_p50_p95_p99_max_s': percentiles(
   80                   [r['imu_stamp_at_processing'] - r['stamp'] for r in timing]),
   81               'image_pending_wall_age_p50_p95_p99_max_s': percentiles(
   82                   [r['decode_begin_wall_monotonic'] - r['arrival_wall_monotonic'] for r in timing]),
   83               'image_shapes': sorted({tuple(r['image_shape']) for r in rows}),
   84               'exposure_sync_verified': False, 'metric_depth_verified': False,
   85               'acceptance_eligible': False,
   86               'analysis_source_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
   87     (run / 'stereo_pairing_audit.json').write_text(json.dumps(result, indent=2, allow_nan=False) + '\n')
   88     return result
