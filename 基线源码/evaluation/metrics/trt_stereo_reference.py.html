   44                 result['forward_raw_max_px']=float(raw_error[finite].max()) if finite.any() else None
   45                 comparisons[name]=result
   46         packets.append({'key':key,'comparisons_with_pytorch_fp32':comparisons,
   47             'prediction_sha256':file_hash(output/(key+'.npz')),'elapsed_with_comparisons_s':time.monotonic()-begin})
   48     report={'packets':packets,'tf32_enabled':False,'mixed_precision':False,'compile_disabled':True,
   49         'same_source_images':True,'torch':torch.__version__,
