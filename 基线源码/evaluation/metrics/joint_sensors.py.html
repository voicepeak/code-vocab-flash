   26 def restore_pose(data):
   27     if data['frame_id'] != 'world_ned' or data['current_control_eligible'] or data['uncertainty_calibrated']:
   28         raise ValueError('unexpected historical pose eligibility')
   29     return HistoricalPose(data['stamp'], data['epoch'], tuple(data['position_ned']),
   30         tuple(data['orientation_wxyz']), tuple(data['bracket_stamps']),
   31         tuple(tuple(d) for d in data['bracket_covariance_diagonals']), data['interpolation_fraction'],
   32         data['position_correction_residual_m'], data['rotation_correction_residual_rad'])

   35 def audit(run):
   36     import cv2
   37     from cv_bridge import CvBridge
   38     from sensor_msgs.point_cloud2 import read_points
   39 
   40     cv2.setNumThreads(1)
   41     manifest = json.loads((run / 'manifest.json').read_text())
   42     summary = json.loads((run / 'joint_summary.json').read_text())
   43     source_rows = [json.loads(s) for s in (run / 'joint_states.jsonl').read_text().splitlines()]
   44     raw_rows = [json.loads(s) for s in (run / 'joint_raw_sensors.jsonl').read_text().splitlines()]
   45     rig = load_nominal_rig(Path(__file__).resolve().parents[2] / 'configs/simulator_settings.json')
   46     if rig.settings_sha256 != manifest['simulator_preflight']['settings_sha256']:
   47         raise ValueError('nominal settings differ from the captured scene')
   48     history = PoseHistory()
   49     source_index = 0
   50     packets = []
   51     used_stamps = {tag: set() for tag in ('left','right','lidar')}
   52     valid = bool(summary['captures']) and summary['fault'] is None and summary['raw_queue_drops'] == 0
   53     for entry in summary['captures']:
   54         relative = Path(entry['directory'])
   55         if relative.is_absolute() or '..' in relative.parts:raise ValueError('invalid packet directory')
   56         directory = run / relative
   57         packet_path = directory / 'packet.json'
   58         if hashlib.sha256(packet_path.read_bytes()).hexdigest() != entry['packet_sha256']:
   59             raise ValueError('joint packet changed')
   60         packet = json.loads(packet_path.read_text())
   61         epoch = packet['epoch']
   62         messages = {tag: load_message(directory / (tag + '.json')) for tag in used_stamps}
   63         exact_images = True
   64         frames = {}
   65         for side in ('left','right'):
   66             frames[side] = decode_image(messages[side], side, epoch, packet['image_age_budget_s'])
   67             independent = CvBridge().imgmsg_to_cv2(messages[side], desired_encoding='passthrough')
   68             if independent.ndim == 2:independent = independent[:, :, None]
   69             exact_images &= np.array_equal(independent, frames[side].pixels)
   70         cloud = decode_xyz(messages['lidar'], epoch)
   71         reference = np.asarray(list(read_points(messages['lidar'], field_names=('x','y','z'), skip_nans=False)), dtype=float).reshape(-1,3)
   72         reference = reference[np.isfinite(reference).all(axis=1)]
   73         exact_cloud = np.array_equal(reference, cloud.points)
   74         while source_index < len(source_rows) and source_rows[source_index]['state']['metadata']['stamp'] <= packet['query_now']:
   75             row = source_rows[source_index];state = state_from_json(row['state'])
   76             if state.metadata.epoch != history.epoch:history.reset(state.metadata.epoch)
   77             history.append(state, row['arrival_wall_monotonic']);source_index += 1
   78         poses = {}
   79         pose_equal = True
   80         for tag, stamp in packet['stamps'].items():
   81             pose = history.lookup(stamp, packet['query_now'], packet['query_wall'], epoch)
   82             pose_equal &= json.loads(json.dumps(asdict(pose))) == packet['poses'][tag]
   83             poses[tag] = restore_pose(packet['poses'][tag])
   84             valid &= stamp == messages[tag].header.stamp.to_sec() and stamp not in used_stamps[tag]
   85             used_stamps[tag].add(stamp)
   86         def metadata(data):
   87             data = dict(data);data['covariance_or_bounds'] = tuple(data['covariance_or_bounds'])
   88             return Metadata(**data)
   89         ml, mr, mc = (metadata(packet[tag + '_source_metadata']) for tag in ('left','right','lidar'))
   90         source_meta_equal = (ml == frames['left'].metadata and mr == frames['right'].metadata and mc == cloud.metadata)
   91         timing_valid = (ml.current(packet['selection_now'], epoch) and mr.current(packet['selection_now'], epoch)
   92             and abs(ml.stamp-mr.stamp) <= .005
   93             and max(abs(mc.stamp-ml.stamp), abs(mc.stamp-mr.stamp)) <= .02
   94             and abs(ml.stamp-mr.stamp) == packet['stereo_header_skew_s']
   95             and abs(mc.stamp-ml.stamp) == packet['left_lidar_header_skew_s']
   96             and abs(mc.stamp-mr.stamp) == packet['right_lidar_header_skew_s']
   97             and mc.current(packet['selection_now'],epoch) == packet['lidar_current_at_selection'])
   98         pair = StereoPair(Metadata(max(ml.stamp,mr.stamp),min(ml.valid_until,mr.valid_until),epoch,
   99                                   'retained_joint_stereo','front_stereo'),frames['left'],frames['right'],abs(ml.stamp-mr.stamp))
  100         disparity = compute_diagnostic_disparity(pair)
  101         luv,lz,ruv,rz,in_view = project_lidar_into_stereo(cloud.points,poses['lidar'],poses['left'],poses['right'],rig)
  102         visible = (sparse_visible(luv,lz,np.isfinite(luv).all(axis=1),rig.left.width,rig.left.height)
  103                    & sparse_visible(ruv,rz,np.isfinite(ruv).all(axis=1),rig.right.width,rig.right.height))
  104         sampled, compared = sample_disparity(luv,disparity.disparity_px,disparity.valid,visible)
  105         expected = np.full(len(luv),np.nan)
  106         expected[in_view] = luv[in_view,0]-ruv[in_view,0]
  107         compared[compared] = expected[compared] > 0
  108         residual = np.full(len(luv),np.nan)
  109         residual[compared] = sampled[compared]-expected[compared]
  110         np.savez_compressed(directory/'projection_diagnostics.npz', left_uv=luv,right_uv=ruv,
  111             in_view=in_view,sparse_visible=visible,comparison_valid=compared,
  112             expected_disparity_px=expected,measured_disparity_px=sampled,residual_px=residual)
  113         values = residual[compared]
  114         packet_valid = bool(exact_images and exact_cloud and pose_equal and source_meta_equal and timing_valid
  115             and not packet['current_control_eligible'] and not packet['per_point_motion_compensation_applied']
  116             and not packet['exposure_sync_verified'])
  117         valid &= packet_valid
  118         packets.append({'directory':entry['directory'],'input_records_valid':packet_valid,
  119             'images_match_cv_bridge':bool(exact_images),'cloud_matches_ros_decoder':bool(exact_cloud),
  120             'poses_match_source_replay':bool(pose_equal),'metadata_matches_sources':bool(source_meta_equal),
  121             'header_timing_valid':bool(timing_valid),'left_lidar_header_skew_s':abs(ml.stamp-mc.stamp),
  122             'right_lidar_header_skew_s':abs(mr.stamp-mc.stamp),'lidar_points':len(cloud.points),
  123             'points_in_both_views':int(in_view.sum()),'sparse_visible_points':int(visible.sum()),
  124             'comparison_points':int(compared.sum()),
  125             'disparity_residual_p10_p50_p90_px':[float(v) for v in np.percentile(values,(10,50,90))] if len(values) else None,
  126             'absolute_disparity_residual_median_px':float(np.median(np.abs(values))) if len(values) else None,
  127             'estimated_header_pose_alignment_applied':True,'per_point_motion_compensation_applied':False,
  128             'metric_calibration_verified':False,'visibility_verified':False,'real_scene_accuracy_verified':False})
  129     graph = summary.get('node_graph') or {}
  130     expected_inputs = sorted(('/airsim_node/initial_pose','/airsim_node/drone_1/gps','/airsim_node/drone_1/imu/imu',
  131         '/airsim_node/drone_1/front_left/Scene','/airsim_node/drone_1/front_right/Scene','/airsim_node/drone_1/lidar'))
  132     valid &= (graph.get('subscriptions') == expected_inputs and not set(graph.get('publishers',[]))-{'/rosout'}
  133               and not summary['control_published'] and not summary['debug_inputs_used']
  134               and len(raw_rows)==summary['raw_sensor_records'] and len(source_rows)==summary['source_states'])
  135     raw_counts = {tag:sum(r['sensor']==tag for r in raw_rows) for tag in ('imu','gps')}
  136     valid &= all(raw_counts[tag]==summary['received'].get(tag,0) for tag in raw_counts)
  137     imu_stamps = [r['stamp'] for r in raw_rows if r['sensor']=='imu']
  138     deltas = np.diff(imu_stamps)
  139     result = {'input_audit_valid':bool(valid),'captures':len(packets),'packets':packets,
  140         'source_states':len(source_rows),'raw_sensor_records':raw_counts,'raw_queue_drops':summary['raw_queue_drops'],
  141         'raw_recording_is_lossy':True,'raw_imu_max_stamp_gap_s':float(deltas.max()) if len(deltas) else None,
  142         'raw_imu_nonincreasing_records':int((deltas<=0).sum()),
  143         'nominal_settings_sha256':rig.settings_sha256,'exposure_sync_verified':False,
  144         'per_point_motion_compensation_applied':False,'metric_calibration_verified':False,
  145         'sparse_visibility_is_complete_proof':False,'world_projection_verified':False,
  146         'current_control_eligible':False,'acceptance_eligible':False,
  147         'interpretation':'Payload and history audit is distinct from geometry accuracy. Residuals compare diagnostic SGBM with nominal LiDAR projection after estimated header-pose alignment; sparse z-buffers do not prove visibility. Point timing, exposure and calibration remain unverified.',
  148         'analysis_source_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
  149     (run/'joint_audit.json').write_text(json.dumps(result,indent=2,allow_nan=False)+'\n')
  150     return result
