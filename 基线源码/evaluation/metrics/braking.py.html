    5 def measure_braking(telemetry,truth,brake_elapsed=6.,stop_speed=.08,hold_seconds=.35):
    6     result={'valid':False,'certified_deceleration_bound':False,
    7             'stop_speed_threshold_m_s':stop_speed,'hold_seconds':hold_seconds}
    8     commands=[r for r in telemetry if r.get('sim_elapsed',-1) >= brake_elapsed]
    9     ordered=sorted({r['stamp']:r for r in truth if r['stamp']>0}.values(),key=lambda r:r['stamp'])
   10     if not commands or len(ordered)<20:
   11         return {**result,'reason':'MISSING_DATA'}
   12     stamp=commands[0]['command']['stamp']
   13     times=np.array([r['stamp'] for r in ordered])
   14     positions=np.array([r['position'] for r in ordered])
   15     roi=(times >= stamp-.1) & (times <= stamp+4.5)
   16     times,positions=times[roi],positions[roi]
   17     if (len(times)<20 or times[0]>stamp or times[-1]<stamp+hold_seconds
   18             or np.max(np.diff(times))>.05):
   19         return {**result,'reason':'TRUTH_COVERAGE_OR_GAP'}
   20     vx=np.gradient(positions[:,0],times)
   21     v0=float(np.interp(stamp,times,vx))
   22     x0=float(np.interp(stamp,times,positions[:,0]))
   23     if v0 < .15:
   24         return {**result,'reason':'INSUFFICIENT_INITIAL_SPEED','initial_speed_m_s':v0}
   25     stop=None
   26     for i,t in enumerate(times):
   27         if t<stamp or t+hold_seconds>times[-1]:
   28             continue
   29         window=(times>=t)&(times<=t+hold_seconds)
   30         if np.all(np.abs(vx[window])<=stop_speed):
   31             stop=i
   32             break
   33     if stop is None:
   34         return {**result,'reason':'DID_NOT_SETTLE_WITHIN_OBSERVATION','initial_speed_m_s':v0}
   35     window=(times>=stamp)&(times<=times[stop]+hold_seconds)
   36     distance=float(max(0.,np.max(positions[window,0])-x0))
   37     if distance<=1e-5:
   38         return {**result,'reason':'INVALID_FORWARD_DISTANCE'}
   39     effective=max(0.,v0*v0-vx[stop]*vx[stop])/(2*distance)
   40     result.update(valid=True,reference_sensor_stamp=stamp,initial_speed_m_s=v0,
   41                   stop_time_s=float(times[stop]-stamp),forward_excursion_m=distance,
   42                   effective_average_deceleration_m_s2=float(effective),
   43                   peak_speed_after_reference_m_s=float(np.max(vx[window])),
   44                   lateral_excursion_m=float(np.ptp(positions[window,1])),
   45                   note='Includes controller/sensing lag; effective average is not an instantaneous lower bound.')
   46     return result
