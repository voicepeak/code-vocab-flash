    1 """Audit recorded online visual-motion timing and subscription boundaries."""
    2 import json
    3 import sys
    4 from collections import Counter
    5 from pathlib import Path
    6 
    7 from perception.stereo_depth.transport import freshness_reason
    8 from interfaces.contracts import Metadata
    9 
   10 
   11 def audit(run):
   12     run = Path(run)
   13     summary = json.loads((run / 'visual_motion_summary.json').read_text())
   14     rows = [json.loads(line) for line in (run / 'visual_motion_results.jsonl').read_text().splitlines()]
   15     errors = []
   16     if len(rows) != summary['results'] or Counter(row['status'] for row in rows) != Counter(summary['statuses']):
   17         errors.append('ROW_COUNT')
   18     for row in rows:
   19         if row['control_eligible'] or row['source_skew_s'] > .005:
   20             errors.append('SOURCE_OR_AUTHORITY')
   21         if row['status'] in ('ESTIMATED', 'GATE_CANDIDATES'):
   22             metadata = Metadata(row['source_stamp'], row['source_valid_until'], row['source_epoch'], 'audit')
   23             reason = freshness_reason(metadata, row['check_imu_stamp'], row['check_epoch'],
   24                 row['clock_fresh_at_check'], row['source_arrival_wall'], row['check_wall'])
   25             if reason != row['timing_rejection'] or row['timing_current'] != (reason is None):
   26                 errors.append('TIMING')
   27             if row['status'] == 'ESTIMATED' and (row['motion']['current_stamp'] != row['source_pair_stamp'] or row['motion']['control_eligible']):
   28                 errors.append('MOTION_IDENTITY')
   29             if row['status'] == 'GATE_CANDIDATES' and any(c['metadata']['stamp'] != row['source_stamp'] or c['metadata']['valid_until'] != row['source_valid_until'] or c['control_eligible'] for c in row['candidates']):
   30                 errors.append('GATE_IDENTITY')
   31     for row in rows:
   32         if 'tracking' not in row or row['tracking']['status'] != 'TRACKED':
   33             continue
   34         tracked = row['tracking']
   35         metadata = Metadata(tracked['source_stamp'], tracked['source_valid_until'], tracked['source_epoch'], 'audit')
   36         reason = freshness_reason(metadata, tracked['check_imu_stamp'], tracked['check_epoch'],
   37                                   tracked['clock_fresh_at_check'], tracked['source_arrival_wall'], tracked['check_wall'])
   38         if (reason is not None or tracked['pose_stamp'] != tracked['source_stamp']
   39                 or tracked['pose_bracket_stamps'][1] > tracked['check_imu_stamp']
   40                 or tracked['control_eligible'] or any(t['control_eligible'] or t['epoch'] != tracked['check_epoch']
   41                     or not t['last_stamp'] <= tracked['check_imu_stamp'] < t['last_valid_until']
   42                     for t in tracked['tracks'])):
   43             errors.append('TRACK_TIMING_OR_IDENTITY')
   44     if summary.get('pipeline') == 'gate_track' and Counter(row['tracking']['status'] for row in rows if 'tracking' in row) != Counter(summary['tracking_statuses']):
   45         errors.append('TRACK_COUNT')
   46     graph = summary.get('node_graph') or {}
   47     expected = sorted(['/airsim_node/drone_1/front_left/Scene', '/airsim_node/drone_1/front_right/Scene',
   48                        '/airsim_node/drone_1/imu/imu'])
   49     if summary.get('pipeline') == 'gate_track':
   50         expected = sorted(expected + ['/airsim_node/initial_pose', '/airsim_node/drone_1/gps'])
   51     if graph.get('subscriptions') != expected or set(graph.get('publishers', [])) - {'/rosout'}:
   52         errors.append('GRAPH')
   53     if summary['faults'] or summary['control_published'] or summary['debug_inputs_used']:
   54         errors.append('FAULT_OR_POLICY')
   55     timed = sum(row.get('timing_current', False) for row in rows)
   56     if timed != summary['timing_current_results']:
   57         errors.append('TIMING_COUNT')
   58     return {'audit_valid': bool(rows and not errors), 'errors': errors, 'rows': len(rows),
   59             'timing_current_results': timed, 'status_counts': dict(Counter(r['status'] for r in rows)),
   60             'control_eligible': False, 'acceptance_eligible': False}
   61 
   62 
   63 def main():
   64     run = Path(sys.argv[1]); result = audit(run)
   65     (run / 'visual_motion_audit.json').write_text(json.dumps(result, indent=2) + '\n')
   66     print(json.dumps(result, indent=2))
   67     return 0 if result['audit_valid'] else 2

   71     raise SystemExit(main())
