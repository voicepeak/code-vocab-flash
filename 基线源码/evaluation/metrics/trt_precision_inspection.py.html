    4 def inspect_geometry_precision(inspector, constraints):
    5     samplers = [row for row in constraints if row['type'] == 'LayerType.GRID_SAMPLE']
    6     if not samplers:
    7         raise ValueError('sampler constraints required')
    8     shapes = {tuple(row['coordinate_shape']) for row in samplers}
    9     expected = {row['name'] for row in samplers}
   10     matched = set()
   11     coordinates = []
   12     for layer in inspector['Layers']:
   13         metadata = layer.get('Metadata', '')
   14         names = {name for name in expected if '[ONNX Layer: ' + name + ']' in metadata}
   15         if not names:
   16             continue
   17         matched.update(names)
   18         for side in ('Inputs', 'Outputs'):
   19             for tensor in layer.get(side, []):
   20                 if tuple(tensor.get('Dimensions', ())) in shapes:
   21                     coordinates.append({'engine_layer': layer['Name'], 'side': side,
   22                         'tensor': tensor['Name'], 'format': tensor.get('Format/Datatype'),
   23                         'sampler_metadata': sorted(names)})
   24     rejected = [row for row in coordinates if row['format'] != 'Float']
   25     return {'expected_samplers': sorted(expected), 'matched_samplers': sorted(matched),
   26         'visible_coordinates': coordinates, 'non_fp32_coordinates': rejected,
   27         'visible_coordinate_formats_verified': bool(coordinates) and not rejected and matched == expected,
   28         'per_operation_internal_precision_trace': False,
   29         'interpretation': 'Visible Float coordinates and metadata coverage are necessary evidence; fused internals are not an instruction trace.'}
