   11 def numerical_agreement(reference,reference_valid,candidate,candidate_valid):
   12     common=reference_valid & candidate_valid;union=reference_valid | candidate_valid
   13     errors=np.abs(reference[common]-candidate[common])
   14     iou=float(common.sum()/union.sum()) if union.any() else 0.
   15     p99=float(np.percentile(errors,99)) if len(errors) else None
   16     return {'common_pixels':int(common.sum()),'mask_iou':iou,
   17         'absolute_median_px':float(np.median(errors)) if len(errors) else None,
   18         'absolute_p99_px':p99,'absolute_max_px':float(errors.max()) if len(errors) else None,
   19         'min_iou_gate':.9,'p99_error_gate_original_px':1.,
   20         'compatible':bool(len(errors) and np.isfinite(errors).all() and iou>=.9 and p99<=1.)}
