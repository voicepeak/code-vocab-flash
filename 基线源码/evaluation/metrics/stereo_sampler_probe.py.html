   10 def reference_1d(image,grid):
   11     """Float64 analytic zero-padded bilinear interpolation for H=1, y=0."""
   12     if (image.ndim!=4 or image.shape[2]!=1 or min(image.shape[:2])<1 or image.shape[3]<2
   13             or grid.ndim!=4 or grid.shape[0]!=image.shape[0] or grid.shape[1]!=1 or grid.shape[2]<1 or grid.shape[3]!=2
   14             or not np.isfinite(image).all() or not np.isfinite(grid).all() or np.any(grid[:,:,:,1]!=0)):
   15         raise ValueError('finite single-row, zero-y geometry sampler inputs required')
   16     n,c,_,width=image.shape
   17     x=(grid[:,0,:,0].astype(np.float64)+1)*(width-1)/2
   18     floor=np.floor(x).astype(np.int64);fraction=x-floor
   19     rows=np.arange(n)[:,None,None];channels=np.arange(c)[None,:,None]
   20     low=image[rows,channels,0,np.clip(floor,0,width-1)[:,None,:]].astype(np.float64)
   21     high=image[rows,channels,0,np.clip(floor+1,0,width-1)[:,None,:]].astype(np.float64)
   22     low*=((floor>=0)&(floor<width))[:,None,:]
   23     high*=((floor+1>=0)&(floor+1<width))[:,None,:]
   24     return ((1-fraction[:,None,:])*low+fraction[:,None,:]*high)[:,:,None,:]

   27 def errors(value,expected):
   28     if not np.isfinite(value).all() or not np.isfinite(expected).all():
   29         return {'mean_abs':None,'p99_abs':None,'max_abs':None,'finite':False,'absolute_gate':.0005,'passed':False}
   30     error=np.abs(value.astype(np.float64)-expected)
   31     return {'mean_abs':float(error.mean()),'p99_abs':float(np.percentile(error,99)),
   32         'max_abs':float(error.max()),'finite':bool(np.isfinite(value).all()),
   33         'absolute_gate':.0005,'passed':bool(np.isfinite(value).all() and error.max()<=.0005)}
