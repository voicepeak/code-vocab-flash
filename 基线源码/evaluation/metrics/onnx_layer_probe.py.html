   15 def difference(a,b):
   16     values=np.abs(a.astype(np.float64)-b.astype(np.float64))
   17     return {'mean_abs':float(values.mean()),'p99_abs':float(np.percentile(values,99)),
   18         'max_abs':float(values.max()),'reference_abs_max':float(np.abs(b).max()),
   19         'finite':bool(np.isfinite(a).all() and np.isfinite(b).all())}
