    5 def freeze_fourier_inputs(model,left,right):
    6     import torch
    7 
    8     selected=[(name,module) for name,module in model.named_modules()
    9         if type(module).__name__=='PositionalEncodingFourier' and hasattr(module,'token_projection')]
   10     if not selected:raise ValueError('no supported shape-only Fourier modules in checkpoint')
   11     captured={};hooks=[]
   12     for name,module in selected:
   13         def shape_hook(_,args,key=name):
   14             captured.setdefault(key,{})['shape']=tuple(int(v) for v in args[0])
   15         def grid_hook(_,args,key=name):
   16             captured.setdefault(key,{})['grid']=args[0].detach().clone()
   17         hooks.append(module.register_forward_pre_hook(shape_hook))
   18         hooks.append(module.token_projection.register_forward_pre_hook(grid_hook))
   19     try:
   20         original=model.forward(left,right,iters=4,test_mode=True,optimize_build_volume='pytorch1')
   21     finally:
   22         for hook in hooks:hook.remove()
   23 
   24     class FrozenFourierInput(torch.nn.Module):
   25         def __init__(self,projection,shape,grid):
   26             super().__init__();self.token_projection=projection;self.expected_shape=shape
   27             # clone() preserves the original channels-last strides and dtype.
   28             self.register_buffer('fourier_input',grid)
   29 
   30         def forward(self,shape):
   31             if not torch.jit.is_tracing() and tuple(int(v) for v in shape)!=self.expected_shape:
   32                 raise ValueError('frozen Fourier input shape changed')
   33             return self.token_projection(self.fourier_input)
   34 
   35     records=[]
   36     for name,module in selected:
   37         data=captured[name];grid=data['grid'];shape=data['shape']
   38         replacement=FrozenFourierInput(module.token_projection,shape,grid)
   39         parent,attribute=name.rsplit('.',1);setattr(model.get_submodule(parent),attribute,replacement)
   40         try:replacement((shape[0],shape[1]+1,shape[2]))
   41         except ValueError:shape_rejected=True
   42         else:shape_rejected=False
   43         if not shape_rejected:raise RuntimeError('frozen shape guard did not reject a different height')
   44         records.append({'module':name,'shape':list(shape),'input_shape':list(grid.shape),
   45             'input_stride':list(grid.stride()),'input_dtype':str(grid.dtype),
   46             'input_sha256':hashlib.sha256(grid.cpu().numpy().tobytes()).hexdigest(),
   47             'original_projection_preserved':replacement.token_projection is module.token_projection,
   48             'shape_mismatch_rejected':True})
   49     model.eval()
   50     return original,records
