   21 def benchmark(run):

   24     root = Path('/workspace/main')

   29     if before != config['input_sha256'] or digest(settings) != config['settings_sha256']:
   30         raise ValueError('historical image input or nominal settings changed')

   37         def image(side):
   38             stamp = row[side + '_stamp']
   39             return ImageFrame(Metadata(stamp, stamp + lifetime, row['epoch'], 'retained_front_' + side,
   40                                       'front_' + side + '_image'), row['encoding'], saved[side])
   41         left_frame, right_frame = image('left'), image('right')
   42     if (left_frame.pixels.shape[:2] != (rig.left.height, rig.left.width)
   43             or right_frame.pixels.shape[:2] != (rig.right.height, rig.right.width)):
   44         raise ValueError('saved image shape differs from nominal settings')
   45     def gray(frame):
   46         if frame.encoding == 'mono8':
   47             return frame.pixels[:, :, 0].copy()

   49                       'bgra8': cv2.COLOR_BGRA2GRAY, 'rgba8': cv2.COLOR_RGBA2GRAY}[frame.encoding]
   50         return cv2.cvtColor(frame.pixels, conversion)
   51     left_gray, right_gray = gray(left_frame), gray(right_frame)
   52     cv2.setNumThreads(1)
   53     outputs, durations = [], []
   54     for _ in range(2):
   55         cv2.setRNGSeed(0)
   56         begin = time.monotonic()
   57         outputs.append(measure_sparse_epipolar(left_gray, right_gray))
   58         durations.append(time.monotonic() - begin)
   59     report, left, right, inliers, fundamental = outputs[0]
   60     repeated = outputs[1]

   62                    and np.array_equal(right, repeated[2]) and np.array_equal(inliers, repeated[3])
   63                    and ((fundamental is None and repeated[4] is None)
   64                         or np.array_equal(fundamental, repeated[4])))

   66     nominal_valid &= inliers

   68     sensitivity[~nominal_valid] = np.nan

   70     np.savez_compressed(run / 'stereo_geometry_points.npz', left_uv=left, right_uv=right,
   71         fundamental_inliers=inliers, nominal_body_frd_m=body, nominal_valid=nominal_valid,
   72         nominal_depth_sensitivity_m_per_px=sensitivity)
   73     def quantiles(values):
   74         return [float(v) for v in np.percentile(values, (10, 50, 90))] if len(values) else None
   75     after = {name: digest(source / name) for name in before}
   76     report.update({'source_run': config['source_run'], 'historical_input_only': True,
   77         'opencv_version': cv2.__version__, 'opencv_threads': cv2.getNumThreads(), 'rng_seed': 0,
   78         'input_sha256': before, 'input_unchanged': before == after,
   79         'settings_unchanged': digest(settings) == config['settings_sha256'], 'nominal_rig': asdict(rig),
   80         'compute_durations_s': durations, 'repeat_deterministic': bool(repeat_same),
   81         'fundamental_matrix': fundamental.tolist() if fundamental is not None else None,
   82         'nominal_local_triangulated_points': int(nominal_valid.sum()),
   83         'nominal_optical_depth_p10_p50_p90_m': quantiles(z),
   84         'nominal_depth_sensitivity_p10_p50_p90_m_per_px': quantiles(sensitivity[nominal_valid]),
   85         'unknown_nominal_points_are_nan': bool(np.isnan(body[~nominal_valid]).all()),
   86         'exposure_sync_verified': False, 'real_scene_metric_accuracy_measured': False,
   87         'current_control_eligible': False, 'world_projection_applied': False,
   88         'interpretation': 'Sparse feature agreement supports an epipolar investigation, not full rectification or metric calibration. Nominal triangulation uses the public settings model. Depth sensitivity is an analytic derivative, not calibrated uncertainty. No ground-truth, LiDAR comparison, occupancy clearing or control use.'})
   89     (run / 'stereo_geometry_report.json').write_text(json.dumps(report, indent=2, allow_nan=False) + '\n')
   90     return 0 if repeat_same and report['input_unchanged'] and report['settings_unchanged'] else 2
