    7 def geometry_precision_scope(network, trt):
    8     """Follow sampling inputs back to learned convolution/matmul boundaries.
    9 
   10     Include coordinate arithmetic before the sampling layer: promoting an already
   11     rounded grid at the final sampler cannot recover its original coordinates.
   12     No learned convolution/matmul is promoted by this traversal.
   13     """
   14     layers = [network.get_layer(i) for i in range(network.num_layers)]
   15     producer = {layer.get_output(j).name: i for i, layer in enumerate(layers)
   16         for j in range(layer.num_outputs)}
   17     reasons = {}
   18     def walk(tensor, reason, seen):
   19         index = producer.get(tensor.name)
   20         if index is None or index in seen:
   21             return
   22         seen.add(index)
   23         layer = layers[index]
   24         if layer.type in (trt.LayerType.CONVOLUTION, trt.LayerType.MATRIX_MULTIPLY):
   25             return
   26         reasons.setdefault(index, set()).add(reason)
   27         for j in range(layer.num_inputs):
   28             value = layer.get_input(j)
   29             if value is not None and value.dtype in (trt.float32, trt.float16):
   30                 walk(value, reason, seen)
   31     for index, layer in enumerate(layers):
   32         if layer.type != trt.LayerType.GRID_SAMPLE:
   33             continue
   34         reasons.setdefault(index, set()).add('geometry_sampling')
   35         walk(layer.get_input(1), 'geometry_coordinates', set())
   36         walk(layer.get_input(0), 'geometry_values', set())
   37     if not any('geometry_sampling' in value for value in reasons.values()):
   38         raise RuntimeError('geometry policy requires actual GridSample layers')
   39     return reasons

   42 def apply_precision_policy(network, config, trt, policy, precision):
   43     if policy not in POLICIES:
   44         raise ValueError('unknown TensorRT precision policy')
   45     if policy == 'default':
   46         return []
   47     if precision != 'fp16':
   48         raise ValueError('non-default policies are mixed FP16/FP32 experiments')
   49 
   50     geometry = geometry_precision_scope(network, trt) if policy == 'sensitive_geometry_fp32' else {}
   51     selected = []
   52     for index in range(network.num_layers):
   53         layer = network.get_layer(index)
   54         layer_type = layer.type
   55         sensitive = layer_type in (trt.LayerType.NORMALIZATION, trt.LayerType.REDUCE, trt.LayerType.SOFTMAX)
   56         if layer_type == trt.LayerType.ACTIVATION:
   57             # get_layer() exposes ILayer in the pinned Python bindings.
   58             layer.__class__ = trt.IActivationLayer
   59             sensitive = layer.type in (trt.ActivationType.SIGMOID, trt.ActivationType.TANH)
   60         reasons = set(geometry.get(index, ()))
   61         if sensitive:
   62             reasons.add('sensitive_operation')
   63         if not reasons or layer_type == trt.LayerType.CONSTANT:
   64             continue
   65         outputs = [i for i in range(layer.num_outputs)
   66             if layer.get_output(i).dtype in (trt.float32, trt.float16)]
   67         if not outputs:
   68             continue  # Integer shape calculations must keep their type.
   69         # Constrain input precision even for movement layers: otherwise the
   70         # builder can round a grid to Half before casting its output to Float.
   71         layer.precision = trt.float32
   72         for output in outputs:
   73             layer.set_output_type(output, trt.float32)
   74         if layer_type == trt.LayerType.NORMALIZATION:
   75             layer.__class__ = trt.INormalizationLayer
   76             layer.compute_precision = trt.float32
   77         if not layer.precision_is_set or layer.precision != trt.float32:
   78             raise RuntimeError('FP32 layer constraint not retained: ' + layer.name)
   79         if any(not layer.output_type_is_set(i) or layer.get_output_type(i) != trt.float32 for i in outputs):
   80             raise RuntimeError('FP32 output constraint not retained: ' + layer.name)
   81         selected.append({'index': index, 'name': layer.name, 'type': str(layer_type),
   82             'fp32_output_indices': outputs, 'requested_precision': 'fp32',
   83             'selection_reasons': sorted(reasons),
   84             'coordinate_shape': list(layer.get_input(1).shape) if layer_type == trt.LayerType.GRID_SAMPLE else None,
   85             'normalization_compute_precision': 'fp32' if layer_type == trt.LayerType.NORMALIZATION else None})
   86     if not selected:
   87         raise RuntimeError('precision policy did not match any floating-point layers')
   88     # Fail a build when constraints cannot be honored; no silent precision fallback.
   89     config.set_flag(trt.BuilderFlag.OBEY_PRECISION_CONSTRAINTS)
   90     config.profiling_verbosity = trt.ProfilingVerbosity.DETAILED
   91     return selected
