   13 def audit(run):
   14     import rospy
   15     from sensor_msgs.msg import PointCloud2, PointField
   16     from sensor_msgs.point_cloud2 import read_points
   17 
   18     summary = json.loads((run / 'lidar_summary.json').read_text())
   19     schema = json.loads((run / 'lidar_first_message.json').read_text())
   20     message = PointCloud2()
   21     message.header.stamp = rospy.Time.from_sec(schema['stamp'])
   22     message.header.frame_id = schema['frame_id']
   23     for key in ('width', 'height', 'row_step', 'point_step', 'is_dense', 'is_bigendian'):
   24         setattr(message, key, schema[key])
   25     message.fields = [PointField(**f) for f in schema['fields']]
   26     message.data = (run / 'lidar_first_message.bin').read_bytes()
   27     rows = [json.loads(s) for s in (run / 'lidar_frames.jsonl').read_text().splitlines()]
   28     actual = decode_xyz(message, rows[0]['epoch'])
   29     reference = np.asarray(list(read_points(message, field_names=('x', 'y', 'z'), skip_nans=False)),
   30                            dtype=float).reshape(-1, 3)
   31     reference = reference[np.isfinite(reference).all(axis=1)]
   32     keys = {tuple(math.floor(float(x) / .25) for x in p) for p in reference
   33             if 0 < math.hypot(*p) < 30}
   34     snapshot = json.loads((run / 'lidar_endpoint_snapshot.json').read_text())
   35     cells = snapshot['cells']
   36     valid = (bool(rows) and summary['fault'] is None
   37              and len(rows) == summary['processed_frames']
   38              and np.array_equal(reference, actual.points)
   39              and len(keys) == rows[0]['occupied_voxels']
   40              and len(cells) == rows[-1]['occupied_voxels']
   41              and snapshot['metadata']['stamp'] == rows[-1]['stamp']
   42              and snapshot['metadata']['epoch'] == rows[-1]['epoch']
   43              and snapshot['metadata']['frame_id'] == 'lidar'
   44              and snapshot['resolution_m'] == .25
   45              and len({tuple(c['key']) for c in cells}) == len(cells)
   46              and all(c['status'] == 'OCCUPIED' and c['metadata'] == snapshot['metadata'] for c in cells)
   47              and all(r['free_voxels'] == 0 and not r['motion_compensated']
   48                      and not r['world_transform_applied'] and r['frame_id'] == 'lidar'
   49                      and r['input_points'] == r['finite_points'] + r['nonfinite_points']
   50                      and r['arrival_wall_monotonic'] <= r['processing_begin_wall_monotonic']
   51                      <= r['processing_end_wall_monotonic'] for r in rows)
   52              and not summary['control_published'] and not summary['debug_inputs_used']
   53              and summary['node_graph']['subscriptions'] == ['/airsim_node/drone_1/lidar']
   54              and not set(summary['node_graph']['publishers']) - {'/rosout'})
   55     durations = [r['processing_end_wall_monotonic'] - r['processing_begin_wall_monotonic'] for r in rows]
   56     arrivals = [r['arrival_wall_monotonic'] for r in rows]
   57     result = {'audit_valid': bool(valid), 'processed_frames': len(rows),
   58               'first_cloud_matches_ros_reference_decoder': bool(np.array_equal(reference, actual.points)),
   59               'first_cloud_finite_points': len(reference),
   60               'first_cloud_reference_endpoint_voxels': len(keys),
   61               'last_cloud_endpoint_voxels': len(cells),
   62               'processing_p50_p95_p99_max_s': [float(x) for x in np.percentile(durations, (50, 95, 99, 100))],
   63               'processed_arrival_span_s': arrivals[-1] - arrivals[0],
   64               'observed_processed_frequency_hz': (len(rows) - 1) / (arrivals[-1] - arrivals[0]) if len(rows) > 1 else None,
   65               'sensor_frame': 'lidar', 'world_transform_verified': False,
   66               'motion_compensated': False, 'free_space_certified': False,
   67               'flight_clearance_eligible': False, 'acceptance_eligible': False,
   68               'header_stamp_to_arrival_latency_calibrated': False,
   69               'analysis_source_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
   70     (run / 'lidar_audit.json').write_text(json.dumps(result, indent=2, allow_nan=False) + '\n')
   71     return result
   72 
   73 
   74 def main():
   75     parser = argparse.ArgumentParser(description=__doc__)
   76     parser.add_argument('run', type=Path)
   77     args = parser.parse_args()
   78     result = audit(args.run)
   79     print(json.dumps(result, indent=2))
