    8 REQUIRED = ('bridge_begin', 'bridge_end', 'parameter_borrow', 'command_constructed',
    9             'command_handoff', 'parameter_free_begin', 'parameter_free_end')

   12 def evaluate(records):
   13     counts = Counter(row['event'] for row in records)
   14     spans = defaultdict(list)
   15     frees = defaultdict(list)
   16     for row in records:
   17         if (row.get('mode') not in ('lifetime','native-input')
   18                 or not 0 <= row['start_monotonic_ns'] <= row['end_monotonic_ns']):
   19             raise ValueError('invalid lifetime record')
   20         if row['bridge_span']:
   21             spans[row['bridge_span']].append(row)
   22         if row['event'] == 'parameter_free_end':
   23             frees[row['parameter_address']].append(row)
   24     complete, handoffs, incomplete = 0, 0, 0
   25     violations = []
   26     for span, rows in spans.items():
   27         begin = [r for r in rows if r['event'] == 'bridge_begin']
   28         end = [r for r in rows if r['event'] == 'bridge_end']
   29         borrowed = [r for r in rows if r['event'] == 'parameter_borrow' and r['parameter_address'] != '0x0']
   30         if len(begin) != 1 or len(end) != 1:
   31             incomplete += 1
   32             continue
   33         complete += 1
   34         for borrow in borrowed:
   35             commands = [r for r in rows if r['event'] == 'command_constructed'
   36                         and r['parameter_address'] == borrow['parameter_address']
   37                         and borrow['end_monotonic_ns'] <= r['start_monotonic_ns']]
   38             for command in commands:
   39                 candidates = [r for r in rows if r['event'] == 'command_handoff'
   40                               and r['parameter_address'] == borrow['parameter_address']
   41                               and r['command_address'] == command['command_address'] != '0x0'
   42                               and r['thread_id'] == command['thread_id'] == borrow['thread_id']
   43                               and command['end_monotonic_ns'] <= r['start_monotonic_ns']
   44                               and begin[0]['start_monotonic_ns'] <= borrow['start_monotonic_ns']
   45                               and r['end_monotonic_ns'] <= end[0]['end_monotonic_ns']]
   46                 for handoff in candidates:
   47                     handoffs += 1
   48                     for free in frees[borrow['parameter_address']]:
   49                         # Strict ordering excludes a free from an earlier reuse of
   50                         # this address and calls overlapping the get() interval.
   51                         if (borrow['end_monotonic_ns'] <= free['start_monotonic_ns']
   52                                 and free['end_monotonic_ns'] <= handoff['start_monotonic_ns']):
   53                             violations.append({'bridge_span': span,
   54                                 'parameter_address': borrow['parameter_address'],
   55                                 'command_address': command['command_address'],
   56                                 'borrow': borrow, 'free': free, 'command_handoff': handoff,
   57                                 'free_to_handoff_ns': handoff['start_monotonic_ns'] - free['end_monotonic_ns']})
   58     return {'event_counts': {name: counts[name] for name in REQUIRED},
   59             'complete_bridge_spans': complete, 'incomplete_bridge_spans': incomplete,
   60             'linked_parameter_handoffs': handoffs,
   61             'freed_parameter_handoffs': len(violations), 'violations': violations,
   62             'hook_coverage_valid': all(counts[name] > 0 for name in REQUIRED) and handoffs > 0,
   63             'record_write_errors': max((r['write_errors'] for r in records), default=0),
   64             'actual_native_consumer_read_observed': False,
   65             'physical_impulse_cause_identified': False, 'acceptance_eligible': False,
   66             'interpretation': 'A violation means a parameter address was freed after get() returned and before its Command shared_ptr get() returned to the original native-call sequence. The probe does not observe the downstream payload read, allocator reuse, or applied actuator input. Instrumentation changes timing.'}

   69 def main():
   70     parser = argparse.ArgumentParser(description=__doc__)
   71     parser.add_argument('run', type=Path)
   72     args = parser.parse_args()
   73     records = [json.loads(line) for line in (args.run / 'bridge_probe.jsonl').read_text().splitlines()]
   74     result = evaluate(records)
   75     (args.run / 'bridge_lifetime_audit.json').write_text(json.dumps(result, indent=2) + '\n')
   76     print(json.dumps({k: v for k, v in result.items() if k != 'violations'}, indent=2))
   77     return 0 if result['hook_coverage_valid'] and not result['record_write_errors'] else 2
