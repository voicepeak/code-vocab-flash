    1 """Offline consistency diagnostics from IMU and recorded PWM commands.
    2 
    3 Motor coefficients and actuator consumption are unverified. Residuals locate discrepancies; they do not
    4 identify collision, wind, numerical faults, or prove an actuator latency bound.
    5 """
    6 import argparse
    7 import json
    8 from pathlib import Path
    9 import numpy as np
   10 from control.motor_allocator.allocator import MotorAllocator

   13 def pwm_samples(rows,model):
   14     samples={}
   15     for row in rows:
   16         data=row.get('data',{})
   17         header=data.get('header',{}).get('stamp',{})
   18         if 'secs' not in header:continue
   19         stamp=header['secs']+header.get('nsecs',0)*1e-9
   20         try:
   21             pwm=np.array([data['rotorPWM'+str(i)] for i in range(4)],dtype=float)
   22         except KeyError:continue
   23         if np.all(np.isfinite(pwm)) and np.all((pwm>=0)&(pwm<=1)):
   24             samples[stamp]=model.B @ (model.max_thrust*pwm)
   25     stamps=np.array(sorted(samples))
   26     return stamps,np.array([samples[t] for t in stamps])

   29 def interval_wrench(stamps,wrenches,start,end,max_gap=.04):
   30     """Integrate recorded PWM as a held model input; reject missing coverage."""
   31     index=int(np.searchsorted(stamps,start,side='right'))-1
   32     if index<0 or not start<end:return None
   33     if end>stamps[-1]+max_gap:return None
   34     total=np.zeros(4)
   35     cursor=start
   36     while cursor<end:
   37         if cursor-stamps[index]>max_gap:return None
   38         right=min(end,stamps[index+1] if index+1<len(stamps) else end)
   39         if right-stamps[index]>max_gap:return None
   40         total+=(right-cursor)*wrenches[index]
   41         cursor=right
   42         index+=1
   43     return total/(end-start)

   46 def evaluate(raw_rows,actuation,config):
   47     model=MotorAllocator(config)
   48     inertia=np.array(config['inertia_kg_m2'])
   49     stamps,wrenches=pwm_samples(actuation,model)
   50     raw=sorted({r['stamp']:r for r in raw_rows}.values(),key=lambda r:r['stamp'])
   51     residuals=[]
   52     for a,b in zip(raw,raw[1:]):
   53         dt=b['stamp']-a['stamp']
   54         if not .002<=dt<=.05 or not len(stamps):continue
   55         mean=interval_wrench(stamps,wrenches,a['stamp'],b['stamp'])
   56         if mean is None:continue
   57         wa,wb=np.array(a['omega_body']),np.array(b['omega_body'])
   58         omega=(wa+wb)*.5
   59         angular=inertia*(wb-wa)/dt+np.cross(omega,inertia*omega)-mean[1:]
   60         # Trapezoidal endpoint approximation, not a collocated force sensor.
   61         specific=(np.array(a['acceleration_body'])+b['acceleration_body'])*.5
   62         linear=specific-np.array((0.,0.,-mean[0]/config['mass_kg']))
   63         residuals.append({'stamp':b['stamp'],'interval_s':dt,
   64             'angular_residual_Nm':angular.tolist(),'angular_norm_Nm':float(np.linalg.norm(angular)),
   65             'specific_force_residual_m_s2':linear.tolist(),'mean_recorded_pwm_model_wrench':mean.tolist(),
   66             'omega_body_rad_s':wb.tolist()})
   67     result={'matched_intervals':len(residuals),'candidate_intervals':max(0,len(raw)-1),
   68         'evaluation_only':True,'cause_identified':False,'motor_model_status':config['parameter_status'],
   69         'actuation_latency_verified':False,'actuator_consumption_verified':False,
   70         'debug_pwm_temporal_fidelity_verified':False,'official_result':'UNKNOWN'}
   71     if residuals:
   72         magnitudes=[r['angular_norm_Nm'] for r in residuals]
   73         result['angular_residual_p50_p95_p99_max_Nm']=np.percentile(magnitudes,[50,95,99,100]).tolist()
   74         result['largest_angular_discrepancies']=sorted(residuals,key=lambda r:r['angular_norm_Nm'],reverse=True)[:5]
   75     return result

   78 def main():
   79     parser=argparse.ArgumentParser(description=__doc__)
   80     parser.add_argument('run',type=Path)
   81     args=parser.parse_args()
   82     def rows(name):return [json.loads(line) for line in (args.run/name).read_text().splitlines()]
   83     imu_file=args.run/'evaluation_imu.jsonl'
   84     raw=rows('evaluation_imu.jsonl') if imu_file.exists() else [r['raw'] for r in rows('pwm_telemetry.jsonl')]
   85     telemetry=rows('pwm_telemetry.jsonl')
   86     start=telemetry[0]['command']['stamp'] if telemetry else float('inf')
   87     end=telemetry[-1]['command']['stamp'] if telemetry else float('-inf')
   88     fault=args.run/'fault_observation.json'
   89     if fault.exists():
   90         value=json.loads(fault.read_text())
   91         if value.get('fresh_read_in_fault_iteration') and value.get('raw'):
   92             raw.append(value['raw']);end=max(end,value['raw']['stamp'])
   93     raw=[r for r in raw if start<=r['stamp']<=end]
   94     root=Path(__file__).resolve().parents[2]
   95     model_path=args.run/'vehicle_model.json'
   96     if not model_path.exists():
   97         # Historical runs did not emit their effective controller model. Refuse
   98         # silent analysis with subsequently changed workspace coefficients.
   99         from evaluation.seed_runner.manifest import file_hash
  100         manifest=json.loads((args.run/'manifest.json').read_text())
  101         model_path=root/'configs/vehicle_identification.json'
  102         expected=manifest['source_sha256_before']['configs/vehicle_identification.json']
  103         if file_hash(model_path)!=expected:raise ValueError('historical motor model changed')
  104     wire=args.run/'evaluation_pwm_commands.jsonl'
  105     result=evaluate(raw,rows(wire.name if wire.exists() else 'evaluation_actuation.jsonl'),json.loads(model_path.read_text()))
  106     result['pwm_source']='independent_ros_command_recorder' if wire.exists() else 'unverified_debug_pwm_record'
  107     result['model_provenance']='controller_emitted_snapshot' if model_path.parent==args.run else 'historical_source_hash_match'
  108     result['imu_source']='independent_full_rate_recorder' if imu_file.exists() else 'controller_sampled_imu'
  109     (args.run/'physics_residual.json').write_text(json.dumps(result,indent=2)+'\n')
  110     print(json.dumps(result,indent=2))
  111     return 0 if result['matched_intervals'] else 2
