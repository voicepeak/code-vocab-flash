    1 """Audit timing and read-only policy of dynamic LiDAR diagnostics."""
    2 from collections import Counter
    3 import json
    4 from pathlib import Path
    5 import sys

    7 from interfaces.contracts import Metadata
    8 from perception.stereo_depth.transport import freshness_reason

   11 def audit(run):
   12     run = Path(run)
   13     summary = json.loads((run / 'dynamic_lidar_summary.json').read_text())
   14     rows = [json.loads(line) for line in (run / 'dynamic_lidar_results.jsonl').read_text().splitlines()]
   15     errors = []
   16     expected = sorted(['/airsim_node/drone_1/lidar', '/airsim_node/drone_1/gps',
   17                        '/airsim_node/drone_1/imu/imu', '/airsim_node/initial_pose'])
   18     if summary.get('planning_diagnostic'): expected.append('/airsim_node/end_goal'); expected.sort()
   19     graph = summary['node_graph'] or {}
   20     if graph.get('subscriptions') != expected or set(graph.get('publishers', [])) - {'/rosout'}:
   21         errors.append('GRAPH')
   22     if (summary['faults'] or summary['control_published'] or summary['debug_inputs_used']
   23             or summary['control_eligible'] or summary['acceptance_eligible']
   24             or len(rows) != summary['results'] or dict(Counter(r['status'] for r in rows)) != summary['statuses']):
   25         errors.append('SUMMARY')
   26     for row in rows:
   27         if row.get('control_eligible') is not False:
   28             errors.append('CONTROL_ELIGIBILITY')
   29         planner = row.get('planner')
   30         if summary.get('planning_diagnostic') and row['status'] == 'PROCESSED' and planner is None:
   31             errors.append('PLANNER_MISSING')
   32         if planner and planner.get('control_eligible', False):
   33             errors.append('PLANNER_CONTROL_ELIGIBILITY')
   34         if planner and 'goal' in planner and (planner['goal']['epoch'] != row['check_epoch']
   35                 or not 0 <= row['check_wall']-planner['goal']['last_arrival_wall'] < .5):
   36             errors.append('PLANNER_GOAL_STALE')
   37         if row['status'] != 'PROCESSED':
   38             continue
   39         meta = Metadata(row['source_stamp'], row['source_valid_until'], row['source_epoch'], 'audit')
   40         if freshness_reason(meta, row['check_imu_stamp'], row['check_epoch'], row['clock_fresh_at_check'],
   41                             row['source_arrival_wall'], row['check_wall']):
   42             errors.append('SOURCE_EXPIRED')
   43         if (row['pose_stamp'] != row['source_stamp'] or row['pose_bracket_stamps'][1] > row['check_imu_stamp']
   44                 or row['dynamic_check']['safety_certified']):
   45             errors.append('POSE_OR_CERTIFICATION')
   46         for track in row['tracks']:
   47             if (track['epoch'] != row['check_epoch'] or not track['stamp'] <= row['check_imu_stamp'] < track['valid_until']
   48                     or track['safety_certified'] or track['control_eligible']):
   49                 errors.append('TRACK_INVALID')
   50     return {'audit_valid': bool(rows and not errors), 'errors': errors,
   51             'rows': len(rows), 'statuses': dict(Counter(r['status'] for r in rows)),
   52             'rejection_reasons': dict(Counter(r['reason'] for r in rows if r['status'] == 'REJECTED')),
   53             'control_eligible': False, 'safety_certified': False}
   54 
   55 
   56 def main():
   57     run = Path(sys.argv[1]); result = audit(run)
   58     (run / 'dynamic_lidar_audit.json').write_text(json.dumps(result, indent=2) + '\n')
   59     print(json.dumps(result, indent=2))
   60     return 0 if result['audit_valid'] else 2
