    7 class BrakingMeasurementTests(unittest.TestCase):
    8     def truth(self):
    9         rows=[]
   10         for t in np.arange(3.,11.,.01):
   11             if t<4:x=0.
   12             elif t<6:x=.4*(t-4)
   13             else:
   14                 dt=min(.8,t-6)
   15                 x=.8+.4*dt-.25*dt*dt
   16             rows.append({'stamp':float(t),'position':(float(x),0.,-1.)})
   17         return rows
   18 
   19     def test_constant_deceleration_response_and_finite_stop(self):
   20         command=[{'sim_elapsed':6.,'command':{'stamp':6.}}]
   21         r=measure_braking(command,self.truth())
   22         self.assertTrue(r['valid'])
   23         self.assertFalse(r['certified_deceleration_bound'])
   24         self.assertAlmostEqual(r['forward_excursion_m'],.16,delta=.01)
   25         self.assertAlmostEqual(r['stop_time_s'],.64,delta=.05)
   26         self.assertGreater(r['effective_average_deceleration_m_s2'],.4)
   27         self.assertLess(r['effective_average_deceleration_m_s2'],.55)
   28 
   29     def test_gaps_and_missing_braking_phase_cannot_create_a_bound(self):
   30         command=[{'sim_elapsed':6.,'command':{'stamp':6.}}]
   31         self.assertFalse(measure_braking([],self.truth())['valid'])
   32         rows=[r for r in self.truth() if not 6.<r['stamp']<6.3]
   33         self.assertFalse(measure_braking(command,rows)['valid'])
   34 
   35     def test_reference_velocity_changes_at_stop_with_continuous_position(self):
   36         before=benchmark_reference((0.,0.,0.),5.999,braking=True)
   37         after=benchmark_reference((0.,0.,0.),6.,braking=True)
   38         self.assertAlmostEqual(after[0][0],.8)
   39         self.assertAlmostEqual(before[1][0],.4)
   40         self.assertEqual(after[1][0],0.)
   41         self.assertLess(abs(before[0][0]-after[0][0]),.001)
