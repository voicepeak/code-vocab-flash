   49     def test_allocator_saturates_and_reports_achieved_wrench(self):
   50         a = MotorAllocator(config())
   51         for thrust, moment in ((8.829, (2., 2., 1.)), (100., (0., 0., 0.)), (0., (1., 0., 0.))):
   52             result = a.allocate(thrust, moment)
   53             self.assertTrue(result.saturated)
   54             self.assertTrue(all(0 <= v <= .6 for v in result.pwm))
   55             np.testing.assert_allclose(result.achieved_wrench, a.B @ (np.array(result.pwm)*a.max_thrust))
   56         result = a.allocate(8.829, (.01, -.01, .005))
   57         self.assertFalse(result.saturated)
   58         np.testing.assert_allclose(result.achieved_wrench, (8.829, .01, -.01, .005), atol=1e-10)

   69     def test_allocator_reports_small_limit_excess_as_saturation(self):
   70         allocator = MotorAllocator(config())
   71         limit = allocator.max_thrust * allocator.ceiling
   72         # Relative closeness of the achieved and requested wrench must not
   73         # conceal a real reduction in collective thrust.
   74         requested = 4. * limit * (1. + 1e-6)
   75         clipped = allocator.allocate(requested, (0., 0., 0.))
   76         self.assertLess(clipped.achieved_wrench[0], requested)
   77         self.assertTrue(clipped.saturated)
   78         self.assertFalse(allocator.allocate(4. * limit, (0., 0., 0.)).saturated)
   79 
   80         # A small moment reduction at a motor boundary is also saturation.
   81         forces = np.array((limit * (1. + 1e-6), limit * .5,
   82                            limit * .5, limit * (1. - 1e-6)))
   83         wrench = allocator.B @ forces
   84         result = allocator.allocate(wrench[0], wrench[1:])
   85         self.assertLess(result.moment_scale, 1.)
   86         self.assertTrue(result.saturated)

   88     def test_small_angle_closed_loop_roll_converges(self):
