   18         return Trajectory(Metadata(10.,end,0,'trajectory'),tuple(TrajectorySample(t,p,(0.,0.,0.),(0.,0.,0.),0.) for t,p in ((10.,a),(end,b))))
   19     def track(self,center=(0.,-2.,0.),velocity=(0.,10.,0.),observations=2):
