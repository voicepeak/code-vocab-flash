   13 def message(points=((1., 2., 3.),), big=False, dtype=7, width=None):
   14     """Owned cloud with reordered fields, point padding, and row padding."""
   15     width = len(points) if width is None else width
   16     height = len(points) // width if width else 1
   17     size = 4 if dtype == 7 else 8
   18     step = 4 * size
   19     row_step = step * width + 8
   20     data = bytearray(row_step * height)
   21     fields = [NS(name=n, offset=o * size, datatype=dtype, count=1)
   22               for n, o in (('z', 0), ('x', 2), ('y', 1))]
   23     for i, (x, y, z) in enumerate(points):
   24         struct.pack_into(('>' if big else '<') + ('4f' if dtype == 7 else '4d'), data,
   25                          (i // width) * row_step + (i % width) * step, z, y, x, 999.)
   26     return NS(header=NS(stamp=NS(to_sec=lambda: 10.), frame_id='lidar'), width=width,
   27               height=height, point_step=step, row_step=row_step, fields=fields,
   28               is_bigendian=big, data=bytes(data), is_dense=True)
