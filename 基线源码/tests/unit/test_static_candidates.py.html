   12 def scene(wall=False, empty=False):
   13     meta = Metadata(10., 30., 0, 'synthetic_static_world')
   14     cells = tuple(Cell(key, Occupancy.OCCUPIED if wall and key == (1, 0, 0) else Occupancy.FREE, meta)
   15                   for key in product(range(-2, 6), range(-4, 6), range(-1, 3)))
   16     grid = OccupancySnapshot(meta, 1., () if empty else cells)
   17     state = StateEstimate(replace(meta, valid_until=10.2), (.5, .5, .5), (0., 0., 0.),
   18         (1., 0., 0., 0.), (0., 0., 0.), Health.OK)
   19     initial = TrajectorySample(10., state.position_ned, state.velocity_ned, (0., 0., 0.), 0.)
   20     limits = LocalPlannerLimits(SweepBounds((.1, .1, .1), .02, 2.), 1.2)
   21     return state, initial, grid, limits

   24 def plan(wall=False, empty=False, changes=None):
   25     state, initial, grid, limits = scene(wall, empty)
   26     values = {'state': state, 'initial_reference': initial, 'snapshot': grid, 'limits': limits,
   27               'goal': (2.5, .5, .5), 'now': 10., 'epoch': 0}
   28     values.update(changes or {})
   29     return plan_static_local(**values)

   33     def test_quintic_preserves_initial_pva_and_stops_at_target(self):
   34         initial = TrajectorySample(10., (.2, -.3, .5), (.3, .1, -.2), (.1, -.2, .3), .4)
   35         curve = QuinticReference.to_rest(initial, (2., 1., .7), 2.)
   36         start, end = curve.sample(10.), curve.sample(12.)
   37         for name in ('position_ned', 'velocity_ned', 'acceleration_ned'):
   38             for actual, expected in zip(getattr(start, name), getattr(initial, name)):
   39                 self.assertAlmostEqual(actual, expected, places=13)
   40         self.assertEqual(end.position_ned, (2., 1., .7))
   41         self.assertEqual(end.velocity_ned, (0., 0., 0.))
   42         self.assertEqual(end.acceleration_ned, (0., 0., 0.))

   52     def test_continuous_derivative_bounds_enclose_dense_samples(self):
   53         initial = TrajectorySample(10., (0., 0., 0.), (.2, -.1, 0.), (.1, .1, 0.), 0.)
   54         curve = QuinticReference.to_rest(initial, (2., 1., -.5), 3.)
   55         speed, acceleration = curve.derivative_bounds()
   56         for i in range(501):
   57             value = curve.sample(curve.stamp + i * 3. / 500)
   58             self.assertLessEqual(math.hypot(*value.velocity_ned), speed + 1e-12)
   59             self.assertLessEqual(math.hypot(*value.acceleration_ned), acceleration + 1e-12)
