   10 class ReferenceIsolationTests(unittest.TestCase):
   11     def test_blocked_process_cannot_refresh_old_output_or_block_control_reader(self):
   12         worker=ReferenceWorker(BenchmarkSpec((0.,0.,-1.),0.),stall_seconds=.3)
   13         worker.start()
   14         try:
   15             deadline=time.monotonic()+3
   16             while time.monotonic()<deadline:
   17                 worker.submit(10.,0,0.)
   18                 packet,reason=worker.snapshot(10.,0)
   19                 if packet is not None:break
   20                 time.sleep(.005)
   21             else:self.fail('reference process did not start')
   22             deadline=time.monotonic()+1.5
   23             durations=[]
   24             while time.monotonic()<deadline and worker.max_compute<.3:
   25                 worker.submit(10.2,0,6.2)
   26                 begin=time.monotonic()
   27                 packet,reason=worker.snapshot(10.5,0)
   28                 durations.append(time.monotonic()-begin)
   29                 self.assertIsNone(packet)
   30                 time.sleep(.005)
   31             self.assertTrue(worker.stall_observed)
   32             self.assertGreaterEqual(worker.max_compute,.3)
   33             self.assertLess(max(durations),.05)
   34             deadline=time.monotonic()+1.
   35             while time.monotonic()<deadline:
   36                 worker.submit(10.5,0,6.5)
   37                 packet,reason=worker.snapshot(10.5,0)
   38                 if packet is not None:break
   39                 time.sleep(.005)
   40             else:self.fail('worker did not recover after bounded stall')
   41             self.assertEqual(worker.snapshot(10.5,1)[1],'REFERENCE_EPOCH')
   42         finally:worker.close()
   43         self.assertFalse(worker.summary()['alive_after_close'])
