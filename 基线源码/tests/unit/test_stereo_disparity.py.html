   10 def pair(left, right):
   11     buffer = StereoPairBuffer()
   12     for side, image in (('left', left), ('right', right)):
   13         frame = ImageFrame(Metadata(10., 10.2, 0, 'synthetic', 'front_' + side + '_image'),
   14                            'mono8', image[:, :, None])
   15         buffer.add(side, frame, 20.)
   16     return buffer.pop(10.01, 20.01)

   19 def shifted(disparity=8):
   20     rng = np.random.RandomState(127)
   21     left = rng.randint(0, 256, (96, 320)).astype(np.uint8)
   22     right = rng.randint(0, 256, left.shape).astype(np.uint8)
   23     right[:, :-disparity] = left[:, disparity:]
   24     return left, right

   27 class StereoDisparityTests(unittest.TestCase):
   28     def test_known_pixel_shift_recovered_without_metric_or_rectification_claim(self):
   29         for shift in (8, 16):
   30             result = compute_diagnostic_disparity(pair(*shifted(shift)), SgbmConfig(num_disparities=32))
   31             region = result.disparity_px[10:-10, 48:-48]
   32             valid = result.valid[10:-10, 48:-48]
   33             self.assertGreater(valid.mean(), .9)
   34             self.assertLess(np.sqrt(np.mean((region[valid] - shift) ** 2)), .2)
   35             self.assertFalse(result.metric_depth_verified)
   36             self.assertFalse(result.rectification_verified)
   37             self.assertFalse(result.acceptance_eligible)
   38             self.assertEqual(result.metadata.valid_until, 10.2)
   39             with self.assertRaises(ValueError): result.disparity_px.setflags(write=True)
   40 
   41     def test_uniform_and_occluded_regions_stay_unknown(self):
   42         blank = np.full((96, 320), 127, np.uint8)
   43         empty = compute_diagnostic_disparity(pair(blank, blank), SgbmConfig(num_disparities=32))
   44         self.assertFalse(empty.valid.any())
   45         self.assertTrue(np.isnan(empty.disparity_px).all())
   46         left, right = shifted()
   47         right[:, 120:180] = 0
   48         result = compute_diagnostic_disparity(pair(left, right), SgbmConfig(num_disparities=32))
   49         self.assertFalse(result.valid[10:-10, 135:175].any())
   50         self.assertTrue(np.isnan(result.disparity_px[10:-10, 135:175]).all())
   51 
   52     def test_inconsistent_vertical_correspondence_is_rejected(self):
   53         left, right = shifted()

   56         self.assertLess(result.valid[10:-10, 48:-48].mean(), .05)
