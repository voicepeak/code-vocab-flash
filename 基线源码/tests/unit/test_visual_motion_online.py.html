   31     def test_expiry_during_inference_clears_anchor(self):
   32         clocks = iter([(1.1,0,True), (1.3,0,True)])
   33         window = MotionWindow(None, estimator=lambda a,b,rig,method:MockMotion(a.metadata.stamp,b.metadata.stamp))

   36         self.assertFalse(result['timing_current']); self.assertEqual(result['timing_rejection'],'SOURCE_LEASE_NOT_CURRENT')
