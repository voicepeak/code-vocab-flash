   15 class StereoGateTests(unittest.TestCase):
   16     def setUp(self):
   17         self.rig = load_nominal_rig(Path(__file__).resolve().parents[2] / 'configs/simulator_settings.json')
   18         self.image = np.zeros((720,960,3),np.uint8)
   19         cv2.rectangle(self.image,(350,250),(355,450),(0,0,255),-1)
   20         cv2.rectangle(self.image,(600,250),(605,450),(0,0,255),-1)
   21         l = ImageFrame(Metadata(10.,10.2,0,'left','front_left_image'),'bgr8',self.image)
   22         r = ImageFrame(Metadata(10.,10.2,0,'right','front_right_image'),'bgr8',self.image)
   23         self.pair = StereoPair(Metadata(10.,10.2,0,'pair','front_stereo'),l,r,0.)
   24         valid = np.ones((720,960),dtype=bool); depth = np.full((720,960),12.,dtype=np.float32)
   25         self.disparity = DisparityObservation(Metadata(10.,10.2,0,'depth','front_left_image'),depth,valid,10.,10.)
   26 
   27     def test_valid_nominal_pair_and_no_control_eligibility(self):
   28         out=stereo_gate_candidates(self.pair,self.disparity,self.rig,10.05,0)
   29         self.assertEqual(len(out),1)
   30         self.assertGreater(out[0].center_body_frd_m[0],20.)
   31         self.assertFalse(out[0].control_eligible);self.assertFalse(out[0].aperture_verified)
   32         self.assertEqual(out[0].metadata.valid_until,10.2)
   33 
   34     def test_shared_edge_cache_matches_and_rejects_different_frame(self):
   35         prepared=prepare_gate_edges(self.pair)
   36         baseline=stereo_gate_candidates(self.pair,self.disparity,self.rig,10.05,0)
   37         shared=stereo_gate_candidates(self.pair,self.disparity,self.rig,10.05,0,prepared=prepared)
   38         self.assertEqual(baseline,shared)
   39         new_pair=replace(self.pair,left=replace(self.pair.left,pixels=self.image))
   40         with self.assertRaisesRegex(ValueError,'another image'):
   41             stereo_gate_candidates(new_pair,self.disparity,self.rig,10.05,0,prepared=prepared)

   43     def test_empty_depth_and_border_cropped_frame_are_not_promoted(self):
