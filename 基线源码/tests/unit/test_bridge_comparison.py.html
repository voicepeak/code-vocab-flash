    9 class BridgeComparisonTests(unittest.TestCase):
   10     def test_rotating_order_keeps_every_seed_mode_and_repeat(self):
   11         protocol = {'schema_version': 1, 'seeds': [2024, 2041], 'repeats': 3,
   12                     'hover_seconds': 14, 'stationary_bootstrap': True}
   13         cases = case_order(protocol)
   14         self.assertEqual(len(cases), 18)
   15         for seed in protocol['seeds']:
   16             for repeat in range(3):
   17                 group = [c['mode'] for c in cases if c['seed'] == seed and c['repeat'] == repeat]
   18                 self.assertEqual(set(group), set(MODES))
   19             first = [c['mode'] for i, c in enumerate(cases) if i % 3 == 0 and c['seed'] == seed]
   20             self.assertEqual(set(first), set(MODES))
   21         for change in ({'repeats': 0}, {'repeats': True}, {'seeds': [True]}, {'seeds': []},
   22                        {'hover_seconds': 300}, {'hover_seconds': float('nan')},
   23                        {'modes': ['lifetime']}, {'modes': ['none', 'none']},
   24                        {'stationary_bootstrap': False}):
   25             with self.subTest(change=change), self.assertRaises(ValueError):
   26                 case_order(dict(protocol, **change))

   37     def test_missing_trace_and_control_fault_are_reported_separately(self):
   38         with tempfile.TemporaryDirectory() as directory:
   39             run = Path(directory)
   40             (run / 'manifest.json').write_text(json.dumps({
   41                 'run_state': 'FAILED', 'observer_exit_code': 0, 'command_trace_audit_exit_code': 2}))
   42             (run / 'pwm_summary.json').write_text(json.dumps({'commands': 100, 'fault': None}))
   43             result = collect_case(run)
   44             self.assertTrue(result['controller_completed'])
   45             self.assertEqual(result['command_trace_audit_exit_code'], 2)
   46             self.assertFalse(result['evaluation_valid'])
   47             (run / 'pwm_summary.json').write_text(json.dumps({'commands': 100, 'fault': 'ANGULAR_RATE_LIMIT'}))
   48             result = collect_case(run)
   49             self.assertFalse(result['controller_completed'])
   50             self.assertEqual(result['controller_fault'], 'ANGULAR_RATE_LIMIT')
   51             (run / 'pwm_summary.json').unlink()
   52             self.assertEqual(collect_case(run)['controller_fault'], 'MISSING_SUMMARY')
