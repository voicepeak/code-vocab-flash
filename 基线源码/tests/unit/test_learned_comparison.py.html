    8 class LearnedComparisonTests(unittest.TestCase):
    9     def test_errors_use_identical_shared_points_without_filtering_large_residuals(self):
   10         uv=np.array(((3.2,3.2),(6.2,6.2),(10.2,10.2)))
   11         expected=np.full(3,10.)

   13         b[10:12,10:12]=100.

   15         a[~va]=np.nan;b[~vb]=np.nan
   16         result,masks=compare_reference(uv,expected,np.ones(3,bool),{'a':(a,va),'b':(b,vb)})
   17         np.testing.assert_array_equal(masks['a'],(True,True,False))
   18         np.testing.assert_array_equal(masks['b'],(False,True,True))
   19         self.assertEqual(result['reference_candidates'],3)

   21         self.assertEqual(result['methods']['b']['shared_support']['count'],1)
   22         self.assertEqual(result['methods']['a']['shared_support']['absolute_median_px'],0.)
   23         self.assertAlmostEqual(result['methods']['b']['shared_support']['absolute_median_px'],2.)
   24         self.assertGreater(result['methods']['b']['own_support']['absolute_p90_px'],80.)
   25
