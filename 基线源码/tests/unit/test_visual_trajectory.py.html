   13 class VisualTrajectoryTests(unittest.TestCase):
   14     def test_composes_body_translation_after_yaw(self):
   15         chain = VisualTrajectory(1., 0)
   16         yaw = ((0.,-1.,0.),(1.,0.,0.),(0.,0.,1.))
   17         chain.append(motion(1., 1.1, (1.,0.,0.), yaw))
   18         result = chain.append(motion(1.1, 1.2, (1.,0.,0.), np.eye(3)))
   19         np.testing.assert_allclose(result.position_initial_body_m, (1.,1.,0.))
   20         self.assertEqual(result.steps, 2)
   21         self.assertFalse(result.control_eligible)

   23     def test_gap_or_invalid_rotation_latches_break(self):
   24         for bad in (motion(1.1,1.2,(0.,0.,0.),np.eye(3)),
   25                     motion(1.,1.1,(0.,0.,0.),np.diag((1.,1.,-1.)))):
   26             chain = VisualTrajectory(1., 0)
   27             with self.assertRaises(ValueError): chain.append(bad)
   28             with self.assertRaises(ValueError): chain.append(motion(1.,1.1,(0.,0.,0.),np.eye(3)))
   29             self.assertEqual(chain.snapshot().steps, 0)
