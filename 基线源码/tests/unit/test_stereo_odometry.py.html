   18     def scene(self):

   22     def test_half_scale_preserves_stereo_metric_geometry(self):
   23         rig = half_scale_rig(self.rig)
   24         points = self.scene()
   25         left, _, visible_left = self.rig.left.project_body(points)
   26         right, _, visible_right = self.rig.right.project_body(points)
   27         visible = visible_left & visible_right
   28         left = (left[visible] + .5) * .5 - .5
   29         right = (right[visible] + .5) * .5 - .5
   30         recovered, valid, _ = rig.triangulate_body(left, right)
   31         self.assertTrue(valid.all())
   32         np.testing.assert_allclose(recovered, points[visible], atol=1e-10)

   34     def test_known_body_motion_and_camera_lever_arm(self):
   35         points = self.scene()
   36         yaw = .05
   37         rotation = np.array(((np.cos(yaw), -np.sin(yaw), 0.), (np.sin(yaw), np.cos(yaw), 0.), (0., 0., 1.)))
   38         displacement = np.array((.25, -.12, .06))
   39         current = (points - displacement) @ rotation
   40         uv, _, valid = self.rig.left.project_body(current)
   41         result, found_rotation, count, p95 = solve_relative(points[valid], uv[valid], self.rig)
   42         np.testing.assert_allclose(result, displacement, atol=.005)
   43         np.testing.assert_allclose(found_rotation, rotation, atol=.001)
   44         self.assertGreater(count, 50)
   45         self.assertLess(p95, .1)

   47     def test_outliers_are_rejected_and_collinear_geometry_fails(self):
   48         points = self.scene()
   49         uv, _, valid = self.rig.left.project_body(points)
   50         points, uv = points[valid], uv[valid]
   51         uv[:20] = np.roll(uv[:20], 7, axis=0)
   52         _, _, count, p95 = solve_relative(points, uv, self.rig)
   53         self.assertLess(count, len(points))
   54         self.assertLess(p95, .1)
   55         with self.assertRaisesRegex(ValueError, 'degenerate'):
   56             solve_relative(np.tile((10., 0., 0.), (20, 1)), np.tile((480., 360.), (20, 1)), self.rig)

   68 class RefineCorrespondencesTests(unittest.TestCase):
   69     def test_subpixel_shift_is_recovered_and_blank_image_rejected(self):
   70         import cv2
   71         from perception.visual_odometry.stereo import refine_correspondences
   72         rng = np.random.RandomState(64)
   73         image = rng.randint(0, 256, (150, 200), dtype=np.uint8)
   74         shifted = cv2.warpAffine(image, np.float32([[1., 0., 2.4], [0., 1., -.7]]), (200, 150))
   75         points = np.array([[40., 40.], [80., 90.], [140., 110.]], dtype=np.float32)
   76         expected = points + (2.4, -.7)
   77         result, valid = refine_correspondences(image, shifted, points, points + (2., -1.), cv2)
   78         self.assertGreaterEqual(int(valid.sum()), 1)
   79         np.testing.assert_allclose(result[valid], expected[valid], atol=.2)
