   13 class OnlineDynamicLidarTests(unittest.TestCase):

   15         self.history = PoseHistory()

   18                 covariance_or_bounds=tuple(np.eye(15).flatten())), (0., 0., 0.), (0., 0., 0.),

   26     def test_live_candidate_is_diagnostic_only(self):
   27         row = self.pipeline.process(self.cloud, 20.01, lambda: (10.01, 0, True, self.state), lambda: 20.011)
   28         self.assertEqual(row['status'], 'PROCESSED', row)
   29         self.assertGreater(len(row['tracks']), 0)
   30         self.assertFalse(row['dynamic_check']['safety_certified'])

   33     def test_joint_planner_keeps_uncertified_space_unusable(self):
   34         pipeline = OnlineDynamicLidar(self.history, self.pipeline.sensor_pose, planning_diagnostic=True)
   35         goal = GoalSnapshot('OBSERVED', (10., 0., 0.), 0, 3, 20.01, 0., '')
   36         row = pipeline.process(self.cloud, 20.01, lambda: (10.01, 0, True, self.state),
   37                                lambda: 20.011, lambda: goal)
   38         self.assertEqual(row['status'], 'PROCESSED', row)
   39         self.assertEqual(row['planner']['reason'], 'NO_STATIC_CANDIDATE')
   40         self.assertFalse(row['planner']['trajectory_produced'])

   56     def test_wait_for_imu_is_bounded_and_causal(self):
   57         wall = [20.01]
   58         def sleep(dt): wall[0] += dt
   59         wait_for_imu(self.cloud, 20.01, lambda: (10., 0, True, self.state),
   60                      lambda: wall[0], sleep, lambda: False)
   61         self.assertGreaterEqual(wall[0] - 20.01, .03)
   62         self.assertLess(wall[0] - 20.01, .031)
   63         wall[0] = 20.01
   64         def snapshot():
   65             return (10.01 if wall[0] >= 20.014 else 10., 0, True, self.state)
   66         wait_for_imu(self.cloud, 20.01, snapshot, lambda: wall[0], sleep, lambda: False)
   67         self.assertLess(wall[0] - 20.01, .01)
   68         row = self.pipeline.process(self.cloud, 20.01, snapshot, lambda: wall[0])
   69         self.assertEqual(row['status'], 'PROCESSED', row)
