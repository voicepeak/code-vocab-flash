   12         for time,z in ((0.,0.),(3.,-1.),(10.,-1.)):
   13             p,v,a=benchmark_reference((0.,0.,0.),time)
   14             np.testing.assert_allclose(p,(0.,0.,z))
   15             np.testing.assert_allclose(v,0.,atol=1e-10)
