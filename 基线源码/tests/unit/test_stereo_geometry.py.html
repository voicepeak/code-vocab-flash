   27     def test_front_projection_and_triangulation_recover_known_body_points(self):
   28         rig = load_nominal_rig(SETTINGS)
   29         body = np.array(((10., 0., 0.), (8., 1., .3), (5., -.5, -.3)))
   30         left, z, vl = rig.left.project_body(body)
   31         right, _, vr = rig.right.project_body(body)
   32         self.assertTrue(vl.all() and vr.all())
   33         np.testing.assert_allclose(z, body[:, 0] - .175)
   34         recovered, valid, sensitivity = rig.triangulate_body(left, right)
   35         self.assertTrue(valid.all())
   36         np.testing.assert_allclose(recovered, body, atol=1e-12)
   37         np.testing.assert_allclose(sensitivity, z ** 2 / (rig.left.fx * .3))

   92 class SparseEpipolarTests(unittest.TestCase):
   93     def images(self, vertical=0):
   94         rng = np.random.RandomState(44)
   95         left = rng.randint(0, 256, (240, 400)).astype(np.uint8)
   96         right = np.roll(left, (-vertical, -12), axis=(0, 1))
   97         return left, right
   98 
   99     def test_known_horizontal_and_vertical_shifts_are_measured_without_vertical_prefilter(self):
  100         for vertical in (0, 8):
  101             result, left, right, mask, fundamental = measure_sparse_epipolar(*self.images(vertical))
  102             self.assertGreater(result['reciprocal_ratio_matches'], 100)
  103             self.assertAlmostEqual(result['vertical_residual_all_p50_p95_p99_max_px'][0], vertical, delta=.2)
  104             self.assertFalse(result['vertical_prefilter_applied'])
  105             self.assertFalse(result['rectification_verified'])
  106             self.assertEqual(left.shape, right.shape)
  107 
  108     def test_blank_images_have_no_invented_correspondences(self):
  109         image = np.zeros((240, 400), np.uint8)
  110         result, left, right, mask, fundamental = measure_sparse_epipolar(image, image)
  111         self.assertEqual(result['reciprocal_ratio_matches'], 0)
  112         self.assertEqual(result['fundamental_inliers'], 0)
  113         self.assertIsNone(result['vertical_residual_all_p50_p95_p99_max_px'])
  114         self.assertIsNone(fundamental)
