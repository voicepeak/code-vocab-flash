   17     def test_rgb_conversion_preserves_declared_channel_order(self):
   18         metadata=Metadata(1.,1.2,0,'test','front_left_image')
   19         image=ImageFrame(metadata,'bgr8',np.array([[[1,2,3]]],dtype=np.uint8))
   20         np.testing.assert_array_equal(rgb_pixels(image),[[[3,2,1]]])
   21         mono=ImageFrame(metadata,'mono8',np.array([[[7]]],dtype=np.uint8))

   24     def test_positive_reverse_magnitude_and_pixel_units_match_known_shift(self):
   25         left,right=self.images();forward=np.full(left.shape,8.,dtype=np.float32)
   26         filtered,common,lr=filter_learned_disparity(forward,forward,left,right)
   27         self.assertTrue(common[3:-3,10:-3].all())
   28         self.assertFalse(lr[:,:8].any())
   29         self.assertTrue(np.isnan(filtered[~common]).all())
   30         self.assertTrue((filtered[common]==8).all())

   32     def test_half_resolution_disparity_is_doubled_in_original_pixel_units(self):

   34         restored=full_resolution_disparity(model_grid,6,4,.5)
   35         np.testing.assert_array_equal(restored,np.repeat(np.repeat(model_grid*2,2,axis=0),2,axis=1))
   36         np.testing.assert_array_equal(full_resolution_disparity(model_grid,3,2,1.),model_grid)

   43     def test_inconsistent_or_nonfinite_reverse_fields_stay_unknown(self):
