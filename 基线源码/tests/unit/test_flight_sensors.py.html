    9 def message(stamp,position=(1.,2.,-1.),q=(1.,0.,0.,0.)):
   10     return Obj(header=Obj(stamp=Obj(to_sec=lambda:stamp)),
   11                pose=Obj(position=Obj(x=position[0],y=position[1],z=position[2]),

   17 class SensorAdapterTests(unittest.TestCase):
   18     def sensors(self):

   25     def test_future_gps_waits_for_imu_then_updates(self):
   26         s=self.sensors()
   27         s.on_gps(message(100.02))
   28         self.assertEqual(s.estimator.last_gps,100.)
   29         self.assertIsNotNone(s.pending_gps)
   30         s.on_imu(message(100.03))
   31         self.assertEqual(s.estimator.last_gps,100.02)
   32         self.assertIsNone(s.pending_gps)
   33         self.assertEqual(s.read()[0].health,Health.OK)
   34 
   35     def test_invalid_queued_gps_cannot_kill_imu_propagation(self):
   36         for position in ((0.,0.,0.),(1.,2.,-1.)):
   37             s=self.sensors()
   38             s.on_gps(message(100.02,position,(0.,0.,0.,0.)))
   39             s.on_imu(message(100.03))
   40             self.assertIsNone(s.fault)
   41             self.assertEqual(s.estimator.last_gps,100.)
   42             self.assertTrue(s.estimator.rejections)
   43             self.assertEqual(s.estimator.stamp,100.03)

   45     def test_optional_history_samples_imu_and_clears_on_clock_reset(self):
