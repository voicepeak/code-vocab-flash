   11 class FlightStereoEvaluationTests(unittest.TestCase):
   12     def test_truth_comparison_interpolates_and_preserves_rejections(self):
   13         estimates = {'source': 'offline', 'results': [
   14             {'previous': 'a', 'current': 'b', 'previous_stamp': 1.01, 'current_stamp': 1.03,
   15              'status': 'ESTIMATED', 'motion': {'translation_previous_body_m': [.02, 0., 0.],
   16              'rotation_previous_from_current_body': np.eye(3).tolist()}},
   17             {'previous': 'b', 'current': 'c', 'status': 'REJECTED', 'reason': 'bad fit'}]}
   18         records = [{'stamp': t, 'position': [t - 1, 0., 0.], 'orientation': [1., 0., 0., 0.]}
   19                    for t in (1., 1.02, 1.04)]
   20         with tempfile.TemporaryDirectory() as tmp:
   21             truth = Path(tmp) / 'truth.jsonl'
   22             truth.write_text(''.join(json.dumps(row) + '\n' for row in records))
   23             output = evaluate(estimates, truth)
   24         self.assertEqual(output['status_counts'], {'COMPARED': 1, 'REJECTED': 1})
   25         self.assertLess(output['results'][0]['translation_error_m'], 1e-10)
   26         self.assertLess(output['results'][0]['rotation_error_rad'], 1e-10)
   27         self.assertEqual(output['results'][1]['reason'], 'bad fit')
   28 
   29     def test_no_truth_extrapolation_or_interpolation_across_large_gap(self):
   30         estimates = {'source': 'offline', 'results': [
   31             {'previous': 'a', 'current': 'b', 'previous_stamp': 1.01, 'current_stamp': 1.5,
   32              'status': 'ESTIMATED', 'motion': {'translation_previous_body_m': [0., 0., 0.],
   33              'rotation_previous_from_current_body': np.eye(3).tolist()}}]}
   34         with tempfile.TemporaryDirectory() as tmp:
   35             truth = Path(tmp) / 'truth.jsonl'
   36             truth.write_text(''.join(json.dumps({'stamp': t, 'position': [0., 0., 0.],
   37                 'orientation': [1., 0., 0., 0.]}) + '\n' for t in (1., 1.02, 2.)))
   38             output = evaluate(estimates, truth)
   39         self.assertEqual(output['status_counts'], {'TRUTH_UNAVAILABLE': 1})
