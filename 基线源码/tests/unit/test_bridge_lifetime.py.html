   13 class BridgeLifetimeAuditTests(unittest.TestCase):
   14     def records(self, free_start=25, free_end=26):
   15         return [row('bridge_begin', 10), row('parameter_borrow', 20, 22),

   20     def test_freed_parameter_before_handoff_is_a_lifetime_violation_not_causal_proof(self):
   21         result = evaluate(self.records())
   22         self.assertEqual(result['freed_parameter_handoffs'], 1)
   23         self.assertEqual(result['linked_parameter_handoffs'], 1)
   24         self.assertTrue(result['hook_coverage_valid'])
   25         self.assertFalse(result['physical_impulse_cause_identified'])
   26         self.assertFalse(result['actual_native_consumer_read_observed'])
   27 
   28     def test_earlier_address_reuse_overlapping_get_and_later_free_cannot_prove_violation(self):
   29         for begin, end in ((1, 2), (21, 25), (45, 46)):

   31                 self.assertEqual(evaluate(self.records(begin, end))['freed_parameter_handoffs'], 0)
   32         records = self.records()
