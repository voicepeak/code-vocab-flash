   19 class PoseHistoryTests(unittest.TestCase):
   20     def history(self, **kwargs):
   21         history = PoseHistory(**kwargs)
   22         for stamp in (10., 10.01, 10.02):
   23             history.append(state(stamp), 20. + stamp - 10.)

   26     def test_interpolates_measurement_time_without_refreshing_current_state(self):
   27         history = self.history()
   28         pose = history.lookup(10.005, 10.02, 20.021, 0)
   29         self.assertAlmostEqual(pose.position_ned[0], .005)
   30         np.testing.assert_allclose(pose.orientation_wxyz, delta_quaternion((0., 0., .001)), atol=1e-12)
   31         self.assertEqual(pose.bracket_stamps, (10., 10.01))
   32         self.assertFalse(pose.current_control_eligible)
   33         self.assertFalse(pose.uncertainty_calibrated)
   34         self.assertFalse(hasattr(pose, 'valid_until'))
   35         with self.assertRaises(ValueError):replace(pose,position_ned=[0.,0.,0.])
   36         with self.assertRaises(ValueError):replace(pose,orientation_wxyz=(0.,0.,0.,0.))

   38     def test_antipodal_quaternions_follow_short_arc(self):
   39         history = PoseHistory()
   40         first, second = state(10.), state(10.01)
   41         second = replace(second, orientation_wxyz=tuple(-np.array(second.orientation_wxyz)))
   42         history.append(first, 20.)

   44         pose = history.lookup(10.005, 10.01, 20.011, 0)

   55     def test_no_extrapolation_gap_or_bad_health_interpolation(self):
   56         history = self.history()

   63         for health in (Health.DEGRADED, Health.LOST):
   64             history = PoseHistory()

   66             history.append(state(10.01), 20.01)

   79     def test_capacity_horizon_and_explicit_reset_do_not_reuse_old_data(self):
   80         history = PoseHistory(capacity=4, horizon_s=.2)
   81         for i in range(30): history.append(state(10. + i * .01), 20. + i * .01)
   82         with self.assertRaises(HistoryUnavailable): history.lookup(10.01, 10.29, 20.291, 0)
   83         with self.assertRaises(HistoryUnavailable): history.lookup(10.25, 10.29, 20.291, 0)
   84         history.reset(1)
   85         with self.assertRaises(HistoryUnavailable): history.lookup(10.28, 10.29, 20.291, 1)
   86         with self.assertRaises(ValueError): history.append(state(10.3), 20.3)
   87         history.append(state(1., epoch=1), 21.)
   88         self.assertEqual(history.lookup(1., 1., 21.001, 1).epoch, 1)
