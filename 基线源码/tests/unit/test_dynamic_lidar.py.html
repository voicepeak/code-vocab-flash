   22     def test_clusters_enclose_sampled_points_and_exclude_sparse_noise(self):
   23         cloud=[(1.+.01*i,2.,3.) for i in range(10)]+[(6.,0.,0.)]
   24         output=cluster_endpoints(observation(10.,cloud),10.,0)
   25         self.assertEqual(len(output.clusters),1);self.assertEqual(output.clusters[0].points,10)
   26         self.assertEqual(output.rejected_components,1);self.assertFalse(output.control_eligible)
   27         c=output.clusters[0]
   28         for point in cloud[:10]:
   29             self.assertTrue(all(abs(a-b)<=h+1e-12 for a,b,h in zip(point,c.center_ned,c.half_extents_m)))

   31     def test_centroid_reduces_voxel_boundary_jump_and_keeps_voxel_enclosure(self):
   32         first=[(.499+.0001*i,2.,3.) for i in range(10)]
   33         second=[(.501+.0001*i,2.,3.) for i in range(10)]
   34         a=cluster_endpoints(observation(10.,first),10.,0,center_mode='centroid').clusters[0]
   35         b=cluster_endpoints(observation(10.1,second),10.1,0,center_mode='centroid').clusters[0]
   36         self.assertLess(abs(b.center_ned[0]-a.center_ned[0]),.01)
   37         for c,points in ((a,first),(b,second)):
   38             self.assertTrue(all(all(abs(x-y)<=h+1e-12 for x,y,h in zip(p,c.center_ned,c.half_extents_m)) for p in points))
   39         default_a=cluster_endpoints(observation(10.,first),10.,0).clusters[0]
   40         default_b=cluster_endpoints(observation(10.1,second),10.1,0).clusters[0]
   41         self.assertAlmostEqual(default_b.center_ned[0]-default_a.center_ned[0],.5)
