    6 class GeometryPrecisionInspectionTests(unittest.TestCase):
    7     constraints = [{'type': 'LayerType.GRID_SAMPLE', 'name': '/GridSample',
    8         'coordinate_shape': [11520, 1, 9, 2]}]
    9 
   10     def inspector(self, source='Float', output='Float'):
   11         return {'Layers': [{'Name': 'sample_fusion', 'Metadata': '[ONNX Layer: /GridSample]',
   12             'Inputs': [{'Name': 'grid', 'Dimensions': [11520, 1, 9, 2], 'Format/Datatype': source}],
   13             'Outputs': [{'Name': 'grid_cast', 'Dimensions': [11520, 1, 9, 2], 'Format/Datatype': output},
   14                 {'Name': 'features', 'Dimensions': [1, 522, 96, 120], 'Format/Datatype': 'Half'}]}]}
   15 
   16     def test_float_coordinate_evidence_does_not_reject_half_feature_output(self):
   17         result = inspect_geometry_precision(self.inspector(), self.constraints)
   18         self.assertTrue(result['visible_coordinate_formats_verified'])
   19         self.assertFalse(result['per_operation_internal_precision_trace'])
   20 
   21     def test_half_source_cannot_be_certified_by_a_final_float_cast(self):
   22         result = inspect_geometry_precision(self.inspector('Half'), self.constraints)
   23         self.assertFalse(result['visible_coordinate_formats_verified'])
   24         self.assertEqual(result['non_fp32_coordinates'][0]['side'], 'Inputs')
   25 
   26     def test_missing_sampler_metadata_does_not_verify_precision(self):
   27         result = inspect_geometry_precision({'Layers': []}, self.constraints)
   28         self.assertFalse(result['visible_coordinate_formats_verified'])
   29 
   30     def test_missing_format_is_not_treated_as_float(self):
   31         result = inspect_geometry_precision(self.inspector(None), self.constraints)
   32         self.assertFalse(result['visible_coordinate_formats_verified'])
