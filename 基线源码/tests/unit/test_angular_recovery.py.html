   18 def simulate_recovery(axis,impulse,lag=.02,delay=0.,inertia_scale=1.,profile='standard'):
   19     cfg=json.loads((Path(__file__).resolve().parents[2]/'configs/vehicle_identification.json').read_text())
   20     inertia=np.array(cfg['inertia_kg_m2'])*inertia_scale;mass=cfg['mass_kg']
   21     controller=Se3Controller(cfg);allocator=MotorAllocator(cfg);supervisor=AngularRecovery(profile)
   22     q=np.array([1.,0.,0.,0.]);p=np.array([0.,0.,-1.]);v=np.zeros(3)
   23     direction=np.eye(3)[axis] if axis<3 else np.array([1.,1.,0.])/np.sqrt(2)
   24     omega=direction*impulse
   25     actual=np.array([mass*9.81,0,0,0]);request=actual.copy()
   26     dt=.0025;max_tilt=0.;max_drift=0.
   27     delay_steps=round(delay/dt)
   28     history=[(p.copy(),v.copy(),q.copy(),np.zeros(3)) for _ in range(delay_steps)]
   29     for i in range(1600):
   30         history.append((p.copy(),v.copy(),q.copy(),omega.copy()))
   31         if i%5==0:
   32             observed=history[len(history)-1-delay_steps]
   33             s=StateEstimate(Metadata(1+i*dt,1.05+i*dt,0,'model'),tuple(observed[0]),tuple(observed[1]),
   34                             tuple(observed[2]),tuple(observed[3]),Health.OK)
   35             decision=supervisor.decide(s)
   36             wrench=controller.step(s,decision.hold_position,yaw=decision.hold_yaw,
   37                 dt=.0125,allow_integral=False,rate_gain_scale=decision.rate_gain_scale,
   38                 attitude_gain_scale=decision.attitude_gain_scale)
   39             request=np.array(allocator.allocate(wrench.thrust,wrench.moment).achieved_wrench)
   40         actual+=(1-np.exp(-dt/lag))*(request-actual)
   41         R=rotation(q)
   42         acceleration=np.array([0,0,9.81])-R[:,2]*actual[0]/mass
   43         alpha=(actual[1:]-np.cross(omega,inertia*omega))/inertia
   44         p+=v*dt+.5*acceleration*dt*dt;v+=acceleration*dt
   45         q=unit(multiply(q,delta_quaternion(omega*dt+.5*alpha*dt*dt)));omega+=alpha*dt
   46         max_tilt=max(max_tilt,float(np.arccos(np.clip(R[2,2],-1,1))))
   47         max_drift=max(max_drift,float(np.linalg.norm(p-[0,0,-1])))
   48     return max_tilt,max_drift,supervisor.phase,float(np.linalg.norm(omega))

   51 class AngularRecoveryTests(unittest.TestCase):
   52     def test_loss_epoch_hard_bound_and_timeout_do_not_resume_normal_commands(self):
   53         for s in (state(health=Health.LOST),state(omega=(0,5.,0))):
   54             with self.assertRaises(RuntimeError):AngularRecovery().decide(s)
   55         r=AngularRecovery();r.decide(state(omega=(0,1.,0)))
   56         with self.assertRaises(RuntimeError):r.decide(state(1.1,epoch=1))
   57         r=AngularRecovery();r.decide(state(omega=(0,1.,0)))
   58         with self.assertRaises(RuntimeError):r.decide(state(4.1,omega=(0,1.,0)))

   60     def test_fast_profile_is_explicit_and_does_not_relax_the_hard_bound(self):
   61         standard=AngularRecovery().decide(state(omega=(0,.6,0)))
   62         fast=AngularRecovery('fast').decide(state(omega=(0,.6,0)))
   63         self.assertEqual((standard.rate_gain_scale,standard.attitude_gain_scale),(2.,1.))
   64         self.assertEqual((fast.rate_gain_scale,fast.attitude_gain_scale),(3.,4.))
   65         with self.assertRaises(RuntimeError):AngularRecovery('fast').decide(state(omega=(0,2.6,0)))
   66         with self.assertRaises(ValueError):AngularRecovery('unsupported')

   68     def test_settling_requires_sustained_window_and_keeps_hold_target(self):
   69         r=AngularRecovery();d=r.decide(state(omega=(0,.6,0)))
   70         self.assertEqual(d.phase,'ANGULAR_RECOVERY')
   71         r.decide(state(1.1));r.decide(state(1.3,omega=(0,.3,0)))
   72         self.assertEqual(r.decide(state(1.4)).phase,'ANGULAR_RECOVERY')
   73         self.assertEqual(r.decide(state(1.76)).phase,'RECOVERY_HOLD')
   74         d=r.decide(replace(state(2.),position_ned=(.1,0.,-1.)))
   75         self.assertEqual(d.hold_position,(0.,0.,-1.))
   76         self.assertEqual(d.phase,'RECOVERY_HOLD')

   78     def test_nominal_rigid_body_impulses_with_80hz_control_and_20ms_motor_lag(self):
   79         for axis in (0,1,2):
   80             for impulse in (1.8,2.3):
   81                 with self.subTest(axis=axis,impulse=impulse):
   82                     max_tilt,max_drift,phase,rate=simulate_recovery(axis,impulse)
   83                     self.assertLess(max_tilt,.35)
   84                     self.assertLess(max_drift,.8)
   85                     self.assertEqual(phase,'RECOVERY_HOLD')
   86                     self.assertLess(rate,.2)
