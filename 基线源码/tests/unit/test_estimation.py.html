   51     def test_gps_loss_degrades_without_teleport_and_recovers(self):

   64     def test_noisy_stationary_covariance_stays_positive(self):
   65         e = self.estimator()
   66         rng = np.random.RandomState(7)
   67         for i in range(1, 501):
   68             stamp = 1+i*.01
   69             e.propagate(stamp, np.array((0., 0., -9.81))+rng.normal(0, .02, 3),
   70                         rng.normal(0, .001, 3))
   71             if i % 10 == 0:
   72                 e.update_gps(stamp, np.array((1., 2., -1.))+rng.normal(0, .03, 3),
   73                              (1., 0., 0., 0.))
   74         self.assertLess(np.linalg.norm(e.p-(1., 2., -1.)), .1)
   75         self.assertLess(np.linalg.norm(e.v), .1)
   76         self.assertGreater(np.linalg.eigvalsh(e.P).min(), 0.)
   77         np.testing.assert_allclose(e.P, e.P.T, atol=1e-12)
