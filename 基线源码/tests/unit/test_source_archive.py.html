   14     def repository(self, root):

   26     def test_dirty_and_untracked_bytes_restore_exactly_with_executable_mode(self):
   27         with tempfile.TemporaryDirectory() as directory:
   28             root = Path(directory).resolve()
   29             self.repository(root)
   30             run, manifest = prepare_run(root, root / 'runs', 1, 'test-digest', {})
   31             snapshot = manifest['source_archive']
   32             with tarfile.open(run / snapshot['path']) as archive:
   33                 self.assertEqual(set(archive.getnames()), set(manifest['source_sha256_before']))
   34                 for member in archive:
   35                     data = archive.extractfile(member).read()
   36                     self.assertEqual(data, (root / member.name).read_bytes())
   37                     self.assertEqual(hashlib.sha256(data).hexdigest(),
   38                                      manifest['source_sha256_before'][member.name])
   39                     self.assertEqual(member.uid, 0)
   40                     self.assertEqual(member.mtime, 0)
   41                 self.assertEqual(archive.getmember('new.sh').mode, 0o755)
   42             other = root / 'runs/second'
   43             other.mkdir()
   44             same = archive_sources(root, other, manifest['source_sha256_before'])
   45             self.assertEqual(snapshot['sha256'], file_hash(other / same['path']))
   46             with self.assertRaises(FileExistsError):
   47                 archive_sources(root, other, manifest['source_sha256_before'])
   48             self.assertEqual(snapshot['sha256'], same['sha256'])
   49             self.assertFalse(list(other.glob('*.tmp')))

   51     def test_changed_source_or_source_set_aborts_before_publication(self):
   52         for change in ('content', 'new_file'):
   53             with self.subTest(change=change), tempfile.TemporaryDirectory() as directory:
   54                 root = Path(directory).resolve()
   55                 self.repository(root)
   56                 expected = source_hashes(root)
   57                 run = root / 'runs/case'
   58                 run.mkdir(parents=True)
   59                 if change == 'content':
   60                     (root / 'control.py').write_text('changed again\n')
   61                 else:
   62                     (root / 'added.py').write_text('added\n')
   63                 with self.assertRaises(RuntimeError):
   64                     archive_sources(root, run, expected)
   65                 self.assertFalse(list(run.iterdir()))
