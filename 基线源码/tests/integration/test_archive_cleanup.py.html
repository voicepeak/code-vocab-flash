   18 class FakeRemote:
   19     revision = "frozen-test-revision"
   20 
   21     def __init__(self, plan, receipt, index):
   22         name = receipt["name"]
   23         self.data = {
   24             "publication_status.json": json.dumps({"state": "COMPLETE", "shards_expected": 1,
   25                                                    "shards_verified": 1}).encode(),
   26             "dataset_summary.json": json.dumps({k: plan[k] for k in ("file_count", "source_bytes")}).encode(),
   27             "shards.json": json.dumps([receipt]).encode(),
   28             "indices/" + index.name: index.read_bytes(),
   29         }
   30         self.info = {
   31             "data/" + name: {"size": receipt["size"], "sha256": receipt["sha256"]},
   32             "indices/" + index.name: {"size": receipt["index_size"], "sha256": receipt["index_sha256"]},
   33         }
   34 
   35     def read(self, path):
   36         return self.data[path]
   37 
   38     def metadata(self, paths):
   39         return {p: self.info[p] for p in paths if p in self.info}

   42 class ArchiveCleanupTests(unittest.TestCase):

   82     def test_only_backed_up_unchanged_files_are_removed(self):
   83         with tempfile.TemporaryDirectory() as temp:
   84             _, run, output, plan, state, remote = self.fixture(Path(temp))
   85             verify_publication(plan, state, remote, output)
   86             # Same byte count, changed contents: stat size alone must not authorize deletion.
   87             original = run / "b.npz"
   88             original.write_bytes(b"x" * original.stat().st_size)
   89             result = delete_verified_members(plan, output)
   90             self.assertEqual(result["deleted_files"], 1)
   91             self.assertEqual(result["preserved_changed_files"], 1)
   92             self.assertFalse((run / "a.jpg").exists())
   93             self.assertTrue(original.exists())
   94             self.assertTrue((run / "unpublished.bin").exists())

  167     def test_retry_after_permission_repair_keeps_prior_deletion_receipt(self):
  168         with tempfile.TemporaryDirectory() as temp:
  169             _, run, output, plan, state, remote = self.fixture(Path(temp))
  170             digest = sha256(output / "plan.json")
  171             size = (run / "a.jpg").stat().st_size
  172             (run / "a.jpg").unlink()
  173             write_json(output / "upload_state.local.json", state)
  174             write_json(output / "per_shard_cleanup_state.local.json", {
  175                 "plan_sha256": digest, "cleaned": {}, "retry_previous": {
  176                     "data-00000.tar": {"deleted_files": 1, "deleted_bytes": size,
  177                         "already_missing_files": 0, "preserved_changed_files": 1}}})
  178             with patch('evaluation.dataset_capture.cleanup_archive.PublicArchive', return_value=remote), \
  179                     redirect_stdout(io.StringIO()):
  180                 incremental_cleanup(plan, output, lambda **kw: None, digest)
  181             saved = json.loads((output / "per_shard_cleanup_state.local.json").read_text())
  182             result = saved["cleaned"]["data-00000.tar"]
  183             self.assertEqual(result["deleted_files"], 2)
  184             self.assertEqual(result["deleted_bytes"], plan["source_bytes"])
  185             self.assertEqual(result["already_missing_files"], 0)
  186             self.assertEqual(result["preserved_changed_files"], 0)
  187             self.assertFalse((run / "b.npz").exists())
