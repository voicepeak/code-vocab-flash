   13 class ReferenceLifetimeTests(unittest.TestCase):
   14     def test_worker_does_not_outlive_a_terminated_flight_parent(self):
   15         script='''
   16 import time
   17 from control.reference_worker import BenchmarkSpec,ReferenceWorker
   18 w=ReferenceWorker(BenchmarkSpec((0.,0.,-1.),0.))
   19 w.start()
   20 deadline=time.monotonic()+3
   21 while time.monotonic()<deadline:
   22     w.submit(10.,0,0.)
   23     if w.snapshot(10.,0)[0] is not None:break
   24     time.sleep(.005)
   25 else:raise RuntimeError('worker not ready')
   26 print(w.process.pid,flush=True)
   27 while True:time.sleep(.1)
   28 '''
   29         root=Path(__file__).resolve().parents[2]
   30         parent=subprocess.Popen([sys.executable,'-c',script],cwd=root,stdout=subprocess.PIPE,
   31                                 stderr=subprocess.PIPE,text=True)
   32         worker_pid=None
   33         def running(pid):
   34             path=Path('/proc')/str(pid)/'stat'
   35             try:return path.read_text().split(') ',1)[1].split()[0] not in ('Z','X')
   36             except FileNotFoundError:return False
   37         try:
   38             with selectors.DefaultSelector() as poll:
   39                 poll.register(parent.stdout,selectors.EVENT_READ)
   40                 self.assertTrue(poll.select(timeout=5.),'flight parent never became ready')
