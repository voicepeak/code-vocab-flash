   46 class BridgeProbeTests(unittest.TestCase):
   47     def test_observer_forwards_and_serializer_mutually_excludes_the_two_original_calls(self):
   48         root=Path(__file__).resolve().parents[2]
   49         build=subprocess.check_output(['bash',str(root/'tools/build_bridge_probe.sh')],cwd=root,text=True).strip()
   50         probe='/workspace/main/'+str(Path(build).relative_to(root))
   51         with tempfile.TemporaryDirectory(prefix='bridge_probe_test_',dir=root/'runs') as temporary:
   52             path=Path(temporary);inside='/workspace/main/'+str(path.relative_to(root))
   53             (path/'mock.cpp').write_text(MOCK);(path/'driver.cpp').write_text(DRIVER)
   54             def run(*args):
   55                 return subprocess.check_output([str(root/'tools/enter_env.sh'),*args],cwd=root,text=True,stderr=subprocess.STDOUT)
   56             run('g++','-std=c++14','-shared','-fPIC','-pthread',inside+'/mock.cpp','-o',inside+'/mock.so')
   57             run('g++','-std=c++14','-pthread',inside+'/driver.cpp','-L'+inside,'-Wl,-rpath,'+inside,
   58                 '-l:mock.so','-o',inside+'/driver')
   59             for mode,expected in (('observe',2),('serialize',1)):
   60                 output=run('env','LD_PRELOAD='+probe,'RMUA_BRIDGE_PROBE_ORIGINAL='+inside+'/mock.so',
   61                     'RMUA_BRIDGE_PROBE_OUTPUT='+inside+'/'+mode+'.jsonl','RMUA_BRIDGE_PROBE_MODE='+mode,
   62                     inside+'/driver')
   63                 self.assertEqual(int(output.strip()),expected)
   64                 records=[json.loads(x) for x in (path/(mode+'.jsonl')).read_text().splitlines()]
   65                 last=records[-1]
   66                 self.assertEqual(last['event'],'exit')
   67                 self.assertEqual((last['pwm_calls'],last['bridge_calls']),(32,32))
   68                 self.assertEqual(last['write_errors'],0)
   69                 self.assertEqual(last['overlap_entries']>0,mode=='observe')
   70                 self.assertTrue(all(r['end_monotonic_ns']>=r['start_monotonic_ns'] for r in records))
   71                 self.assertTrue(all(r['thread_id']!=r['other_thread_id'] and r['other_thread_id']>0
   72                                     for r in records if r['event']=='overlap'))
