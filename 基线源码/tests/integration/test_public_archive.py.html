   24     def test_credentials_snapshots_and_symlinks_are_excluded(self):
   25         with tempfile.TemporaryDirectory() as temp:
   26             source, run = self.fixture(Path(temp))
   27             (run / "secret.log").write_text("hf_" + "a" * 30)
   28             (run / "controller_sources").mkdir()
   29             (run / "controller_sources/manifest.json").write_text("{}")
   30             (run / "private_key.txt").write_text("sensitive")
   31             (run / "outside.jpg").symlink_to("/etc/passwd")
   32             files, excluded, _ = scan(source)
   33             self.assertEqual({Path(row["path"]).name for row in files}, {"frame.jpg", "telemetry.json"})
   34             self.assertIn("possible_credential_content", {r["reason"] for r in excluded})

   36     def test_tar_and_per_file_index_are_deterministic_and_match_content(self):
   37         with tempfile.TemporaryDirectory() as temp:
   38             base = Path(temp)
   39             source, _ = self.fixture(base)
   40             first, second = base / "a", base / "b"
   41             with redirect_stdout(io.StringIO()):
   42                 plan = prepare(source, first, 1024 ** 2)
   43             second.mkdir()
   44             a, index, receipt = pack(plan, first, plan["shards"][0])
   45             _, _, other = pack(plan, second, plan["shards"][0])
   46             self.assertEqual(receipt["sha256"], other["sha256"])
   47             self.assertEqual(receipt["index_sha256"], other["index_sha256"])
   48             with gzip.open(index, "rt") as f:
   49                 rows = [json.loads(line) for line in f]
   50             with tarfile.open(a) as tar:
   51                 self.assertEqual(len(tar.getmembers()), 2)
   52                 for row in rows:
   53                     member = tar.getmember(row["path"])
   54                     self.assertFalse(member.issym())
   55                     self.assertFalse(Path(member.name).is_absolute())
   56                     self.assertEqual(hashlib.sha256(tar.extractfile(member).read()).hexdigest(), row["sha256"])
