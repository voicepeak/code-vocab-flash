   21 class CommandGuard:
   22     def __init__(self, epoch, minimum_period=.01, input_timeout=.1):
   23         if not (math.isfinite(minimum_period) and minimum_period >= .01
   24                 and math.isfinite(input_timeout) and input_timeout > 0):
   25             raise ValueError('invalid command guard timing')
   26         self.epoch = epoch
   27         self.minimum_period = minimum_period
   28         self.input_timeout = input_timeout
   29         self.last_wall = None
   30         self.last_sequence = -1
   31         self.last_stamp = None
   32         self.fault = None
   33 
   34     def fail(self, reason):
   35         self.fault = self.fault or reason
   36         raise RuntimeError(self.fault)
   37 
   38     def accept(self, command, clock, wall_now):
   39         if self.fault:
   40             self.fail(self.fault)
   41         if not math.isfinite(wall_now):
   42             self.fail('INVALID_WALL_CLOCK')
   43         if command.epoch != self.epoch or clock.epoch != self.epoch:
   44             self.fail('CLOCK_RESET')
   45         if (not clock.fresh(wall_now, self.input_timeout) or clock.stamp is None
   46                 or not 0 <= clock.stamp-command.stamp <= .05):
   47             self.fail('STALE_IMU_OR_COMMAND')
   48         if command.sequence <= self.last_sequence:
   49             self.fail('COMMAND_REPLAY')
   50         if self.last_stamp is not None and command.stamp < self.last_stamp:
   51             self.fail('COMMAND_TIME_ROLLBACK')
   52         if self.last_wall is not None:
   53             elapsed = wall_now-self.last_wall
   54             if elapsed < 0:
   55                 self.fail('WALL_CLOCK_ROLLBACK')
   56             if elapsed < self.minimum_period:
   57                 return False  # Rate limit; no stale command replay to fill gaps.
   58         self.last_wall, self.last_sequence, self.last_stamp = wall_now, command.sequence, command.stamp
   59         return True
   60 
   61     def published(self, wall_now):
   62         """Space the next command from completion even if ROS publish blocked."""
   63         if (self.last_wall is None or not math.isfinite(wall_now)
   64                 or wall_now < self.last_wall):
   65             self.fail('INVALID_PUBLICATION_TIME')
   66         self.last_wall = wall_now
