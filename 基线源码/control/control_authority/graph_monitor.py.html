    2 from collections import deque
    3 import math
    4 import threading
    5 import time

    8 class GraphMonitor:
    9     def __init__(self, probe, interval=.1, stale_after=.3, clock=time.monotonic):
   10         if not (math.isfinite(interval) and math.isfinite(stale_after) and 0 < interval < stale_after):
   11             raise ValueError('invalid graph observation timing')
   12         self.probe, self.interval, self.stale_after, self.clock = probe, interval, stale_after, clock
   13         self.lock = threading.Lock()
   14         self.stop_event = threading.Event()
   15         self.last_verified = None
   16         self.fault = None
   17         self.durations = deque(maxlen=1000)
   18         self.count = 0
   19         self.thread = None
   20 
   21     def _once(self):
   22         start = self.clock()
   23         try:
   24             self.probe()
   25             end = self.clock()
   26             with self.lock:
   27                 # The returned graph could have been sampled at request time.
   28                 self.last_verified = start
   29                 self.durations.append(end-start)
   30                 self.count += 1
   31         except Exception as error:
   32             with self.lock:
   33                 if not self.stop_event.is_set():
   34                     self.fault = str(error)
   35 
   36     def start(self):
   37         self._once()  # Startup verification precedes any flight command.
   38         self.check()
   39         def worker():
   40             while not self.stop_event.wait(self.interval):
   41                 self._once()
   42         self.thread = threading.Thread(target=worker, daemon=True)
   43         self.thread.start()
   44 
   45     def check(self):
   46         now = self.clock()
   47         with self.lock:
   48             if self.fault:
   49                 raise RuntimeError('ROS_GRAPH:'+self.fault)
   50             if self.last_verified is None or not 0 <= now-self.last_verified <= self.stale_after:
   51                 raise RuntimeError('ROS_GRAPH_STALE')
   52 
   53     def close(self):
   54         self.stop_event.set()
   55         if self.thread:
   56             self.thread.join(timeout=.2)
   57 
   58     def summary(self):
   59         with self.lock:
   60             return {'successful_checks':self.count,'fault':self.fault,
   61                     'max_rpc_wall_s':max(self.durations,default=0.),
   62                     'stale_after_wall_s':self.stale_after}
