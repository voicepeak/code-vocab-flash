    9 class Wrench:

   11     moment: tuple

   16 class Se3Controller:
   17     def __init__(self, config):
   18         self.mass = float(config['mass_kg'])
   19         self.inertia = np.asarray(config['inertia_kg_m2'], dtype=float)
   20         self.integral = np.zeros(3)
   21         self.kp = np.array((1.5, 1.5, 2.5))
   22         self.kd = np.array((2., 2., 2.))
   23         self.ki = np.array((.1, .1, .2))
   24         self.kr = np.array((.12, .16, .08))
   25         self.kw = np.array((.035, .045, .035))
   26 
   27     def step(self, state, position, velocity=(0., 0., 0.), acceleration=(0., 0., 0.),
   28              yaw=0., dt=.0125, allow_integral=True, yaw_rate=0.,rate_gain_scale=1.,attitude_gain_scale=1.):
   29         reference = np.r_[position, velocity, acceleration, yaw, dt, yaw_rate]
   30         if (reference.shape != (12,) or not np.all(np.isfinite(reference))
   31                 or not 0 < dt <= .05 or not math.isfinite(rate_gain_scale)
   32                 or not 1.<=rate_gain_scale<=3. or not math.isfinite(attitude_gain_scale)
   33                 or not 1.<=attitude_gain_scale<=4.):
   34             raise ValueError('invalid controller reference or dt')
   35         R = rotation(state.orientation_wxyz)
   36         omega = np.asarray(state.angular_velocity_body)
   37         error = np.asarray(position)-state.position_ned
   38         v_error = np.asarray(velocity)-state.velocity_ned
   39         if allow_integral:
   40             self.integral = np.clip(self.integral + error*dt, -1., 1.)
   41         a = np.asarray(acceleration)+self.kp*error+self.kd*v_error+self.ki*self.integral
   42         a[2] = np.clip(a[2], -2., 2.)
   43         horizontal = np.linalg.norm(a[:2])
   44         if horizontal > 1.5:
   45             a[:2] *= 1.5/horizontal
   46         force = self.mass*(a-(0., 0., 9.81))
   47         b3 = -force/np.linalg.norm(force)
   48         heading = np.array((math.cos(yaw), math.sin(yaw), 0.))
   49         b2 = np.cross(b3, heading)
   50         b2 /= np.linalg.norm(b2)
   51         Rd = np.column_stack((np.cross(b2, b3), b2, b3))
   52         error_matrix = .5*(Rd.T @ R - R.T @ Rd)
   53         attitude_error = np.array((error_matrix[2, 1], error_matrix[0, 2], error_matrix[1, 0]))
   54         desired_body_rate = R.T @ (0., 0., yaw_rate)
   55         moment = -attitude_gain_scale*self.kr*attitude_error-rate_gain_scale*self.kw*(omega-desired_body_rate)+np.cross(omega, self.inertia*omega)
   56         thrust = max(0., float(-force @ R[:, 2]))
   57         return Wrench(thrust, tuple(moment), tuple(a), tuple(self.integral))
