   45 def offer_latest(queue,item):
   46     try:
   47         queue.put_nowait(item)
   48         return True
   49     except Full:
   50         try:queue.get_nowait()
   51         except Empty:return False
   52         try:
   53             queue.put_nowait(item)
   54             return True
   55         except Full:return False

   58 def process_main(spec,requests,results,stop,stall_seconds,parent_pid):
   59     # Linux process lifetime belongs to the flight invocation even if the host
   60     # has to terminate its parent before Python finally blocks can execute.
   61     import ctypes
   62     import signal
   63     libc=ctypes.CDLL(None,use_errno=True)
   64     if libc.prctl(1,signal.SIGTERM,0,0,0)!=0:
   65         offer_latest(results,{'error':'PARENT_DEATH_SIGNAL_UNAVAILABLE'})
   66         return
   67     if os.getppid()!=parent_pid:return
   68     stalled=False
   69     max_compute=0.
   70     while not stop.is_set() and os.getppid()==parent_pid:
   71         try:request=requests.get(timeout=.05)
   72         except Empty:continue
   73         stamp,epoch,elapsed,requested_wall=request
   74         begin=time.monotonic()
   75         try:
   76             if stall_seconds and not stalled and elapsed>=6.:
   77                 stalled=True
   78                 # Deliberately block only this worker. The PWM loop receives no
   79                 # special injection notification and must detect stale output.
   80                 time.sleep(stall_seconds)
   81             p,v,a=benchmark_reference(spec.origin,elapsed,spec.step,spec.braking)
   82             yaw,rate=benchmark_yaw(spec.yaw,elapsed,spec.yaw_motion)
   83             max_compute=max(max_compute,time.monotonic()-begin)
   84             packet=ReferencePacket(stamp,epoch,requested_wall,tuple(p),tuple(v),tuple(a),
   85                                    yaw,rate,max_compute,stalled)
   86             offer_latest(results,packet)
   87         except Exception as error:
   88             offer_latest(results,{'error':str(error)})
   89             return

   92 class ReferenceWorker:
   93     def __init__(self,spec,stall_seconds=0.,sensor_ttl=.08,wall_ttl=.15):
   94         finite_vec3(spec.origin)
   95         if (not math.isfinite(spec.yaw) or not 0<=stall_seconds<=1.
   96                 or not 0<sensor_ttl<=.1 or not 0<wall_ttl<=.3):
   97             raise ValueError('invalid reference worker configuration')
   98         context=mp.get_context('spawn')
   99         self.requests=context.Queue(maxsize=1)
  100         self.results=context.Queue(maxsize=1)
  101         self.stop=context.Event()
  102         self.process=context.Process(target=process_main,args=(spec,self.requests,self.results,self.stop,stall_seconds,os.getpid()),
  103                                      name='rmua_reference_worker',daemon=True)
  104         self.sensor_ttl,self.wall_ttl=sensor_ttl,wall_ttl
  105         self.latest=None
  106         self.error=None
  107         self.submitted=self.dropped=self.received=self.stale_observations=0
  108         self.max_compute=0.
  109         self.stall_observed=False
  110 
  111     def start(self):self.process.start()
  112 
  113     def submit(self,stamp,epoch,elapsed):
  114         if not all(math.isfinite(v) for v in (stamp,elapsed)) or stamp<=0 or epoch<0:
  115             raise ValueError('invalid reference request')
  116         if offer_latest(self.requests,(stamp,epoch,elapsed,time.monotonic())):self.submitted+=1
  117         else:self.dropped+=1
  118 
  119     def snapshot(self,stamp,epoch):
  120         # Each mailbox holds one result. Never drain an unbounded producer loop.
  121         try:
  122             value=self.results.get_nowait()
  123             if isinstance(value,ReferencePacket):
  124                 self.latest=value;self.received+=1
  125                 self.max_compute=max(self.max_compute,value.max_compute_wall_s)
  126                 self.stall_observed=self.stall_observed or value.stall_performed
  127             else:self.error=value.get('error','INVALID_REFERENCE_RESULT')
  128         except Empty:pass
  129         if self.error:return None,'WORKER_ERROR'
  130         if not self.process.is_alive():return None,'WORKER_EXITED'
  131         p=self.latest
  132         if p is None:return None,'NO_REFERENCE'
  133         if p.epoch!=epoch:return None,'REFERENCE_EPOCH'
  134         if not 0<=stamp-p.stamp<=self.sensor_ttl or not 0<=time.monotonic()-p.requested_wall<=self.wall_ttl:
  135             self.stale_observations+=1
  136             return None,'REFERENCE_STALE'
  137         return p,None
  138 
  139     def close(self):
  140         self.stop.set()
  141         if self.process.pid is not None:
  142             self.process.join(timeout=.2)
  143             if self.process.is_alive():
  144                 self.process.terminate()
  145                 self.process.join(timeout=1.)
  146         for queue in (self.requests,self.results):
  147             queue.cancel_join_thread();queue.close()
  148 
  149     def summary(self):
  150         return {'pid':self.process.pid,'submitted':self.submitted,'requests_dropped':self.dropped,
  151                 'received':self.received,'stale_observations':self.stale_observations,
  152                 'max_compute_wall_s':self.max_compute,'stall_observed':self.stall_observed,
  153                 'fault':self.error,'sensor_ttl_s':self.sensor_ttl,'wall_ttl_s':self.wall_ttl,
  154                 'alive_after_close':self.process.is_alive()}
