   16 def check_inputs(state: StateEstimate, trajectory: Trajectory, now: float,
   17                  epoch: int, pwm_publishers: int, vel_publishers: int) -> PreflightResult:
   18     reasons = []
   19     if not state.metadata.current(now, epoch):
   20         reasons.append("STATE_TIME_OR_EPOCH")
   21     if state.health != Health.OK:
   22         reasons.append("STATE_HEALTH")
   23     if not trajectory.metadata.current(now, epoch):
   24         reasons.append("TRAJECTORY_TIME_OR_EPOCH")
   25     if state.metadata.frame_id != trajectory.metadata.frame_id:
   26         reasons.append("FRAME_MISMATCH")
   27     if pwm_publishers != 1 or vel_publishers != 0:
   28         reasons.append("CONTROL_PUBLISHER_CONFLICT")
   29     return PreflightResult(tuple(reasons))
