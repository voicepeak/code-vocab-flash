    1 """Experimental local attitude recovery, requiring fresh healthy localization.
    2 
    3 This does not reverse a competition collision or authorize continued mission
    4 flight. After settling, hold at the trigger position until the experiment ends.
    5 """
    6 from dataclasses import dataclass
    7 import math
    8 import numpy as np
    9 from estimation.geometry import rotation
   10 from interfaces.contracts import Health
   11 
   12 
   13 @dataclass(frozen=True)
   14 class AngularDecision:
   15     phase: str
   16     hold_position: tuple
   17     hold_yaw: float
   18     rate_gain_scale: float
   19     attitude_gain_scale: float
   20 
   21 
   22 class AngularRecovery:
   23     def __init__(self,profile='standard'):
   24         if profile not in ('standard','fast'):raise ValueError('unknown angular recovery profile')
   25         self.profile=profile
   26         self.phase='NORMAL'
   27         self.position=None
   28         self.yaw=0.
   29         self.trigger_stamp=None
   30         self.settled_since=None
   31         self.previous_stamp=None
   32         self.epoch=None
   33         self.transitions=[]
   34 
   35     def decide(self,state):
   36         stamp=state.metadata.stamp
   37         if state.health!=Health.OK:raise RuntimeError('ANGULAR_RECOVERY_LOCALIZATION')
   38         if self.epoch is not None and state.metadata.epoch!=self.epoch:
   39             raise RuntimeError('ANGULAR_RECOVERY_EPOCH')
   40         if self.previous_stamp is not None and stamp<self.previous_stamp:
   41             raise RuntimeError('ANGULAR_RECOVERY_CLOCK')
   42         self.previous_stamp=stamp
   43         self.epoch=state.metadata.epoch
   44         R=rotation(state.orientation_wxyz)
   45         tilt=math.acos(float(np.clip(R[2,2],-1,1)))
   46         rate=float(np.linalg.norm(state.angular_velocity_body))
   47         if tilt>.35 or rate>2.5:raise RuntimeError('ANGULAR_RECOVERY_HARD_BOUND')
   48         previous=self.phase
   49         if self.phase=='NORMAL' and (rate>.35 or tilt>.18):
   50             self.phase='ANGULAR_RECOVERY'
   51             self.position=state.position_ned
   52             self.yaw=math.atan2(R[1,0],R[0,0])
   53             self.trigger_stamp=stamp
   54         if self.phase!='NORMAL':
   55             if np.linalg.norm(np.array(state.position_ned)-self.position)>.8:
   56                 raise RuntimeError('ANGULAR_RECOVERY_POSITION_BOUND')
   57             settled=(rate<.2 and tilt<.12 and np.linalg.norm(state.velocity_ned)<.25)
   58             if self.phase=='ANGULAR_RECOVERY':
   59                 if settled:
   60                     if self.settled_since is None:self.settled_since=stamp
   61                     if stamp-self.settled_since>=.35:self.phase='RECOVERY_HOLD'
   62                 else:self.settled_since=None
   63                 if self.phase!='RECOVERY_HOLD' and stamp-self.trigger_stamp>3.:
   64                     raise RuntimeError('ANGULAR_RECOVERY_TIMEOUT')
   65             elif rate>.35 or tilt>.18:
   66                 # A second disturbance gets a new bounded recovery interval,
   67                 # while keeping the original safe hold target.
   68                 self.phase='ANGULAR_RECOVERY'
   69                 self.trigger_stamp=stamp
   70                 self.settled_since=None
   71         if previous!=self.phase:
   72             self.transitions.append({'stamp':stamp,'from':previous,'to':self.phase,
   73                 'rate_rad_s':rate,'tilt_rad':tilt})
   74         return AngularDecision(self.phase,self.position or state.position_ned,self.yaw,
   75                                (3. if self.profile=='fast' else 2.) if self.phase=='ANGULAR_RECOVERY' else 1.,
   76                                4. if self.phase=='ANGULAR_RECOVERY' and self.profile=='fast' else 1.)
