    6 def benchmark_reference(origin, elapsed, step=False, braking=False):
    7     if not math.isfinite(elapsed) or elapsed < 0:
    8         raise ValueError('invalid reference time')
    9     origin = np.asarray(origin, dtype=float)
   10     if origin.shape != (3,) or not np.all(np.isfinite(origin)):
   11         raise ValueError('invalid reference origin')
   12     s = min(1., elapsed/3.)
   13     h = 10*s**3-15*s**4+6*s**5
   14     dh = (30*s**2-60*s**3+30*s**4)/3
   15     ddh = (60*s-180*s**2+120*s**3)/9 if s < 1 else 0.
   16     position = origin+(0., 0., -h)
   17     if step and elapsed >= 4:
   18         position[0] += .4 if elapsed < 8 else -.4
   19     vx = 0.
   20     if braking:
   21         # Deliberate velocity steps identify response; not a mission trajectory.
   22         position[0] += .4*min(2.,max(0.,elapsed-4.))
   23         vx = .4 if 4 <= elapsed < 6 else 0.
   24     return tuple(position), (vx, 0., -dh), (0., 0., -ddh)
