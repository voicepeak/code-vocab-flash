    1 """Collective-priority bounded allocator with explicit achieved wrench feedback."""
    2 from dataclasses import dataclass
    3 import math
    4 import numpy as np
    5 
    6 
    7 @dataclass(frozen=True)
    8 class Allocation:
    9     pwm: tuple
   10     achieved_wrench: tuple
   11     saturated: bool
   12     moment_scale: float
   13     min_margin: float
   14 
   15 
   16 class MotorAllocator:
   17     def __init__(self, config):
   18         arm = float(config['arm_radius_m']) / math.sqrt(2)
   19         rps_squared = (float(config['max_rpm'])/60)**2
   20         self.max_thrust = float(config['thrust_coefficient_rps2'])*rps_squared
   21         ratio = float(config['torque_coefficient_rps2'])/float(config['thrust_coefficient_rps2'])
   22         self.ceiling = float(config['local_pwm_ceiling'])
   23         signs = np.asarray(config['yaw_torque_signs'], dtype=float)
   24         if (not all(math.isfinite(v) and v > 0 for v in (arm, self.max_thrust, ratio))
   25                 or not 0 < self.ceiling <= 1 or signs.shape != (4,)
   26                 or not np.all(np.isin(signs, (-1, 1)))):
   27             raise ValueError('invalid motor model')
   28         # F_body=(0,0,-f). Torque is r cross F; yaw signs are from local pulses.
   29         self.B = np.array(((1., 1., 1., 1.), (-arm, arm, arm, -arm),
   30                            (arm, -arm, arm, -arm), tuple(ratio*signs)))
   31         self.inverse = np.linalg.inv(self.B)
   32 
   33     def allocate(self, thrust, moment):
   34         request = np.r_[thrust, moment]
   35         if request.shape != (4,) or not np.all(np.isfinite(request)) or thrust < 0:
   36             raise ValueError('invalid wrench')
   37         limit = self.max_thrust*self.ceiling
   38         collective = min(float(thrust)/4, limit)
   39         differential = self.inverse @ np.r_[0., moment]
   40         scale = 1.
   41         for value in differential:
   42             if value > 0:
   43                 scale = min(scale, (limit-collective)/value)
   44             elif value < 0:
   45                 scale = min(scale, collective/-value)
   46         forces = np.clip(collective+scale*differential, 0., limit)
   47         pwm = forces/self.max_thrust
   48         achieved = self.B @ forces
   49         # Saturation describes a limit applied by the allocator, not whether the
   50         # resulting wrench is numerically close to the request. A relative
   51         # tolerance can hide small but real clipping near a motor limit and
   52         # incorrectly leave the controller's integral enabled.
   53         saturated = float(thrust) > 4. * limit or scale < 1.
   54         margin = float(np.minimum(pwm, self.ceiling-pwm).min())
   55         return Allocation(tuple(pwm), tuple(achieved), saturated, float(scale), margin)
