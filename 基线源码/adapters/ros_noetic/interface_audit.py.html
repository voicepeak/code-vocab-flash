   11 def audit(config, duration):
   12     import rosgraph
   13     import roslib.message
   14     import rospy
   15     import rosservice
   16 
   17     rospy.init_node("rmua_next_interface_audit", anonymous=True, disable_signals=True)
   18     master = rosgraph.Master(rospy.get_name())
   19     types = dict(master.getTopicTypes())
   20     publishers, subscribers, _ = master.getSystemState()
   21     pub_map, sub_map = dict(publishers), dict(subscribers)
   22     samples = {topic: deque(maxlen=10000) for topic in config["sensors"]}
   23     listeners = []
   24     lock = threading.Lock()
   25     observed = {}
   26 
   27     def receive(topic, msg):
   28         arrival = time.monotonic()
   29         header = getattr(msg, "header", None)
   30         stamp = header.stamp.to_sec() if header else None
   31         # ROS now - stamp is diagnostic clock difference, not proven actuation latency.
   32         ros_now = rospy.Time.now().to_sec()
   33         age = ros_now - stamp if stamp is not None and stamp > 0 else None
   34         with lock:
   35             samples[topic].append((arrival, stamp, age))
   36 
   37     for topic in config["sensors"]:
   38         if "/debug/" in topic:
   39             raise ValueError("debug topics are excluded from algorithm inputs")
   40         type_name = types.get(topic)
   41         cls = roslib.message.get_message_class(type_name) if type_name else None
   42         observed[topic] = {"type": type_name, "publishers": pub_map.get(topic, []),
   43                            "message_class_available": cls is not None}
   44         if cls is not None:
   45             listeners.append(rospy.Subscriber(topic, cls, lambda msg, t=topic: receive(t, msg),
   46                                               queue_size=1))
   47     controls = {topic: {"type": types.get(topic), "publishers": pub_map.get(topic, []),
   48                         "subscribers": sub_map.get(topic, [])} for topic in config["control_topics"]}
   49     services = {name: rosservice.get_service_type(name) for name in config["services"]}
   50     try:
   51         deadline = time.monotonic() + duration
   52         while time.monotonic() < deadline and not rospy.is_shutdown():
   53             time.sleep(min(0.05, max(0.0, deadline - time.monotonic())))
   54     finally:
   55         for listener in listeners:
   56             listener.unregister()
   57     with lock:
   58         captured = {topic: list(rows) for topic, rows in samples.items()}
   59     for topic, rows in captured.items():
   60         stamps = [r[1] for r in rows if r[1] is not None]
   61         ages = [r[2] for r in rows if r[2] is not None and math.isfinite(r[2])]
   62         span = rows[-1][0] - rows[0][0] if len(rows) > 1 else 0
   63         observed[topic].update({
   64             "received_count": len(rows),
   65             "receiver_hz": (len(rows) - 1) / span if span > 0 else None,
   66             "stamp_regressions": sum(b < a for a, b in zip(stamps, stamps[1:])),
   67             "ros_clock_difference_median_s": statistics.median(ages) if ages else None,
   68         })
   69     return {"topics": observed, "controls": controls, "services": services,
   70             "missing_topics": [t for t in config["sensors"] if not observed[t]["received_count"]],
   71             "formal_server_permissions_verified": False,
   72             "pwm_range_and_motor_signs_verified": False,
   73             "clock_note": "receiver rate is wall time; ROS clock difference is not end-to-end control latency"}

   76 def main():
   77     parser = argparse.ArgumentParser(description=__doc__)
   78     parser.add_argument("--duration", type=float, default=10.0)
   79     parser.add_argument("--run-dir", type=Path, required=True,
   80                         help="existing exclusive run prepared on the host")
   81     args = parser.parse_args()
   82     if not math.isfinite(args.duration) or not 0 < args.duration <= 120:
   83         parser.error("duration must be in (0, 120] seconds")
   84     run = args.run_dir
   85     manifest = json.loads((run / "manifest.json").read_text())
   86     config = manifest["config"]
   87     print("Run directory:", run, flush=True)
   88     code = 1
   89     try:
   90         report = audit(config, args.duration)
   91         (run / "interface_audit.json").write_text(json.dumps(report, indent=2) + "\n")
   92         manifest["topic_manifest"] = "interface_audit.json"
   93         manifest["audit_state"] = "INCOMPLETE" if report["missing_topics"] else "OBSERVED"
   94         code = 2 if report["missing_topics"] else 0
   95     except Exception as error:
   96         manifest["audit_state"] = "FAILED"
   97         manifest["error"] = str(error)
   98         print("Audit failed:", error, flush=True)
   99     finally:
  100         (run / "manifest.json").write_text(json.dumps(manifest, indent=2) + "\n")
  101     return code
