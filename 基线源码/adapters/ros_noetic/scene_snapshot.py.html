    1 """Read-only local diagnostics from official cameras and LiDAR, no debug inputs."""
    2 import argparse
    3 import json
    4 from pathlib import Path
    5 import numpy as np

    8 def main():
    9     import cv2
   10     import rospy
   11     from cv_bridge import CvBridge
   12     from sensor_msgs.msg import Image,PointCloud2
   13     from sensor_msgs import point_cloud2
   14     parser=argparse.ArgumentParser(description=__doc__)
   15     parser.add_argument('--run-dir',type=Path,required=True)
   16     args=parser.parse_args()
   17     rospy.init_node('rmua_scene_snapshot',anonymous=True,disable_signals=True)
   18     bridge=CvBridge()
   19     result={'input_policy':'official_sensors_only','cameras':{}}
   20     for camera in ('front_left','front_right','back_left','back_right'):
   21         msg=rospy.wait_for_message('/airsim_node/drone_1/'+camera+'/Scene',Image,timeout=5)
   22         frame=bridge.imgmsg_to_cv2(msg,desired_encoding='bgr8')
   23         path=args.run_dir/(camera+'.png')
   24         if path.exists():raise RuntimeError('snapshot already exists')
   25         if not cv2.imwrite(str(path),frame):raise RuntimeError('image write failed')
   26         result['cameras'][camera]={'stamp':msg.header.stamp.to_sec(),'frame_id':msg.header.frame_id}
   27     msg=rospy.wait_for_message('/airsim_node/drone_1/lidar',PointCloud2,timeout=5)
   28     points=np.asarray(list(point_cloud2.read_points(msg,field_names=('x','y','z'),skip_nans=True)),dtype=float).reshape(-1,3)
   29     np.savez_compressed(str(args.run_dir/'scene_lidar.npz'),points=points)
   30     result['lidar']={'stamp':msg.header.stamp.to_sec(),'frame_id':msg.header.frame_id,'points':len(points)}
   31     (args.run_dir/'scene_snapshot.json').write_text(json.dumps(result,indent=2)+'\n')
   32     return 0
   33 
   34
