   36 def decode_xyz(message, epoch, ttl_s=.2, expected_frame='lidar'):
   37     """Preserve the sensor frame and stamp; drop nonfinite returns as UNKNOWN.
   38 
   39     Handles row/point padding and both byte orders. No dense/is_dense assumption,
   40     ray timing, no-return inference, or body/world coordinate conversion.
   41     """
   42     stamp = message.header.stamp.to_sec()
   43     if not math.isfinite(stamp) or stamp <= 0:
   44         raise ValueError('invalid lidar timestamp')
   45     if not math.isfinite(ttl_s) or not 0 < ttl_s <= .5:
   46         raise ValueError('invalid lidar lifetime')
   47     if not expected_frame or message.header.frame_id != expected_frame:
   48         raise ValueError('unexpected lidar frame')
   49     width = integer(message.width, 'width')
   50     height = integer(message.height, 'height')
   51     step = integer(message.point_step, 'point_step', 1)
   52     row_step = integer(message.row_step, 'row_step')
   53     if (width * height > 250000 or step > 1024 or row_step < width * step
   54             or row_step * height > 16 * 1024 * 1024):
   55         raise ValueError('oversized or inconsistent cloud layout')
   56     if len(message.data) != row_step * height:
   57         raise ValueError('truncated or oversized cloud data')
   58     data = bytes(message.data)
   59     fields = {}
   60     sizes = {1: 1, 2: 1, 3: 2, 4: 2, 5: 4, 6: 4, 7: 4, 8: 8}
   61     for field in message.fields:
   62         offset = integer(field.offset, 'field offset')
   63         count = integer(field.count, 'field count', 1)
   64         datatype = integer(field.datatype, 'field datatype', 1)
   65         if (not field.name or field.name in fields or datatype not in sizes
   66                 or offset + count * sizes[datatype] > step):
   67             raise ValueError('invalid or duplicate point field')
   68         fields[field.name] = field
   69     endian = '>' if message.is_bigendian else '<'
   70     points = np.empty((height * width, 3), dtype=np.float64)
   71     for axis, name in enumerate(('x', 'y', 'z')):
   72         field = fields.get(name)
   73         if field is None or field.count != 1 or field.datatype not in (7, 8):
   74             raise ValueError('XYZ requires scalar FLOAT32 or FLOAT64 fields')
   75         if width * height:
   76             view = np.ndarray((height, width), dtype=endian + ('f4' if field.datatype == 7 else 'f8'),
   77                               buffer=data, offset=field.offset, strides=(row_step, step))
   78             points[:, axis] = view.reshape(-1)
   79     finite = np.isfinite(points).all(axis=1)
   80     metadata = Metadata(stamp, stamp + ttl_s, integer(epoch, 'epoch'),
   81                         'official_lidar_xyz', message.header.frame_id)
   82     return LidarCloud(metadata, points[finite], len(points), int((~finite).sum()))
