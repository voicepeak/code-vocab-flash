   19 def run_experiment(run, duration, step,yaw_motion=False,braking=False):
   20     import rosgraph
   21     import rospy
   22     from airsim_ros.msg import VelCmd
   23     rospy.init_node('rmua_next_vel_benchmark', anonymous=True, disable_signals=True)
   24     master = rosgraph.Master(rospy.get_name())
   25     rows, fault = [], None
   26     latest_state,latest_raw,latest_elapsed=None,None,None
   27     fresh_read=False
   28     fault_observation=None
   29     last_published_wall=None
   30     connections = 0
   31     def graph(require_self):
   32         pubs, _, _ = master.getSystemState()
   33         check_graph(dict(pubs), rospy.get_name(), 'VEL', require_self)
   34     with ControlLease(rospy.get_param('/run_id'), 'VEL'):
   35         graph(False)
   36         sensors = FlightSensors()
   37         publisher = None
   38         monitor = None
   39         try:
   40             deadline = time.monotonic()+5
   41             while time.monotonic() < deadline:
   42                 try:
   43                     state, raw = sensors.read()
   44                     if state.health == Health.OK:
   45                         break
   46                 except (ValueError, RuntimeError):
   47                     pass
   48                 time.sleep(.01)
   49             else:
   50                 raise RuntimeError('NO_HEALTHY_STATE')
   51             publisher = rospy.Publisher('/airsim_node/drone_1/vel_body_cmd', VelCmd, queue_size=1)
   52             deadline = time.monotonic()+2
   53             while publisher.get_num_connections() != 1 and time.monotonic() < deadline:
   54                 time.sleep(.01)
   55             connections = publisher.get_num_connections()
   56             if connections != 1:
   57                 raise RuntimeError('VEL_SUBSCRIBER_MD5_OR_COUNT')
   58             graph(True)
   59             _, subscribers, _ = master.getSystemState()
   60             inputs = sorted(topic for topic, nodes in subscribers if rospy.get_name() in nodes)
   61             config = json.loads((Path(__file__).resolve().parents[2]/'configs/official_baseline.yaml').read_text())
   62             if any(topic not in config['sensors'] for topic in inputs):
   63                 raise RuntimeError('INPUT_WHITELIST_POLICY')
   64             (run/'runtime_input_audit.json').write_text(json.dumps({'node':rospy.get_name(),
   65                 'topics':inputs,'debug_inputs':False},indent=2)+'\n')
   66             monitor=GraphMonitor(lambda:graph(True))
   67             monitor.start()
   68             state, raw = sensors.read()
   69             origin = np.array(state.position_ned)
   70             R = rotation(state.orientation_wxyz)
   71             yaw = math.atan2(R[1, 0], R[0, 0])
   72             guard = CommandGuard(sensors.clock.epoch, minimum_period=.0105)
   73             start = time.monotonic()
   74             start_stamp = state.metadata.stamp
   75             next_tick = start
   76             sequence = 0
   77             while time.monotonic()-start < duration:
   78                 now = time.monotonic()
   79                 fresh_read=False
   80                 state, raw = sensors.read()
   81                 latest_state,latest_raw,latest_elapsed=state,raw,now-start
   82                 fresh_read=True
   83                 if state.health != Health.OK:
   84                     raise RuntimeError('STATE_HEALTH_'+state.health.value)
   85                 if rotation(state.orientation_wxyz)[2,2] < math.cos(.35):
   86                     raise RuntimeError('TILT_LIMIT')
   87                 if np.linalg.norm(np.array(state.position_ned)-origin) > 1.8:
   88                     raise RuntimeError('POSITION_LIMIT')
   89                 if np.linalg.norm(state.velocity_ned) > 4:
   90                     raise RuntimeError('SPEED_LIMIT')
   91                 if np.linalg.norm(state.angular_velocity_body) > 1.:
   92                     raise RuntimeError('ANGULAR_RATE_LIMIT')
   93                 monitor.check()
   94                 sim_elapsed = state.metadata.stamp-start_stamp
   95                 reference, velocity, _ = benchmark_reference(origin, sim_elapsed, step,braking)
   96                 yaw_command,yaw_rate=benchmark_yaw(yaw,sim_elapsed,yaw_motion)
   97                 values = body_velocity(state, reference, velocity, yaw_command,yaw_rate)
   98                 command = VelocityCommand(state.metadata.stamp, state.metadata.epoch, sequence, values)
   99                 with sensors.lock:
  100                     accepted_wall = time.monotonic()
  101                     allowed = guard.accept(command, sensors.clock, accepted_wall)
  102                 if allowed:
  103                     msg = VelCmd()
  104                     msg.header.stamp = rospy.Time.from_sec(command.stamp)
  105                     msg.header.seq = sequence
  106                     msg.vx, msg.vy, msg.vz, msg.yawRate = values
  107                     msg.va, msg.stop = 4, 0
  108                     publisher.publish(msg)
  109                     published_wall=time.monotonic()
  110                     last_published_wall=published_wall
  111                     guard.published(published_wall)
  112                     rows.append({'wall_elapsed':now-start,'guard_accept_wall_elapsed':accepted_wall-start,
  113                         'publish_wall_elapsed':published_wall-start, 'command':asdict(command),
  114                         'sim_elapsed':sim_elapsed,
  115                         'yaw_reference':yaw_command,'yaw_rate_reference':yaw_rate,
  116                         'raw':raw,'reference':reference,
  117                         'state':{k:v for k,v in asdict(state).items() if k != 'metadata'}})
  118                     sequence += 1
  119                 next_tick += .02
  120                 delay = next_tick-time.monotonic()
  121                 if delay < -.1:
  122                     raise RuntimeError('CONTROL_DEADLINE')
  123                 time.sleep(max(0.,delay))
  124         except Exception as error:
  125             fault = str(error)
  126             failed_wall=time.monotonic()
  127             fault_observation={'reason':fault,'fresh_read_in_fault_iteration':fresh_read,
  128                 'wall_elapsed_at_last_read':latest_elapsed,
  129                 'wall_since_last_publication_s':None if last_published_wall is None else failed_wall-last_published_wall,
  130                 'state':asdict(latest_state) if latest_state else None,'raw':latest_raw}
  131         finally:
  132             if monitor:
  133                 monitor.close()
  134             if publisher:
  135                 publisher.unregister()
  136             sensors.close()
  137     (run/'vel_telemetry.jsonl').write_text(''.join(json.dumps(row)+'\n' for row in rows))
  138     if fault_observation is not None:
  139         (run/'fault_observation.json').write_text(json.dumps(fault_observation,indent=2)+'\n')
  140     result = {'commands':len(rows),'fault':fault,'control_mode':'VEL',
  141               'connected_subscribers':connections,'message_md5':VelCmd._md5sum,
  142               'duration_s':duration,'step':step,'official_result':'UNKNOWN',
  143               'reference_clock':'IMU',
  144               'graph_monitor':monitor.summary() if monitor else None,
  145               'yaw_motion':yaw_motion,
  146               'braking':braking,
  147               'host_scene_stop_required':True}
  148     (run/'vel_summary.json').write_text(json.dumps(result,indent=2)+'\n')
  149     print(json.dumps(result),flush=True)
  150     return 0 if rows and fault is None else 2

  153 def main():
  154     parser = argparse.ArgumentParser(description=__doc__)
  155     parser.add_argument('--run-dir',type=Path,required=True)
  156     parser.add_argument('--hover-seconds',type=float,required=True)
  157     parser.add_argument('--step',action='store_true')
  158     parser.add_argument('--yaw-motion',action='store_true')
  159     parser.add_argument('--braking',action='store_true')
  160     args=parser.parse_args()
  161     if not 3 <= args.hover_seconds <= 20 or (args.step and args.hover_seconds < 10):
  162         parser.error('bounded VEL duration must be [3,20], step >=10')
  163     if (args.braking or args.yaw_motion) and args.hover_seconds < 10:
  164         parser.error('braking/yaw identification requires >=10s')
  165     if args.braking and args.step:
  166         parser.error('braking and position step are separate experiments')
  167     return run_experiment(args.run_dir,args.hover_seconds,args.step,args.yaw_motion,args.braking)
