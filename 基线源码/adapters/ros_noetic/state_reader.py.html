   14 def pose_values(pose):
   15     p, q = pose.position, pose.orientation
   16     return (p.x, p.y, p.z), (q.w, q.x, q.y, q.z)

   19 def observe(run, duration):
   20     import rospy
   21     from geometry_msgs.msg import PoseStamped
   22     from sensor_msgs.msg import Imu
   23 
   24     rospy.init_node("rmua_next_state_reader", anonymous=True, disable_signals=True)
   25     lock = threading.RLock()
   26     clock = SensorClock()
   27     estimator = ImuGpsEstimator()
   28     events = Counter()
   29     initial = None
   30     health = Counter()
   31     count = 0
   32     with (run / "state.jsonl").open("x") as output:
   33         def initial_pose(msg):
   34             nonlocal initial
   35             with lock:
   36                 try:
   37                     initial = pose_values(msg.pose)
   38                 except ValueError:
   39                     events["INVALID_INITIAL"] += 1
   40 
   41         def imu(msg):
   42             nonlocal estimator, initial, count
   43             with lock:
   44                 try:
   45                     stamp = msg.header.stamp.to_sec()
   46                     previous = clock.stamp
   47                     if clock.observe(stamp, time.monotonic()):
   48                         estimator = ImuGpsEstimator(clock.epoch)
   49                         initial = None  # Require a newly received initial pose after reset.
   50                         events["CLOCK_RESET"] += 1
   51                     elif previous == stamp:
   52                         events["IMU_DUPLICATE"] += 1
   53                         return
   54                     if initial is not None and not estimator.initialized:
   55                         estimator.initialize(*initial)
   56                     a, w = msg.linear_acceleration, msg.angular_velocity
   57                     estimator.propagate(stamp, (a.x, a.y, a.z), (w.x, w.y, w.z))
   58                     if estimator.initialized:
   59                         state = estimator.snapshot()
   60                         output.write(json.dumps(asdict(state), allow_nan=False)+"\n")
   61                         health[state.health.value] += 1
   62                         count += 1
   63                 except (ValueError, ArithmeticError) as error:
   64                     events["INVALID_IMU:"+str(error)] += 1
   65 
   66         def gps(msg):
   67             with lock:
   68                 try:
   69                     stamp = msg.header.stamp.to_sec()
   70                     if stamp <= 0 or not clock.fresh(time.monotonic()):
   71                         events["GPS_NO_VALID_CLOCK"] += 1
   72                         return
   73                     estimator.update_gps(stamp, *pose_values(msg.pose))
   74                 except (ValueError, ArithmeticError) as error:
   75                     events["INVALID_GPS:"+str(error)] += 1
   76 
   77         subscribers = [
   78             rospy.Subscriber("/airsim_node/initial_pose", PoseStamped, initial_pose, queue_size=1),
   79             rospy.Subscriber("/airsim_node/drone_1/imu/imu", Imu, imu, queue_size=1),
   80             rospy.Subscriber("/airsim_node/drone_1/gps", PoseStamped, gps, queue_size=1),
   81         ]
   82         deadline = time.monotonic()+duration
   83         try:
   84             while time.monotonic() < deadline and not rospy.is_shutdown():
   85                 time.sleep(.02)
   86         finally:
   87             for subscriber in subscribers:
   88                 subscriber.unregister()
   89         with lock:
   90             result = {"state_count": count, "health_counts": dict(health),
   91                       "events": dict(events), "gps_rejections": estimator.rejections,
   92                       "epoch": clock.epoch, "control_published": False,
   93                       "noise_calibrated": False, "official_result": "UNKNOWN"}
   94             if count:
   95                 result["last_state"] = asdict(estimator.snapshot())
   96             (run / "state_summary.json").write_text(json.dumps(result, indent=2)+"\n")
   97     return 0 if count else 2

  101     parser = argparse.ArgumentParser(description=__doc__)
