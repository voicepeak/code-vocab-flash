   36     def on_initial(self, msg):

   41     def on_imu(self, msg):
   42         callback_arrival=time.monotonic()
   43         with self.lock:
   44             try:
   45                 stamp = msg.header.stamp.to_sec()
   46                 if self.local_injection is not None:
   47                     if self.local_injection.affects('IMU_DROP', stamp):
   48                         return
   49                     if self.local_injection.affects('IMU_ROLLBACK', stamp):
   50                         stamp -= 1.
   51                 previous = self.clock.stamp
   52                 if self.clock.observe(stamp, time.monotonic()):
   53                     if self.pose_history is not None:self.pose_history.reset(self.clock.epoch)
   54                     self.fault = 'IMU_CLOCK_RESET'
   55                     return
   56                 if previous == stamp:
   57                     return
   58                 if self.initial is not None and not self.estimator.initialized:
   59                     self.estimator.initialize(*self.initial)
   60                 a, w = msg.linear_acceleration, msg.angular_velocity
   61                 acceleration, omega = (a.x, a.y, a.z), (w.x, w.y, w.z)
   62                 self.estimator.propagate(stamp, acceleration, omega)
   63                 if self.pending_gps is not None and self.pending_gps[0] <= stamp:
   64                     try:
   65                         if self.estimator.update_gps(*self.pending_gps):
   66                             self.bootstrap_gps.append(self.pending_gps)
   67                     except (ValueError, ArithmeticError) as error:
   68                         self.estimator.reject('INVALID_GPS:'+str(error))
   69                     self.pending_gps = None
   70                 self.raw = {'stamp': stamp, 'acceleration_body': acceleration,
   71                             'omega_body': omega}
   72                 attitude = getattr(msg,'orientation',None)
   73                 if attitude is not None:
   74                     # Record for calibration only; it does not enter the filter.
   75                     self.raw['imu_orientation_wxyz']=(attitude.w,attitude.x,attitude.y,attitude.z)
   76                 if not self.bootstrap_applied:self.bootstrap_imu.append(dict(self.raw))
   77                 if self.pose_history is not None and self.estimator.initialized:
   78                     self.pose_history.append(self.estimator.snapshot(), callback_arrival)
   79             except (ValueError, ArithmeticError) as error:
   80                 self.fault = str(error)

   82     def on_gps(self, msg):
   83         with self.lock:
   84             try:
   85                 stamp = msg.header.stamp.to_sec()
   86                 if stamp > 0 and self.clock.fresh(time.monotonic()):
   87                     position, attitude = pose_values(msg.pose)
   88                     if self.local_injection is not None and self.local_injection.affects('GPS_ZERO',stamp):
   89                         position, attitude = (0.,0.,0.), (0.,0.,0.,0.)
   90                     if self.clock.stamp < stamp <= self.clock.stamp+.15:
   91                         # ROS callback arrival order is not sensor measurement order.
   92                         # Defer future GPS to the first IMU propagation covering it.
   93                         self.pending_gps = (stamp, position, attitude)
   94                     else:
   95                         if self.estimator.update_gps(stamp, position, attitude):
   96                             self.bootstrap_gps.append((stamp,position,attitude))
   97             except (ValueError, ArithmeticError) as error:
   98                 self.estimator.reject('INVALID_GPS:'+str(error))

  100     def read(self):
  101         with self.lock:
  102             if self.fault:
  103                 raise RuntimeError(self.fault)
  104             if not self.clock.fresh(time.monotonic()):
  105                 raise RuntimeError('STALE_IMU')
  106             raw = dict(self.raw)
  107             raw['last_gps_stamp'] = self.estimator.last_gps
  108             raw['gyro_bias_body'] = tuple(self.estimator.bg)
  109             raw['accel_bias_body'] = tuple(self.estimator.ba)
  110             return self.estimator.snapshot(), raw

  112     def bootstrap_at_rest_before_control(self):
  113         """Caller must not have created a control publisher; never call in flight."""
  114         from estimation.stationary_bootstrap import resting_window
  115         with self.lock:
  116             if self.bootstrap_applied:raise RuntimeError('BOOTSTRAP_ALREADY_APPLIED')
  117             if self.fault or self.initial is None or not self.clock.fresh(time.monotonic()):return None
  118             result=resting_window(self.bootstrap_imu,self.bootstrap_gps,self.initial[1],self.clock.stamp)
  119             if result is None:return None
  120             e=self.estimator
  121             e.p=np.array(result['position']);e.v=np.zeros(3)
  122             e.q=np.array(self.initial[1],dtype=float);e.q/=np.linalg.norm(e.q)
  123             e.bg=np.array(result['gyro_bias']);e.ba=np.zeros(3)
  124             # Retain conservative GPS/attitude/bias uncertainty; averaging does
  125             # not establish independent measurement noise or identify accel bias.
  126             e.P=np.diag([.01]*3+[.01]*3+[.01]*3+[.01]*3+[.001]*3)
  127             e.omega=np.array(self.raw['omega_body'])-e.bg
  128             self.bootstrap_applied=True
  129             return result
