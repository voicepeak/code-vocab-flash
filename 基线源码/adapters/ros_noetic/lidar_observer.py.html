   20 def observe(run, duration):

   22     import rospy
   23     from sensor_msgs.msg import PointCloud2

   25     rospy.init_node('rmua_lidar_endpoints', anonymous=True, disable_signals=True)
   26     lock = threading.Lock()
   27     pending = None
   28     received = 0
   29 
   30     def receive(message):
   31         nonlocal pending, received
   32         with lock:
   33             pending = (message, time.monotonic())
   34             received += 1
   35

   39     previous = None
   40     latest = None
   41     schemas = []

   45     graph = None

   48         with (run / 'lidar_frames.jsonl').open('x') as output:
   49             while time.monotonic() - started < duration and not rospy.is_shutdown():
   50                 now = time.monotonic()

   52                     raise RuntimeError('STALE_LIDAR_STREAM')
   53                 if last_processed_arrival is None and now - started > 5:
   54                     raise RuntimeError('NO_VALID_LIDAR_STREAM')

   56                     item, pending = pending, None
   57                 if item is None:
   58                     time.sleep(.002)
   59                     continue

   61                 begin = time.monotonic()

   63                     events['STALE_PENDING'] += 1
   64                     continue
   65                 try:
   66                     cloud = decode_xyz(message, epoch)
   67                     stamp = cloud.metadata.stamp
   68                     if previous is not None and stamp <= previous:
   69                         if stamp == previous:
   70                             events['DUPLICATE'] += 1
   71                             continue
   72                         epoch += 1
   73                         latest = None
   74                         events['CLOCK_RESET'] += 1
   75                         cloud = decode_xyz(message, epoch)
   76                     previous = stamp
   77                     snapshot = endpoint_snapshot(cloud, stamp, epoch)
   78                 except (ValueError, ArithmeticError) as error:
   79                     events['REJECTED:' + str(error)] += 1
   80                     continue
   81                 schema = {'frame_id': message.header.frame_id,
   82                           'point_step': message.point_step, 'is_bigendian': message.is_bigendian,
   83                           'fields': [{'name': f.name, 'offset': f.offset, 'datatype': f.datatype,
   84                                       'count': f.count} for f in message.fields]}

   86                     schemas.append(schema)
   87                 if count == 0:
   88                     with (run / 'lidar_first_message.bin').open('xb') as raw:
   89                         raw.write(bytes(message.data))

   91                                  row_step=message.row_step, is_dense=message.is_dense)
   92                     (run / 'lidar_first_message.json').write_text(json.dumps(first, indent=2) + '\n')

   94                 row = {'stamp': stamp, 'epoch': epoch, 'frame_id': cloud.metadata.frame_id,
   95                        'arrival_wall_monotonic': arrival, 'processing_begin_wall_monotonic': begin,
   96                        'processing_end_wall_monotonic': time.monotonic(),
   97                        'input_points': cloud.input_points, 'finite_points': len(cloud.points),
   98                        'nonfinite_points': cloud.nonfinite_points,
   99                        'unconfirmed_zero_or_max_range_points': int(((ranges <= 0) | (ranges >= 30)).sum()),
  100                        'occupied_voxels': len(snapshot.cells), 'free_voxels': 0,
  101                        'motion_compensated': False, 'world_transform_applied': False}
  102                 output.write(json.dumps(row, allow_nan=False) + '\n')
  103                 latest = snapshot
  104                 last_processed_arrival = arrival
  105                 count += 1
  106             if last_processed_arrival is None or time.monotonic() - last_processed_arrival > .5:
  107                 raise RuntimeError('STALE_LIDAR_STREAM')
  108             # Only the observer's registrations are audited; the simulator's
  109             # debug publishers are not algorithm subscriptions.
  110             pubs, subs, _ = rosgraph.Master(rospy.get_name()).getSystemState()
  111             node = rospy.get_name()
  112             graph = {'subscriptions': sorted(t for t, nodes in subs if node in nodes),
  113                      'publishers': sorted(t for t, nodes in pubs if node in nodes)}
  114             if graph['subscriptions'] != [TOPIC] or set(graph['publishers']) - {'/rosout'}:
  115                 raise RuntimeError('LIDAR_OBSERVER_GRAPH_POLICY')

  117         fault = str(error)
  118     finally:
  119         listener.unregister()
  120         if latest is not None:
  121             (run / 'lidar_endpoint_snapshot.json').write_text(json.dumps(asdict(latest), indent=2) + '\n')
  122         summary = {'processed_frames': count, 'received_frames': received, 'events': dict(events),
  123                    'fault': fault, 'schemas': schemas, 'node_graph': graph,
  124                    'control_published': False, 'debug_inputs_used': False,
  125                    'frame_semantics': 'frozen_sensor_frame_at_scan_stamp',
  126                    'motion_compensated': False, 'free_space_certified': False,
  127                    'flight_clearance_eligible': False, 'official_result': 'UNKNOWN'}
  128         (run / 'lidar_summary.json').write_text(json.dumps(summary, indent=2) + '\n')
  129     return 0 if count and fault is None else 2
