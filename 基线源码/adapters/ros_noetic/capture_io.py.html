    8 def save_message(directory, prefix, message, kind):
    9     if prefix not in ('left', 'right', 'lidar') or kind not in ('image', 'lidar'):
   10         raise ValueError('unsupported captured sensor')
   11     directory = Path(directory)
   12     data = bytes(message.data)
   13     if len(data) > 32 * 1024 * 1024:
   14         raise ValueError('oversized captured payload')
   15     stamp = message.header.stamp
   16     document = {'kind': kind, 'stamp_secs': stamp.secs, 'stamp_nsecs': stamp.nsecs,
   17                 'stamp': stamp.to_sec(), 'sequence': message.header.seq,
   18                 'frame_id': message.header.frame_id, 'width': message.width, 'height': message.height,
   19                 'is_bigendian': message.is_bigendian, 'payload_file': prefix + '.bin.gz',
   20                 'raw_bytes': len(data), 'raw_sha256': hashlib.sha256(data).hexdigest()}
   21     if kind == 'image':
   22         document.update(encoding=message.encoding, step=message.step)
   23     else:
   24         document.update(point_step=message.point_step, row_step=message.row_step, is_dense=message.is_dense,
   25                         fields=[{'name': f.name, 'offset': f.offset, 'datatype': f.datatype, 'count': f.count}
   26                                 for f in message.fields])
   27     with (directory / document['payload_file']).open('xb') as raw:
   28         with gzip.GzipFile(filename='', mode='wb', fileobj=raw, compresslevel=1, mtime=0) as output:
   29             output.write(data)
   30     with (directory / (prefix + '.json')).open('x') as output:
   31         output.write(json.dumps(document, indent=2, allow_nan=False) + '\n')
   32     return document

   35 def read_payload(path):
   36     path = Path(path)
   37     document = json.loads(path.read_text())
   38     size = document['raw_bytes']
   39     name = document['payload_file']
   40     if (not isinstance(size, int) or not 0 <= size <= 32 * 1024 * 1024
   41             or Path(name).name != name or name in ('.', '..')):
   42         raise ValueError('invalid retained sensor descriptor')
   43     with gzip.open(path.parent / name, 'rb') as source:
   44         payload = source.read(size + 1)
   45     if len(payload) != size or hashlib.sha256(payload).hexdigest() != document['raw_sha256']:
   46         raise ValueError('captured payload differs from its descriptor')
   47     return document, payload

   50 def load_message(path):
   51     import rospy
   52     from sensor_msgs.msg import Image, PointCloud2, PointField
   53 
   54     document, payload = read_payload(path)
   55     if document['kind'] == 'image':
   56         message = Image()
   57         keys = ('width', 'height', 'step', 'is_bigendian', 'encoding')
   58     elif document['kind'] == 'lidar':
   59         message = PointCloud2()
   60         keys = ('width', 'height', 'point_step', 'row_step', 'is_bigendian', 'is_dense')
   61         message.fields = [PointField(**f) for f in document['fields']]
   62     else:
   63         raise ValueError('unknown captured sensor kind')
   64     message.header.stamp = rospy.Time(document['stamp_secs'], document['stamp_nsecs'])
   65     message.header.seq = document['sequence']
   66     message.header.frame_id = document['frame_id']
   67     for key in keys:
   68         setattr(message, key, document[key])
   69     message.data = payload
   70     if message.header.stamp.to_sec() != document['stamp']:
   71         raise ValueError('captured timestamp did not reconstruct exactly')
   72     return message
