   20 def observe(run, duration):
   21     import rosgraph
   22     import rospy
   23     from sensor_msgs.msg import Image
   24 
   25     rospy.init_node('rmua_pose_history_observer', anonymous=True, disable_signals=True)
   26     history = PoseHistory()
   27     sensors = FlightSensors(pose_history=history)
   28     lock = threading.Lock()
   29     pending = None
   30 
   31     def receive(message):
   32         nonlocal pending
   33         with lock:
   34             pending = (message.header.stamp.to_sec(), time.monotonic(), message.header.frame_id)
   35 
   36     listener = rospy.Subscriber(CAMERA, Image, receive, queue_size=1, buff_size=4000000)
   37     started = time.monotonic()
   38     dumped = None
   39     count = matched = 0
   40     reasons = Counter()
   41     fault = None
   42     graph = None
   43     source_count = 0
   44     last_image_arrival = None
   45     with (run / 'history_states.jsonl').open('x') as sources, (run / 'camera_history.jsonl').open('x') as output:
   46         def dump_sources():
   47             nonlocal dumped, source_count
   48             for state, arrival in history.snapshot_sources():
   49                 key = (state.metadata.epoch, state.metadata.stamp)
   50                 if dumped is None or key > dumped:
   51                     sources.write(json.dumps({'state': asdict(state), 'arrival_wall_monotonic': arrival},
   52                                              allow_nan=False) + '\n')
   53                     dumped = key
   54                     source_count += 1
   55 
   56         try:
   57             while time.monotonic() - started < duration and not rospy.is_shutdown():
   58                 with sensors.lock:
   59                     if sensors.fault:
   60                         raise RuntimeError(sensors.fault)
   61                     ready = sensors.estimator.initialized and sensors.raw is not None
   62                     if ready and not sensors.clock.fresh(time.monotonic()):
   63                         raise RuntimeError('STALE_IMU')
   64                 dump_sources()
   65                 with lock:
   66                     item, pending = pending, None
   67                 if not ready or item is None:
   68                     if time.monotonic() - started > 5 and (not ready or last_image_arrival is None):
   69                         raise RuntimeError('NO_STATE_OR_CAMERA')
   70                     if last_image_arrival is not None and time.monotonic() - last_image_arrival > .5:
   71                         raise RuntimeError('STALE_CAMERA')
   72                     time.sleep(.003)
   73                     continue
   74                 stamp, arrival, header_frame = item
   75                 last_image_arrival = arrival
   76                 begin = time.monotonic()
   77                 pose = None
   78                 reason = None
   79                 with sensors.lock:
   80                     if sensors.fault:
   81                         raise RuntimeError(sensors.fault)
   82                     now, epoch = sensors.clock.stamp, sensors.clock.epoch
   83                     query_wall = time.monotonic()
   84                     try:
   85                         pose = history.lookup(stamp, now, query_wall, epoch)
   86                     except HistoryUnavailable as error:
   87                         reason = str(error)
   88                 elapsed = time.monotonic() - begin
   89                 output.write(json.dumps({'image_stamp': stamp, 'image_header_frame_id': header_frame,
   90                     'image_arrival_wall_monotonic': arrival, 'query_wall_monotonic': query_wall,
   91                     'query_now': now, 'epoch': epoch, 'lookup_elapsed_wall_s': elapsed,
   92                     'reason': reason, 'historical_pose': asdict(pose) if pose else None,
   93                     'world_projection_applied': False, 'current_control_eligible': False}, allow_nan=False) + '\n')
   94                 count += 1
   95                 if pose is not None:
   96                     matched += 1
   97                 else:
   98                     reasons[reason] += 1
   99                 time.sleep(.003)
  100         except Exception as error:
  101             fault = str(error)
  102         finally:
  103             try:
  104                 pubs, subs, _ = rosgraph.Master(rospy.get_name()).getSystemState()
  105                 node = rospy.get_name()
  106                 graph = {'subscriptions': sorted(t for t, nodes in subs if node in nodes),
  107                          'publishers': sorted(t for t, nodes in pubs if node in nodes)}
  108                 if graph['subscriptions'] != INPUTS or set(graph['publishers']) - {'/rosout'}:
  109                     fault = fault or 'HISTORY_OBSERVER_GRAPH_POLICY'
  110             except Exception as error:
  111                 fault = fault or 'GRAPH_AUDIT_FAILED:' + str(error)
  112             listener.unregister()
  113             sensors.close()
  114             dump_sources()
  115     summary = {'images': count, 'historical_matches': matched, 'unavailable_reasons': dict(reasons),
  116                'source_states': source_count, 'fault': fault, 'node_graph': graph,
  117                'horizon_s': history.horizon_s, 'max_bracket_s': history.max_bracket_s,
  118                'control_published': False, 'debug_inputs_used': False,
  119                'image_exposure_time_calibrated': False, 'world_projection_applied': False,
  120                'historical_uncertainty_calibrated': False, 'acceptance_eligible': False}
  121     (run / 'pose_history_summary.json').write_text(json.dumps(summary, indent=2) + '\n')
  122     return 0 if matched and fault is None else 2
