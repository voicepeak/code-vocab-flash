    1 """Bounded official-sensor scene recorder, isolated from the control process.
    2 
    3 Callbacks retain only the latest message. Compression and disk writes run here,
    4 never on the flight loop. No control publishers, task requests or debug inputs.
    5 """
    6 import argparse
    7 import json
    8 from pathlib import Path
    9 import threading
   10 import time
   11 import numpy as np
   12 
   13 
   14 def main():
   15     import cv2
   16     import rospy
   17     import rosgraph
   18     from cv_bridge import CvBridge
   19     from sensor_msgs.msg import Image, PointCloud2
   20     from sensor_msgs import point_cloud2
   21     parser=argparse.ArgumentParser(description=__doc__)
   22     parser.add_argument('--run-dir',type=Path,required=True)
   23     args=parser.parse_args()
   24     output=args.run_dir/'scene_trace'
   25     output.mkdir(exist_ok=False)
   26     rospy.init_node('rmua_scene_trace',anonymous=True,disable_signals=True)
   27     graph_types=rosgraph.Master(rospy.get_name()).getTopicTypes()
   28     bridge=CvBridge()
   29     lock=threading.Lock()
   30     pending={}
   31     last={}
   32     rows=[]
   33     counts={}
   34     def receive(name,msg):
   35         stamp=msg.header.stamp.to_sec()
   36         with lock:
   37             if stamp>last.get(name,float('-inf')):
   38                 pending[name]=(msg,time.monotonic())
   39     cameras=('front_left','front_right','back_left','back_right')
   40     listeners=[rospy.Subscriber('/airsim_node/drone_1/'+name+'/Scene',Image,
   41                lambda msg,n=name:receive(n,msg),queue_size=1,buff_size=4000000) for name in cameras]
   42     listeners.append(rospy.Subscriber('/airsim_node/drone_1/lidar',PointCloud2,
   43                      lambda msg:receive('lidar',msg),queue_size=1,buff_size=4000000))
   44     deadline=time.monotonic()+40
   45     fault=None
   46     try:
   47         while time.monotonic()<deadline and not rospy.is_shutdown() and not (args.run_dir/'scene_stop').exists():
   48             with lock:
   49                 batch=pending.copy()
   50                 pending.clear()
   51             for name,(msg,arrival) in batch.items():
   52                 stamp=msg.header.stamp.to_sec()
   53                 period=.09 if name=='lidar' else .45
   54                 if stamp-last.get(name,float('-inf'))<period or counts.get(name,0)>=400:
   55                     continue
   56                 index=counts.get(name,0)
   57                 relative=name+'_'+str(index).zfill(4)+('.npz' if name=='lidar' else '.png')
   58                 if name=='lidar':
   59                     points=np.asarray(list(point_cloud2.read_points(msg,field_names=('x','y','z'),
   60                                       skip_nans=True)),dtype=np.float32).reshape(-1,3)
   61                     np.savez_compressed(str(output/relative),points=points)
   62                 else:
   63                     frame=bridge.imgmsg_to_cv2(msg,desired_encoding='bgr8')
   64                     if not cv2.imwrite(str(output/relative),frame):
   65                         raise RuntimeError('scene image write failed')
   66                 rows.append({'sensor':name,'stamp':stamp,'frame_id':msg.header.frame_id,
   67                              'arrival_wall_monotonic':arrival,'file':relative})
   68                 last[name]=stamp
   69                 counts[name]=index+1
   70             if all(counts.get(n,0)>0 for n in ('lidar',)+cameras):
   71                 (args.run_dir/'scene_ready').touch(exist_ok=True)
   72             time.sleep(.005)
   73     except Exception as error:
   74         fault=str(error)
   75     finally:
   76         for listener in listeners:listener.unregister()
   77         (output/'index.json').write_text(json.dumps({'input_policy':'official_sensors_only',
   78             'advertised_topic_types_for_interface_audit':graph_types,
   79             'recording_is_lossy':True,'latest_message_only':True,'fault':fault,
   80             'counts':counts,'frames':rows},indent=2)+'\n')
   81     return 0 if fault is None and all(counts.get(n,0)>0 for n in ('lidar',)+cameras) else 2
   82 
   83 
   84 if __name__=='__main__':
   85     raise SystemExit(main())
