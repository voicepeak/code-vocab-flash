   23 def observe(run, duration, image_ttl, settings, method='orb_lk', pipeline='visual_motion'):
   24     import cv2
   25     import rosgraph
   26     import rospy
   27     from sensor_msgs.msg import Image, Imu
   28     cv2.setNumThreads(1)
   29     rospy.init_node('rmua_visual_motion_observer', anonymous=True, disable_signals=True)
   30     lock = threading.Lock(); clock = SensorClock(); pending = {}; received = Counter(); overwritten = Counter()
   31     faults = []; slot = LatestSlot(); stop = threading.Event(); rows = []
   32     rig = load_nominal_rig(settings)
   33     history = PoseHistory() if pipeline == 'gate_track' else None
   34     sensors = FlightSensors(pose_history=history) if history is not None else None
   35     tracking = OnlineGateTracks(history) if history is not None else None
   36     window = GateWindow(rig) if pipeline in ('gate_roi', 'gate_track') else MotionWindow(rig, method=method)
   37     def receive(side, message):
   38         arrival = time.monotonic()
   39         with lock:
   40             received[side] += 1
   41             if side == 'imu':
   42                 try:
   43                     if clock.observe(message.header.stamp.to_sec(), arrival): faults.append('IMU_CLOCK_RESET')
   44                 except ValueError as error: faults.append(str(error))
   45             else:
   46                 if side in pending: overwritten[side] += 1
   47                 pending[side] = (message, arrival)
   48     def snapshot():
   49         with lock:
   50             if sensors is not None:
   51                 with sensors.lock:
   52                     return (sensors.clock.stamp, sensors.clock.epoch,
   53                             sensors.clock.fresh(time.monotonic()) and not sensors.fault and not faults)
   54             return clock.stamp, clock.epoch, clock.fresh(time.monotonic()) and not faults
   55     listeners = [rospy.Subscriber(topic, Imu if side == 'imu' else Image,
   56                  lambda msg, s=side: receive(s, msg), queue_size=1, buff_size=4000000)
   57                  for side, topic in TOPICS.items() if sensors is None or side != 'imu']
   58     def worker():
   59         try:
   60             with (run / 'visual_motion_results.jsonl').open('x') as output:
   61                 while not stop.is_set():
   62                     item = slot.take()
   63                     if item is None:
   64                         time.sleep(.002); continue
   65                     pair, arrival = item
   66                     started = time.monotonic()
   67                     row = window.process(pair, arrival, snapshot, time.monotonic)
   68                     if tracking is not None:
   69                         row['tracking'] = tracking.process(row, pair, arrival, snapshot, time.monotonic)
   70                     row.update(processing_s=time.monotonic()-started, source_pair_stamp=pair.metadata.stamp,
   71                                source_epoch=pair.metadata.epoch, source_skew_s=pair.skew_s)
   72                     output.write(json.dumps(row, allow_nan=False) + '\n'); output.flush(); rows.append(row)
   73         except Exception as error:
   74             with lock: faults.append('WORKER:' + str(error))
   75     thread = threading.Thread(target=worker, daemon=True); thread.start()
   76     buffer = StereoPairBuffer(); arrivals = {side: OrderedDict() for side in ('left', 'right')}
   77     started = time.monotonic(); pairs = 0; graph = None
   78     try:
   79         while time.monotonic() - started < duration and not rospy.is_shutdown():
   80             with lock:
   81                 batch = dict(pending); pending.clear(); errors = list(faults)
   82             if errors: raise RuntimeError(errors[0])
   83             now, epoch, fresh = snapshot()
   84             if now is None:
   85                 if time.monotonic() - started > 5: raise RuntimeError('NO_IMU_CLOCK')
   86                 time.sleep(.002); continue
   87             if not fresh: raise RuntimeError('STALE_IMU_CLOCK')
   88             for side, (message, arrival) in batch.items():
   89                 frame = decode_image(message, side, epoch, image_ttl)
   90                 if buffer.add(side, frame, arrival): arrivals[side][frame.metadata.stamp] = arrival
   91                 while len(arrivals[side]) > 8: arrivals[side].popitem(last=False)
   92             pair = buffer.pop(now, time.monotonic())
   93             if pair is not None:
   94                 source_arrival = min(arrivals[side][getattr(pair, side).metadata.stamp] for side in arrivals)
   95                 slot.put((pair, source_arrival)); pairs += 1
   96             time.sleep(.002)
   97     except Exception as error:
   98         with lock: faults.append(str(error))
   99     finally:
  100         stop.set(); slot.close(); thread.join(10.)
  101         if thread.is_alive(): faults.append('WORKER_DID_NOT_STOP')
  102         try:
  103             pubs, subs, _ = rosgraph.Master(rospy.get_name()).getSystemState(); node = rospy.get_name()
  104             graph = {'subscriptions': sorted(t for t, nodes in subs if node in nodes),
  105                      'publishers': sorted(t for t, nodes in pubs if node in nodes)}
  106             expected = list(TOPICS.values())
  107             if sensors is not None: expected.extend(['/airsim_node/initial_pose','/airsim_node/drone_1/gps'])
  108             if graph['subscriptions'] != sorted(expected) or set(graph['publishers']) - {'/rosout'}:
  109                 faults.append('GRAPH_POLICY')
  110         except Exception as error: faults.append('GRAPH_AUDIT:' + str(error))
  111         for listener in listeners: listener.unregister()
  112         if sensors is not None: sensors.close()
  113         summary = {'faults': faults, 'pairs': pairs, 'received': dict(received), 'overwritten': dict(overwritten),
  114                    'pairing_events': dict(buffer.events), 'slot': slot.stats(), 'results': len(rows),
  115                    'statuses': dict(Counter(row['status'] for row in rows)),
  116                    'tracking_statuses': dict(Counter(row['tracking']['status'] for row in rows if 'tracking' in row)),
  117                    'timing_current_results': sum(row.get('timing_current', False) for row in rows),
  118                    'node_graph': graph, 'image_age_budget_s': image_ttl,
  119                    'source_settings_sha256': rig.settings_sha256, 'method': method, 'pipeline': pipeline, 'control_published': False,
  120                    'debug_inputs_used': False, 'control_eligible': False, 'acceptance_eligible': False}
  121         (run / 'visual_motion_summary.json').write_text(json.dumps(summary, indent=2, allow_nan=False) + '\n')
  122     return 0 if pairs and rows and not faults else 2
