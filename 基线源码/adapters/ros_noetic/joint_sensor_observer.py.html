   27 def raw_record(tag, message, arrival):
   28     def vector(v):return (v.x, v.y, v.z)
   29     def quaternion(q):return (q.w, q.x, q.y, q.z)
   30     record = {'sensor': tag, 'stamp': message.header.stamp.to_sec(),
   31               'stamp_secs': message.header.stamp.secs, 'stamp_nsecs': message.header.stamp.nsecs,
   32               'sequence': message.header.seq, 'frame_id': message.header.frame_id,
   33               'arrival_wall_monotonic': arrival}
   34     if tag == 'imu':
   35         record.update(acceleration_body=vector(message.linear_acceleration),
   36                       omega_body=vector(message.angular_velocity), orientation_wxyz=quaternion(message.orientation),
   37                       acceleration_covariance=tuple(message.linear_acceleration_covariance),
   38                       omega_covariance=tuple(message.angular_velocity_covariance),
   39                       orientation_covariance=tuple(message.orientation_covariance))
   40     else:
   41         record.update(position=vector(message.pose.position), orientation_wxyz=quaternion(message.pose.orientation))
   42     finite = True
   43     def sanitize(value):
   44         nonlocal finite
   45         if isinstance(value, (tuple, list)):return [sanitize(v) for v in value]
   46         if isinstance(value, float) and not math.isfinite(value):
   47             finite = False
   48             return None
   49         return value
   50     record = {key: sanitize(value) for key, value in record.items()}
   51     record['raw_fields_finite'] = finite
   52     return record

   55 def observe(run, duration, image_ttl_s):
   56     import rosgraph
   57     import rospy
   58     from geometry_msgs.msg import PoseStamped
   59     from sensor_msgs.msg import Image, Imu, PointCloud2
   60 
   61     rospy.init_node('rmua_joint_sensors', anonymous=True, disable_signals=True)
   62     history = PoseHistory()
   63     sensors = FlightSensors(pose_history=history)
   64     pairs = StereoPairBuffer()
   65     clouds = LidarAssociationBuffer()
   66     lock = threading.Lock()
   67     pending, last_received = {}, {}
   68     raw_pending = deque()
   69     received = Counter()
   70     raw_drops = 0
   71 
   72     def receive(tag, message):
   73         with lock:
   74             arrival = time.monotonic()
   75             pending[tag] = (message, arrival)
   76             last_received[tag] = arrival
   77             received[tag] += 1
   78 
   79     def receive_raw(tag, message):
   80         nonlocal raw_drops
   81         with lock:
   82             if len(raw_pending) == 512:
   83                 raw_pending.popleft()
   84                 raw_drops += 1
   85             raw_pending.append((tag, message, time.monotonic()))
   86             received[tag] += 1
   87 
   88     listeners = [rospy.Subscriber(topic, PointCloud2 if tag == 'lidar' else Image,
   89                  lambda msg, t=tag: receive(t, msg), queue_size=1, buff_size=4000000)
   90                  for tag, topic in TOPICS.items()]
   91     listeners += [rospy.Subscriber('/airsim_node/drone_1/imu/imu', Imu,
   92                                   lambda msg: receive_raw('imu', msg), queue_size=1),
   93                   rospy.Subscriber('/airsim_node/drone_1/gps', PoseStamped,
   94                                    lambda msg: receive_raw('gps', msg), queue_size=1)]
   95     original = {tag: OrderedDict() for tag in TOPICS}
   96     packet_root = run / 'joint'
   97     packet_root.mkdir(exist_ok=False)
   98     captures, reasons = [], Counter()
   99     source_count = raw_count = stereo_count = 0
  100     dumped = None
  101     last_capture = None
  102     fault = graph = None
  103     started = time.monotonic()
  104     with (run / 'joint_states.jsonl').open('x') as states, (run / 'joint_raw_sensors.jsonl').open('x') as raw:
  105         def dump_sources():
  106             nonlocal source_count, raw_count, dumped
  107             for state, arrival in history.snapshot_sources():
  108                 key = (state.metadata.epoch, state.metadata.stamp)
  109                 if dumped is None or key > dumped:
  110                     states.write(json.dumps({'state': asdict(state), 'arrival_wall_monotonic': arrival}, allow_nan=False) + '\n')
  111                     dumped = key
  112                     source_count += 1
  113             with lock:
  114                 batch = list(raw_pending)
  115                 raw_pending.clear()
  116             for tag, message, arrival in batch:
  117                 raw.write(json.dumps(raw_record(tag, message, arrival), allow_nan=False) + '\n')
  118                 raw_count += 1
  119 
  120         try:
  121             while time.monotonic() - started < duration and not rospy.is_shutdown():
  122                 dump_sources()
  123                 with sensors.lock:
  124                     if sensors.fault:raise RuntimeError(sensors.fault)
  125                     ready = sensors.estimator.initialized and sensors.raw is not None
  126                     if ready and not sensors.clock.fresh(time.monotonic()):raise RuntimeError('STALE_IMU')
  127                     now, epoch = sensors.clock.stamp, sensors.clock.epoch
  128                 with lock:
  129                     batch = dict(pending);pending.clear()
  130                     latest = dict(last_received)
  131                 if time.monotonic() - started > 5:
  132                     if not ready or set(latest) != set(TOPICS):raise RuntimeError('JOINT_STREAM_NOT_READY')
  133                     if any(time.monotonic() - t > .5 for t in latest.values()):raise RuntimeError('STALE_JOINT_STREAM')
  134                     if not captures:raise RuntimeError('NO_JOINT_CAPTURE_WITHIN_LIMITS')
  135                 if not ready:
  136                     time.sleep(.003)
  137                     continue
  138                 for tag, (message, arrival) in batch.items():
  139                     if tag == 'lidar':
  140                         decoded = decode_xyz(message, epoch)
  141                         accepted = clouds.add(decoded, arrival)
  142                     else:
  143                         decoded = decode_image(message, tag, epoch, ttl_s=image_ttl_s)
  144                         accepted = pairs.add(tag, decoded, arrival)
  145                     if accepted:
  146                         original[tag][decoded.metadata.stamp] = message
  147                         while len(original[tag]) > 8:original[tag].popitem(last=False)
  148                 with sensors.lock:
  149                     if sensors.fault:raise RuntimeError(sensors.fault)
  150                     now, epoch = sensors.clock.stamp, sensors.clock.epoch
  151                 pair = pairs.pop(now, time.monotonic())
  152                 if pair is None:
  153                     time.sleep(.003)
  154                     continue
  155                 stereo_count += 1
  156                 if len(captures) >= 6 or (last_capture is not None and time.monotonic() - last_capture < 1.5):
  157                     continue
  158                 selection_wall = time.monotonic()
  159                 association = clouds.select(pair, now, selection_wall, epoch)
  160                 if association is None:
  161                     reasons['NO_LIDAR_ASSOCIATION'] += 1
  162                     continue
  163                 stamps = {'left': pair.left.metadata.stamp, 'right': pair.right.metadata.stamp,
  164                           'lidar': association.cloud.metadata.stamp}
  165                 try:
  166                     with sensors.lock:
  167                         if sensors.fault:raise RuntimeError(sensors.fault)
  168                         query_now, query_wall = sensors.clock.stamp, time.monotonic()
  169                         poses = {tag: history.lookup(stamp, query_now, query_wall, epoch) for tag, stamp in stamps.items()}
  170                 except HistoryUnavailable as error:
  171                     reasons['HISTORY:' + str(error)] += 1
  172                     continue
  173                 messages = {tag: original[tag].get(stamp) for tag, stamp in stamps.items()}
  174                 if any(message is None for message in messages.values()):
  175                     reasons['RAW_MESSAGE_EVICTED'] += 1
  176                     continue
  177                 directory = packet_root / str(len(captures)).zfill(4)
  178                 directory.mkdir(exist_ok=False)
  179                 for tag, message in messages.items():
  180                     save_message(directory, tag, message, 'lidar' if tag == 'lidar' else 'image')
  181                 packet = {'stamps': stamps, 'epoch': epoch, 'query_now': query_now, 'query_wall': query_wall,
  182                           'selection_now': now, 'selection_wall': selection_wall,
  183                           'poses': {tag: asdict(pose) for tag, pose in poses.items()},
  184                           'stereo_header_skew_s': pair.skew_s,
  185                           'left_lidar_header_skew_s': association.left_skew_s,
  186                           'right_lidar_header_skew_s': association.right_skew_s,
  187                           'left_source_metadata': asdict(pair.left.metadata),
  188                           'right_source_metadata': asdict(pair.right.metadata),
  189                           'lidar_source_metadata': asdict(association.cloud.metadata),
  190                           'lidar_current_at_selection': association.lidar_current_at_selection,
  191                           'image_age_budget_s': image_ttl_s, 'capture_wall': time.monotonic(),
  192                           'current_control_eligible': False, 'per_point_motion_compensation_applied': False,
  193                           'exposure_sync_verified': False, 'world_projection_applied': False}
  194                 packet_path = directory / 'packet.json'
  195                 packet_path.write_text(json.dumps(packet, indent=2, allow_nan=False) + '\n')
  196                 captures.append({'directory': str(directory.relative_to(run)),
  197                     'packet_sha256': hashlib.sha256(packet_path.read_bytes()).hexdigest()})
  198                 last_capture = time.monotonic()
  199             if not captures:raise RuntimeError('NO_JOINT_CAPTURES')
  200         except Exception as error:
  201             fault = str(error)
  202         finally:
  203             try:
  204                 pubs, subs, _ = rosgraph.Master(rospy.get_name()).getSystemState();node = rospy.get_name()
  205                 graph = {'subscriptions': sorted(t for t, nodes in subs if node in nodes),
  206                          'publishers': sorted(t for t, nodes in pubs if node in nodes)}
  207                 if graph['subscriptions'] != INPUTS or set(graph['publishers']) - {'/rosout'}:
  208                     fault = fault or 'JOINT_OBSERVER_GRAPH_POLICY'
  209             except Exception as error:
  210                 fault = fault or 'GRAPH_AUDIT_FAILED:' + str(error)
  211             for listener in listeners:listener.unregister()
  212             sensors.close()
  213             dump_sources()
  214     summary = {'captures': captures, 'stereo_pairs': stereo_count, 'received': dict(received),
  215                'source_states': source_count, 'raw_sensor_records': raw_count, 'raw_queue_drops': raw_drops,
  216                'rejections': dict(reasons), 'stereo_events': dict(pairs.events), 'lidar_events': dict(clouds.events),
  217                'fault': fault, 'node_graph': graph, 'image_age_budget_s': image_ttl_s,
  218                'max_stereo_skew_s': .005, 'max_lidar_skew_s': clouds.max_skew_s,
  219                'raw_recording_is_lossy': True, 'payloads_retained_exactly_for_selected_captures': True,
  220                'control_published': False, 'debug_inputs_used': False,
  221                'per_point_motion_compensation_applied': False, 'acceptance_eligible': False}
  222     (run / 'joint_summary.json').write_text(json.dumps(summary, indent=2) + '\n')
  223     return 0 if captures and fault is None and not raw_drops else 2

  226 def main():
  227     parser = argparse.ArgumentParser(description=__doc__)
  228     parser.add_argument('--run-dir', type=Path, required=True)
  229     parser.add_argument('--duration', type=float, default=10.)
  230     parser.add_argument('--image-age-budget-ms', type=int, choices=(150, 200), default=150)
  231     args = parser.parse_args()
  232     if not math.isfinite(args.duration) or not 0 < args.duration <= 30:
  233         parser.error('duration must be in (0,30] seconds')
  234     return observe(args.run_dir, args.duration, args.image_age_budget_ms / 1000.)
