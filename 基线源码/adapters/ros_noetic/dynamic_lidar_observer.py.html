   19 def wait_for_imu(cloud, arrival, snapshot, wall_clock, sleep, stopped, max_wait_s=.03):
   20     """Wait briefly for causal IMU coverage without extending source lifetime."""
   21     while not stopped():
   22         now, epoch, fresh, _ = snapshot()
   23         elapsed = wall_clock() - arrival
   24         if (not fresh or epoch != cloud.metadata.epoch or now is None
   25                 or now >= cloud.metadata.stamp or elapsed < 0 or elapsed >= max_wait_s):
   26             return
   27         sleep(min(.001, max_wait_s - elapsed))
