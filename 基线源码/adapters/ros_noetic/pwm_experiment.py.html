   31 def motor_pattern():
   32     # Lift gently, then probe one motor at a time with very small positive pulses.
   33     pattern = [(.20,)*4]*4 + [(.176,)*4]*2
   34     for motor in range(4):
   35         values = [.176]*4
   36         values[motor] += .003
   37         pattern.append(tuple(values))
   38         pattern.append((.176,)*4)
   39     return tuple(pattern)

   42 def run_experiment(run, levels, pulse_seconds, motor_pulses=False, hover_seconds=0, step=False,
   43                    sensor_fault=None, fault_seconds=.15, gps_bridge=False, yaw_motion=False,braking=False,
   44                    stationary_bootstrap=False,angular_recovery=False,angular_pulse=False,
   45                    isolated_reference=False,reference_stall_seconds=0.,collective_pulse_pwm=None,
   46                    angular_pulse_moment=.5,angular_recovery_profile='standard'):
   47     import rospy
   48     import rosgraph
   49     from airsim_ros.msg import RotorPWM
   50     rospy.init_node('rmua_next_pwm_identification', anonymous=True, disable_signals=True)
   51     master = rosgraph.Master(rospy.get_name())
   52     scene_id = rospy.get_param('/run_id')

   55     fresh_read=False
   56     fault_observation=None
   57     last_published_wall=None
   58     def graph(require_self):

   62     with ControlLease(scene_id, 'PWM'):
   63         graph(False)
   64         sensors = FlightSensors()
   65         publisher = None
   66         monitor = None
   67         recovery = HoverRecovery()
   68         angular_supervisor=AngularRecovery(angular_recovery_profile)
   69         pulse_sent=False
   70         reference_worker=None
   71         reference_hold=ReferenceHold()
   72         collective_pulse=CollectivePulse(collective_pulse_pwm) if collective_pulse_pwm is not None else None
   73         try:
   74             deadline = time.monotonic()+5
   75             while time.monotonic() < deadline:
   76                 try:
   77                     state, raw = sensors.read()
   78                     bootstrap=None
   79                     if state.health == Health.OK and stationary_bootstrap:
   80                         bootstrap=sensors.bootstrap_at_rest_before_control()

   82                         if bootstrap is not None:
   83                             (run/'stationary_bootstrap.json').write_text(json.dumps(bootstrap,indent=2)+'\n')
   84                             state,raw=sensors.read()
   85                         break
   86                 except (ValueError, RuntimeError):
   87                     pass
   88                 time.sleep(.01)
   89             else:
   90                 raise RuntimeError('NO_HEALTHY_STATE')
   91             origin = np.array(state.position_ned)
   92             publisher = rospy.Publisher('/airsim_node/drone_1/rotor_pwm_cmd', RotorPWM, queue_size=1)
   93             expected_connections,backend_nodes=expected_control_connections(master.getSystemState(),
   94                                                '/airsim_node/drone_1/rotor_pwm_cmd')
   95             deadline = time.monotonic()+2
   96             while publisher.get_num_connections() < expected_connections and time.monotonic() < deadline:
   97                 time.sleep(.01)
   98             if publisher.get_num_connections() < expected_connections:
   99                 raise RuntimeError('PWM_SUBSCRIBER_COUNT')
  100             graph(True)
  101             _, subscribers, _ = master.getSystemState()
  102             runtime_inputs = sorted(topic for topic, nodes in subscribers if rospy.get_name() in nodes)
  103             official_config = json.loads((Path(__file__).resolve().parents[2]/'configs/official_baseline.yaml').read_text())
  104             if any(topic not in official_config['sensors'] for topic in runtime_inputs):
  105                 raise RuntimeError('INPUT_WHITELIST_POLICY')
  106             (run/'runtime_input_audit.json').write_text(json.dumps({'node': rospy.get_name(),
  107                 'topics': runtime_inputs, 'debug_inputs': False,'expected_control_connections':expected_connections,
  108                 'control_connections':publisher.get_num_connections(),'backend_subscriber_nodes':backend_nodes}, indent=2)+'\n')
  109             monitor = GraphMonitor(lambda: graph(True))
  110             monitor.start()
  111             guard = CommandGuard(sensors.clock.epoch, minimum_period=.0105)
  112             pattern = motor_pattern() if motor_pulses else tuple((v,)*4 for v in levels)
  113             config = json.loads((Path(__file__).resolve().parents[2]/'configs/vehicle_identification.json').read_text())
  114             (run/'vehicle_model.json').write_text(json.dumps(config,indent=2)+'\n')
  115             allocator, controller = MotorAllocator(config), Se3Controller(config)
  116             previous_saturated = False
  117             yaw_reference = math.atan2(rotation(state.orientation_wxyz)[1, 0],
  118                                        rotation(state.orientation_wxyz)[0, 0])
  119             state, raw = sensors.read()
  120             if isolated_reference:
  121                 reference_worker=ReferenceWorker(BenchmarkSpec(tuple(origin),yaw_reference,step,braking,yaw_motion),
  122                                                   stall_seconds=reference_stall_seconds)
  123                 reference_worker.start()
  124                 deadline=time.monotonic()+3.
  125                 while time.monotonic()<deadline:
  126                     state,raw=sensors.read()
  127                     reference_worker.submit(state.metadata.stamp,state.metadata.epoch,0.)
  128                     packet,reason=reference_worker.snapshot(state.metadata.stamp,state.metadata.epoch)
  129                     if packet is not None:break
  130                     time.sleep(.005)
  131                 else:raise RuntimeError('REFERENCE_WORKER_STARTUP')
  132             start = time.monotonic()
  133             state, raw = sensors.read()
  134             start_stamp = state.metadata.stamp
  135             if sensor_fault:
  136                 sensors.local_injection = LocalSensorFault(sensor_fault, start_stamp+6., fault_seconds)
  137             next_tick = start
  138             previous_control_stamp = start_stamp-.01
  139             sequence = 0
  140             while True:
  141                 now = time.monotonic()
  142                 index = int((now-start)/pulse_seconds)
  143                 if (hover_seconds and now-start >= hover_seconds) or (not hover_seconds and index >= len(pattern)):
  144                     break
  145                 fresh_read=False
  146                 state, raw = sensors.read()
  147                 latest_state,latest_raw,latest_elapsed=state,raw,now-start
  148                 fresh_read=True
  149                 R = rotation(state.orientation_wxyz)
  150                 if state.health != Health.OK and not (hover_seconds and gps_bridge):
  151                     raise RuntimeError('STATE_HEALTH_'+state.health.value)
  152                 angular_decision=angular_supervisor.decide(state) if angular_recovery else None
  153                 if not angular_recovery and R[2, 2] < math.cos(.25):
  154                     raise RuntimeError('TILT_LIMIT')
  155                 if np.linalg.norm(state.velocity_ned) > 4:
  156                     raise RuntimeError('SPEED_LIMIT')
  157                 if not angular_recovery and np.linalg.norm(state.angular_velocity_body) > 1.:
  158                     raise RuntimeError('ANGULAR_RATE_LIMIT')
  159                 if np.linalg.norm(np.array(state.position_ned)-origin) > (1.8 if hover_seconds else 1.2):
  160                     raise RuntimeError('POSITION_LIMIT')
  161                 monitor.check()
  162                 details = {}
  163                 if hover_seconds:
  164                     sim_elapsed = state.metadata.stamp-start_stamp
  165                     if reference_worker is None:
  166                         reference, velocity, acceleration = benchmark_reference(origin, sim_elapsed, step,braking)
  167                         yaw_command,yaw_rate = benchmark_yaw(yaw_reference,sim_elapsed,yaw_motion)
  168                     allow_integral = not previous_saturated
  169                     phase = 'NORMAL'
  170                     rate_gain_scale=1.
  171                     attitude_gain_scale=1.
  172                     if reference_worker is not None:
  173                         reference_worker.submit(state.metadata.stamp,state.metadata.epoch,sim_elapsed)
  174                         packet,reason=reference_worker.snapshot(state.metadata.stamp,state.metadata.epoch)
  175                         selected=reference_hold.select(state,packet,reason)
  176                         reference,velocity,acceleration=selected.position,selected.velocity,selected.acceleration
  177                         yaw_command,yaw_rate=selected.yaw,selected.yaw_rate
  178                         phase=selected.phase
  179                         allow_integral=allow_integral and selected.allow_integral
  180                     if gps_bridge:
  181                         gps_stamp = raw['last_gps_stamp']
  182                         gps_age = math.inf if gps_stamp is None else state.metadata.stamp-gps_stamp
  183                         decision = recovery.decide(state,gps_age,(reference,velocity,acceleration))
  184                         reference,velocity,acceleration = decision.position,decision.velocity,decision.acceleration
  185                         allow_integral = allow_integral and decision.allow_integral
  186                         phase = decision.phase
  187                         if phase != 'NORMAL':
  188                             yaw_command,yaw_rate = yaw_reference,0.
  189                     if angular_decision is not None and angular_decision.phase!='NORMAL':
  190                         reference=angular_decision.hold_position
  191                         velocity=acceleration=(0.,0.,0.)
  192                         yaw_command,yaw_rate=angular_decision.hold_yaw,0.
  193                         rate_gain_scale=angular_decision.rate_gain_scale
  194                         attitude_gain_scale=angular_decision.attitude_gain_scale
  195                         allow_integral=False
  196                         phase=angular_decision.phase
  197                     dt = state.metadata.stamp-previous_control_stamp
  198                     if dt <= 0:
  199                         time.sleep(.002)
  200                         continue
  201                     if dt > .05:
  202                         raise RuntimeError('CONTROL_STATE_GAP')
  203                     wrench = controller.step(state, reference, velocity, acceleration,
  204                                              yaw_command, dt, allow_integral,yaw_rate=yaw_rate,
  205                                              rate_gain_scale=rate_gain_scale,attitude_gain_scale=attitude_gain_scale)
  206                     moment=np.array(wrench.moment)
  207                     pulse_this_tick=(angular_pulse and not pulse_sent and sim_elapsed>=5.
  208                                      and phase=='NORMAL')
  209                     if pulse_this_tick:moment[1]+=angular_pulse_moment
  210                     allocation = allocator.allocate(wrench.thrust, moment)
  211                     collective_values=collective_pulse.request(state,sim_elapsed) if collective_pulse else None
  212                     if collective_values is not None:
  213                         achieved=allocator.B@(allocator.max_thrust*np.array(collective_values))
  214                         allocation=Allocation(collective_values,tuple(achieved),True,0.,
  215                                               min(collective_values[0],1-collective_values[0]))
  216                     previous_saturated = allocation.saturated
  217                     values = allocation.pwm
  218                     details = {'reference': tuple(reference), 'wrench': asdict(wrench),
  219                                'allocation': asdict(allocation), 'sim_elapsed': sim_elapsed,'supervisor_phase':phase,
  220                                'yaw_reference':yaw_command,'yaw_rate_reference':yaw_rate}
  221                     details['angular_pulse_applied']=pulse_this_tick
  222                     details['rate_gain_scale']=rate_gain_scale
  223                     details['attitude_gain_scale']=attitude_gain_scale
  224                     details['moment_requested_including_experiment_pulse']=moment.tolist()
  225                     details['collective_pulse_applied']=collective_values is not None
  226                 else:
  227                     values = pattern[index]
  228                 command = PwmCommand(state.metadata.stamp, state.metadata.epoch, sequence, values)
  229                 with sensors.lock:
  230                     accepted_wall = time.monotonic()
  231                     allowed = guard.accept(command, sensors.clock, accepted_wall)
  232                 if allowed:
  233                     msg = RotorPWM()
  234                     msg.header.stamp = rospy.Time.from_sec(command.stamp)
  235                     msg.header.seq = sequence
  236                     msg.rotorPWM0, msg.rotorPWM1, msg.rotorPWM2, msg.rotorPWM3 = command.values
  237                     publish_begin_wall=time.monotonic()
  238                     publisher.publish(msg)
  239                     published_wall = time.monotonic()
  240                     last_published_wall=published_wall
  241                     guard.published(published_wall)
  242                     rows.append({'wall_elapsed': now-start, 'command': asdict(command),
  243                                  'guard_accept_wall_elapsed': accepted_wall-start,
  244                                  'publish_wall_elapsed': published_wall-start,
  245                                  'publish_begin_wall_monotonic':publish_begin_wall,
  246                                  'publish_end_wall_monotonic':published_wall,
  247                                  'raw': raw, 'state': {k:v for k,v in asdict(state).items() if k != 'metadata'},
  248                                  **details})
  249                     sequence += 1
  250                     if details.get('angular_pulse_applied'):pulse_sent=True
  251                     if details.get('collective_pulse_applied'):collective_pulse.published(command.stamp)
  252                 previous_control_stamp = state.metadata.stamp
  253                 next_tick += 1/80.
  254                 delay = next_tick-time.monotonic()
  255                 if delay < -.1:
  256                     raise RuntimeError('CONTROL_DEADLINE')
  257                 time.sleep(max(0., delay))
  258             if collective_pulse is not None and not collective_pulse.sent:
  259                 raise RuntimeError('COLLECTIVE_EXCITATION_NOT_SENT')
  260             if angular_pulse and not pulse_sent:raise RuntimeError('ANGULAR_EXCITATION_NOT_SENT')
  261             if angular_pulse and angular_supervisor.phase!='RECOVERY_HOLD':
  262                 raise RuntimeError('ANGULAR_RECOVERY_NOT_SETTLED')
  263         except Exception as error:
  264             fault = str(error)
  265             failed_wall=time.monotonic()
  266             fault_observation={'reason':fault,'fresh_read_in_fault_iteration':fresh_read,
  267                 'failure_wall_monotonic':failed_wall,
  268                 'wall_elapsed_at_last_read':latest_elapsed,
  269                 'wall_since_last_publication_s':None if last_published_wall is None else failed_wall-last_published_wall,
  270                 'state':asdict(latest_state) if latest_state else None,'raw':latest_raw}
  271         finally:
  272             control_end_wall=time.monotonic()
  273             if monitor:
  274                 monitor.close()
  275             if publisher:
  276                 publisher.unregister()
  277             if reference_worker:reference_worker.close()
  278             sensors.close()
  279     (run/'pwm_telemetry.jsonl').write_text(''.join(json.dumps(row)+'\n' for row in rows))
  280     if fault_observation is not None:
  281         (run/'fault_observation.json').write_text(json.dumps(fault_observation,indent=2)+'\n')
  282     result = {'commands': len(rows), 'fault': fault, 'levels': levels,
  283               'control_end_wall_monotonic':control_end_wall,
  284               'motor_pulses': motor_pulses,
  285               'hover_seconds': hover_seconds,
  286               'step': step,
  287               'reference_clock': 'IMU',
  288               'gps_rejections': sensors.estimator.rejections,
  289               'injection': asdict(sensors.local_injection) if sensors.local_injection else None,
  290               'graph_monitor':monitor.summary() if monitor else None,
  291               'gps_bridge_enabled':gps_bridge,'recovery_transitions':recovery.transitions,
  292               'yaw_motion':yaw_motion,
  293               'braking':braking,
  294               'angular_recovery_enabled':angular_recovery,
  295               'angular_recovery_phase':angular_supervisor.phase,
  296               'angular_recovery_profile':angular_recovery_profile,
  297               'angular_recovery_transitions':angular_supervisor.transitions,
  298               'angular_pulse_sent':pulse_sent,
  299               'angular_pulse_moment_Nm':angular_pulse_moment if angular_pulse else None,
  300               'reference_worker':reference_worker.summary() if reference_worker else None,
  301               'reference_hold_transitions':reference_hold.transitions,
  302               'collective_excitation':None if collective_pulse is None else vars(collective_pulse),
  303               'pulse_seconds': pulse_seconds,
  304               'stage': 'LOCAL_PWM_CLOSED_LOOP_HOVER' if hover_seconds else 'LOCAL_OPEN_LOOP_IDENTIFICATION',
  305               'official_result': 'UNKNOWN', 'host_scene_stop_required': True}
  306     (run/'pwm_summary.json').write_text(json.dumps(result, indent=2)+'\n')
  307     print(json.dumps(result), flush=True)
  308     return 0 if rows and fault is None else 2

  311 def main():
  312     parser = argparse.ArgumentParser(description=__doc__)
  313     parser.add_argument('--run-dir', type=Path, required=True)
  314     parser.add_argument('--levels', default='.10,.15,.20,.25,.30,.35,.40')
  315     parser.add_argument('--pulse-seconds', type=float, default=.3)
  316     parser.add_argument('--motor-pulses', action='store_true')
  317     parser.add_argument('--hover-seconds', type=float, default=0.)
  318     parser.add_argument('--step', action='store_true')
  319     parser.add_argument('--sensor-fault', choices=('GPS_ZERO','IMU_DROP','IMU_ROLLBACK'))
  320     parser.add_argument('--fault-seconds',type=float,default=.15)
  321     parser.add_argument('--gps-bridge',action='store_true')
  322     parser.add_argument('--yaw-motion',action='store_true')
  323     parser.add_argument('--braking',action='store_true')
  324     parser.add_argument('--stationary-bootstrap',action='store_true')
  325     parser.add_argument('--angular-recovery',action='store_true')
  326     parser.add_argument('--angular-recovery-profile',choices=('standard','fast'),default='standard')
  327     parser.add_argument('--angular-pulse',action='store_true',help='single bounded pitch moment pulse at 5s')
  328     parser.add_argument('--angular-pulse-moment',type=float)
  329     parser.add_argument('--isolated-reference',action='store_true')
  330     parser.add_argument('--reference-stall-seconds',type=float,default=0.)
  331     parser.add_argument('--collective-pulse-pwm',type=float)
  332     args = parser.parse_args()
  333     levels = tuple(float(value) for value in args.levels.split(','))
  334     if (not levels or len(levels) > 20 or not all(0 < v <= .6 for v in levels)
  335             or not .05 <= args.pulse_seconds <= 1 or not 0 <= args.hover_seconds <= 20
  336             or (args.hover_seconds and args.motor_pulses)):
  337         parser.error('bounded identification requires <=20 levels in (0,.6], pulses [.05,1]s')
  338     if args.step and args.hover_seconds < 10:
  339         parser.error('step profile requires hover-seconds >=10')
  340     if (args.braking or args.yaw_motion) and args.hover_seconds < 10:
  341         parser.error('braking/yaw identification requires >=10s hover experiment')
  342     if args.braking and args.step:
  343         parser.error('braking and position step are separate experiments')
  344     if args.sensor_fault and (args.hover_seconds < 8 or not 0 < args.fault_seconds <= 1):
  345         parser.error('local sensor injection requires >=8s hover and duration (0,1]')
  346     if args.angular_recovery and (args.hover_seconds<10 or args.gps_bridge or args.sensor_fault):
  347         parser.error('angular recovery requires >=10s healthy-state hover without GPS/fault modes')
  348     if args.angular_recovery_profile!='standard' and not args.angular_recovery:
  349         parser.error('fast recovery profile requires angular-recovery')
  350     if args.angular_pulse and (not args.angular_recovery or args.step or args.braking or args.yaw_motion):
  351         parser.error('angular pulse requires isolated angular-recovery hover')
  352     if args.angular_pulse_moment is not None and (not args.angular_pulse or not 0<args.angular_pulse_moment<=.8):
  353         parser.error('angular-pulse-moment requires angular-pulse, moment (0,.8] Nm')
  354     if args.isolated_reference and (args.hover_seconds<10 or args.angular_recovery or args.gps_bridge or args.sensor_fault):
  355         parser.error('isolated reference requires >=10s healthy-state hover without other fault modes')
  356     if args.reference_stall_seconds and (not args.isolated_reference or not .1<=args.reference_stall_seconds<=1.):
  357         parser.error('reference stall requires isolated reference and duration [.1,1]')
  358     if args.collective_pulse_pwm is not None and (not 0<=args.collective_pulse_pwm<=1
  359             or args.hover_seconds<10 or args.step or args.braking or args.yaw_motion or args.sensor_fault
  360             or args.gps_bridge or args.angular_recovery or args.isolated_reference):
  361         parser.error('collective pulse requires isolated >=10s healthy-state hover, PWM in [0,1]')
  362     return run_experiment(args.run_dir, levels, args.pulse_seconds, args.motor_pulses, args.hover_seconds, args.step,
  363                           args.sensor_fault,args.fault_seconds,args.gps_bridge,args.yaw_motion,args.braking,
  364                           args.stationary_bootstrap,args.angular_recovery,args.angular_pulse,
  365                           args.isolated_reference,args.reference_stall_seconds,args.collective_pulse_pwm,
  366                           .5 if args.angular_pulse_moment is None else args.angular_pulse_moment,
  367                           args.angular_recovery_profile)
